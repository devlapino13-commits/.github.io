CREATE PROCEDURE PDN_SAVE_JSON(
    OBJ_ID D_ID,
    UN_ID D_ID,
    ACTION_DATA D_STR_8100,
    ACTION_KIND D_STR_CODE)
AS
declare variable survey_json d_str_8100;
declare variable jso1 d_integer;
declare variable jso2 d_integer;
declare variable gdoc d_smallint;
declare variable income decimal(18,2);
declare variable taxrate d_smallint;
declare variable incomewt decimal(18,2);
declare variable nmonth d_smallint;
declare variable incomeavg decimal(18,2);
declare variable data_count d_smallint;
declare variable dtjob d_str_20;
declare variable k_id d_id;
declare variable c_id d_id;
declare variable acl_id d_id;
declare variable scoring_cnt d_smallint;
declare variable pdn_limit d_currency;
declare variable curr_reg d_str_100;
declare variable curr_rosstat_income d_currency;
declare variable curr_anketa_income d_currency;
declare variable curr_rosstat_income_m d_currency;
declare variable curr_anketa_income_m d_currency;
declare variable sp_new d_currency;
declare variable sp_old d_currency;
declare variable sp d_currency;
declare variable pdn d_currency;
declare variable pdn_new d_currency;
declare variable income_new d_currency;
declare variable income_other d_currency;
declare variable income_all d_currency;
declare variable kc_id d_id;
declare variable kc_date d_date;
declare variable ki_id d_id;
declare variable multiplier d_currency;
begin
    survey_json = '{}';
    scoring_cnt = null;
    pdn_limit = null;
    curr_reg = null;
    curr_rosstat_income = null;
    curr_anketa_income = null;
    sp_new = null;
    sp_old = null;
    sp = null;
    pdn = null;
    pdn_new = null;
    income_new = null;
    income_other = null;
    income_all =  null;
    kc_date =  null;

    select c_id, k_id, acl_id, kc_id, survey_json
    from pdn_info_action('miniappclient_income', :obj_id)
    into :c_id, :k_id, :acl_id, :kc_id, :survey_json;

    if (:action_kind = 'send_kk') then
    begin
        update or insert into kk_income (
            ki_from_k_id,
            ki_from_c_id,
            ki_from_acl_id,
            ki_from_un_id,
            ki_verify_income,
            ki_is_manual,
            ki_survey)
        values (
            :k_id,
            :c_id,
            :acl_id,
            :un_id,
            f_json_blob_get_value(:survey_json, 'incomeavg', '0'),
            0,
            :survey_json)
        matching (ki_from_k_id, ki_from_acl_id, ki_from_c_id);
    
        :survey_json = '{}';

        update miniappclient_income maci
        set maci.maci_is_blocked = 1
        where maci.maci_from_mac_id = :obj_id;
    end
    else
    if (:action_kind = 'kont_to_ma') then
    begin
        select ki_survey
        from kk_income
        where ki_from_k_id = :k_id
          and ki_from_c_id = :c_id
        into :survey_json;
    end
    else
    if(:action_kind = 'send_to_revision') then
    begin
        select ki.ki_id, ki.ki_survey
        from kk_income ki
        where ki.ki_from_k_id = :k_id
          and ki.ki_from_c_id = :c_id
          and ki.ki_from_acl_id = :acl_id
        into :ki_id, :survey_json;

        survey_json = coalesce(:survey_json, '{}');

        update kk_income
        set ki_survey = '{}'
        where ki_id = :ki_id;

        update miniappclient_income
        set maci_is_blocked = 0
        where maci_from_mac_id = :obj_id;
    end
    else
    begin
        jso1 = json.parse(:action_data);
        jso2 = json.parse(:survey_json);
    
        if (:action_kind = 'employer_save') then
        begin
            select scoring_cnt,
                   pdn_limit,
                   curr_reg,
                   curr_rosstat_income,
                   curr_anketa_income, 
                   curr_rosstat_income_m,
                   curr_anketa_income_m,
                   sp_new,
                   sp_old,
                   sp,
                   pdn,
                   pdn_new,
                   income_new,
                   income_other,
                   income_all,
                   kc_date,
                   multiplier
            from hp_pdn_getinfo_add(:kc_id)
            into :scoring_cnt,
                 :pdn_limit,
                 :curr_reg,
                 :curr_rosstat_income,
                 :curr_anketa_income, 
                 :curr_rosstat_income_m,
                 :curr_anketa_income_m,
                 :sp_new,
                 :sp_old,
                 :sp,
                 :pdn,
                 :pdn_new,
                 :income_new,
                 :income_other,
                 :income_all,
                 :kc_date,
                 :multiplier;

            gdoc   = json.obj_get_val(jso1, 'gdoc', -1);
            income = json.obj_get_val(jso2, 'income', 0);
            dtjob = json.obj_get_val(jso1, 'dtjob', null);

            json.obj_set_i(jso2, 'gdoc', :gdoc);
            json.obj_set_s(jso2, 'dtjob', :dtjob);
            json.obj_set_i(jso2, 'dtlastm', json.obj_get_val(jso1, 'dtlastm', null));
            json.obj_set_s(jso2, 'dtpdn', f_date_format('dd.mm.yyyy', f_date_month_begin(:kc_date)));
            json.obj_set_s(jso2, 'dtjobb', f_date_format('dd.mm.yyyy', f_date_month_begin(dateadd(-1 year to f_date_month_begin(:kc_date)))));
            json.obj_set_s(jso2, 'dtjobe', f_date_format('dd.mm.yyyy', f_date_month_end(dateadd(-4 month to f_date_month_begin(:kc_date)))));
            json.obj_set_s(jso2, 'employer', json.obj_get_val(jso1, 'employer', ''));
            json.obj_set_i(jso2, 'scoring_cnt', :scoring_cnt);
            json.obj_set_f(jso2, 'pdn_limit', :pdn_limit);
            json.obj_set_f(jso2, 'koef', :multiplier);
            json.obj_set_s(jso2, 'curr_reg', :curr_reg);
            json.obj_set_f(jso2, 'curr_rosstat_income', :curr_rosstat_income);
            json.obj_set_f(jso2, 'curr_anketa_income', :curr_anketa_income); 
            json.obj_set_f(jso2, 'curr_rosstat_income_m', :curr_rosstat_income_m);
            json.obj_set_f(jso2, 'curr_anketa_income_m', :curr_anketa_income_m);
            json.obj_set_f(jso2, 'sp_new', :sp_new);
            json.obj_set_f(jso2, 'sp_old', :sp_old);
            json.obj_set_f(jso2, 'sp', :sp);
            json.obj_set_f(jso2, 'pdn', :pdn);
            json.obj_set_f(jso2, 'pdn_new', :pdn_new);
            json.obj_set_f(jso2, 'income_new', :income_new);
            json.obj_set_f(jso2, 'income_other', :income_other);
            json.obj_set_f(jso2, 'income_all', :income_all);
    
            taxrate  = iif(:gdoc = 2, 0, 13);
            incomewt = :income * (1 - (:taxrate * 0.01));
    
            if (coalesce(:dtjob, '') = '') then
                data_count = 12;
            else
                data_count = f_json_blob_get_count(:survey_json, 'data');
    
            nmonth     = iif(:data_count > 12, 12, iif(:data_count < 1, 1, :data_count));
            incomeavg  = iif(:nmonth > 0, :incomewt / :nmonth, 0);
    
            json.obj_set_i(jso2, 'taxrate',   :taxrate);
            json.obj_set_f(jso2, 'incomewt',  :incomewt);
            json.obj_set_i(jso2, 'nmonth',    :nmonth);
            json.obj_set_f(jso2, 'incomeavg', :incomeavg);

        end
        else if (:action_kind = 'monthly_save') then
        begin
            json.obj_set_arr(jso2, 'data', json.obj_get_arr(jso1, 'data'));
            json.obj_set_f(jso2, 'income', cast(json.obj_get_val(jso1, 'income', 0) as decimal(18,4)));
    
            gdoc   = json.obj_get_val(jso2, 'gdoc', -1);
            income = json.obj_get_val(jso2, 'income', 0);
            dtjob = json.obj_get_val(jso2, 'dtjob', null);
    
            taxrate  = iif(:gdoc = 2, 0, 13);
            incomewt = :income * (1 - (:taxrate * 0.01));
    
            if (coalesce(:dtjob, '') = '') then
                data_count = 12;
            else
                data_count = f_json_blob_get_count(:action_data, 'data');
    
            nmonth     = iif(:data_count > 12, 12, iif(:data_count < 1, 1, :data_count));
            incomeavg  = iif(:nmonth > 0, :incomewt / :nmonth, 0);
    
            json.obj_set_i(jso2, 'taxrate',   :taxrate);
            json.obj_set_f(jso2, 'incomewt',  :incomewt);
            json.obj_set_i(jso2, 'nmonth',    :nmonth);
            json.obj_set_f(jso2, 'incomeavg', :incomeavg);
        end
    
        survey_json = json.obj_to_json(jso2);
    end
    
    update miniappclient_income
    set maci_survey = :survey_json,
        maci_from_un_id = :un_id
    where maci_from_mac_id = :obj_id;
end
