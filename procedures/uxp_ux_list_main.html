CREATE PROCEDURE UXP$UX_LIST_MAIN(
    MAIN_UA_UUID D_UUID = null,
    ONLY_OWN D_SMALLINT = 0,
    SHOW_MENU D_SMALLINT = 0)
RETURNS (
    RES_ID D_STR_255,
    RES_PARENT_ID D_STR_255,
    RES_KIND D_STR_CODE,
    RES_IMAGE D_SMALLINT,
    RES_NAME D_STR_100,
    RES_FCODE D_STR_100,
    RES_MEMO D_STR_LONG,
    RES_KIND_CAPTION D_STR_CODE,
    RES_DATETIME D_DATETIME,
    RES_SOURCE_UUID D_UUID,
    RES_WARNING_STATE D_SMALLINT,
    RES_UUID D_UUID,
    RES_UUID_STR D_STR_CODE,
    RES_SCODE D_STR_CODE,
    RES_CAPTION D_STR_100,
    RES_SOURCE_CODE D_STR_CODE,
    RES_READONLY D_BOOL)
AS
declare variable ux_root d_str_code;
declare variable ux_root_favorite d_str_code;
declare variable ux_root_proc d_str_code;
declare variable us_is_table d_bool;
declare variable ux_root_main d_str_code;
declare variable is_lookup d_bool;
declare variable has_items d_bool;
declare variable us_in_ui d_bool;
declare variable us_kind d_smallint;
declare variable user_name d_str_code;
declare variable ux_root_extform d_str_code;
declare variable is_pro d_boolean;
declare variable us_is_view d_bool;
declare variable is_superuser d_boolean;
declare variable ux_root_view d_str_255;
begin
  user_name = current_user;

  if (only_own is null) then
    only_own = 0;

  if (show_menu is null) then
    show_menu = 0;

  if (main_ua_uuid is not null) then
  begin
    if (not exists(select * from ux$uniq
    where uu_id = :main_ua_uuid)) then
      main_ua_uuid = null;
  end

  is_superuser = (user_name in ('sysdba', 'sislamov'));

  if (only_own is null) then
    only_own = 0;

  is_pro = uxf$pro();
  ux_root_view = '';

  if (main_ua_uuid is not null) then
  begin
    res_warning_state = 0;
    res_readonly = 0;

    select us_uuid, us_code, uxa_name, uu_datetime, uuid_to_char(ux_draft_uuid),
           trim(iif(us_is_table = 1, 'table', 'procedure')),
           trim(iif(us_is_table = 1, 'Таблица', 'Процедура'))
    from ux$action, ux$alias, ux$code, ux$source, ux$uniq
    where ua_uuid = :main_ua_uuid and
          ua_uxa_uuid = uxa_uuid and
          uxa_owner = ux_uuid and
          ux_us_uuid = us_uuid and
          us_uuid = uu_id
    into res_source_uuid, res_source_code, res_caption, res_datetime, res_fcode, res_kind,
         res_kind_caption;
  
    res_uuid = res_source_uuid;
    res_uuid_str = uuid_to_char(res_source_uuid);

    res_image = 89; -- Звездочка избранных
    res_parent_id = null;
    res_id = 'action_' || lower(uuid_to_char(main_ua_uuid));
    res_scode = '';
    res_name = res_caption;
    res_memo = 'Выбранное действие для просмотра';
    suspend;
  end

  if (show_menu = 1) then
  begin
    for select res_id, res_parent_id, res_kind, res_image, res_name, res_fcode, res_memo, res_kind_caption, res_datetime,
               res_source_uuid, res_warning_state, res_uuid, res_uuid_str, res_scode, res_caption, res_source_code,
               res_readonly
        from uxp$ux_list_menu(:only_own)
    into :res_id, :res_parent_id, :res_kind, :res_image, :res_name, :res_fcode, :res_memo, :res_kind_caption,
         :res_datetime, :res_source_uuid, :res_warning_state, :res_uuid, :res_uuid_str, :res_scode, :res_caption,
         :res_source_code, :res_readonly
    do
      suspend;

    exit;
  end

  res_warning_state = 0;
  res_readonly = 0;

  res_source_uuid = null;
  res_source_code = '';

  res_kind_caption = '';
  res_datetime = null;
  res_uuid = null;
  res_uuid_str = '';

  if (exists(select * from ux$userfav
  where ufv_user = :user_name)) then
  begin
    res_image = 89; -- Звездочка избранных
    res_kind = 'table.favorite';
    res_parent_id = null;
    res_id = 'table.favorite';
    res_fcode = '';
    res_scode = '';
    res_name = '';
    res_memo = 'Избранные источники пользователя';
    res_caption = 'Избранные источники';
    ux_root_favorite = res_id;
    suspend;
  end

  if (is_pro) then
  begin
    res_image = 84; -- Обычная таблица
    res_kind = 'view.main';
    res_parent_id = null;
    res_id = 'view.main';
    res_fcode = '';
    res_scode = '';
    res_name = '';
    res_memo = 'Представления (VIEW)';
    res_caption = 'Представления (VIEW)';
    ux_root_view = res_id;
    suspend;
  end

  res_image = 84; -- Обычная таблица
  res_kind = 'table.main';
  res_parent_id = null;
  res_id = 'table.main';
  res_fcode = '';
  res_scode = '';
  res_name = '';
  res_memo = 'Таблицы';
  res_caption = 'Таблицы';
  ux_root_main = res_id;
  suspend;

  res_image = 75; -- Процедура
  res_kind = 'proc.ui';
  res_parent_id = null;
  res_id = 'proc.ui';
  res_fcode = '';
  res_scode = '';
  res_name = '';
  res_memo = 'Процедуры, участвующие в интерфейсах или формах поисках';
  res_caption = 'Процедуры';
  ux_root_proc = res_id;
  suspend;

  if (not is_pro) then
  begin
    res_image = 132; -- Внешние формы FinStart
    res_kind = 'extform.ui';
    res_parent_id = null;
    res_id = 'extform.ui';
    res_fcode = '';
    res_scode = '';
    res_name = '';
    res_memo = 'Формы FinStart вызывающие динамические формы';
    res_caption = 'Действия FinStart форм';
    ux_root_extform = res_id;
    suspend;
  end

  res_image = 79; -- зеленая кника
  res_kind = 'object.main';
  res_parent_id = null;
  res_id = 'object.main';
  res_fcode = '';
  res_scode = '';
  res_name = '';
  res_memo = 'Служебные объекты';
  res_caption = 'Объекты';
  suspend;

  res_parent_id = res_id;

  if (not is_pro) then
  begin
    res_image = 43; -- Картинка портфеля
    res_kind = 'mainmenu';
    res_id = 'mainmenu';
    res_scode = '';
    res_name = '';
    res_memo = 'Главное меню приложения';
    res_caption = 'Главное меню';

    res_uuid = uxf$mainmenu();
    res_fcode = uxf$code(res_uuid, 'fcode');
  
    res_uuid_str = uuid_to_char(res_uuid);
    suspend;
  end

  res_image = 72; -- Картинка с Fx
  res_source_uuid = uxf$sys_id();
  res_source_code = uxf$code(res_source_uuid, 'code');

  res_kind = 'uxframe';

  select ufr_uuid, ufr_fcode, ufr_code
  from ux$code, ux$form, ux$frame
  where ux_us_uuid = :res_source_uuid and
        ux_uuid = ufm_ux_uuid and
        ufm_uuid = ufr_owner
  into res_uuid, res_fcode, res_scode;

  res_id = res_fcode;
  res_memo = 'Шаблоны редакторов';
  res_caption = 'Пользовательские редакторы';
  res_name = 'Пользовательские редакторы';
  res_uuid_str = uuid_to_char(res_uuid);

  suspend;

  res_source_uuid = null;
  res_source_code = '';
  res_uuid = null;
  res_uuid_str = '';

  if (not is_pro) then
  begin
    res_image = 29; -- Картинка FRX
    res_kind = 'report_sync';
    res_id = 'report_sync';
    res_fcode = '';
    res_scode = '';
    res_name = '';
    res_memo = 'Бланки и отчеты FinStart';
    res_caption = 'Бланки и отчеты FinStart';
    suspend;

    res_image = 112; -- Картинка формы
    res_kind = 'unit';
    res_id = 'unit';
    res_fcode = '';
    res_scode = '';
    res_name = '';
    res_memo = 'Формы из проекта FinStart (Finance)';
    res_caption = 'Формы FinStart';
    suspend;

    res_image = 131; -- Refdata
    res_kind = 'eref.cfg';
    res_id = 'eref.cfg';
    res_fcode = '';
    res_scode = '';
    res_name = '';
    res_memo = 'Настройка отображения данных из справочника RefData';
    res_caption = 'Справочники RefData';
    suspend;
  end

  res_image = 71; -- Enum
  res_kind = 'enum.main';
  res_id = 'enum';
  res_fcode = '';
  res_scode = '';
  res_name = '';
  res_memo = 'Список перечислений';
  res_caption = 'Перечисления';
  suspend;

  res_image = 135; -- Картинка с квадратными скобками
  res_kind = 'jsoneditor';
  res_id = 'jsoneditor';
  res_fcode = '';
  res_scode = '';
  res_name = '';
  res_memo = 'JSON просмотр и редакторирование';
  res_caption = 'JSON редактор';
  suspend;

  if (not is_pro) then
  begin
    res_image = 104; -- Окно с настройкой
    res_kind = 'eventcfg';
    res_id = 'eventcfg';
    res_fcode = '';
    res_scode = '';
    res_name = '';
    res_memo = 'Настройка связок элементов управления с событиями';
    res_caption = 'Настройка событий';
    suspend;
  end

  res_image = 102; -- Картинка с картинкой
  res_kind = 'images';
  res_id = 'images';
  res_fcode = '';
  res_scode = '';
  res_name = '';
  res_memo = 'Библиотека картинок для действий и меню';
  res_caption = 'Библиотека иконок';
  suspend;

  res_image = 113; -- Картинка треугольный цикл
  res_kind = 'class';
  res_id = 'class';
  res_fcode = '';
  res_scode = '';
  res_name = '';
  res_memo = 'Классы используемые в Delphi';
  res_caption = 'Классы объектов';
  suspend;

  res_image = 109; -- Картинка треугольный цикл
  res_kind = 'tree';
  res_id = 'tree';
  res_fcode = '';
  res_scode = '';
  res_name = '';
  res_memo = 'Структурная информация в виде дерева';
  res_caption = 'Структурные данные';
  suspend;

  res_kind = 'table';
  res_kind_caption = 'Таблица';
  res_memo = '';

  for select us_uuid, us_is_table, iif(us_different = 0 , 0, 1), us_memo, us_code, us_name,
             iif(coalesce(us_lookup_text, '') <> '', 1, 0),
             iif(us_fields + us_params <> 0, 1, 0),
             us_is_view, us_in_ui, us_kind, uu_datetime
  from ux$source, ux$uniq
  where us_kind in (0, 1) and -- кроме SQL запросов и служебных таблиц
        us_uuid = uu_id
  order by us_code
  into :res_uuid, :us_is_table, res_warning_state, res_memo, :res_fcode, :res_caption,
       is_lookup, has_items, us_is_view, us_in_ui, us_kind, res_datetime
  do
  begin
    res_parent_id = ux_root_main;

    if (only_own = 1 and uxf$own(res_fcode, user_name) = 0) then
      continue;

    res_scode = res_fcode;
    res_name = res_caption;

    if (not(is_superuser) and is_pro and res_fcode containing '$') then
      continue;

    if (us_is_table = 1) then
    begin
      res_kind = 'table';
      res_kind_caption = 'Таблица';
      res_image = 76; -- Обычная таблица
      ux_root = ux_root_main;

      if (us_is_view = 1) then
        res_kind_caption = 'View';
--      else
--        if (is_pro and usf$view_uuid(res_fcode) is not null) then
--          Continue;

      if (us_kind = 0) then
      begin
        if (is_lookup = 1) then
          res_image = 83; -- Таблица без интерфейса, с Lookup
        else
        if (has_items = 1) then
          res_image = 85; -- Таблица без интерфейса, с полями
  
        if (us_in_ui = 1) then
        begin
          res_image = 21; -- Таблица с интерфейсом

          if (is_lookup = 1) then
            res_image = 82; -- Таблица с интерфейсом, с Lookup
        end
      end else
      if (us_kind = 1) then
        res_image = 86; -- Виртуальная, Draft таблица
      else
      if (us_kind = 2) then
        res_image = 87; -- Служебная таблица
    end
    else
    begin
      res_kind = 'procedure';
      res_kind_caption = 'Процедура';
      res_image = 75; -- Процедура
      ux_root = ux_root_proc;
    end

    res_id = ux_root || '.' || res_fcode;
    res_parent_id = ux_root;

    if (res_caption <> '' and res_caption <> res_fcode)
    then res_caption = f_format('%1 (%2)', res_fcode, res_caption);
    else res_caption = res_fcode;

    res_id = lower(res_id);
    res_parent_id = lower(res_parent_id);

    if (us_is_view = 1 and ux_root_view <> '') then
      res_parent_id = ux_root_view;

    res_source_uuid = res_uuid;
    res_source_code = uxf$code(res_source_uuid, 'code');
    res_uuid_str = uuid_to_char(res_uuid);

    suspend;

    -- Добавление в избранные
    if (exists(select * from ux$userfav
    where ufv_user = :user_name and
          ufv_uuid = :res_uuid)) then
    begin
      ux_root = ux_root_favorite;
      res_id = ux_root || '.' || res_fcode;
      res_parent_id = ux_root;

      res_id = lower(res_id);
      res_parent_id = lower(res_parent_id);

      suspend;
    end
  end

  if (not is_pro) then
  begin
    res_warning_state = 0;
    res_kind = 'extform';
    res_kind_caption = 'Форма FinStart';
    res_image = 132;
    res_source_uuid = Null;
    res_source_code = '';
  
    ux_root = ux_root_extform;
  
    for select efm_uuid, efm_memo, efm_code, efm_name, uu_datetime
    from ux$extform, ux$uniq
    where efm_uuid = uu_id
    order by efm_code
    into :res_uuid, res_memo, :res_fcode, :res_caption, res_datetime
    do
    begin
      res_scode = res_fcode;
      res_name = res_caption;
  
      res_id = ux_root || '.' || res_fcode;
      res_parent_id = ux_root;
  
      if (res_caption <> '' and res_caption <> res_fcode)
      then res_caption = f_format('%1 (%2)', res_fcode, res_caption);
      else res_caption = res_fcode;
  
      res_id = lower(res_id);
      res_parent_id = lower(res_parent_id);
  
      res_uuid_str = uuid_to_char(res_uuid);
  
      suspend;
    end
  end
