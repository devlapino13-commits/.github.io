CREATE PROCEDURE HP_EDOC_SIGN_CHECK_KONT(
    K_ID ID_TYPE,
    DISB_KIND D_STR_100 = 'all')
RETURNS (
    KF_CLIENT_FIO VARCHAR(100),
    KF_CLIENT_PASS_NO VARCHAR(30),
    KF_DISB_DATE DATE,
    KC_AMOUNT DECIMAL(18,4),
    KF_ERROR_TEXT VARCHAR(255),
    KF_ERROR_IS_CRITICAL INTEGER)
AS
declare variable c_id id_type;
declare variable ma_id d_integer;
declare variable check_date d_date;
declare variable pr_code d_str_100;
begin
 if (k_id = 109010199148900101) then  exit;
 if (k_id = 109010199564600101) then  exit;
 if (k_id = 109010196226600101) then  exit; --от чолпон 21,10,22
 if (k_id = 109010209274400101) then  exit; --от чолпон 29,11,22
 if (k_id = 109010209366000101) then  exit; --от чолпон 02,11,22
 if (k_id = 109010205351400101) then  exit; --от Чолпон 11.11.22
 if (k_id = 109010186467000201) then  exit; --заявка в ОТРС от Улана 18.11.2022
 if (k_id = 109010212558300101) then  exit; --заявка в ОТРС от Улана 30.11.2022
 if (k_id = 109010215045700101) then  exit; --от Чолпон 23.12.22
 if (k_id = 109010227421600101) then  exit; -- Jira 283 (Чолпон 23.03.23)
 if (k_id = 109010227345900101) then  exit; --jira 316
 if (k_id = 109010236957300101 ) then exit; --jira 335
 if (K_ID = 109010239773700101) then exit; ---RFCKG-613
    /*
        disb_kind:
        disb - выдача
        approve - утверждение
        all - все
    */

  -- если ИПФ и по ИФП eDoc отключен, тогда дальше не проверяем
  if (f_tmp_const('edoc_ipf_inactive') = 1) then
  begin
      if (exists(select kont.k_id
                  from kontract kont
                  inner join product pr on pr.pr_id = kont.k_from_pr_id
                  where kont.k_id = :k_id
                  and pr.pr_code like '%.ipf.%')) then
      begin
        exit;
      end
  end

  -- Проверка наличия подписанных электронных документов
  select k_disb_date, pr.pr_code
  from kontract kt
  inner join product pr on pr.pr_id = kt.k_from_pr_id
  where k_id = :k_id
  into check_date, pr_code;

  if (pr_code containing '.installment_goods.' or pr_code containing '.sugarmill.') then
    exit;

  if (disb_kind in ('disb', 'all')) then
  begin
      -- Договор займа
      kf_error_text = f_edoc_check('loan.agreement', k_id, check_date);
      if (kf_error_text <> '') then
      begin
        kf_error_is_critical = 2;
        suspend;
      end
  end

  select k_from_ma_app_id from kontract
  where k_id = :k_id
  into ma_id;

  select ma_date_app
  from miniapp
  where ma_id = :ma_id
  order by ma_id desc, ma_number desc
  rows 1
  into check_date;

  kf_error_text = f_edoc_check('loan.application', ma_id, check_date);
  if (kf_error_text <> '') then
  begin
    kf_error_is_critical = 2;
    suspend;
  end

  -- Договор АСП
  for select kc_from_c_id, kc_amount_approved, k_disb_date, c_name_fio, c_pass_no
      from knclient, kontract, clients
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0 and
            kc_from_k_id = k_id and
            kc_from_c_id = c_id
      into c_id, kc_amount, kf_disb_date, kf_client_fio, kf_client_pass_no
  do
  begin
    if (kf_disb_date >= '10.01.2020') then
    begin
      kf_error_text = f_edoc_check('asp.agreement', c_id, kf_disb_date);
      if (kf_error_text <> '') then
      begin
        kf_error_is_critical = 2;
        suspend;
      end
    end
  end
end
