CREATE PROCEDURE PRINT_CLIENT_FINANCES(
    K_ID D_ID,
    UN_ID D_ID)
RETURNS (
    CA_DATE D_DATE,
    C_NAME_FIO D_STR_255,
    MA_NO D_INTEGER,
    KAL_NEW_KF_NAME D_STR_100,
    KC_AMOUNT_APPROVED D_CURRENCY,
    C_POSITION D_STR_255,
    CONTRACT_SALARY D_INTEGER,
    CL_INCOME_ASP D_INTEGER,
    CL_INCOME_PENSION D_INTEGER,
    SALARY_STATEMENT D_INTEGER,
    AVG_MONTHLY_INCOME D_INTEGER,
    SH_ALL_WITH_TAX D_CURRENCY,
    KDD D_CURRENCY,
    KDN D_CURRENCY)
AS
declare variable KL_DATE D_DATE;
declare variable C_ID D_ID;
declare variable MA_ID D_ID;
declare variable CA_ID D_ID;
declare variable CP_ID D_ID;
declare variable MAC_ID D_ID;
begin
-- 1. Основные данные
for select
    k.k_from_ma_app_id, 
    c.c_id, 
    c.c_name_fio, 
    kc.kc_amount_approved,
    kal.kal_new_kf_name
from kontract k
join knclient kc on kc.kc_from_k_id = k.k_id
join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id
join clients c on c.c_id = kc.kc_from_c_id
where kc.kc_from_k_id = :k_id
    and kc.kc_amount_approved > 0
    and kal.is_delete=0
    and current_date between kal.kal_date and kal.kal_date_end
into
    :ma_id,
    :c_id,
    :c_name_fio,
    :kc_amount_approved,
    :kal_new_kf_name
do
    begin
    -- Номер заявки
    select
        ma.ma_no,
        mac.mac_id
    from miniapp ma
    join miniappclient mac on mac.mac_from_ma_id = ma.ma_id
    where ma.ma_id = :ma_id
        and mac.mac_from_c_id = :c_id
        and ma.ma_date_end = '31.12.9999'
    into
        :ma_no,
        :mac_id;
    -- Дата последнего статуса контракта
    select max(kl.kl_datetime)
    from kontlog kl
    where kl.kl_from_k_id = :k_id
        and kl.kl_state = 1
    into
        :kl_date;
    -- Анкета клиента
    select
      ca.ca_id,
      ca.ca_date,
      ca.ca_from_cp_id
    from cfp_anketa ca
    join miniappclient mac on mac.mac_id = ca.ca_object_id
    where ca.ca_object_kind = f_objkind('miniappclient', 'id')
    and ca.ca_object_id = :mac_id
    order by ca.ca_date desc
    rows 1
    into :ca_id, :ca_date, :cp_id;
    
    -- Если анкета не найдена — fallback по дате
    if (ca_id is null) then
    begin
      ca_date = coalesce(kl_date, current_date);
      c_position = null;
      avg_monthly_income = 0;
    end
    else
    begin
      select first 1 coalesce(cp.cp_name_ru, cp.cp_name_kg)
      from cfp_positions cp
      where cp.cp_id = :cp_id
      into
        :c_position;
          -- Доходы по анкете
      select
        max(case when cq.cq_code_name = 'contract_salary' then coalesce(caa.caa_value, 0) end),
        max(case when cq.cq_code_name = 'cl_income_asp' then coalesce(caa.caa_value, 0) end),
        max(case when cq.cq_code_name = 'cl_income_pension' then coalesce(caa.caa_value, 0) end),
        max(case when cq.cq_code_name = 'salary_statement' then coalesce(caa.caa_value, 0) end)
      from cfp_anketa_answers caa
      join cfp_questions cq on cq.cq_id = caa.caa_from_cq_id
      where caa.caa_from_ca_id = :ca_id
      and cq.cq_code_name in (
          'contract_salary',
          'cl_income_asp',
          'cl_income_pension',
          'salary_statement')
      into :contract_salary, :cl_income_asp, :cl_income_pension, :salary_statement;

      -- RFCKZ-924: Средний ежемесячный доход
      avg_monthly_income = coalesce(salary_statement, 0) + coalesce(contract_salary, 0);
    end
    -- Ежемесячный платёж
    select sh_all_with_tax
    from fm_schedule_list(11, :ma_id)
    where sh_id = 1
    into :sh_all_with_tax;
    
    -- КДН и КДД по анкете
    select cakdn.cakdn_kdn, cakdn.cakdn_kdd
    from cfp_anketa_kdn cakdn
    join miniappclient mac on mac.mac_id = cakdn.cakdn_from_mac_id
    where cakdn.cakdn_from_mac_id = :mac_id
    order by cakdn.cakdn_change_datetime desc
    rows 1
    into :kdn, :kdd;
    end
    suspend;
end
