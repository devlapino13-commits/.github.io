CREATE PROCEDURE HP_IS_CLIENT_CORRESPONDS_PP(
    C_ID D_ID)
RETURNS (
    IS_CLIENT_CORRESPONDS_PP D_SMALLINT)
AS
declare variable date_end d_date;
declare variable close_date d_date;
declare variable last_k_id d_id;
declare variable db_id d_integer;
declare variable sh_date d_date;
declare variable ol_delinq_pay_dd d_date;
begin
  --если клиент соответсвует условию ПП то возвращаем 1 иначе 0
  -- Условия участия в партнерской программе
  -- Клиент должен быть активным или когда либо бравший кредит
  db_id = f_get_db_id();
  if (db_id <> 81001) then
  begin
    is_client_corresponds_pp = 0;
    suspend;
    exit;
  end
  is_client_corresponds_pp = 1;
  close_date = null;
  date_end = current_date;

  select ks_date_end, ks_from_k_id
  from kontstate, knclient
  where ks_ol_is_delete = 0 and
        ks_state = 1 and
        ks_from_k_id = kc_from_k_id and
        kc_amount_approved > 0 and
        kc_from_c_id = :c_id
  order by ks_date_end desc
  rows 1
  into close_date, last_k_id;

  if (close_date is null) then
    is_client_corresponds_pp = 0; -- если не имеется кр. история то отменяется
  else
  if (dateadd(month, -6, :date_end) > :close_date) then
    is_client_corresponds_pp = 0; -- если после закрытия последнего контракта прошло более 6 мес.
  else
  if (exists(select *
             from moratory
             where m_is_black = 2 and
                   m_is_active = 1 and
                   m_from_c_id = :c_id)) then
    is_client_corresponds_pp = 0; -- если в Стоп-листе
  else
  begin
    /* if (exists(select *
             from oksummary
             where oks_from_k_id = :last_k_id and
                   oks_close_state = 0 and
                   ((oks_delinq_day > 30 and
                   oks_date_begin <= cast(:date_end as date)) or
                   (:date_end between oks_date_begin and oks_date_end and
                   cast(:date_end as date) - oks_delinq_date > 30)))) then
    is_client_corresponds_pp = 0;  */

    for select distinct ok_sh_date
        from operkred
        where ok_ol_is_active = 1 and
              ok_from_k_id = :last_k_id and
              ok_ol_date <= :date_end and
              ok_from_c_id = 0
        into :sh_date
    do
    begin
      ol_delinq_pay_dd = null;
      select max(ok_ol_date)
      from operkred
      where ok_ol_is_active = 1 and
            ok_ol_date <= :date_end and
            ok_sh_date < ok_ol_date and
            ok_sh_date = :sh_date and
            ok_from_k_id = :last_k_id and
            ok_type = 4 and
            (ok_os + ok_perc) <= -1 and
            ok_from_c_id = 0
      into :ol_delinq_pay_dd;
      if (ol_delinq_pay_dd is not null) then
      begin
        if ((ol_delinq_pay_dd - sh_date) > 30) then
        begin
          is_client_corresponds_pp = 0;
          suspend;
          exit;
        end
      end
    end
    suspend;
  end
end
