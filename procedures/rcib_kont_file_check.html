CREATE PROCEDURE RCIB_KONT_FILE_CHECK(
    RCIB_KONT_NO STR_100)
RETURNS (
    ERR STR_255,
    KONTRACT_NO_NEW STR_100,
    KONT_STATUS STR_100)
AS
declare variable pos_1 integer;
declare variable pos_2 integer;
declare variable kontract_no str_100;
declare variable kncl_no integer;
declare variable k_id id_type;
declare variable c_id id_type;
declare variable double_kncl smallint;
declare variable k_id_new id_type;
declare variable kc_number_new integer;
begin
  pos_1 = position('(' in rcib_kont_no);
  pos_2 = position(')' in rcib_kont_no);

  kontract_no = substring(rcib_kont_no from 1 for pos_1 - 1);

  kncl_no = substring(rcib_kont_no from pos_1 + 1 for (pos_2 - 1) - (pos_1));

  k_id = null;
  kontract_no_new = null;
  err = null;
  kont_status = null;

  select kont.k_id
  from kontract kont
  where kont.k_kontract_no = :kontract_no
  into :k_id;

  if (k_id is null) then
  begin
    err = 'Контракт не найден';
  end
  else
  begin
    select kncl.kc_from_c_id
    from knclient kncl
    where kncl.kc_from_k_id = :k_id and
          kncl.kc_number = :kncl_no and
          kncl.kc_amount_approved > 0
    into :c_id;

    if (c_id is null) then
    begin
      err = 'Клиент, в контракте, не найден';

      kontract_no_new = null;
      double_kncl = 0;
      k_id_new = null;

      for select kont.k_kontract_no, kont.k_id
          from kontdevides kd
          inner join kontract kont on kd.kd_k_new = kont.k_id
          inner join knclient kncl on kncl.kc_from_k_id = kont.k_id
          where kd.kd_k_orig = :k_id and
                kncl.kc_amount_approved > 0 and
                kncl.kc_number = :kncl_no
          into :kontract_no_new, :k_id_new
      do
      begin
        double_kncl = double_kncl + 1;
      end

      if ((k_id_new is not null) and
          (double_kncl = 0)) then
        err = 'Клиент в другом контракте';
      else
      begin
        k_id_new = null;

        select rck.rck_from_c_id
        from rcib_kontract rck
        where rck.rck_cred_no = :rcib_kont_no and
              rck.rck_is_poruch = 0
        order by rck.rck_from_cp_id desc
        rows 1
        into :c_id;

        kc_number_new = null;

        select kont.k_kontract_no, kont.k_id, kncl.kc_number
        from kontdevides kd
        inner join kontract kont on kd.kd_k_new = kont.k_id
        inner join knclient kncl on kncl.kc_from_k_id = kont.k_id
        where kd.kd_k_orig = :k_id and
              kncl.kc_amount_approved > 0 and
              kncl.kc_from_c_id = :c_id
        into :kontract_no_new, :k_id_new, :kc_number_new;

        kc_number_new = coalesce(kc_number_new, '');

        if (k_id_new is not null) then
        begin
          err = 'Клиент в другом контракте (kc_number - изменен или есть дубль) kc_number = ' || kc_number_new;
        end
      end

      if (k_id_new is not null) then
      begin
        select (case
                  when ks.ks_state = 0 then 'В процессе'
                  when ks.ks_state = 1 then 'Активный'
                  when ks.ks_state = 2 then 'закрыт'
                end)
        from kontstate ks
        where ks.ks_from_k_id = :k_id_new and
              ks.ks_ol_is_delete = 0 and
              current_date between ks.ks_date_begin and ks.ks_date_end
        into :kont_status;
      end
    end
  end

  if (err is not null) then
  begin
    suspend;
  end
end
