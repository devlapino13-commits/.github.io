CREATE PROCEDURE HP_OPERDELMAIN_EXEC_DELETE(
    IN_ODM_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable k_id d_id;
declare variable main_ol_id d_id;
declare variable loan_ol_id d_id;
declare variable oper_type d_integer;
declare variable oper_subtype d_integer;
declare variable res_code d_str_code;
declare variable trans_ol_id d_id;
declare variable odl_id d_integer;
declare variable transfer_kont_count d_integer;
declare variable odm_main_ol_id d_id;
declare variable is_delete d_bool;
declare variable odm_need_reverse d_bool;
begin
  res_is_error = 0;
  res_text = '';

  delete from operdellist
  where odl_from_odm_id = :in_odm_id and
        odl_state = -1;

  select odm_main_ol_id, is_delete, odm_need_reverse
  from operdelmain, operlist
  where odm_id = :in_odm_id and
        odm_main_ol_id = ol_id
  into odm_main_ol_id, is_delete, odm_need_reverse;

  if (row_count = 0) then
    res_text = 'Заявка на удаление операции не найдена: ' || coalesce(in_odm_id, '<Null>');

  if (res_text = '' and is_delete = 1) then
    res_text = 'Операция по заявке уже удалена: ' || coalesce(in_odm_id, '<Null>');

  if (res_text <> '') then
  begin
    res_is_error = 1;
    suspend;
    exit;
  end

  for select odk_from_k_id
  from operdelkont
  where odk_from_odm_id = :in_odm_id
  order by iif(odk_loan_ol_id is not null, 0, 1), odk_id
  into :k_id
  do
  begin
    for select odl_id, odl_main_ol_id, odl_loan_ol_id, coalesce(ol_oper_type, 0), coalesce(ol_oper_subtype, 0)
    from operdellist, operdelmain, operlist ol
    where odl_from_odm_id = :in_odm_id and
          odl_from_k_id = :k_id and
          odl_from_odm_id = odm_id and
          odl_main_ol_id <> odm_main_ol_id and
          odl_loan_ol_id = ol.ol_id and
          odl_state = 0 and
          ol.is_delete = 0
    order by ol_date desc, odl_id desc
    into :odl_id, :main_ol_id, loan_ol_id, oper_type, oper_subtype
    do
    begin
      res_code = '';

      select res_code
      from hp_operdel_operkind(:loan_ol_id, :oper_type, :oper_subtype)
      into :res_code;

      /*
      when 1 then 'disb'  X
      when 2 then 'repay'
      when 3 then 'writeoff'
      when 4 then 'reschedule'  X
      when 9 then 'transfer'
      when 10 then 'overpayback'
      when 11 then 'overpayoff'
      when 12 then 'floatperc'
      when 13 then 'reverse'  X
      */

      if (res_code = 'transfer') then
      begin
        select count(distinct ok_from_k_id)
        from operkred
        where ok_from_ol_id = :loan_ol_id
        into transfer_kont_count;

        if (transfer_kont_count > 1) then
        begin
          trans_ol_id = null;
          insert into operlist (ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                                ol_status, is_delete, ol_datetime, ol_user_id, ol_document_client, ol_from_mo_id,
                                ol_is_mo_modified, ol_document_office, ol_acc_no_client, ol_acc_no_office, ol_oper_subtype,
                                ol_date_oper_day, ol_pocket, ol_doc_client_name, ol_is_client, ol_year, ol_month,
                                ol_manager_c_id, ol_from_rd_id)
          select ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo, ol_status,
                 is_delete, ol_datetime, ol_user_id, ol_document_client, ol_from_mo_id, ol_is_mo_modified, ol_document_office,
                 ol_acc_no_client, ol_acc_no_office, ol_oper_subtype, ol_date_oper_day, ol_pocket, ol_doc_client_name,
                 ol_is_client, ol_year, ol_month, ol_manager_c_id, ol_from_rd_id
          from operlist
          where ol_id = :loan_ol_id
          returning ol_id
          into trans_ol_id;

          update operkred
          set ok_from_ol_id = :trans_ol_id
          where ok_from_ol_id = :loan_ol_id and
                ok_from_k_id = :k_id;

          update operdellist
          set odl_loan_ol_id = :trans_ol_id,
              odl_main_ol_id = :trans_ol_id
          where odl_id = :odl_id;

          main_ol_id = trans_ol_id;
          loan_ol_id = trans_ol_id;
        end
      end

      update operlist
      set is_delete = 1
      where ol_id = :main_ol_id;

      update operdellist
      set odl_state = -1
      where odl_id = :odl_id;
    end
  end

  if (odm_need_reverse = 0) then
  begin
    update operlist
    set is_delete = 1
    where ol_id = :odm_main_ol_id;
  end

  update operdellist
  set odl_state = -1
  where odl_from_odm_id = :in_odm_id and
        odl_main_ol_id = :odm_main_ol_id;

  suspend;
end
