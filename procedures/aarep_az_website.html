CREATE PROCEDURE AAREP_AZ_WEBSITE(
    REP_REGION_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    CL_FIO D_STR_100,
    PHONE_NUMBER D_STR_20,
    AW_DATE D_DATE,
    AW_DATETIME D_DATETIME,
    REG_NAME D_STR_100,
    UN_FIO D_STR_100,
    AW_BLOCK_DATETIME D_DATETIME,
    AW_STATUS D_STR_20,
    DECLINE_REASON D_STR_255)
AS
declare variable UN_ID D_ID;
declare variable REG_ID D_INTEGER;
declare variable AW_ID D_ID;
begin
  for select 
        aw.aw_id,
        aw.aw_cl_fio,
        aw.aw_phone,
        aw.aw_date,
        aw.aw_datetime,
        t.id_list,
        aw.aw_from_un_id,
        aw.aw_block_datetime,
        (case aw_status when 0 then 'Заявка в процессе'
                        when 1 then 'Заявка одобрена'
                        when -1 then 'Заявка отклонена'
                        end)
      from az_website aw
      inner join region_list(:rep_region_id, 1, 3, 1) t 
          on t.id_list = aw.aw_from_reg_id
      where aw.aw_date between :rep_date_begin and :rep_date_end and
            aw.aw_id not in (4, 5, 6, 7, 8, 9, 15)
      into 
        aw_id,
        cl_fio,
        phone_number,
        aw_date,
        aw_datetime,
        reg_id,
        un_id,
        aw_block_datetime,
        aw_status
  do
  begin
    reg_name = null;
    decline_reason = null;

    select rg.reg_name
    from region rg
    where rg.reg_unicode = :reg_id
    into reg_name;

    un_fio = null;
    select cl.c_name_fio
    from username un
    inner join clients cl on cl.c_id = un.un_from_c_id
    where un.un_id = :un_id
    into un_fio;

    select first 1 kr.kr_text
    from kontreslist krl
    join kontres kr on kr.kr_id = krl.krl_from_kr_id
    where krl.krl_ma_id = :aw_id
    order by krl.krl_datetime desc
    into decline_reason;

    suspend;
  end
end
