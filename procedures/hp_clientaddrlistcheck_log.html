CREATE PROCEDURE HP_CLIENTADDRLISTCHECK_LOG(
    C_ID ID_TYPE,
    ADL_ID ID_TYPE,
    ADL_FROM_PL_ID ID_TYPE,
    ADL_DEPART_NO D_INTEGER,
    ADL_DEPART_PREF D_STR_20,
    ADL_FLAT_NO D_INTEGER,
    ADL_BUILDING_NO D_INTEGER,
    ADL_SECTION_NO D_INTEGER,
    ADL_ADDRESS_STR D_STR_255,
    ADL_COMMENT D_STR_255,
    ADL_POST_INDEX D_INTEGER,
    ADL_IS_EDITED D_INTEGER,
    ADL_STREET_ID ID_TYPE,
    ADL_IS_WITHOUT_STREET D_INTEGER)
AS
begin
  if (not exists(
       select * from clientaddrlistcheck_log
       where acl_c_id = :c_id)) then
  begin
    insert into clientaddrlistcheck_log (
      acl_adl_id, acl_c_id, acl_from_pl_id, acl_depart_no, acl_depart_pref,
      acl_flat_no, acl_building_no, acl_section_no, acl_address_str, acl_comment,
      acl_post_index, acl_is_edited, acl_street_id, acl_is_without_street)
    values (
      :adl_id, :c_id, :adl_from_pl_id, :adl_depart_no, :adl_depart_pref,
      :adl_flat_no, :adl_building_no, :adl_section_no, :adl_address_str, :adl_comment,
      :adl_post_index, :adl_is_edited, :adl_street_id, :adl_is_without_street);
  end
end
