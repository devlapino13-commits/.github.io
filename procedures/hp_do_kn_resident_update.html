CREATE PROCEDURE HP_DO_KN_RESIDENT_UPDATE(
    K_ID D_ID,
    REP_DATE D_DATE,
    CURR_CITIZEN_ID D_INTEGER,
    DOC_VNJ_ID D_INTEGER,
    UN_ID D_ID)
AS
declare variable KC_ID D_ID;
declare variable C_ID D_ID;
declare variable KNR_DATE_BEGIN D_DATE;
declare variable KNR_NON_RESIDENT D_SMALLINT;
declare variable C_NON_RESIDENT D_SMALLINT;
declare variable CDS_ID D_INTEGER;
declare variable ACC_ID D_ID;
declare variable ACC_FROM_SAB_ID D_ID;
declare variable AMOUNT_SUM D_CURRENCY;
declare variable AMOUNT_TAX D_CURRENCY;
declare variable REG_ID D_INTEGER;
declare variable CLIENT_TYPE D_INTEGER;
declare variable NEW_ACC_ID D_ID;
declare variable IS_DISB D_SMALLINT;
declare variable ACC_DT D_ID;
declare variable ACC_KT D_ID;
declare variable OL_ID D_ID;
declare variable KONTRACT_NO D_STR_255;
declare variable KONTRACT_DATE D_DATE;
declare variable OL_MEMO D_STR_LONG;
begin
-- если дата не указана, значит это выдача и нужно очистить и создать только одну запись
  if (rep_date is null) then
    is_disb = 1;
  else
    is_disb = 0;

  for select kc_id, kc_from_c_id, k_fact_disb_reg_id, iif(:is_disb = 0, :rep_date, k_disb_date), k_kontract_no,
             k_kontract_date, kc_non_resident
      from knclient, kontract
      where k_id = :k_id and
            kc_from_k_id = k_id and
            kc_amount_approved > 0
      into :kc_id, :c_id, :reg_id, :rep_date, :kontract_no, kontract_date, knr_non_resident
  do
  begin
    ol_memo = 'Перенос остатка договору займа ' || kontract_no || ' от ' || f_datetostr(:kontract_date) || ' г.';

    knr_date_begin = null;
    cds_id = null;

    if (is_disb = 1) then
    begin
      delete from kn_resident
      where knr_from_kc_id = :kc_id;
    end
    else
    begin
-- получаем последний статус резидентства на дату
      select knr_date_begin, knr_non_resident
      from kn_resident
      where knr_from_kc_id = :kc_id and
            knr_date_begin <= :rep_date
      order by knr_from_kc_id desc, knr_date_begin desc
      rows 1
      into knr_date_begin, knr_non_resident;
    end

-- получаем актуальный статус резидентства клиента из документов на дату
    select res_client_type, res_non_resident, res_from_cds_id
    from hp_get_knclient_type(:kc_id, :curr_citizen_id, :doc_vnj_id, :rep_date) -- изменено от 02.04.2026 Улан, было  hp_get_client_type(:c_id, :curr_citizen_id, :doc_vnj_id, :rep_date)
    into client_type, c_non_resident, cds_id;

--    if (knr_non_resident = c_non_resident) then
--      continue;

    if (knr_date_begin = rep_date) then -- если до этого была создана запись
    begin
-- обновляем данные на дату
      update kn_resident
      set knr_non_resident = :c_non_resident,
          knr_from_cds_id = :cds_id
      where knr_from_kc_id = :kc_id and
            knr_date_begin = :knr_date_begin;
    end
    else
    begin
-- вставляем новую запись
--      insert into kn_resident (knr_from_kc_id, knr_date_begin, knr_non_resident, knr_from_cds_id)
--      values (:kc_id, :rep_date, :c_non_resident, :cds_id);

      update or insert into kn_resident (knr_from_kc_id, knr_date_begin, knr_non_resident, knr_from_cds_id)
      values (:kc_id, :rep_date, :c_non_resident, :cds_id)
      matching (knr_from_kc_id, knr_date_begin);
    end

    if (is_disb = 1) then
      continue;

    if (ol_id is null) then
    begin
      insert into operlist (ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind, ol_doc_no,
                            ol_doc_no_str, ol_memo, ol_parent_id, is_delete, ol_user_id)
      values (:reg_id, 2, 31, :rep_date, 9, 0, '', :ol_memo, null, 0, :un_id)
      returning ol_id
      into :ol_id;
    end

    update kn_resident
    set knt_from_ol_id = :ol_id
    where knr_from_kc_id = :kc_id and
          knr_date_begin = :rep_date;

-- переносим остаток и закрепим OL_ID к записи
    for select acc_id, acc_from_sab_id
        from ref_account
        where acc_object_kind = 8 and
              acc_object_id = :kc_id and
              acc_non_resident <> :c_non_resident
        into acc_id, acc_from_sab_id
    do
    begin
      amount_sum = 0;
      amount_tax = 0;
      select kobs_bal_sum, kobs_bal_tax
      from kn_obsummary
      where kobs_acc_id = :acc_id and
            kobs_date <= :rep_date and
            kobs_date_end >= :rep_date
      order by kobs_acc_id desc, kobs_date desc
      rows 1
      into :amount_sum, amount_tax;
      if (amount_sum is null) then
        amount_sum = 0;
      if (amount_tax is null) then
        amount_tax = 0;
      if (amount_sum = 0 and
          amount_tax = 0) then
        continue;

      new_acc_id = null;
      select acc_id
      from get_ref_account_knclient(:acc_from_sab_id, '', :kc_id, :c_non_resident, :client_type, :rep_date, :reg_id)
      into new_acc_id;

      if (acc_id = new_acc_id) then
        continue;

      if (amount_sum > 0) then
      begin
        acc_dt = new_acc_id;
        acc_kt = acc_id;
      end
      else
      begin
        acc_dt = acc_id;
        acc_kt = new_acc_id;
        amount_sum = -amount_sum;
        amount_tax = -amount_tax;
      end

      execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :acc_dt, :acc_kt, null, null, :amount_sum, 0, :amount_tax,
          :rep_date, :reg_id, 1, 'Перенос остатка из-за смены резидентства');
    end
  end
end
