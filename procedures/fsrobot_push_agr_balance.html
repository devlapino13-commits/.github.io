CREATE PROCEDURE FSROBOT_PUSH_AGR_BALANCE
AS
declare variable mcl_id d_id;
declare variable push_text d_str_long;
declare variable ag_id d_id;
declare variable c_id d_id;
declare variable c_name_fio d_str_100;
declare variable epay_saldo d_currency;
begin
  /*  Уведомить клиентов Партнеров о накопленных бонусах
Разослать уведомления Партнерам, у кого есть накопленные бонусы по Партнерской Программе.


*/
  --date_begin = current_date - extract(day from current_date) + 1;
  for select ag_id, c_id, c_name_fio --, ag_date_begin, ag_date_end
      from agreements, clients
      where ag_from_rd_id = 7940 and
            ag_status = 1 and
            ag_object_kind = 0 and
            ag_object_id = c_id and
            ag_date_begin <= current_date --and AG_ID=5058
      into :ag_id, c_id, c_name_fio
  do
  begin
    epay_saldo= 0;
    select sum((ogr_amount + ogr_tax_fl) * ogr_sign * f_loyal(ogr_from_ol_id, current_date)) -- Остаток накопленных баллов
    from operagr
    where ogr_ol_date <= current_date and
          ogr_ol_is_active = 1 and
          ogr_from_ag_id = :ag_id
    into :epay_saldo;

    if (epay_saldo in (600, 1200)) then -- RFCRU-6095
    begin
      mcl_id = 0;
      select coalesce(ml.mcl_id, 0)
      from mc_login ml
      where ml.mcl_from_c_id = :c_id and
            ml.mcl_is_delete = 0
      order by mcl_last_login_datetime desc
      rows 1
      into :mcl_id;

      if (mcl_id > 0) then
      begin
        push_text = 'Уважаемый Партнер, благодарим вас за сотрудничество.' || ascii_char(10) || 'У вас есть накопленные бонусы. Для того, чтобы вывести их на карту, надо порекомендовать еще несколько друзей.' || ascii_char(10) || 'Как только они получает заем и сделают первую выплату, вам будут начислены бонусы';

        execute procedure mc_send_push('mc_login', :mcl_id, 'Получите бонусы!', :push_text, 7);
      end
    end
  end
end
