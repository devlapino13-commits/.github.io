CREATE PROCEDURE OKT_GET_REGION_KM_BY_REG_ID(
    REG_ID INTEGER)
RETURNS (
    C_ID ID_TYPE,
    C_NAME_FIO STR_100,
    C_PLASTCARD_NO VARCHAR(20))
AS
begin
  for select c_id, c_name_fio,
             (select cont_value
              from hp_contdata_get_cell_main(0, c_id))
      from hr_pstatus hps, joblist jl, jobkind jk, clients cl
      where current_date between ps_date_begin and ps_date_end and
            jl.jl_id = ps_from_jl_id and
            jk.jk_id = jl.jl_kind and
            cl.c_id = hps.ps_from_c_id and
            ps_is_working = 1 and
            ps_status = 1 and
            jk.jk_system_code = 'JobSystem.Region_KM' and
            hps.ps_from_reg_id = :reg_id
      into :c_id, :c_name_fio, :c_plastcard_no
  do
  begin
    suspend;
  end
end
