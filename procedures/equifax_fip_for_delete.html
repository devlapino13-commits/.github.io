CREATE PROCEDURE EQUIFAX_FIP_FOR_DELETE(
    FHOW_FIXED D_SMALLINT = 0)
RETURNS (
    RES_STR D_STR_255)
AS
declare variable APPLICATION_NUM D_STR_100;
declare variable APPLICATION_DATE D_DATETIME;
declare variable DOC_NO D_STR_100;
declare variable APPLICATION_DATE_STR D_STR_100;
declare variable SEC D_INTEGER;
declare variable SEC_STR D_STR_10;
declare variable FOR_DEL D_SMALLINT;
declare variable FIA_DEL_REASON D_STR_255;
begin
  for_del = 1;
  if (fhow_fixed = 1) then for_del = -1;

  for select fa.fia_application_num,
             fa.fia_application_date,
             fa.fia_doc_no,
             fa.fia_del_reason
  from equifax_fip_actuality fa
  where fa.fia_for_del = :for_del
  into :application_num, :application_date, :doc_no, :fia_del_reason
  do
  begin
    sec =  extract(second from :application_date);
    sec_str = right('0' || sec, 2);
    application_date_str = right('0' || extract(day from :application_date), 2) || '.' || right('0' || extract(month from :application_date), 2) || '.' || extract(year from :application_date) || ' ' || right('0' || extract(hour from :application_date), 2) || ':' || right('0' || extract(minute from :application_date), 2) || ':' || sec_str;

    res_str = application_num||';810;'||application_date_str||';;1;'||doc_no||';'||:fia_del_reason||';;1';
    suspend;
  end
end
