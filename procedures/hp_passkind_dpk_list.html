CREATE PROCEDURE HP_PASSKIND_DPK_LIST(
    DP_ID D_INTEGER,
    ONLY_LINKED D_BOOL,
    ONLY_ACTIVE D_BOOL = null,
    UN_ID D_ID = null)
RETURNS (
    PK_REC_ID D_INTEGER,
    PK_ID D_INTEGER,
    PK_NAME D_STR_100,
    PK_COUNTRY D_STR_LANG,
    PK_MEMO D_STR_255,
    PK_IS_DELETE D_BOOL,
    PK_IS_DELETE_STR D_STR_CODE,
    PK_CATEGORY D_STR_CODE,
    ORIG_GROUP_NO D_INTEGER,
    DPK_GROUP_NO D_INTEGER,
    DPK_CHECKED_0 D_INTEGER,
    DPK_CHECKED_1 D_INTEGER,
    DPK_CHECKED_2 D_INTEGER)
AS
declare variable citizen_kind d_integer;
declare variable valid_kind d_integer;
begin
  if (dp_id is null) then
    dp_id = 0;

  if (only_active is null) then
    only_active = 0;

  for select pk_id, pk_name, pk_country, pk_is_delete,
             trim(iif(pk_is_other_category = 0, 'Основные', 'Доп.документы')),
             pdt_code || ', ' || pdt_name
  from passkind, passdoctype
  where pk_from_pdt_id = pdt_id and
        pk_id > 0
  order by pk_is_delete, pk_is_other_category, pk_id
  into pk_id, pk_name, pk_country, pk_is_delete,
       pk_category, pk_memo
  do
  begin
    if (only_active = 1 and pk_is_delete = 1) then
      Continue;

    pk_is_delete_str = '';
    if (pk_is_delete = 1) then
      pk_is_delete_str = 'Неактивный';

    for select 0
    from rdb$database
    union
    select distinct dpk_group_no
    from docproppk
    where dpk_from_dp_id = :dp_id and
          dpk_from_pk_id = :pk_id and
          dpk_group_no <> 0 and
          dpk_is_delete = 0
    into dpk_group_no
    do
    begin
      dpk_checked_0 = -1;
      dpk_checked_1 = -1;
      dpk_checked_2 = -1;

      pk_rec_id = pk_id * 100 + dpk_group_no;

      for select dpk_citizen_kind, dpk_valid_kind
      from docproppk
      where dpk_from_dp_id = :dp_id and
            dpk_from_pk_id = :pk_id and
            dpk_group_no = :dpk_group_no and
            dpk_is_delete = 0
      into citizen_kind, valid_kind
      do
      begin
        if (citizen_kind = 0) then
          dpk_checked_0 = valid_kind;
        if (citizen_kind = 1) then
          dpk_checked_1 = valid_kind;
        if (citizen_kind = 2) then
          dpk_checked_2 = valid_kind;
      end

      orig_group_no = dpk_group_no;

      if (dpk_group_no = 0) then
        dpk_group_no = null;

      if (only_linked = 0 or
          dpk_checked_0 <> -1 or
          dpk_checked_1 <> -1 or
          dpk_checked_2 <> -1) then
        suspend;
    end
  end
end
