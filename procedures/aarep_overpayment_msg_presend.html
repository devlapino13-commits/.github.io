CREATE PROCEDURE AAREP_OVERPAYMENT_MSG_PRESEND(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    KONTRACT_NO D_STR_20,
    KC_ID D_ID,
    FIO D_STR_255,
    DISB_DATE D_DATE,
    FACT_CLOSE_DATE D_DATE,
    OVERPAY_DATE D_DATE,
    OVERPAY_SUM DECIMAL(15,2),
    BLOCK_REASON D_STR_255)
AS
declare variable k_id d_id;
declare variable ovm_status d_integer;
declare variable ovm_id d_id;
begin
  rep_date_begin = coalesce(rep_date_begin, '01.01.1900');
  rep_date_end = coalesce(rep_date_end, '31.12.9999');

  for select kc_id, k_kontract_no, c.c_name_fio, k_disb_date, ovm.ovm_over_date, k_id, ovm_status, ovm_id
  from overpayment_msg ovm, knclient kc, kontract k, clients c
  where ovm_status <> 0
    and ovm.ovm_from_msg_id is null  -- не отправлено
    and ovm.ovm_over_date between :rep_date_begin and :rep_date_end
    and ovm.ovm_is_delete = 0
    and kc_id = ovm.ovm_from_kc_id
    and k_id = kc_from_k_id
    and c_id = kc_from_c_id
  order by ovm_over_date
  into kc_id, kontract_no, fio, disb_date, overpay_date, k_id, ovm_status, ovm_id
  do
  begin
    select max(ol_date)
    from  (select ok.ok_ol_date as ol_date
          from operkred ok
          inner join operlist ol on ol.ol_id = ok.ok_from_ol_id
          where ok.ok_from_k_id = :k_id
            and ok.ok_type = 4
            and ok.ok_ol_is_active = 1
            and ok_from_c_id = 0
            and -(ok.ok_os + ok.ok_perc + ok.ok_fine + ok.ok_tax + ok.ok_kommis) > 0
        union all
          select oko_ol_date
          from operkroff oko
          where oko_ol_is_active = 1
          and oko_from_k_id = :k_id
          and oko_type = 4
    )
    into fact_close_date;

    overpay_sum = (select -sum(ok_os)
                  from operkred
                  where ok_from_k_id = :k_id
                    and ok_ol_is_active = 1
                    and ok_type = 9);

    overpay_sum = coalesce(overpay_sum, 0);

    -- перепроверим
    if (ovm_status = 999 and overpay_sum < 1) then
      ovm_status = -1;

    -- критерий: 6.  По займу не было ни одной рассылки СМС о переплате.
    if (ovm_status = 999 and exists(select ovm_id
                                    from overpayment_msg ovm
                                    where ovm.ovm_from_kc_id = :kc_id
                                      and ovm.ovm_from_msg_id is not null  -- отправлено
                                      and ovm.ovm_is_delete = 0
      )) then
        ovm_status = -6;

   block_reason = case ovm_status
                    when -6 then 'уже было отправлено СМС сообщение'
                    when -5 then 'есть задолженность'
                    when -4 then 'не пройдена проверка ' || (select omm_memo from overpayment_msg_memo where omm_id = :ovm_id) -- ПОДФТ
                    when -3 then 'начислялась ГП'
                    when -2 then 'нет основного номера телефона'
                    when -1 then 'сумма меньше 1р'
                    else ''
                  end;

      suspend;
  end 
end
