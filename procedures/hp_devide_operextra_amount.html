CREATE PROCEDURE HP_DEVIDE_OPEREXTRA_AMOUNT(
    K_ID ID_TYPE,
    OL_DATE D_DATE,
    OX_AMOUNT_PLUS CURR_TYPE,
    OX_AMOUNT_MINUS CURR_TYPE,
    OX_AMOUNT_WRITEOFF CURR_TYPE)
RETURNS (
    KC_ID ID_TYPE,
    RES_OX_PLUS CURR_TYPE,
    RES_OX_MINUS CURR_TYPE,
    RES_OX_WRITEOFF CURR_TYPE)
AS
declare variable k_amount curr_type;
declare variable kc_amount curr_type;
declare variable curr_line d_integer;
declare variable koef double precision;
declare variable round_scale integer;
declare variable gr_bal_law curr_type;
declare variable arr_zad varchar(400);
declare variable k_count integer;
declare variable gr_new_saldo curr_type;
declare variable bal_law curr_type;
declare variable saldo_ox_plus curr_type;
declare variable saldo_ox_minus curr_type;
declare variable saldo_ox_writeoff curr_type;
begin
  round_scale = 2;

  if (ox_amount_minus <> 0 and
      ox_amount_writeoff <> 0) then
    exception e_default 'Repay and write off amount at same line';

  select k_amount_approved, k_kolich_client_approved
  from kontract
  where k_id = :k_id
  into k_amount, k_count;

-- если участников группы только 1, возвращаем как есть
  if (k_count = 1) then
  begin
    select kc_id
    from knclient
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0
    into :kc_id;

    res_ox_plus = ox_amount_plus;
    res_ox_minus = ox_amount_minus;
    res_ox_writeoff = ox_amount_writeoff;

    suspend;
    exit;

    when any do
      exception e_default 'Client count is differ: kontract: 1, knclient > 1';
  end

  gr_bal_law = 0;
  arr_zad = '';

  if (ox_amount_minus <> 0 or ox_amount_writeoff <> 0) then
  begin
    for select kc_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into kc_id
    do
    begin
      bal_law = 0;
      select oks_bal_law
      from kn_oksummary
      where oks_from_kc_id = :kc_id and
            oks_date_begin <= :ol_date and
            oks_date_end >= :ol_date
      into bal_law;
      if (bal_law is null) then
        bal_law = 0;

      gr_bal_law = gr_bal_law + bal_law;
      arr_zad = arr_zad || right('                    ' || bal_law, 20);
    end

    gr_new_saldo = gr_bal_law + ox_amount_plus - ox_amount_minus - ox_amount_writeoff;
  end

-- максимально допустимые отклонения, меньше которой будем выравнивать
  saldo_ox_plus = ox_amount_plus;
  saldo_ox_minus = ox_amount_minus;
  saldo_ox_writeoff = ox_amount_writeoff;
  curr_line = 0;

  for select kc_id, kc_amount_approved
      from knclient
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0
      into kc_id, kc_amount
  do
  begin
    curr_line = curr_line + 1;

    if (curr_line <> k_count) then
    begin
      bal_law = 0;
      if (arr_zad <> '') then
        bal_law = substring(arr_zad from ((curr_line - 1) * 20 + 1) for 20);

      koef = cast(kc_amount as double precision) / k_amount;

      res_ox_plus = 0;
      res_ox_minus = 0;
      res_ox_writeoff = 0;

      if (ox_amount_plus <> 0) then -- начисление
        res_ox_plus = round(koef * ox_amount_plus, round_scale);

-- погашения
      if (ox_amount_minus <> 0) then
        res_ox_minus = bal_law - round(koef * gr_new_saldo, round_scale);
      else
      if (ox_amount_writeoff <> 0) then
        res_ox_writeoff = bal_law - res_ox_minus - round(koef * gr_new_saldo, round_scale);

      saldo_ox_plus = saldo_ox_plus - res_ox_plus;
      saldo_ox_minus = saldo_ox_minus - res_ox_minus;
      saldo_ox_writeoff = saldo_ox_writeoff - res_ox_writeoff;
    end
    else
    begin
      res_ox_plus = saldo_ox_plus;
      res_ox_minus = saldo_ox_minus;
      res_ox_writeoff = saldo_ox_writeoff;
    end

    suspend;
  end -- KnClient
end
