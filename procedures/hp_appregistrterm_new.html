CREATE PROCEDURE HP_APPREGISTRTERM_NEW(
    C_ID D_ID NOT NULL,
    DT_APP D_DATE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    LAST_ITA_ID D_ID,
    NOSIGN_LST D_STR_8100,
    ERR_CODE D_INTEGER,
    ERR_TEXT D_STR_255)
AS
declare variable num_term d_integer;
declare variable ito_id d_integer;
declare variable trm_id d_integer;
declare variable trm_type d_smallint;
declare variable trm_name d_str_255;
declare variable trm_name_lat d_str_255;
declare variable trm_object_id d_id;
declare variable adl_id d_id;
declare variable partner_id d_id;
declare variable reg_id d_integer;
declare variable res_text d_str_255;
declare variable rd_code d_str_255;
begin
  err_code = 0;
  err_text = '';
  nosign_lst = '';
  num_term = 0;
  for
    select
      ito_id,
      trm_id,
      trm_type,
      ito_object_id
    from ipc_terminal_ownership
      inner join ipc_terminals
        on trm_id = ito_trm_id
          and trm_actual = 1
    where ito_status = 0
  into
    :ito_id,
    :trm_id,
    :trm_type,
    :trm_object_id
  do
  begin
    if (trm_type = 2) then
    begin
      trm_name = 'М Булак';
      trm_name_lat = 'M Bulak';
      reg_id = trm_object_id;
      adl_id = (select reg_address_fact from region where reg_unicode = :reg_id);
    end
    else
    begin
      select
        cs_from_c_id,
        cs_name,
        cs_name_lat,
        cs_from_adl_id,
        rd_code
      from clientshops
      left join refdata on cs_from_rd_id = rd_id
      where cs_id = :trm_object_id
      into
        :partner_id,
        :trm_name,
        :trm_name_lat,
        :adl_id,
        :rd_code;

      reg_id = null;
      select
        err_text,
        reg_id
      from hp_clientshops_checktsp(:partner_id, :rd_code, :dt_app)
      into
        :res_text,
        :reg_id;
      if (:res_text <> '') then
      begin
        nosign_lst = nosign_lst || iif(nosign_lst = '', '', ';') || trm_id;
        continue;
      end
    end

    insert into ipc_terminal_app (
      ita_doc_date,
      ita_from_trm_id,
      ita_type,
      ita_term_name,
      ita_term_name_lat,
      ita_from_reg_id,
      ita_from_adl_id,
      ita_contact_person,
      ita_from_un_id
    )
    values (
      :dt_app,
      :trm_id,
      :trm_type,
      :trm_name,
      :trm_name_lat,
      :reg_id,
      :adl_id,
      :c_id,
      :un_id
    )
    returning ita_id
    into :last_ita_id;
    num_term = num_term + 1;

    update ipc_terminal_ownership set
      ito_status = 1
    where ito_id = :ito_id;
  end

  if (num_term = 0) then
  begin
    err_code = 1;
    err_text = 'Терминалы для создания заявок на регистрацию не найдены';
  end else err_text = f_format('Было создано заявок на регистрацию терминалов: %1 ', num_term);
  suspend;
end
