CREATE PROCEDURE HP_MILITARY_QUERY_STATUS(
    C_ID D_ID)
RETURNS (
    STATUS D_SMALLINT,
    QUERY_DATE D_DATE,
    ANSWER D_DATE,
    MCQ_RECEIVED D_SMALLINT)
AS
declare variable EIL_ID D_ID;
declare variable C_SEX D_SMALLINT;
declare variable AGE D_INTEGER;
declare variable eil_status d_smallint;
begin
  -- RFCKG-4079 Процедура вернет статус запроса проверки статуса воинской службы

  select c.c_sex,
         f_age(c.c_birth_date, current_date)
  from clients c
  where c.c_id = :c_id
  into c_sex, age;

  if (coalesce(age, '') = '') then
    exception e_message('Не указана дата рождения клиента!');

  if (age not between 18 and 27 -- клиент младше 18 ИЛИ старше 27
      or (c_sex <> 1) ) then    -- клиент не мужчина
  begin
    status = 0;
    query_date = current_date;
    suspend;
    exit;
  end

  select mcq.mcq_status,
         cast(mcq.mcq_status_datetime as d_date),
         mcq.mcq_answer_date,
         mcq.mcq_from_eil_id,
         mcq.mcq_received
  from military_check_query mcq
  where mcq.mcq_from_c_id = :c_id
  order by mcq.mcq_status_datetime desc
  rows 1
  into status,
       query_date,
       answer,
       eil_id,
       mcq_received;

  if(coalesce(eil_id, 0) = 0) then
    status = -3; -- запроса еще не было

  if (coalesce(eil_id, 0) <> 0) then
  begin
    select eil.eil_status
    from extinfolist eil
    where eil.eil_id = :eil_id
    into eil_status;

    if (eil_status in (-1, 0)) then
      status = -4; -- ждем ответ от сервиса
  end

  suspend;
end
