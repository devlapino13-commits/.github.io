CREATE PROCEDURE AAREP_SSVZ_MONTH(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    KF_OFFICER VARCHAR(100),
    SVZ_STAGE_MONTH INTEGER,
    DISB_KOLICH_KRED_LAST INTEGER,
    DISB_KOLICH_KRED INTEGER,
    CLOSED_KOLICH_GR INTEGER,
    DOLYA_UDERJ CURR_TYPE,
    DISB_KOLICH_PARALEL INTEGER,
    PAR7_KAT_1 CURR_TYPE,
    PAR7_KAT_1_2 CURR_TYPE,
    DELINQ_BALANCE_KAT_1_2_LAST CURR_TYPE,
    DELINQ_BALANCE_KAT_1_2 DECIMAL(18,4),
    KOL_DELINQ_GR INTEGER,
    BALANCE_KAT_1_2 DECIMAL(18,4),
    KOL_CLI_KAT_1_2 INTEGER,
    PRIROST_CLI INTEGER)
AS
declare variable rep_date_begin date;
declare variable svz_id id_type;
declare variable rep_date_end_last date;
declare variable rep_date_begin_last date;
declare variable svz_stage_date date;
declare variable disb_kolich_gr integer;
declare variable balance_kat_1 curr_type;
declare variable delinq_balance_kat_1 curr_type;
declare variable kol_cli_kat_1_2_last integer;
begin
-- начало мемяца
  rep_date_begin = :rep_date_end - extract(day from :rep_date_end) + 1;
-- конец предыдущего месяца
  rep_date_end_last = :rep_date_begin - 1;
-- начало предыдущего месяца
  rep_date_begin_last = :rep_date_end_last - extract(day from :rep_date_end_last) + 1;

  for
-- Вытаскивает всех СВЗ кроме декретников
      select ps_from_c_id
      from jobkind, joblist, hr_pstatus, region_list(:rep_region_id, 1, 3, 1)
      where jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
            jk_id = jl_kind and
            jl_id = ps_from_jl_id and
            ps_status = 1 and
            ps_is_working = 1 and
            :rep_date_end between ps_date_begin and ps_date_end and
            ps_from_reg_id = id_list
-- Исключаем декретников
            and
            not coalesce((select sawt_id
                          from hr_voctravel, hr_orders, sa_worktype
                          where hrv_from_ho_id = ho_id and
                                ho_order_type = sawt_id and
                                hrv_from_c_id = ps_from_c_id and
                                :rep_date_end between hrv_date_begin and hrv_date_end_fact
                          order by hrv_date_begin desc
                          rows 1), 0) in (16, 17, 18)
      into svz_id
  do
  begin
--    ФИО СВЗ
    kf_officer = (select c_name_fio
                  from clients
                  where c_id = :svz_id);
--    Стаж в месяцах
    svz_stage_date = null;
    select ps_date_begin
    from hr_pstatus --, hr_orders , joblist
    where ps_from_c_id = :svz_id and
          ps_date_begin is not null and
          ps_status = 1
    order by ps_date_begin
    rows 1
    into :svz_stage_date;
    if (svz_stage_date is null) then
      svz_stage_date = :rep_date_end;
    svz_stage_month = datediff(day, :svz_stage_date, :rep_date_end) / 30;
--   Кол-во выдач за предыдущий месяц
    disb_kolich_kred_last = null;
    select sum(k_kolich_client_approved)
    from kontstate, kontract, product
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          cast(ks_date_begin as date) between :rep_date_begin_last and :rep_date_end_last and
          ks_kc_id = :svz_id and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind <> 2
    into disb_kolich_kred_last;
    if (disb_kolich_kred_last is null) then
      disb_kolich_kred_last = 0;
--   Кол-во выдач за текущий месяц
    disb_kolich_kred = null;
    select sum(k_kolich_client_approved)
    from kontstate, kontract, product
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
          ks_kc_id = :svz_id and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind <> 2
    into disb_kolich_kred;
    if (disb_kolich_kred is null) then
      disb_kolich_kred = 0;
--  Кол-во закрытых групп за текущий месяц
    closed_kolich_gr = null;
    select count(k_id)
    from kontstate, kontract, product
    where ks_ol_is_delete = 0 and
          ks_state = 2 and
          cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
          ks_kc_id = :svz_id and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind <> 2
    into closed_kolich_gr;
    if (closed_kolich_gr is null) then
      closed_kolich_gr = 0;
--  Кол-во выданных групп для расчета % удержание групп
    disb_kolich_gr = null;
    select count(k_id)
    from kontstate, kontract, product
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
          ks_kc_id = :svz_id and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind <> 2 and
          k_circle > 1 -- для расчете удержание будем брать только повтоно выданные
    into disb_kolich_gr;
    if (disb_kolich_gr is null) then
      disb_kolich_gr = 0;
    if (closed_kolich_gr <> 0) then
      dolya_uderj = cast(disb_kolich_gr as double precision) / closed_kolich_gr;
    else
      dolya_uderj = 0;
--  Выдача параллельных кредитов
    disb_kolich_paralel = null;
    select sum(k_kolich_client_approved)
    from kontstate, kontract, product
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
          ks_kc_id = :svz_id and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind = 2
    into disb_kolich_paralel;
    if (disb_kolich_paralel is null) then
      disb_kolich_paralel = 0;
--  Расчет пара по категории 1
-- Вычисление ПАР
    select sum(iif((select aa_category_id
                    from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) = 1, oks_balance, 0)) balance_kat_1, sum(iif((select aa_category_id
                                                                                                                                                                from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) = 1 and :rep_date_end - 7 >= oks_delinq_date, oks_balance, 0)) delinq_balance_kat_1,
           sum(oks_balance) balance_kat_1_2,
           sum(iif(:rep_date_end - 7 >= oks_delinq_date, oks_balance, 0)) delinq_balance_kat_1_2
    from oksummary
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_kc_id = :svz_id and
          oks_close_state = 0 and
          (select aa_category_id
           from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2)
    into balance_kat_1, delinq_balance_kat_1, balance_kat_1_2, delinq_balance_kat_1_2;
    if (balance_kat_1 is null) then
      balance_kat_1 = 0;
    if (delinq_balance_kat_1 is null) then
      delinq_balance_kat_1 = 0;
    if (balance_kat_1_2 is null) then
      balance_kat_1_2 = 0;
    if (delinq_balance_kat_1_2 is null) then
      delinq_balance_kat_1_2 = 0;

    if (balance_kat_1 > 0) then
      par7_kat_1 = delinq_balance_kat_1 / balance_kat_1;
    else
      par7_kat_1 = 0;
    if (balance_kat_1_2 > 0) then
      par7_kat_1_2 = delinq_balance_kat_1_2 / balance_kat_1_2;
    else
      par7_kat_1_2 = 0;
--  Баланс просрочки за предыдущий месяц по категории 1 и 2
    select sum(oks_balance)
    from oksummary
    where :rep_date_end_last between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_kc_id = :svz_id and
          oks_close_state = 0 and
          :rep_date_end - 7 >= oks_delinq_date and
          (select aa_category_id
           from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2)
    into delinq_balance_kat_1_2_last;
    if (delinq_balance_kat_1_2_last is null) then
      delinq_balance_kat_1_2_last = 0;
--    Кол-во просроченных групп
    kol_delinq_gr = null;
    select count(oks_from_k_id)
    from oksummary, kontract, product
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_kc_id = :svz_id and
          oks_close_state = 0 and
          :rep_date_end - 30 >= oks_delinq_date and
          (select aa_category_id
           from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2) and
          oks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind <> 2
    into kol_delinq_gr;
    if (kol_delinq_gr is null) then
      kol_delinq_gr = 0;
--    Активные клиенты категории 1 и 2
    kol_cli_kat_1_2 = null;
    select sum(ac_is_client)
    from aarep_cli_active_list(:rep_date_end)
    where ac_kc_id = :svz_id and
          ac_is_client = 1 and
          (select aa_category_id
           from aarep_get_category_id(ac_kontract_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2)
    into kol_cli_kat_1_2;
    if (kol_cli_kat_1_2 is null) then
      kol_cli_kat_1_2 = 0;
--    Активные клиенты категории 1 и 2 на предыдущий месяц
    kol_cli_kat_1_2_last = null;
    select sum(ac_is_client)
    from aarep_cli_active_list(:rep_date_end_last)
    where ac_kc_id = :svz_id and
          ac_is_client = 1 and
          (select aa_category_id
           from aarep_get_category_id(ac_kontract_id, :rep_date_end_last, null, null, null, null, null, null)) in (1, 2)
    into kol_cli_kat_1_2_last;
    if (kol_cli_kat_1_2_last is null) then
      kol_cli_kat_1_2_last = 0;
    prirost_cli = kol_cli_kat_1_2 - kol_cli_kat_1_2_last;

    suspend;
  end
end
