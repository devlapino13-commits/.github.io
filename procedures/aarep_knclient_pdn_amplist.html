CREATE PROCEDURE AAREP_KNCLIENT_PDN_AMPLIST(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_K_ID D_ID = null,
    REPEATED D_BOOL = null)
RETURNS (
    LINE D_STR_10,
    LINE_TYPE D_STR_10,
    LOAN_NO D_INTEGER,
    AVG_REQ_DATE D_DATE,
    K_KONTRACT_NO D_STR_20,
    KC_ID D_ID,
    K_DISB_DATE D_DATE,
    UID D_STR_CODE,
    KBKI_STR D_STR_CODE,
    KBKI_SOURCE_STR D_STR_CODE,
    BKI_SOURCE_STR D_STR_CODE,
    CALC_DATE D_DATE,
    AVG_SUM D_CURRENCY,
    AVG_CURRENCY D_STR_10,
    IS_ACTUAL_STR D_STR_10,
    AVG_SUM_FINAL D_CURRENCY,
    INNER_K_KONTRACT_NO D_STR_20,
    INNER_KC_ID D_ID,
    K_ID D_ID,
    KBKI_INFO1 D_STR_CODE,
    KBKI_INFO2 D_STR_CODE,
    KBKI_INFO3 D_STR_CODE,
    KBKI_INFO4 D_STR_CODE,
    CLIENT_TYPE D_STR_20)
AS
declare variable is_actual boolean;
declare variable uid_temp d_str_code;
declare variable calc_date_temp d_date;
declare variable avg_sum_temp d_currency;
declare variable avg_sum_final_sum d_currency;
declare variable kcp_id d_id;
declare variable rec_no d_integer;
declare variable kbki_addinfo_jso d_integer;
declare variable kbki_addinfo d_str_255;
declare variable kbki_eil_id d_id;
declare variable pr_code d_str_100;
declare variable is_msp d_bool;

declare function getcatval (
kind d_str_code,
id d_id)
returns d_str_255
as
declare variable res d_str_255;
begin
res = (select bct_name
from bki_catalog
where bct_kind = :kind and
bct_id = :id);
return iif(res is null, id, res);
end


begin
  if (repeated is null) then
    repeated = 0;

  loan_no = 0;

  for select distinct new_kc_id, f_uuid_cbr(f_uuid('knclient', new_kc_id))
      from hp_pdn_get_loanlist(:rep_date_begin, :rep_date_end, :rep_k_id, :repeated)
      into kc_id, uid_temp
  do
  begin
    select k_kontract_no, k_disb_date, k_id, p.pr_code
    from knclient, kontract k, product p
    where kc_id = :kc_id and
          k_id = kc_from_k_id and
          p.pr_id = k.k_from_pr_id
    into k_kontract_no, k_disb_date, k_id, pr_code;

    client_type = 'ФЛ';
    if (pr_code containing '.selfemployed.') then
      client_type = 'СЗ';
    else if (pr_code containing '.business.') then
    begin
      client_type = 'ИП-неМСП';
      
      select is_msp
      from hp_kontur_ip_get_la_info('knclient', :kc_id)
      into is_msp;

      if (is_msp = 1) then
        client_type = 'ИП-МСП';
    end

    for select kcp_id, kc_date, kpl_month_repay_final, getcatval('bki_ogrn', kcp_kbki_host), kcp_kbki_addinfo,
               kcp_kbki_eil_id
        from knclient_pdn, knclient_pdnlist
        where kc_id = :kc_id and
              kpl_from_kcp_id = kcp_id and
              kpl_rec_no = 0 and
              kcp_kind = :repeated
        order by kcp_id desc
        into kcp_id, calc_date_temp, avg_sum_temp, kbki_str, kbki_addinfo, kbki_eil_id
    do
      kbki_info1 = null;
    kbki_info2 = null;
    kbki_info3 = null;
    kbki_info4 = null;
    kbki_addinfo_jso = null;
    avg_req_date = (select eil_date
                    from extinfolist
                    where eil_id = :kbki_eil_id);

    if (coalesce(kbki_addinfo, '') <> '') then
    begin
      kbki_addinfo_jso = json.parse(kbki_addinfo);
      kbki_info1 = json.obj_get_val(kbki_addinfo_jso, '1247700058319'); -- СБ
      kbki_info2 = json.obj_get_val(kbki_addinfo_jso, '1057746710713'); -- НБКИ
      kbki_info3 = json.obj_get_val(kbki_addinfo_jso, '1047796788819'); -- ОКБ
      kbki_info4 = json.obj_get_val(kbki_addinfo_jso, '1057747734934'); -- Кинфо
    end

    -- наш займ
    rec_no = 0;
    loan_no = loan_no + 1;
    line = loan_no;
    line_type = 'Новый';
    uid = uid_temp;
    calc_date = calc_date_temp;
    avg_sum = avg_sum_temp;
    avg_sum_final = avg_sum;
    avg_sum_final_sum = avg_sum_final;
    kbki_source_str = '-';
    bki_source_str = '-';
    inner_k_kontract_no = null;
    inner_kc_id = null;
    suspend;

    line_type = 'ССП';
    kbki_info1 = null;
    kbki_info2 = null;
    kbki_info3 = null;
    kbki_info4 = null;

    for select kpa_uid, getcatval('bki_ogrn', kpa_kbki), getcatval('bki_ogrn', kpa_bki), kpa_calc_date, kpa_sum,
               kpa_currency, kpa_is_actual = 1, kpa_from_kc_id
        from knclient_pdn_amplist, knclient_pdn
        where kcp_id = kpa_from_kcp_id and
              kcp_id = :kcp_id
        order by kpa_uid, kpa_calc_date desc, kpa_kbki
        into uid, kbki_source_str, bki_source_str, calc_date, avg_sum, avg_currency, is_actual, inner_kc_id
    do
    begin
      rec_no = rec_no + 1;
      line = loan_no || '.' || rec_no;
      avg_sum_final = iif(is_actual, avg_sum, null);
      avg_sum_final_sum = avg_sum_final_sum + coalesce(avg_sum_final, 0);
      is_actual_str = iif(is_actual, 'Да', 'Нет');
      inner_k_kontract_no = null;

      if (inner_kc_id is not null) then
      begin
        select k_kontract_no
        from knclient, kontract
        where kc_id = :inner_kc_id and
              k_id = kc_from_k_id
        into inner_k_kontract_no;
      end

      suspend;
    end

    line = loan_no;
    line_type = 'Итог';
    uid = uid_temp;
    avg_currency = null;
    is_actual_str = null;
    calc_date = calc_date_temp;
    avg_sum = avg_sum_temp;
    avg_sum_final = avg_sum_final_sum;
    kbki_source_str = '-';
    bki_source_str = '-';
    inner_k_kontract_no = null;
    inner_kc_id = null;

    if (kbki_addinfo_jso is not null) then
    begin
      kbki_info1 = json.obj_get_val(kbki_addinfo_jso, '1247700058319'); -- СБ
      kbki_info2 = json.obj_get_val(kbki_addinfo_jso, '1057746710713'); -- НБКИ
      kbki_info3 = json.obj_get_val(kbki_addinfo_jso, '1047796788819'); -- ОКБ
      kbki_info4 = json.obj_get_val(kbki_addinfo_jso, '1057747734934'); -- Кинфо
    end

    suspend;
  end
end
