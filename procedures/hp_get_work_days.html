HP_GET_WORK_DAYS

Возвращает количество рабочих дней для расчета просрочки. / Рабочие дни - не субботы и не воскресенья

CREATE PROCEDURE HP_GET_WORK_DAYS(
    DATE_BEGIN DATE,
    DATE_END DATE,
    PR_IS_FINE_WEEKEND INTEGER)
RETURNS (
    WD_COUNT INTEGER)
AS
declare variable wd_begin integer;
declare variable wd_end integer;
declare variable w1d_count integer;
declare variable w2d_count integer;
declare variable d date;
declare variable hol_count integer;
begin
  /*
  PR_IS_FINE_WEEKEND
  0 - не начисляем в выходные, не начисляем в праздники
  1 -    начисляем в выходные, не начисляем в праздники
  2 - не начисляем в выходные,    начисляем в праздники
  3 -    начисляем в выходные,    начисляем в праздники
  */
  wd_count = 0;
  if (date_end > date_begin) then
  begin
    if (pr_is_fine_weekend in (0, 2)) then
    begin
      if (date_end - date_begin < 8) then
      begin
        /* Даты находятся близко */
        d = date_begin;
        while (d < date_end) do
        begin
          if (extract(weekday from d) not in (0, 6)) then
            wd_count = wd_count + 1;
          d = d + 1;
        end
      end
      else
      begin
        /* Даты находятся в разных неделях */
        wd_begin = extract(weekday from date_begin);
        if (wd_begin = 0) then
          wd_begin = 7;
        wd_end = extract(weekday from date_end);
        if (wd_end = 0) then
          wd_end = 7;
        w1d_count = 0;
        if (wd_begin < 6) then
          w1d_count = 6 - wd_begin;
        w2d_count = 5;
        if (wd_end < 6) then
          w2d_count = wd_end;
        wd_count = ((date_end - wd_end) - (date_begin + 7 - wd_begin)) / 7 * 5 + w1d_count + w2d_count;
        if (wd_end < 6) then
          wd_count = wd_count - 1;
      end
    end
    else
      wd_count = date_end - date_begin;

    if (pr_is_fine_weekend in (0, 1)) then
    begin
      hol_count = 0;/* Отнимаем праздничние дни  */
      select count(*)
      from sa_calendays
      where scd_date between :date_begin and :date_end - 1 and
            scd_work_hours = 0 and
            extract(weekday from scd_date) not in (0, 6)
      into :hol_count;
      if (hol_count is null) then
        hol_count = 0;
      wd_count = wd_count - hol_count;
    end
    if (wd_count < 0) then
      wd_count = 0;
  end
  suspend;

end
