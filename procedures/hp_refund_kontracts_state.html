CREATE PROCEDURE HP_REFUND_KONTRACTS_STATE(
    OBJ_ID D_ID,
    OBJ_KIND D_INTEGER)
RETURNS (
    REFUND_KONTRACTS_STR D_STR_LONG,
    ERROR_TEXT D_STR_255,
    ERROR_STATUS D_SMALLINT)
AS
declare variable k_kontract_no d_str_20;
declare variable k_id d_id;
declare variable pr_code d_str_100;
declare variable zad_os d_currency;
declare variable zad_perc d_currency;
declare variable zad_total d_currency;
declare variable ma_summa_minus d_currency;
declare variable non_refinanceable_kontract d_str_long;
declare variable disb_date d_date;
declare variable error_loan_tmp d_smallint;
declare variable ma_id d_id;
declare variable c_id d_id;
declare variable days_activ_pros d_integer;
declare variable uuid d_str_code;
declare variable ma_office_id d_integer;
begin
  -- RFCRU-3486 Изменение порядка рефинансирования займов
  -- 1) Процедура Проверяет условия для рефинансирования см. табл. CONDITIONMAIN
  -- 2) Возвращает рефинансируемые контракты клиента по заявке refund_kontracts_str для HP_HTML_GET_LOAN_STATUS
  refund_kontracts_str = '<ul style = "padding-left: 15px;">';
  non_refinanceable_kontract = '';
  error_text = '';
  zad_total = 0; -- задолжность всех контрактов ОД + %
  error_status = 0;

  if (obj_kind = f_objkind('kontract', 'id')) then
  begin
    select k.k_from_ma_app_id
    from kontract k
    where k.k_id = :obj_id rows 1
    into ma_id;
  end
  if (obj_kind = f_objkind('miniapp', 'id')) then
    ma_id = obj_id;

  select ma.ma_summa_minus, pr.pr_code, ma_office_id
  from miniapp ma
  inner join product pr on pr.pr_id = ma.ma_from_pr_id
  where ma.ma_id = :ma_id and
        ma_refund = 1 and
        ma.ma_is_active = 1 and
        ma.ma_refund = 1
  order by ma.ma_number desc rows 1
  into ma_summa_minus, pr_code, ma_office_id;

  select mac.mac_from_c_id
  from miniappclient mac
  where mac.mac_from_ma_id = :ma_id
  and mac.mac_amount > 0
  rows 1
  into c_id;

  ma_summa_minus = coalesce(ma_summa_minus, 0);

  for select k.k_kontract_no, k.k_id, k.k_disb_date
      from kontrefinance kr
      inner join kontstate ks on kr.kr_from_k_id = ks.ks_from_k_id
      inner join kontract k on k.k_id = kr.kr_from_k_id
      where kr.kr_from_ma_id = :ma_id and
            ks.ks_ol_is_delete = 0 and
            ks.ks_state in (1, 2) and -- RFCRU-4385
            current_date between ks.ks_date_begin and ks.ks_date_end
      into k_kontract_no, k_id, disb_date
  do
  begin
    error_loan_tmp = 0;
    -- основная сумма, проценты
    select sum(zad_os), sum(sh_calc_perc)
    from aarep_make_zad_list(:k_id, current_date, 1)
    into zad_os, zad_perc;

    zad_os = coalesce(zad_os, 0);
    zad_perc = coalesce(zad_perc, 0);

    zad_total = zad_total + zad_os + zad_perc;

    if (pr_code containing '.reliable_client.' or pr_code containing '.loyal.') then -- ЕСЛИ невыполненны условия для "Надежный клиент" и "Параллельный"
    begin

      select
            iif((cast(:disb_date as date) - oks.oks_delinq_date < 0), 0, cast(:disb_date as date) -  oks.oks_delinq_date) -- числоднейактивнойпросрочки
      from oksummary oks
      where :disb_date between oks.oks_date_begin and oks.oks_date_end     -- RFCRU-4042
      and oks.oks_from_k_id = :k_id
      and oks_close_state = 0
      into days_activ_pros;

      -- у клиента нет активной просрочки по текущему займу в Компании
      if (days_activ_pros > 0) then
        error_loan_tmp = 0;

      -- У клиента не более 8ми баллов, к рефинансированию не подлежит
      if (f_point_get(11, ma_id, c_id, current_date) < 8) then  -- RFCRU-4053
      begin
        error_loan_tmp = 1;
      end

      if (error_loan_tmp = 1) then
      begin
        non_refinanceable_kontract = non_refinanceable_kontract || f_format('%1, ', :k_kontract_no);
        error_status = 1;
      end
    end

    refund_kontracts_str = refund_kontracts_str || f_format('<li>%1</li>', k_kontract_no);
  end

  -- в список добавляем рефинансированные внешние займы клиента (RFCRU-4021)
  uuid = null;
  zad_os = null;
  for select kr_loan_uuid, kr_balance
      from kontrefinance
      where kr_from_ma_id = :ma_id and
            kr_from_k_id is null and
            kr_loan_uuid is not null
      into uuid, zad_os
  do
  begin
    zad_total = zad_total + zad_os;
    refund_kontracts_str = refund_kontracts_str || f_format('<li>%1</li>', uuid);
  end

  if (refund_kontracts_str containing '<li>') then
    refund_kontracts_str = refund_kontracts_str || '</ul>';
  else
    refund_kontracts_str = '';

  if (error_status = 1) then
  begin
    non_refinanceable_kontract = left(non_refinanceable_kontract, char_length(non_refinanceable_kontract) - 2);
    error_text = f_format('Выбранный заем/мы «%1» не может быть рефинансирован', non_refinanceable_kontract);
  end

  if (ma_summa_minus < zad_total) then
  begin
    error_status = 1;
    error_text = 'Суммы нового займа недостаточно для рефинансирования всех выбранных займов, измените выбор';
  end

  suspend;
end
