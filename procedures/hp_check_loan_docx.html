CREATE PROCEDURE HP_CHECK_LOAN_DOCX(
    K_ID ID_TYPE)
RETURNS (
    KF_CLIENT_FIO VARCHAR(100),
    KF_CLIENT_PASS_NO VARCHAR(30),
    KF_DISB_DATE DATE,
    KC_AMOUNT DECIMAL(18,4),
    KF_ERROR_TEXT VARCHAR(255),
    KF_ERROR_IS_CRITICAL INTEGER)
AS
declare variable C_ID ID_TYPE;
declare variable PDT_CODE varchar(100);
declare variable REGISTRATION_FIZ smallint;
declare variable MAGRATION_CARD smallint;
declare variable REGISTRATION_RVP smallint;
declare variable REGISTRATION_VNJ smallint;
declare variable WORK_PATENT smallint;
declare variable KVIT_PATENT smallint;
declare variable WORK_PERMIT smallint;
declare variable PK_NAME varchar(100);
declare variable SUPDP smallint;
declare variable JOB_AGREEMENT smallint;
declare variable CERTIFICATE_APPROVING_DOC smallint;
declare variable CERTIFICATE_2NDFL smallint;
declare variable CERTIFICATE_IP smallint;
declare variable CERTIFICATE_3NDFL smallint;
declare variable DOOKU smallint;
declare variable REGISTRATION_PERMANENT smallint;
declare variable C_CITIZEN integer;
declare variable RD_CODE varchar(255);
declare variable RD_NAME varchar(255);
declare variable PR_CODE varchar(100);
declare variable IS_DISB smallint;
declare variable CDS_DOC_DATE_BEGIN date;
declare variable CDS_DOC_DATE_END date;
declare variable MAGRATION_CARD_DATE_BEGIN date;
declare variable MAGRATION_CARD_DATE_END date;
declare variable WORK_PATENT_DATE_BEGIN date;
declare variable WORK_PATENT_DATE_END date;
declare variable CERTIFICATE_APPROVING_DOC_BEGIN date;
declare variable REGISTRATION_YUR smallint;
declare variable CERTIFICATE_DRIVER_RF smallint;
declare variable CERTIFICATE_DRIVER_KG smallint;
declare variable CERTIFICATE_DRIVER_KG_MN smallint;
declare variable K_KREDLINE_END date;
declare variable REGISTRATION_RVP_DATE_BEGIN date;
declare variable REGISTRATION_RVP_DATE_END date;
declare variable REGISTRATION_VNJ_DATE_BEGIN date;
declare variable REGISTRATION_VNJ_DATE_END date;
declare variable REGISTRATION_FIZ_DATE_END date;
declare variable SUPDP_DATE_END date;
declare variable KVIT_PATENT_DATE_END date;
declare variable REGISTRATION_FIZ_DATE_BEGIN date;
declare variable KONTRACT_NA_OBUCHENIE smallint;
declare variable SPRAVKA_OB_UCHEBE smallint;
declare variable KVIT_OB_OPLATE_ZA_UCHEB smallint;
declare variable SPRAVKA_OB_UCHEBE_BEGIN date;
declare variable CERTIFICATE_RUS_LANG smallint;
declare variable JOB_CERTIFICATE D_SMALLINT;
declare variable STUDENT_CARD D_SMALLINT;
declare variable STUDENT_CARD_DATE_END date;
declare variable MEDIKAL_CERTIFICATE smallint;
begin
  for select kc_from_c_id, k_disb_date, kc_amount_approved, pr_code, k_kredline_end
      from kontract, product, knclient
      where k_id = :k_id and
            k_from_pr_id = pr_id and
            k_id = kc_from_k_id and
            kc_amount_approved > 0
      into c_id, kf_disb_date, kc_amount, pr_code, k_kredline_end
  do
  begin
    select c_name_fio, c_pass_no, c_citizen
    from clients
    where c_id = :c_id
    into kf_client_fio, kf_client_pass_no, c_citizen;

    select rd_code, rd_name
    from refdata
    where rd_id = :c_citizen
    into rd_code, rd_name;

    registration_fiz = 0; --  временная регистрация, физ.лицо
    registration_yur = 0; --  временная регистрация, физ.лицо
    registration_permanent = 0; -- постоянная регистрация
    magration_card = 0; -- миграционная карта
    registration_rvp = 0; --рвп - разрешение на временное проживание
    registration_vnj = 0; -- внж - вид на жительство
    work_patent = 0; -- патент
    kvit_patent = 0; -- квитанция патент
    work_permit = 0; -- разрешение наработу
    supdp = 0; -- свидетельство участника программы добровольного переселения
    job_agreement = 0; -- трудовой договор
    certificate_approving_doc = 0; -- справка о подаче на патент/рвп/внж
    certificate_2ndfl = 0; -- справка 2ндфл
    certificate_ip = 0; -- свидетельство ип
    certificate_3ndfl = 0; -- справка 3ндфл
    dooku = 0; -- договор об оказании консультационных услуг
    magration_card_date_begin = '01.01.1990';
    magration_card_date_end = '01.01.1990';
    work_patent_date_begin = '01.01.1990';
    certificate_approving_doc_begin = '01.01.1990';
    certificate_driver_rf = 0;
    certificate_driver_kg = 0;
    certificate_driver_kg_mn = 0;
    registration_rvp_date_begin = '01.01.1990';
    registration_rvp_date_end = '01.01.1990';
    registration_vnj_date_begin = '01.01.1990';
    registration_vnj_date_end = '01.01.1990';
    registration_fiz_date_begin = '31.12.2999';
    registration_fiz_date_end = '01.01.1990';
    supdp_date_end = '01.01.1990';
    kvit_patent_date_end = '01.01.1990';
    kontract_na_obuchenie = 0;
    spravka_ob_uchebe = 0;
    kvit_ob_oplate_za_ucheb = 0;
    spravka_ob_uchebe_begin = '01.01.1990';
    certificate_rus_lang = 0;
    student_card = 0;
    student_card_date_end = '01.01.1990';

    -- RFCRU-4085
    medikal_certificate = 0;

    for select pdt_code, pk_name, cds_doc_date_begin, cds_doc_date_end
        from clientdocs, passkind, passdoctype
        where cds_from_c_id = :c_id and
              cds_is_delete = 0 and
              coalesce(cds_doc_blob_1, 0) > 0 and
              cds_from_pk_id = pk_id and
              pk_from_pdt_id = pdt_id and
              :kf_disb_date between cds_doc_date_begin and coalesce(cds_doc_date_end, '31.12.2999')

        into pdt_code, pk_name, cds_doc_date_begin, cds_doc_date_end
    do
    begin
      if (pdt_code = 'registration_fiz') then
      begin
        registration_fiz = 1;
        registration_fiz_date_begin = cds_doc_date_begin;
        registration_fiz_date_end = cds_doc_date_end;
      end
      if (pdt_code = 'registration_yur') then
      begin
        registration_yur = 1;
        registration_fiz_date_begin = cds_doc_date_begin;
        registration_fiz_date_end = cds_doc_date_end;
      end
      if (pdt_code = 'registration_permanent') then
        registration_permanent = 1;
      if (pdt_code = 'magration_card' and
          magration_card_date_begin < cds_doc_date_begin) then
      begin
        magration_card = 1;
        magration_card_date_begin = cds_doc_date_begin;
        magration_card_date_end = cds_doc_date_end;
      end
      if (pdt_code = 'registration_rvp') then
      begin
        registration_rvp = 1;
        registration_rvp_date_begin = cds_doc_date_begin;
        registration_rvp_date_end = cds_doc_date_end;
      end
      if (pdt_code = 'registration_vnj') then
      begin
        registration_vnj = 1;
        registration_vnj_date_begin = cds_doc_date_begin;
        registration_vnj_date_end = cds_doc_date_end;
      end
      if (pdt_code = 'work_patent' and
          pk_name not in ('свидетельство участника программы добровольного переселения', 'договор об оказании консультационных услуг') and
          work_patent_date_begin < cds_doc_date_begin) then
      begin
        work_patent = 1;
        work_patent_date_begin = cds_doc_date_begin;
        work_patent_date_end = cds_doc_date_end;
      end
      if (pdt_code = 'kvit_patent') then
      begin
        kvit_patent = 1;
        kvit_patent_date_end = cds_doc_date_end;
      end
      if (pdt_code = 'work_permit') then
        work_permit = 1;
      if (pk_name = 'свидетельство участника программы добровольного переселения') then
      begin
        supdp = 1;
        supdp_date_end = cds_doc_date_end;
      end
      if (pdt_code in ('job_agreement', 'job_agreement_2')) then
        job_agreement = 1;
      if (pdt_code in ('job_certificate') and
          cds_doc_date_begin >= dateadd(month, -1, kf_disb_date)) then
        job_certificate = 1; --справка с места работы
      if (pdt_code = 'certificate_approving_doc' and
          certificate_approving_doc_begin < cds_doc_date_begin) then
      begin
        certificate_approving_doc = 1;
        certificate_approving_doc_begin = cds_doc_date_begin;
      end
      if (pdt_code = 'certificate_2ndfl') then
        certificate_2ndfl = 1;
      if (pdt_code = 'certificate_ip') then
        certificate_ip = 1;
      if (pdt_code = 'certificate_3ndfl') then
        certificate_3ndfl = 1;
      if (pk_name = 'договор об оказании консультационных услуг') then
        dooku = 1;
      if (pk_name = 'водител. удостов. рф') then
        certificate_driver_rf = 1;
      if (pk_name = 'водител. удостов. кг') then
        certificate_driver_kg = 1;
      if (pk_name = 'международ.водител.права') then
        certificate_driver_kg_mn = 1;
      if (pdt_code = 'docs_vuz_suz') then
      begin
        if (pk_name = 'контракт на обучение в вуз/суз') then
          kontract_na_obuchenie = 1;
        if (pk_name = 'справка об учебе в вуз/суз') then
        begin
          spravka_ob_uchebe = 1;
          spravka_ob_uchebe_begin = cds_doc_date_begin;
        end
        if (pk_name = 'квитанция об оплате за учебу в вуз/суз') then
          kvit_ob_oplate_za_ucheb = 1;
      end
      if (pdt_code = 'certificate_rus_lang') then
      begin
        certificate_rus_lang = 1;
      end
      if (pdt_code = 'student_card') then
      begin
        student_card = 1;
        student_card_date_end = cds_doc_date_end;
      end
      if (pdt_code = 'medikal_certificate') then
      begin
        medikal_certificate = 1;
      end
    end
    --  проверка регистрации
    if (registration_fiz = 0 and
        registration_yur = 0 and
        registration_permanent = 0) then
    begin
      kf_error_text = 'не имеется регистрация клиента.';
      kf_error_is_critical = 2;
      suspend;
    end
    --  для иностранных граждан должно быть миграционная карта или рвп или внж
    if (rd_code not in ('citizen.ru')) then
    begin
      if (magration_card = 0 and
          registration_rvp = 0 and
          registration_vnj = 0) then
      begin
        if (not(pr_code containing '.express_online.' and rd_code in ('citizen.uz', 'citizen.tj'))) then
        begin
          kf_error_text = 'для иностранных граждан необхимо один из след. документво: мирг. карта или рвп или внж.';
          kf_error_is_critical = 2;
          suspend;
        end
      end
    end

    if (pr_code containing '.zalog_car.') then
    begin
      if (registration_permanent = 0) then
      begin
        if (exists(select *
                   from zalog -- если клиент является залогодателем
                   where z_from_k_id = :k_id and
                         z_type = 1 and
                         z_from_c_id = :c_id)) then
        begin
          if (dateadd(month, 3, kf_disb_date) > registration_fiz_date_end) then
          begin
            kf_error_text = 'регистрация, должно быть не менее 3 месяцев от планируемой даты получения займа.';
            kf_error_is_critical = 2;
            suspend;
          end
        end
        else
        begin -- если залогодателем является третье лицо
          -- если залогодатель не регистрирован то сработает не сработает первая проверка но сработает блок по залогодателю где id = 144
          if (kf_disb_date not between registration_fiz_date_begin and registration_fiz_date_end) then
          begin
            kf_error_text = 'регистрация, должно быть действительная на планируемую дату получения займа.';
            kf_error_is_critical = 2;
            suspend;
          end
        end
      end

      if (rd_code in ('citizen.uz', 'citizen.tj') or (rd_code = 'citizen.kg' and
          (registration_rvp = 1 or registration_vnj = 1))) then
      begin
        is_disb = 0;
        --      e)  граждане рт или ру должны иметь рвп или внж либо свидетельство участника программы добровольного переселения,
        --      до конца срока действия которых остается не менее 3 месяцев от выдачи займа
        if (dateadd(month, 3, kf_disb_date) <= maxvalue(registration_rvp_date_end, registration_vnj_date_end, supdp_date_end)) then
          is_disb = 1;
        if (is_disb = 0) then
        begin
          --        либо действующий патент, выданный не ранее чем за 2 месяца до даты получения займа и оплаченный за 10 месяцев, следующих за датой выдачи займа.
          if (work_patent = 1 --and work_patent_date_begin between dateadd(month, -2, kf_disb_date) and kf_disb_date  письмо зары от 03.05.19
              and
              dateadd(month, 3, kf_disb_date) <= kvit_patent_date_end) then
          begin
            is_disb = 1;
          end
        end
        if (is_disb = 0) then
        begin
          kf_error_text = 'срок действия внж/рвп/свидетельства/патента не соответствует требованиям.';
          kf_error_is_critical = 2;
          suspend;
        end
      end
    end

    /*--  проверка по экспресс    --  письмо от Зары 15.09.20 время 17:40
    if (pr_code containing '.express.') then
    begin
      is_disb = 0;
--    от гражданина рф ничего не требуется.
      if (rd_code in ('citizen.ru')) then is_disb = 1;
      else
--    ели у иностранного гражданина нет рвп или внж, а срок миграционной карты истек, то требуется один из документов
      if (registration_rvp = 0 and registration_vnj = 0
          and magration_card = 1 and magration_card_date_begin + 89 < :kf_disb_date) then
      begin
--      патент и квитанция (для граждан уз и тж)
        if (rd_code in ('citizen.uz', 'citizen.tj') and work_patent = 1 and kvit_patent = 1) then is_disb = 1;
        else
--      разрешение на работу (для граждан уз и тж)
        if (rd_code in ('citizen.uz', 'citizen.tj') and work_permit = 1) then is_disb = 1;
        else
--      свидетельство участника программы переселения (для уз, тж, кр)
        if (rd_code in ('citizen.kg','citizen.uz', 'citizen.tj') and supdp = 1) then is_disb = 1;
        else
--      трудовой договор (для кр)
        if (rd_code in ('citizen.kg') and (job_agreement = 1 or magration_card_date_end > kf_disb_date)) then is_disb = 1;
      end else
      if (magration_card = 1 or registration_rvp = 1 or registration_vnj = 1) then is_disb = 1;
      if (is_disb = 0) then
      begin
        kf_error_text = 'продукт "экспресс", не приложен документ продлевающий пребывание в рф';
        kf_error_is_critical = 2;
        suspend;
      end 
    end  */
    --  продукт шанс
    if (pr_code containing '.shans.') then
    begin
      if (rd_code in ('citizen.kg', 'citizen.ru')) then
      begin
        kf_error_text = 'продукт "шанс", не предназначен для граждан кр и рф';
        kf_error_is_critical = 2;
        suspend;
        exit;
      end
      if (registration_rvp = 1 or registration_vnj = 1 or supdp = 1 or work_permit = 1) then
      begin
        kf_error_text = 'продукт "шанс", не предназначен с рвп или внж либо со свидетельством программы добровольного переселения или с разрешением на работу';
        kf_error_is_critical = 2;
        suspend;
        exit;
      end

      -- RFCRU-4085
      if (magration_card = 1) then
      begin
        if (datediff(day, :magration_card_date_begin, :kf_disb_date) between 20 and 25) then
        begin
            is_disb = 0;
            -- Если НЕ приложены документы: медицинская справка, сертификат о знании русского языка, тогда блок срабатывает
            if (medikal_certificate = 1 and certificate_rus_lang = 1) then
                is_disb = 1;
            -- RFCRU-4119
            if (certificate_approving_doc = 1) then
                is_disb = 1;

            if (is_disb = 0) then
            begin
                kf_error_text = 'продукт "шанс", не приложены сертификат на русский язык и/или мед справка';
                kf_error_is_critical = 2;
                suspend;
            end
        end

        if (datediff(day, :magration_card_date_begin, :kf_disb_date) > 25) then
        begin
            is_disb = 0;

            -- справка о подаче на патент или разрешение на работу или рвп или внж.
            if (certificate_approving_doc = 1) then
                is_disb = 1;

            -- патент и квитанция, до конца срока действия патента должно оставаться не менее 15 рабочих дней
            if (work_patent = 1 and kvit_patent = 1 and :kf_disb_date <= dateadd(year, 1, work_patent_date_begin) - 15) then
                is_disb = 1;

            if (is_disb = 0) then
            begin
              kf_error_text = 'продукт "шанс", не приложен документ о подаче на патент и т.п.';
              kf_error_is_critical = 2;
              suspend;
            end
        end

      end
      else
      if (magration_card = 0) then
      begin
        kf_error_text = 'продукт "шанс", не имеется действующей миграционной карты';
        kf_error_is_critical = 2;
        suspend;
      end
    end
    --  продукт шанс плюс
    if (pr_code containing '.shans plus.') then
    begin
      if (not(certificate_approving_doc = 1 and
          certificate_approving_doc_begin <= kf_disb_date)) then
      begin
        kf_error_text = 'продукт "шанс плюс", не имеется справка о подаче на патент';
        kf_error_is_critical = 2;
        suspend;
      end
      if (certificate_rus_lang = 0) then
      begin
        kf_error_text = 'продукт "шанс плюс", не имеется "сертификат о знании русского языка"';
        kf_error_is_critical = 2;
        suspend;
      end
    end

    --  продукт благо
    if (pr_code containing '.blago.' or pr_code containing '.education.' or pr_code containing '.float_percent.' or pr_code containing '.express.' or pr_code containing '.reliable_client.' or pr_code containing '.par.loyal.') then
    begin
      --    нет рвп или внж,  дополнительно требуется один из документов
      if (registration_rvp = 0 and
          registration_vnj = 0) then
      begin
        is_disb = 0;
        if (rd_code in ('citizen.uz', 'citizen.tj')) then
        begin
          --        патент и квитанция
          if (work_patent = 1 and
              kvit_patent = 1) then
            is_disb = 1;
          else

          --        разрешение на работу
          if (work_permit = 1) then
            is_disb = 1;
          else

          --        свидетельство участника программы переселения
          if (supdp = 1) then
            is_disb = 1;
          if (pr_code containing '.blago.' or pr_code containing '.float_percent.' or pr_code containing '.express.' or pr_code containing '.reliable_client.' or pr_code containing '.par.loyal.') then
          begin
            if (spravka_ob_uchebe = 1) then
              is_disb = 1;
            if (student_card = 1 and
                kf_disb_date <= student_card_date_end) then
              is_disb = 1;
          end

        end
        else
        if (rd_code in ('citizen.kg')) then
        begin
          if (magration_card = 1) then
          is_disb = 1;
          else
          --        трудовой договор
          if (job_agreement = 1) then
            is_disb = 1;
          else

          --        свидетельство участника программы переселения
          if (supdp = 1) then
            is_disb = 1;
          if (pr_code containing '.blago.' or pr_code containing '.float_percent.' or pr_code containing '.express.' or pr_code containing '.reliable_client.' or pr_code containing '.par.loyal.') then
          begin
            if (spravka_ob_uchebe = 1) then
              is_disb = 1;
            if (student_card = 1 and
                kf_disb_date <= student_card_date_end) then
              is_disb = 1;
            --     если есть мигр.карта и студ.билет и срок дейст больше выдачи (jira - 2245)
            if (magration_card = 1 and
                magration_card_date_end > kf_disb_date and
                student_card = 1 and
                student_card_date_end > kf_disb_date) then
              is_disb = 1;
          end
        end
        else
        if (rd_code in ('citizen.ru')) then
          is_disb = 1;
        if (is_disb = 0) then
        begin
          if (pr_code containing '.blago.' or pr_code containing '.float_percent.' or pr_code containing '.express.' or pr_code containing '.reliable_client.' or pr_code containing '.par.loyal.') then
            kf_error_text = 'Не приложен документ дающий право на работу в РФ';
          else

          kf_error_is_critical = 2;
          suspend;
        end

        begin
          kf_error_text = 'дата справки с места учебы, должно быть выданная не ранее чем за 30 дней до планируемой даты получения займа';

          is_disb = 1;

          is_disb = 1;

            kf_disb_date <= student_card_date_end) then
          is_disb = 1;

          suspend;
        end
      end
      is_disb = 0;
      if (/*job_agreement = 1 and */ certificate_2ndfl = 1) then
        is_disb = 1; -- job_agreement = 1 убран от 01.11.2018, письмо от зары
      else
      if (/*certificate_ip = 1 and */ certificate_3ndfl = 1) then
        is_disb = 1; -- certificate_ip = 1 убран от 01.11.2018, письмо от зары
      else
      if (job_certificate = 1) then
        is_disb = 1; --от 27.07.2020, письмо от зары
      else
      if (spravka_ob_uchebe = 1) then
        is_disb = 1;
      else
      if (student_card = 1 and
          kf_disb_date <= student_card_date_end) then
        is_disb = 1;
      if (is_disb = 0) then
      begin
        kf_error_text = 'продукт "Надежный клиент плюс", не приложен документ потверждающий наличие работы или бизнеса';
        kf_error_is_critical = 2;
        suspend;
      end
    end

    --  продукт успех
    if (pr_code containing '.uspeh.') then

      if (rd_code in ('citizen.uz', 'citizen.tj') and
          registration_rvp = 0 and

      begin
        is_disb = 0;
        --        патент и квитанция

            kvit_patent = 1) then

          is_disb = 1;

        if (spravka_ob_uchebe = 1) then
          is_disb = 1;

            kf_disb_date <= student_card_date_end) then

        suspend;
      end
    end

    if (pr_code containing '.patent plus.' and
        pr_code not containing '.par.') then
    begin
      if (rd_code in ('citizen.kg', 'citizen.ru')) then
      begin
        kf_error_text = 'продукт "патент", не предназначен для граждан кр и рф';
        kf_error_is_critical = 2;
        suspend;
        exit;
      end
      if (registration_rvp = 1 or registration_vnj = 1 or supdp = 1 or work_permit = 1) then
      begin
        kf_error_text = 'продукт "патент", не предназначен с рвп или внж либо со свидетельством программы добровольного переселения или с разрешением на работу';

      if (certificate_approving_doc = 1 and

      if (registration_vnj = 1 or registration_rvp = 1) then

      if (is_disb = 0) then
      begin
        kf_error_text = 'продукт "статус", не приложен справка о подаче на рвп(с даты выдачи не более чем 55 дней) или документ внж';
