CREATE PROCEDURE FM_HTML_REPAY_DETAIL(
    UN_ID DECIMAL(18,0),
    REP_DATE_END DATE,
    REP_OBJECT_KIND VARCHAR(100),
    REP_OBJECT_ID DECIMAL(18,0))
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable LINE_STR D_STR_4096;
declare variable DB_ID integer;
declare variable K_ID decimal(18,0);
declare variable OK_FROM_OL_ID decimal(18,0);
declare variable REPAY_KIND varchar(100);
declare variable OL_ID decimal(18,0);
declare variable OKS_BALANCE decimal(18,4) = 0;
declare variable C_ID ID_TYPE;
declare variable LOAN_FOUND integer;
declare variable KD_PERCENT D_CURRENCY;
declare variable PERC_SUFFIX D_STR_CODE;
declare variable KC_ID D_ID;
declare variable K_NO D_STR_20;
declare variable MATLIST_STR D_STR_2048;
declare variable TEMP_STR D_STR_2048;
declare variable OPERKROFF_DEBT decimal(18,4);
begin
  res_text = (select res_style
              from fm_html_styles) || '<body>';

  db_id = f_get_db_id();
  k_id = null;
  c_id = null;
  if (rep_object_kind in ('9', 'kontract')) then
    k_id = rep_object_id;
  if (rep_object_kind in ('0', 'clients')) then
    c_id = rep_object_id;

  loan_found = 0;

  for select distinct(ks_from_k_id)
      from (select ks_from_k_id, ks_date_begin
            from kontstate, knclient knc
            where ks_state = 1 and
                  ks_ol_is_delete = 0 and
                  ks_from_k_id = kc_from_k_id and
                  kc_from_c_id = :rep_object_id and
                  (:c_id is not null) and
                  kc_amount_approved > 0
            union
            select okf.oko_from_k_id, k.k_fact_disb_date
            from operkroff okf
            inner join kontract k on k.k_id = okf.oko_from_k_id
            where okf.oko_from_c_id = :rep_object_id and
                  okf.oko_ol_is_active = 1 and
                  (:c_id is not null) and
                  :db_id = 81001
            union all
            select ks_from_k_id, ks_date_begin
            from kontstate
            where ks_state = 1 and
                  ks_ol_is_delete = 0 and
                  ks_from_k_id = :rep_object_id and
                  (:c_id is null)) a
      order by ks_date_begin desc
      into :k_id
  do
  begin
    loan_found = 1;

    perc_suffix = '%';
    if (db_id = 81001) then
    begin
      select kd_percent
      from kontract, product, kontdata
      where k_id = :k_id and
            k_from_pr_id = pr_id and
            pr_percent_kind = 1 and
            k_disb_date >= '08.09.2021' and
            kd_from_k_id = k_id and
            k_disb_date + 1 between kd_ol_date and kd_date_end and
            kd_percent <> k_percent
      into kd_percent;

      if (kd_percent is not null) then
      begin
        -- Кредит с базовой и пониженной ставкой
        perc_suffix = f_format('%-базовая, (%1%- пониженная)', round(kd_percent, 2));
      end
    end

    oks_balance = 0.0;
    select oks_balance
    from oksummary
    where oks_from_k_id = :k_id and
          oks_date_begin <= :rep_date_end and
          oks_date_end >= :rep_date_end and
          oks_close_state = 0
    into :oks_balance;

    operkroff_debt = 0;

    if (db_id = 81001) then
    begin
      select sum(oko_os + oko_sh_os + oko_perc + oko_sh_perc + oko_other + oko_sh_other)
      from operkroff okf
      where okf.oko_from_k_id = :k_id and
            okf.oko_ol_is_active = 1
      into :operkroff_debt;

      if (coalesce(operkroff_debt, 0) > 0) then
        oks_balance = oks_balance + operkroff_debt;
    end

    res_text = blob_append(res_text, f_format('<h1>%1</h1>(%2)', iif(db_id <> 94401, 'История погашений', 'Ödəniş tarixçəsi'), iif(db_id <> 94401, 'последний цикл', 'sonuncu dövr')));

    -- Общие условия выдачи
    if (db_id <> 94401) then
      res_text = blob_append(res_text, '<table width="100%">' || '<col width="20%"><col width="80%">' || '<tr>' || f_format('<th>%1</th><th>%2</th>', iif(db_id <> 94401, 'Инфо', 'Məlumat'), iif(db_id <> 94401, 'Условия выдачи', 'Verilmə şərtləri')) || '</tr>');

    line_str = '';
    matlist_str = '';

    if (db_id in (39801, 41701)) then
    begin
      for select iif(coalesce(ml.ml_model, '') <> '<>', '<td align="left">' || trim('Наим. товара') || ': <b>' || '</td>' || '<td align="left">' || ml.ml_model || '</b><br>' || '</td>' || '</tr>', '')
          from kontmat kml
          inner join matlist ml on ml.ml_id = kml.kml_from_ml_id
          where kml.kml_object_kind = 9 and
                kml.kml_object_id = :k_id
          into temp_str
      do
      begin
        matlist_str = matlist_str || temp_str;
      end
    end

    select '<tr>' || '<td align="left">' || trim(iif(:db_id <> 94401, 'Офис', 'Filial')) || ': <b>' || '</td>' || '<td align="left">' || reg_code || ' ' || reg_name || '</b><br>' || '</td>' || '</tr>' ||

           '<tr>' || '<td align="left">' || trim(iif(:db_id <> 94401, 'Специалист', 'Mütəxəssis')) || ': <b>' || '</td>' || '<td align="left">' || trim(iif(:db_id in (39801, 41701) and pr_code containing '.online_loan.', (select c_name_fio from clients where c_id = kal_new_kf_officer), (select c_name_fio from clients where c_id = ks_kc_id))) || '</b><br>' || '</td>' || '</tr>' ||

           '<tr>' || '<td align="left">' || trim(iif(:db_id <> 94401, 'Кр.папка', 'Kredit qovluğu')) || ': <b>' || '</td>' || '<td align="left">' || kal_new_kf_name || '</b><br>' || '</td>' || '</tr>' ||

           '<tr>' || '<td align="left">' || trim(iif(:db_id <> 94401, 'Договор', 'Müqavilə')) || ': <b>' || '</td>' || '<td align="left">' || coalesce(k_kontract_no, 'не указан') || '</b><br>' || '</td>' || '</tr>' ||

           '<tr>' || '<td align="left">' || trim(iif(:db_id <> 94401, 'Выдан', 'Verilib')) || ': <b>' || '</td>' || '<td align="left">' || right('0' || extract(day from ks_date_begin), 2) || '.' || right('0' || extract(month from ks_date_begin), 2) || '.' || extract(year from ks_date_begin) || '</b><br>' || '</td>' || '</tr>' ||

           '<tr>' || '<td align="left">' || trim(iif(:db_id <> 94401, 'Продукт', 'Məhsul')) || ': <b>' || '</td>' || '<td align="left">' || pr_name || '</b><br>' || '</td>' || '</tr>' ||

           iif(:matlist_str <> '', :matlist_str, '') ||

           iif(:db_id in (39801, 41701), '<tr>' || '<td align="left">' || trim('Способ выдачи') || ': <b>' || '</td>' || '<td align="left">' || (case 
                                                                                                                                                             when k_disb_kind = 1 then 'Наличными'
                                                                                                                                                             when k_disb_kind = 6 then 'Карта Мбулак'
                                                                                                                                                             when k_disb_kind = 5 and :db_id = 39801 then 'Банковская карта'
                                                                                                                                                             when k_disb_kind = 5 then 'Сторонняя карта Элкарт'    -- RFCKG-2729
                                                                                                                                                             else 'Неизвестно'
                                                                                                                                                           end) ||
           '</b><br>' || '</td>' || '</tr>', '') ||

           '<tr>' || '<td align="left">' || trim(iif(:db_id <> 94401, 'Сумма', 'Məbləğ')) || ': <b>' || '</td>' || '<td align="left">' || round(k_amount_approved, 2) || '</b><br>' || '</td>' || '</tr>' ||

           '<tr>' || '<td align="left">' || f_format('%1 (%2): <b>', trim(iif(:db_id <> 94401, 'Срок', 'Müddət')), trim(iif(:db_id <> 94401, 'в мес.', 'ay'))) || '</td>' || '<td align="left">' || k_kolich_repay || '</b><br>' || '</td>' || '</tr>' ||

           '<tr>' || '<td align="left">' || trim(iif(:db_id <> 94401, 'Проц.ставка', 'Faiz dərəcəsi')) || ': <b>' || '</td>' || '<td align="left">' || round(k_percent - k_percent_lgota, 2) || :perc_suffix || '</b><br>' || '</td>' || '</tr>'
    from region, kontstate, kontract, product, knaktlist
    where ks_region_id = reg_unicode and
          ks_from_k_id = :k_id and
          ks_state = 1 and
          ks_ol_is_delete = 0 and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          kal_from_kf_id = k_from_kf_id and
          knaktlist.is_delete = 0 and
          coalesce(:rep_date_end, current_date) between kal_date and kal_date_end
    into :line_str;
    line_str = line_str || '<tr>' || '<td align="left">' || iif(db_id <> 94401, 'Текущий баланс', 'Cari balans') || ': <b>' || '</td>' || '<td align="left">' || round(oks_balance, 2) || '</b><br>' || '</td>' || '</tr>';

    res_text = blob_append(res_text, line_str || '<br><br>');
    res_text = blob_append(res_text, '</table><br>');

    -- Состав группы
    res_text = blob_append(res_text, '<br><br>');
    res_text = blob_append(res_text, '<table width="100%">
       <col width="70%">
       <col width="10%">
       <col width="20%">
       <tr>
         <th>' || iif(db_id <> 94401, 'Участники в договоре', 'Qrup iştirakçıları') || '</th>
         <th>' || iif(db_id <> 94401, 'Цикл', 'Dövr') || '</th>
         <th>' || iif(db_id <> 94401, 'Сумма', 'Məbləğ') || '</th>
       </tr>');

    line_str = '';
    for select '<tr>' || '<td align="left">' || c_name_fio || '</td>' || '<td align="center">' || coalesce(kc_circle, 1) || '</td>' || '<td align="right">' || round(kc_amount_approved, 0) || '</td>' || '</tr>'
        from knclient, clients
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0 and
              kc_from_c_id = c_id
        into line_str
    do
    begin
      res_text = blob_append(res_text, line_str);
    end
    res_text = blob_append(res_text, '</table>');

    -- История платежей
    res_text = blob_append(res_text, '<br><br>');

    res_text = blob_append(res_text, '<table width="100%">
       <col width="20%">
       <col width="25%">
       <col width="20%">
       <col width="15%">
       <col width="20%">
       <tr>' || f_format('<th>%1/<br>%2</th><th><b>%3</b>/<br>%4</th><th>%5/', iif(:db_id <> 94401, 'День оплаты', 'Ödəniş günü'), iif(:db_id <> 94401, 'Тип платежа', 'Ödəniş növü'), iif(:db_id <> 94401, 'Всего', 'Cəmi'), iif(:db_id <> 94401, 'ОС', 'Əsas məbləğ'), iif(:db_id <> 94401, 'Проц', 'Faiz')) || f_format('<br>%1</th><th>%2/<br>%3</th>', iif(:db_id <> 94401, 'Доп проц.', 'Komis. faizi'), trim(iif(:db_id <> 94401, 'Пеня', 'Cərimə')), trim(iif(:db_id <> 94401, 'Дни проср', 'Gecikmə gün.'))) || case db_id
                                                                                                                                                                                                                                                                                                                                              when 41701 then f_format('<th>%1/<br>%2</th>', 'НсП', 'Вид оплаты')
                                                                                                                                                                                                                                                                                                                                              when 39801 then f_format('<th>%1/<br>%2</th>', 'НсП', 'Вид оплаты')
                                                                                                                                                                                                                                              when 94401 then f_format('<th>%1</th>', 'Ödəniş üsulu')
                                                                                                                                                                                                                                              else f_format('<th>%1</th>', 'Способ оплаты')
                                                                                                                                                                                                                                            end || '</tr>');

    line_str = '';

    for with tab
        as (select ok_ol_date as ok_ol_date,
                   (
                   case
                      when ok_type = 2 then iif(:db_id <> 94401, 'Реструктр.', 'Restruk.')
                      when ok_type in (4, 9) then iif(:db_id <> 94401, 'Погашение', 'Ödəniş')
                      when ok_type in (5, 6) then iif(:db_id <> 94401, 'Списание', 'Silinmə')
                      when ok_type = 8 then iif(:db_id <> 94401, 'Выдача', 'Buraxılış')
                    end) as ok_type,
                   ok_from_ol_id, -sum(ok_os) as ok_os, -sum(ok_perc) as ok_perc, -sum(ok_fine) as ok_fine,
                   -sum(ok_kommis) as ok_kommis, -sum(ok_tax) as ok_tax,
                   -sum(ok_os + ok_perc + ok_fine + ok_kommis + ok_tax) as ok_all,
                   max(iif(ok_ol_date > ok_sh_date and (ok_os + ok_perc) <> 0.0, ok_ol_date - ok_sh_date, 0)) as delinq_days
            from operkred
            where ok_from_k_id = :k_id and
                  ok_ol_is_active = 1 and
                  ok_type in (2, 4, 5, 6, 8, 9)
            group by 1, 2, 3
            
            union all

           select
            oko.oko_ol_date as ok_ol_date,
            'Погашение (В/У)' as ok_type,
            oko.oko_from_ol_id as ok_from_ol_id, 
            -coalesce(sum(oko.oko_os), 0) as ok_os, 
            -coalesce(sum(oko.oko_perc), 0) as ok_perc, 
            -coalesce(sum(oko.oko_sh_os + oko.oko_sh_perc + oko.oko_other + oko.oko_sh_other), 0) as ok_fine,
            0 as ok_kommis, 
            0 as ok_tax,
            -coalesce(sum(oko.oko_os + oko.oko_sh_os + oko.oko_perc + oko.oko_sh_perc + oko.oko_other + oko.oko_sh_other), 0) as ok_all,
            0 as delinq_days
        from operkroff oko
        where oko.oko_from_k_id = :k_id and
              oko.oko_ol_is_active = 1 and
              :db_id = 81001 and
              oko.oko_type = 4
        group by 1, 2, 3)

        select ok_from_ol_id,
               '<tr>' || '<td align="left">' || f_datetostr(ok_ol_date) || '<br>' || ok_type || '</td>' || '<td align="right"><b>' || round(ok_all, 0) || '</b><br>' || round(ok_os, 0) || '</td>' || '<td align="right">' || round(ok_perc, 0) || '<br>' || round(ok_kommis, 0) || '</td>' || '<td align="right">' || round(ok_fine, 0) || '<br>' || iif(delinq_days <> 0, delinq_days || '<wbr>д.', '-') || '</td>' || '<td align="right">' || iif(ok_tax <> 0, round(ok_tax, 0), '') || '<br>'
        from tab
        into ok_from_ol_id, :line_str
    do
    begin
      ol_id = null;
      select ol_parent_id
      from operlist
      where ol_id = :ok_from_ol_id
      into ol_id;
      if (ol_id is null) then
        ol_id = ok_from_ol_id;

      repay_kind = '';
      select iif(ol_doc_kind in (0, 1), iif(:db_id <> 94401, 'Наличн.', 'Nəğd'), iif(ol_doc_kind in (6, 7), iif(:db_id <> 94401, 'Банк', 'Bank'), ''))
      from operlist
      where ol_id = :ol_id
      into repay_kind;

      if (repay_kind = '') then
      begin
        select esm_project
        from extsysoper, extsysmain
        where eso_from_ol_id = :ol_id and
              eso_from_esm_id = esm_id
        into repay_kind;
      end
      if (repay_kind = '') then
        repay_kind = iif(db_id <> 94401, 'Прочие', 'Digər');

      res_text = blob_append(res_text, line_str || repay_kind || '</td>');
    end

    res_text = blob_append(res_text, '</table>');
    res_text = blob_append(res_text, '</body>');

    -- RFCKG-107
    if (:db_id in (39801, 41701)) then
    begin
      select kc.kc_id, k.k_kontract_no
      from knclient kc
      inner join kontract k on k.k_id = kc.kc_from_k_id
      where kc.kc_from_k_id = :k_id and
            kc.kc_from_c_id = :c_id rows 1
      into :kc_id, :k_no;

        res_text = blob_append(res_text, f_uri_sql('Создать повторный целевой мониторинг для ' || :k_no, 'insert_new_monitoring', 'KnClient', :kc_id, 'none', 'Подтверждаете создание задачи?', '', 0, 0, '', '', 'dialog', '') || '<br>');
    end
    -- RFCKG-107 END
  end

  if (loan_found = 0) then
  begin
    res_text = iif(db_id <> 94401, 'У клиента нет кредитной истории', 'Müştərinin kredit tarixçəsi mövcüd deyil');
  end
  suspend;
end
