CREATE PROCEDURE HP_CHECK_LOAN_KONDITIONS_SAVE(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    CHECK_KIND D_STR_CODE,
    UN_ID D_ID)
RETURNS (
    CCS_ID D_INTEGER,
    CCS_WARNINGS D_INTEGER,
    CCS_ERRORS D_INTEGER,
    CCS_IGNORES D_INTEGER,
    RES_IS_ERROR D_SMALLINT,
    RES_NEED_SHOW D_SMALLINT)
AS
declare variable k_id d_id;
declare variable ma_id d_id;
declare variable ccr_info d_str_255;
declare variable ccr_message d_str_long;
declare variable ccr_is_critical d_smallint;
declare variable cdm_id d_integer;
declare variable check_kind_code d_str_code;
declare variable obj_kind_id d_integer;
declare variable check_date d_date;
begin
  k_id = 0;
  ma_id = 0;

  res_is_error = 0;

  if (trim(coalesce(obj_kind, '')) = '') then
    exception e_default 'Empty obj_kind';

  obj_kind = f_objkind(obj_kind, 'name');
  if (obj_kind = 'kontract')
  then k_id = obj_id;
  else
    if (obj_kind = 'miniapp')
    then ma_id = obj_id;
    else
      exception e_default 'Unregistered obj_kind: ' || obj_kind;

  check_kind = f_enum('condition_check_kind', check_kind, 'id');
  check_kind_code = f_enum('condition_check_kind', check_kind, 'code');

  obj_kind_id = f_objkind(obj_kind, 'id');

  select blank_date
  from hp_edoc_object_info(:obj_kind, :obj_id, :un_id)
  into :check_date;
  if (check_date is null) then
    check_date = current_date;

  select ccs_id
  from condcheckstate
  where ccs_object_kind = :obj_kind_id and
        ccs_object_id = :obj_id and
        ccs_check_kind = :check_kind
  into ccs_id;

  if (ccs_id is null) then
  begin
    insert into condcheckstate (ccs_object_kind, ccs_object_id, ccs_check_kind)
    values (:obj_kind_id, :obj_id, :check_kind)
    returning ccs_id
    into :ccs_id;
  end else
  begin
    -- удаляем результаты последней проверки
    delete from condcheckresult
    where ccr_from_ccs_id = :ccs_id;
  end

  ccs_warnings = 0;
  ccs_errors = 0;
  ccs_ignores = 0;

  for select kf_client_fio, kf_error_text, kf_error_is_critical, sql_id
  from hp_check_loan_konditions(:k_id, 0, :ma_id, :check_kind_code, :un_id)
  where kf_error_is_critical > 0
  into :ccr_info, :ccr_message, :ccr_is_critical, :cdm_id
  do
  begin
    if (ccr_is_critical > 0) then
    begin
      if (exists(select * from condcheckignore
      where cci_object_kind = :obj_kind_id and
            cci_object_id = :obj_id and
            cci_from_cdm_id = :cdm_id and
            cci_date_end >= :check_date and
            cci_state = 1)) then
      begin
        ccr_is_critical = 0;
        ccs_ignores = ccs_ignores + 1;
      end
      else
        ccs_errors = ccs_errors + 1;
    end
    else
      ccs_warnings = ccs_warnings + 1;

    insert into condcheckresult (ccr_from_ccs_id, ccr_from_cdm_id, ccr_info, ccr_message, ccr_is_critical, ccr_from_un_id)
    values (:ccs_id, :cdm_id, :ccr_info, :ccr_message, :ccr_is_critical, :un_id);
  end

  update condcheckstate
  set ccs_is_actual = 1,
      ccs_last_datetime = current_timestamp,
      ccs_warnings = :ccs_warnings,
      ccs_errors = :ccs_errors,
      ccs_ignores = :ccs_ignores
  where ccs_id = :ccs_id;

  if (ccs_errors > 0) then
    res_is_error = 1;

  res_need_show = 0;
  if ((ccs_ignores + ccs_errors + ccs_warnings) > 0) then
    res_need_show = 1;

  suspend;
end
