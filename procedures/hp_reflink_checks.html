CREATE PROCEDURE HP_REFLINK_CHECKS(
    MASTER_CATEGORY VARCHAR(255),
    REF_CATEGORY VARCHAR(255),
    REF_OBJ_ID DECIMAL(18,0),
    KIND VARCHAR(100))
RETURNS (
    RD_ID INTEGER,
    RD_NAME VARCHAR(255),
    RD_MEMO VARCHAR(255),
    RD_STR_ID VARCHAR(255),
    RD_IS_ACTIVE INTEGER,
    RD_CHECKED INTEGER)
AS
declare variable ref_table_name varchar(100);
declare variable obj_kind integer;
begin
  if (coalesce(master_category, '') = '') then
    exit;

  if (coalesce(ref_category, '') = '') then
    ref_category = master_category;
  select ref_table_name
  from hp_reflink_category(0)
  where ref_name = :master_category
  into :ref_table_name;

  obj_kind = f_objkind(ref_table_name, 'id');

  if (kind = 'all_in_category') then
  begin
    for select rd_id, rd_name, rd_memo, rd_str_id, rd_is_active
        from refdata
        where rd_main_code = 'reflink' and
              rd_memo = :ref_category
        into :rd_id, rd_name, rd_memo, rd_str_id, rd_is_active
    do
    begin
      rd_checked = 0;
      if (exists(select *
                 from reflink
                 where rlk_object_kind = :obj_kind and
                       rlk_object_id = :ref_obj_id and
                       rlk_from_rd_id = :rd_id)) then
        rd_checked = 1;

      suspend;
    end
  end
  if (kind = 'all_checked') then
  begin
    rd_checked = 1;
    for select rd_id, rd_name, rd_memo, rd_str_id, rd_is_active
        from refdata, reflink
        where rd_main_code = 'reflink' and
              rlk_object_kind = :obj_kind and
              rlk_object_id = :ref_obj_id and
              rlk_from_rd_id = rd_id
        into :rd_id, rd_name, rd_memo, rd_str_id, rd_is_active
    do
    begin
      suspend;
    end
  end

end
