CREATE PROCEDURE MKR_FILL_INDIV_POKET_TEMP(
    REPORT_DATE_END DATE,
    KC_FROM_K_ID ID_TYPE,
    KC_NUMBER ID_TYPE,
    REPORT_CATEG SMALLINT)
RETURNS (
    CR_TYPE SMALLINT,
    KNC_ID ID_TYPE,
    ID STR_255,
    NAME STR_255,
    PINCODE STR_255,
    COUNTRYCODE VARCHAR(3),
    DATEOFBIRTH DATE,
    ACCOUNTNO STR_255,
    PURPOSEOFCREDIT VARCHAR(3),
    CURRENCYOFCREDIT VARCHAR(3),
    INITIALAMOUNTOFCREDIT CURR_TYPE,
    DISOUTAMOUNTOFCREDIT CURR_TYPE,
    ANNUALINTERESTRATE DECIMAL(18,7),
    DATEOFGRANT DATE,
    DUETIMEFIRSTCONTRACT DATE,
    DUETIMELASTCONTRACT DATE,
    LASTPAYMENTDATE DATE,
    MONTHLYPAYMENTAMOUNT CURR_TYPE,
    DAYSMAINSUMISOVERDUE INTEGER,
    DAYSINTERESTISOVERDUE INTEGER,
    OIAFORREPPERIOD CURR_TYPE,
    OIAFORLASTTHREEMONTHS CURR_TYPE,
    NUMBEROFPROLONGS INTEGER,
    CREDITCLASSCODE VARCHAR(3),
    CREDITSTATUSCODE VARCHAR(3),
    CREDITPERIODINMONTHS INTEGER,
    REP_DATE_TEPM DATE)
AS
declare variable C_DOC_TYPE integer;
declare variable K_CURRENCY varchar(3);
declare variable KNC_BUSINESS_SUBTYPE_ID integer;
declare variable K_AMOUNT_APPROVED CURR_TYPE;
declare variable K_BALANCE CURR_TYPE;
declare variable IS_FINISH integer;
declare variable SH_DATE_TMP date;
declare variable SUMM_TMP CURR_TYPE;
declare variable IS_SECOND_CIRCLE smallint;
declare variable MONTHLYPAYMENTAMOUNT_GROUP CURR_TYPE;
declare variable OIAFORREPPERIOD_GROUP CURR_TYPE;
declare variable OIAFORLASTTHREEMONTHS_GROUP CURR_TYPE;
declare variable LASTPAYMENTDATE_OPERKROFF date;
declare variable PR_ID integer;
declare variable K_KOLICH_CLIENT_APPROVED integer;
declare variable OKF_DEBT_OS CURR_TYPE;
declare variable OKF_DEBT_PERC CURR_TYPE;
declare variable OKS_DELINQ_DATE date;
declare variable MIN_SH_DATE date;
declare variable LOAN_CLOSED_DATE date;
declare variable BASE_K_ID D_ID;
declare variable OLD_DISOUTAMOUNTOFCREDIT CURR_TYPE;
declare variable DAYSMAINSUMISOVERDUE_FIX D_INTEGER;
begin
  is_second_circle = 0;
  if (report_categ is null) then
    report_categ = 0;

  for select ks_state as cr_type,
             c_pass_no,
             kc_from_c_id,

             c_name_fio,
             c_doc_type,
             trim(coalesce(c_pincode, '')) as pincode,

             c_birth_date,
             cast(iif(k_base_k_id is null, k_id, k_base_k_id) as varchar(255)) || '-' || cast(kc_number as varchar(255)),
             kc.kc_busuness_subtype,

             kt.k_currency,
             coalesce(kc_amount_approved, 0) as initialamountofcredit,
             k_amount_approved,

             k_disb_date,
             k_kredline_end,
             k_period,

             pr.pr_id,
             kt.k_kolich_client_approved,
             kt.k_percent_per_day * 360.0000000,

             kt.k_base_k_id
      from clients cl, knclient kc, kontstate ks, kontract kt, product pr
      where c_id = kc_from_c_id and
            kc_from_k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            ((ks_state = :report_categ) or
            (:report_categ = 0)) and
            ks_from_k_id = k_id and
            ((kc_number = :kc_number) or
            (:kc_number is null)) and
            :report_date_end between ks_date_begin and ks_date_end and
            k_id = :kc_from_k_id and
            kc.kc_amount_approved > 0 and
            pr.pr_id = kt.k_from_pr_id
      into cr_type,
           id,
           knc_id,

           name,
           c_doc_type,
           pincode,

           dateofbirth,
           accountno,
           knc_business_subtype_id,

           k_currency,
           initialamountofcredit,
           k_amount_approved,

           dateofgrant,
           duetimelastcontract,
           creditperiodinmonths,

           pr_id,
           k_kolich_client_approved,
           annualinterestrate,

           base_k_id
  do
  begin
    countrycode = '';

    select iif(pk_country = 'AZN', 'AZE', upper(pk_country))
    from passkind
    where pk_id = :c_doc_type
    into :countrycode;

    select purposeofcredit
    from mkr_purposeofcredit_get(:pr_id, :knc_business_subtype_id, :dateofgrant)
    into :purposeofcredit;

    if (is_second_circle = 0) then
    begin
       -- Дата погашения кредита с первоначальным договором
      select max(sh.sh_date)
      from schedule sh
      where sh.sh_from_k_id = :kc_from_k_id
      into :duetimefirstcontract;  

      select cu_symbol as currencyofcredit
      from currlist
      where cu_id = :k_currency
      into currencyofcredit;

      select oks_balance,
             iif((cast(:report_date_end as date) - oks_delinq_date) < 0, 0, cast(:report_date_end as date) - oks_delinq_date) as daysmainsumisoverdue,
             oks_delinq_date
      from oksummary
      where oks_from_k_id = :kc_from_k_id and
            :report_date_end between oks_date_begin and oks_date_end and
            oks_close_state = 0
      into k_balance,
           daysmainsumisoverdue,
           oks_delinq_date;
    end

    daysinterestisoverdue = daysmainsumisoverdue;

    if (k_kolich_client_approved > 1) then
      disoutamountofcredit = round(k_balance / k_amount_approved * initialamountofcredit,2);
    else
      disoutamountofcredit = round(k_balance, 2);

    -- Если это разделенный кредит, необходимо провериь какая сумма долга по ОС отправлялась по клиенту до разделения,
    -- т.к. если отпаравить в МКР сумму большую чем отправлялась ранее, МКР запись не примет.
    -- Поэтому если разница между суммой которая отправлялась до разделения и текущей суммой долга по ОС в пределах 0.1 AZN тогда отправляем старую сумму.
    if (base_k_id is not null) then
    begin
        old_disoutamountofcredit = 0;
    
        select round(mcc.mc_disoutamountofcredit,2)
        from mkr_credit mcc
        where mcc.mc_from_k_id = :base_k_id
        and mcc.mc_from_c_id = :knc_id
        order by mcc.mc_from_mp_id desc
        rows 1
        into :old_disoutamountofcredit;
    
        if (abs(disoutamountofcredit - old_disoutamountofcredit) <= 0.1) then
            disoutamountofcredit = old_disoutamountofcredit;
    end

    if (is_second_circle = 0) then
    begin
      select max(ok_ol_date)
      from operkred ok
      where ok_type = 4 and
            ok_ol_is_active = 1 and
            ok_from_k_id = :kc_from_k_id and
            ok_ol_date <= :report_date_end and
            ok_kommis = 0 and
            ok_from_c_id = 0 and
            (ok.ok_os + ok.ok_perc) <> 0
      into :lastpaymentdate;
    end

    monthlypaymentamount = 0;
    is_finish = 0;

    if (is_second_circle = 0) then
    begin
      monthlypaymentamount_group = 0;
      for select ok.ok_sh_date, sum(ok.ok_os + ok.ok_sh_os + ok.ok_perc + ok.ok_sh_perc) as summ
          from operkred ok
          where ok.ok_from_k_id = :kc_from_k_id and
                ok.ok_type in (1, 2, 3) and
                ok.ok_from_c_id = 0 and
                ok.ok_ol_is_active = 1 and
                ok.ok_ol_date <= :report_date_end
          group by 1
          having sum(ok.ok_os + ok.ok_sh_os + ok.ok_perc + ok.ok_sh_perc) <> 0
          into :sh_date_tmp, :summ_tmp
      do
      begin
        if (is_finish = 0) then
        begin
          if (sh_date_tmp >= :report_date_end) then
          begin
            monthlypaymentamount_group = summ_tmp;
            is_finish = 1;
          end
          else
          begin
            monthlypaymentamount_group = summ_tmp;
            is_finish = 0;
          end
        end
      end
    end

    if (k_kolich_client_approved > 1) then
      monthlypaymentamount = monthlypaymentamount_group / k_amount_approved * initialamountofcredit;
    else
      monthlypaymentamount = monthlypaymentamount_group;
    oiaforrepperiod = null;

    if (is_second_circle = 0) then
    begin
      oiaforrepperiod_group = null;

      select sum(ok_perc + ok_sh_perc)
      from operkred
      where ok_type < 7 and
            ok_ol_is_active = 1 and
            ok_from_c_id = 0 and
            ok_from_k_id = :kc_from_k_id and
            ok_ol_date <= :report_date_end and
            ok_sh_date <= :report_date_end
      into :oiaforrepperiod_group;

      if (k_kolich_client_approved > 1) then
        oiaforrepperiod_group = oiaforrepperiod_group / :k_amount_approved;

      if (cr_type <> 1) then
        oiaforrepperiod_group = 0;
    end

    if (k_kolich_client_approved > 1) then
      oiaforrepperiod = oiaforrepperiod_group * initialamountofcredit;
    else
      oiaforrepperiod = oiaforrepperiod_group;

    if (oiaforrepperiod is null) then
      oiaforrepperiod = 0; --
    if (oiaforrepperiod < 0) then
      oiaforrepperiod = 0;

    rep_date_tepm = iif(duetimelastcontract < report_date_end, duetimelastcontract, report_date_end);

    oiaforlastthreemonths = null;

    if (is_second_circle = 0) then
    begin
      oiaforlastthreemonths_group = null;
      select sum(ok_perc + ok_sh_perc)
      from operkred
      where ok_type < 7 and
            ok_ol_is_active = 1 and
            ok_from_c_id = 0 and
            ok_from_k_id = :kc_from_k_id and
            ok_ol_date <= :report_date_end and
            ok_sh_date between :rep_date_tepm - 90 and :rep_date_tepm
      into :oiaforlastthreemonths_group;

      if (k_kolich_client_approved > 1) then
        oiaforlastthreemonths_group = oiaforlastthreemonths_group / k_amount_approved;
      else
        oiaforlastthreemonths_group = oiaforlastthreemonths_group;
    end

    if (k_kolich_client_approved > 1) then
      oiaforlastthreemonths = oiaforlastthreemonths_group * initialamountofcredit;
    else
      oiaforlastthreemonths = oiaforlastthreemonths_group;

    if (oiaforlastthreemonths is null) then
      oiaforlastthreemonths = 0;
    if (oiaforlastthreemonths < 0) then
      oiaforlastthreemonths = 0;

    if (is_second_circle = 0) then
    begin
      numberofprolongs = null;
      select count(*) numberofprolongs
      from operrest, operlist
      where or_from_k_id = :kc_from_k_id and
            or_from_ol_id = ol_id and
            or_kind > 2 and
            operlist.is_delete = 0 and
            ol_date <= :report_date_end and
            or_date_end >= :report_date_end
      into numberofprolongs;

      if (numberofprolongs is null) then
        numberofprolongs = 0;
      if (daysmainsumisoverdue is null) then
        creditclasscode = '002';
      else
      begin
        if (daysmainsumisoverdue <= 30) then
          creditclasscode = '002';
        if ((daysmainsumisoverdue > 30) and
            (daysmainsumisoverdue <= 90)) then
          creditclasscode = '003';
        if ((daysmainsumisoverdue > 90) and
            (daysmainsumisoverdue <= 180)) then
          creditclasscode = '004';
        if ((daysmainsumisoverdue > 180) and
            (daysmainsumisoverdue <= 360)) then
          creditclasscode = '005';
        if (daysmainsumisoverdue > 360) then
          creditclasscode = '006';
      end
    end
    if (cr_type = 1) then
    /* активные кредиты */
    begin
      if (daysmainsumisoverdue < 90) then
        creditstatuscode = '007'; -- iwlek - рабочие кредиты - актиные кредиты с просрочкой до 89 дней
      else
      if (daysmainsumisoverdue >= 90) then
        creditstatuscode = '008'; -- qeyri iwlek - нерабочие - активные кредиты с просрочкой 90 или более дней

      /* закомментировал igusev, в соответствии с запросом в письме от Азера. Тема: Pandemiya ilə bağlı tövsiyənin qüvvədən düşməsi
      if ((daysmainsumisoverdue > 0) and
           (oks_delinq_date between cast('01.03.2020' as date) and cast('30.09.2020' as date))) then -- если просрочка за период с 1.03.2020 по 30.09.2020тогда код = 009 "Обязательство, просроченное в связи с пандемией"
        creditstatuscode = '009';
      */

      if ((oiaforrepperiod <= 0) and
          (disoutamountofcredit <= 0)) then -- если заем активный но нет долга не по ос, не по %, тогда передаем что он закрыт
        creditstatuscode = '001';
    end
    else -- если закрытый заем
    begin
      okf_debt_os = 0;
      okf_debt_perc = 0;

      select sum(okf.oko_os + okf.oko_sh_os),
             sum(okf.oko_perc + okf.oko_sh_perc)
      from operkroff okf
      inner join operlist ol on okf.oko_from_ol_id = ol.ol_id
      where okf.oko_from_k_id = :kc_from_k_id and
            okf.oko_from_c_id = :knc_id and
            ol.is_delete = 0 and
            ol.ol_date <= :report_date_end
      into :okf_debt_os, -- задолжность по ос
           :okf_debt_perc; -- задолжность по %

      okf_debt_os = coalesce(okf_debt_os, 0);
      okf_debt_perc = coalesce(okf_debt_perc, 0);

      if ((okf_debt_os > 0) or (okf_debt_perc > 0))  then
      begin
        disoutamountofcredit = okf_debt_os; -- задолжность по ОС
        oiaforrepperiod = okf_debt_perc;    -- задолжность по %

        creditclasscode = '006';
        creditstatuscode = '003'; -- silinmis - списанные - списанные с баланса компании

        if (oiaforrepperiod < 0) then
          oiaforrepperiod = 0;

        daysmainsumisoverdue = null;

        /*
        select :report_date_end - min(ok_sh_date)
        from kontract k, operkred ok
        where k_id = :kc_from_k_id and
            ok.ok_from_k_id = k.k_id and
            k.k_fact_closed_ol_id = ok_from_ol_id and
            ok_type = 5 and
            ok_sh_date < ok_ol_date and
            ok_ol_is_active = 1
        into daysmainsumisoverdue;
        */
        loan_closed_date = null;

        select ks.ks_date_begin
        from kontstate ks
        where ks.ks_state = 2
        and ks.ks_from_k_id = :kc_from_k_id
        and ks.ks_ol_is_delete = 0
        and :report_date_end  between ks.ks_date_begin and ks.ks_date_end
        into :loan_closed_date;

        if (:loan_closed_date is not null) then
        begin
            daysmainsumisoverdue_fix = null;

            select :report_date_end - (mdd.mdd_last_renewal - mdd.mdd_delinq_days)
            from mkr_delinq_days mdd
            where mdd.mdd_account_no = :accountno
            into :daysmainsumisoverdue_fix;

            if (daysmainsumisoverdue_fix is not null) then
                daysmainsumisoverdue = daysmainsumisoverdue_fix;
            else
            begin
                select :report_date_end - oks.oks_delinq_date
                from oksummary oks
                where oks.oks_from_k_id = :kc_from_k_id
                and oks.oks_close_state = 0
                and :loan_closed_date - 1 between oks.oks_date_begin and oks.oks_date_end
                into daysmainsumisoverdue;
            end
        
            daysinterestisoverdue = daysmainsumisoverdue;
        end

        /*
        if (daysmainsumisoverdue is null) then -- если кредит не был закрыт спианием
        begin
            creditstatuscode = '001';
            disoutamountofcredit = 0;
            oiaforrepperiod = 0;
            daysmainsumisoverdue = 0;
        end
        */
        daysmainsumisoverdue = coalesce(daysmainsumisoverdue, 0);
        daysinterestisoverdue = daysmainsumisoverdue;

        lastpaymentdate_operkroff = null;

        select ol.ol_date
        from operkroff okf
        inner join operlist ol on ol.ol_id = okf.oko_from_ol_id
        where okf.oko_from_k_id = :kc_from_k_id and
              okf.oko_from_c_id = :knc_id and
              ol.is_delete = 0 and
              ol_date <= :report_date_end and
              okf.oko_type = 4 and
              okf.oko_os + okf.oko_perc <> 0
        order by ol.ol_date desc
        rows 1
        into :lastpaymentdate_operkroff;-- последняя дата оплаты

        lastpaymentdate_operkroff = coalesce(lastpaymentdate_operkroff, cast('01.01.1900' as date));
        if (lastpaymentdate_operkroff > lastpaymentdate) then
          lastpaymentdate = lastpaymentdate_operkroff;
      end
      else
        creditstatuscode = '001'; -- bagl? - закрытые кредиты

        -- если кредиты закрылся и на дату закрытия был закрыт платеж, просрочка по которому за период с 01.03.2020 по 30.09.2020 тогда статус  010 - Кредит, просроченный в связи с пандемией и полностью оплаченный
        /*

        закомментировал igusev, в соответствии с запросом в письме от Азера. Тема: Pandemiya ilə bağlı tövsiyənin qüvvədən düşməsi

        min_sh_date = null;

        select min(ok.ok_sh_date)
        from operkred ok
        where ok.ok_from_k_id = :kc_from_k_id
        and ok.ok_type = 4
        and ok.ok_ol_date = :lastpaymentdate
        and ok.ok_sh_date < :lastpaymentdate -- Важно!
        and ok.ok_ol_is_active = 1
        and ((ok.ok_os < 0) or (ok.ok_perc < 0) or (ok.ok_fine < 0))
        into :min_sh_date;

        if ((min_sh_date is not null) and (min_sh_date between cast('01.03.2020' as date) and cast('30.09.2020' as date))) then
        begin
          creditstatuscode = '010';
          daysmainsumisoverdue = lastpaymentdate - min_sh_date;
          daysinterestisoverdue = daysmainsumisoverdue;
        end
        */
    end

    is_second_circle = 1;

    disoutamountofcredit = coalesce(disoutamountofcredit, 0);
    oiaforrepperiod = coalesce(oiaforrepperiod, 0);
    daysinterestisoverdue = coalesce(daysinterestisoverdue, 0);

    if (disoutamountofcredit > initialamountofcredit) then
      disoutamountofcredit = initialamountofcredit;
    if (disoutamountofcredit < 0) then
      disoutamountofcredit = 0;

     /*
    if (not exists(select *
                   from kib_ignore ki
                   where ki.ki_from_k_id = :kc_from_k_id)) then
    */
      suspend;
  end
end
