CREATE PROCEDURE CM_KG_0011_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_from_c_id id_type;
declare variable par_c_fio varchar(100);
declare variable par_c_birth_date date;
declare variable par_c_pass_no varchar(30);
declare variable par_c_inn varchar(30);
declare variable par_c_cell_no varchar(20);
declare variable m_date_begin date;
declare variable m_date_end date;
begin
  if (ma_id in (4675283, 4710478, 4717106, 4717469, 4724366, 4732462, 4739544,
                4744229)) then
    exit; --RFCKG-3741

  for select c_name_fio, mac_from_c_id, c_name_fio, c_birth_date, coalesce(c_pass_no, ''), coalesce(c_inn, ''),
             coalesce(c_plastcard_no, '')
      from miniappclient, clients
      where mac_from_ma_id = :ma_id and
            mac_amount > 0 and
            c_type = 0 and
            mac_from_c_id = c_id
      into res_info_text, kc_from_c_id, par_c_fio, par_c_birth_date, par_c_pass_no, par_c_inn, par_c_cell_no
  do
  begin
    res_error_text = '';
    select left(c_comment || ', ' || bl_comment, 255)
    from hp_blacklist_search(:kc_from_c_id, :par_c_fio, :par_c_birth_date, :par_c_pass_no, :par_c_inn, :par_c_cell_no, 1, 0, 0)
    where c_comment_id = 1
    order by c_is_black desc, m_datetime desc
    rows 1
    into :res_error_text;
    if (res_error_text <> '') then
    begin
      res_error_text = ' клиент находится в стоп-листе'; -- RFCKG-5062
      res_error_is_critical = 1;
      suspend;
    end
    else
    begin
      select left(c_comment || ', ' || bl_comment, 255), m_date_begin, m_date_end
      from hp_blacklist_search(:kc_from_c_id, :par_c_fio, :par_c_birth_date, :par_c_pass_no, :par_c_inn, :par_c_cell_no, 0, 1, 0)
      where c_comment_id = 1 rows 1
      into res_error_text, m_date_begin, m_date_end;
      if (res_error_text <> '') then
      begin
        res_error_text = ' клиент находится в стоп-листе'; -- RFCKG-5062
        m_date_end = coalesce(m_date_end, '31.12.2999');
        if (m_date_begin = m_date_end or current_date between m_date_begin and m_date_end) then
        begin
          res_error_is_critical = 1;
        end
        else
        begin
          res_error_is_critical = 0;
        end
        suspend;
      end
    end
  end
end
