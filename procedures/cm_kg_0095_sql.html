CREATE PROCEDURE CM_KG_0095_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable kf_client_pass_no varchar(30);
declare variable pr_code varchar(100);
declare variable c_id id_type;
declare variable res_str_values str_100;
begin
    select pr_code,
           k.k_fact_disb_date
    from kontract k
    inner join Product pr on k.K_from_pr_id = pr.Pr_id
    where k_id = :k_id
    into pr_code,
         kf_disb_date;

    if (pr_code containing '.auto_kk.' and check_kind = '.send_kk.') then
      exit;

    if (pr_code containing '.budget.') then
    begin
        for select kc_from_c_id,
                   kc.kc_amount_approved
        from knclient kc
        where kc_from_k_id = :k_id
        and kc_amount_approved > 0
        into c_id,
             kc_amount
        do
        begin
            res_str_values = (select coalesce(res_str_values, '')
                              from HP_OBJECTPROPS_ANKETA_GET(0, :c_id, -1,  'Organization_type', null, null));
            if (res_str_values <> 'Бюджетная') then
            begin
                res_error_text = 'Продукт Бюджетник предназначен только для  клиентов из бюджетных организаций';
                res_error_is_critical = 1;
                select c.c_name_fio,
                       c.c_pass_no
                from clients c
                where c.c_id=:c_id
                into res_info_text,
                     kf_client_pass_no;
                suspend;
            end
        end
    end
end
