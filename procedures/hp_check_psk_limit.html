CREATE PROCEDURE HP_CHECK_PSK_LIMIT(
    CHECK_ITEM D_STR_CODE,
    CHECK_ID D_ID,
    CHECK_DATE D_DATE)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable res_psk_limit d_currency;
declare variable disb_amount d_currency;
declare variable total_income d_currency;
declare variable cml_object_kind d_integer;
declare variable k_id d_id;
declare variable payed_plus d_currency;
declare variable min_sh_date d_date;
declare variable max_sh_date d_date;
declare variable pr_is_secured d_bool;
declare variable k_disb_date d_date;
declare variable psk_calc_date d_date;
declare variable psk_calc_perc d_percent;
declare variable psk_limit d_percent;
declare variable pr_code d_str_255;
begin
  if (check_item not in ('schedule', 'com_list')) then
    exception e_default 'Unknown check_item: ' || coalesce(check_item, '<Null>');

  res_text = '';

  select res_psk_limit
  from hp_calcperc_psk_limit(:check_date)
  into :res_psk_limit;

  if (res_psk_limit > 0) then
  begin
    if (check_item = 'schedule') then
    begin
      k_id = check_id;
      select sum(sh_os), sum(sh_perc)
      from schedule
      where sh_from_k_id = :k_id
      into disb_amount, total_income;

      select pr_code
      from kontract, product
      where k_id = :k_id and
            k_from_pr_id = pr_id
      into pr_code;

      if (total_income > round(disb_amount * res_psk_limit) and pr_code not containing '.selfemployed.') then
        res_text = 'Процентный доход превышает ' || round(res_psk_limit * 100.00) || '% от суммы выдачи';
    end else
    if (check_item = 'com_list') then
    begin
      select cml_object_kind, cml_object_id
      from com_list
      where cml_id = :check_id
      into cml_object_kind, k_id;

      if (cml_object_kind <> f_objkind('kontract', 'id')) then
        res_text = 'В заявке отсутствует договор займа';

      if (res_text = '') then
      begin
        disb_amount = 0;
        payed_plus = 0;
        total_income = 0;

        select k_amount_approved, pr_code
        from kontract, product
        where k_id = :k_id and
              k_from_pr_id = pr_id
        into disb_amount, pr_code;

        select -sum(ok_perc + ok_fine)
        from operkred
        where ok_from_k_id = :k_id and
              ok_type = 4 and
              ok_ol_is_active = 1 and
              ok_ol_date <= :check_date
        into payed_plus;

        if (payed_plus is null) then
          payed_plus = 0;

        select sum(cmr_sh_perc)
        from com_list_resh
        where cmr_from_cml_id = :check_id
        into total_income;

        if (total_income is null) then
          total_income = 0;

        total_income = total_income + payed_plus;
        if (k_id not in (101010009792002101)) then
          if (total_income > round(disb_amount * res_psk_limit) and pr_code not containing '.selfemployed.') then
            res_text = 'Превышен ограничитель ' || round(res_psk_limit * 100.00) || '% по соотношению дохода по займу к сумме займа, измените график';
      end
    end

    if (res_text = '') then
    begin
      -- Расчет и контроль лимита по ПСК
      -- k_id и disb_amount уже получены ранее
      select pr_is_secured, k_disb_date, pr_code
      from kontract, product
      where k_id = :k_id
        and k_from_pr_id = pr_id
      into pr_is_secured, k_disb_date, pr_code;
  
      min_sh_date = k_disb_date;
  
      if (check_item = 'schedule') then
      begin
        psk_calc_date = k_disb_date;
        select max(sh_date)
        from schedule
        where sh_from_k_id = :k_id
        into max_sh_date;
  
        select effect_rate
        from hp_effect_rate_common(:k_id)
        into psk_calc_perc;
      end
      else
      if (check_item = 'com_list') then
      begin
        select cml_date
        from com_list
        where cml_id = :check_id
        into psk_calc_date;
  
        select max(cmr_sh_date)
        from com_list_resh
        where cmr_from_cml_id = :check_id
        into max_sh_date;
  
        select effect_rate
        from hp_effect_rate_reschedule(:k_id, :psk_calc_date, :check_id)
        into psk_calc_perc;
      end
  
      select limit
      from hp_kontpsk_limit(:psk_calc_date, :max_sh_date - :min_sh_date, :disb_amount, :pr_is_secured)
      into psk_limit;
  
      if (psk_calc_perc > psk_limit and pr_code not containing '.selfemployed.') then
        res_text = f_format('Превышен ограничитель по ПСК: фактическое значение ПСК = %1, предельное значение ограничителя = %2. Измените график',
          f_numtostr(psk_calc_perc, 3), f_numtostr(psk_limit, 3));
    end
  end

  suspend;
end
