CREATE PROCEDURE UXP$SELECT_ITEMS(
    LIST_KIND D_STR_CODE,
    ITEM_CODE D_STR_100,
    ITEM_UUID D_UUID)
RETURNS (
    ID D_STR_255,
    ID_VISIBLE D_STR_255,
    TEXT D_STR_255,
    ID_UUID D_UUID)
AS
declare variable dont_show_fields d_bool;
declare variable tmp_uuid d_uuid;
declare variable uf_uuid d_uuid;
declare variable ux_uuid d_uuid;
declare variable field_name d_str_code;
declare variable tmp_source d_str_code;
declare variable ux_kind d_str_code;
declare variable parent_uuid d_uuid;
declare variable ufi_kind d_integer;
declare variable is_param d_bool;
declare variable ml_id d_id;
declare variable rep_kind d_integer;
declare variable ufi_control d_str_code;
declare variable tmp_count d_integer;
declare variable tmp_text d_str_255;
declare variable ufm_uuid d_uuid;
declare variable is_preblank d_bool;
declare variable mm_uuid d_uuid;
begin
  if (trim(coalesce(list_kind, '')) = '') then
    exit;

  if (list_kind is null) then
    list_kind = '';

  item_code = trim(coalesce(item_code, ''));

  if (list_kind = 'ufp_buttonedit_uxa') then
  begin
    for select distinct id, id_visible, text, id_uuid
    from uxp$buttonedit_aliases(char_to_uuid(:item_code))
    into :id, :id_visible, :text, :id_uuid
    do
      suspend;

    exit;
  end

  if (list_kind = 'ufp_combobox_code') then
  begin
    for select distinct ucl_uf_uuid
    from ux$combolist
    into uf_uuid
    do
    begin
      id = '';
      select us_code || '.' || uf_field_name
      from ux$field, ux$source
      where uf_uuid = :uf_uuid and
            uf_owner = us_uuid
      into id;

      id_visible = id;
      text = uxf$code(uf_uuid, 'name');

      suspend;
    end
    exit;
  end

  if (list_kind = 'uxframeitem_add_frames') then
  begin
    if (coalesce(item_uuid, '') = '') then
      exit;

    for select ufr_uuid, ufr_fcode, ufr_code, ufr_name
    from ux$frame
    where ufr_owner = :item_uuid
    into id_uuid, id, id_visible, text
    do
    begin
      if (exists(select * from ux$fritem
      where ufi_owner = :item_uuid and
            ufi_kind = 1 and
            ufi_ufr_uuid = :id_uuid)) then
        continue;

      suspend;
    end

    exit;
  end


  if (list_kind = 'frame_master_fields') then
  begin
    if (coalesce(item_uuid, '') = '') then
      exit;

    for select uf_field_name,
               uf_field_name,
               iif(uf_pk_field = 0, '', 'pk' || uf_pk_field ||'  ') ||
               iif(uf_name <> uf_field_name,
                   uf_field_name || '  - ' || uf_name, uf_field_name)
    from ux$frparam, ux$field
    where ufp_ufr_uuid = :item_uuid and
          ufp_uf_uuid = uf_uuid
    into id, id_visible, text
    do
      suspend;

    exit;
  end

  if (list_kind = 'frame_detail_fields') then
  begin
    if (coalesce(item_uuid, '') = '') then
      exit;

    select us_code
    from ux$frame, ux$source
    where ufr_uuid = :item_uuid and
          ufr_us_uuid = us_uuid
    into tmp_source;

    -- Дочерное поле для связки мастер-деталь
    -- Отображаем только тех у кого есть ссылка на другие таблицы
    for select uf_uuid,
               uf_field_name,
               uf_field_name,
               uf_is_param,
               iif(uf_pk_field = 0, '', 'pk' || uf_pk_field ||'  ') ||
               iif(uf_name <> uf_field_name,
                   uf_field_name || '  - ' || uf_name, uf_field_name)
    from ux$frparam, ux$field
    where ufp_ufr_uuid = :item_uuid and
          ufp_uf_uuid = uf_uuid
    into uf_uuid, id, id_visible, is_param, text
    do
    begin
      if (is_param = 1 or (exists(select * from ux$index
      where uix_source = :tmp_source and
            uix_fields = :id and
            uix_fk_source <> ''))) then
        suspend;
    end

    exit;
  end

  if (list_kind = 'action_form_open') then
  begin
    ux_kind = uxf$kind('uxform', 'id');

    for select uxa_uuid, uxa_fcode, uxa_fcode,
               uxa_name || '  - ' || Upper(us_code)
    from ux$alias, ux$code, ux$source, ux$uniq
    where uxa_owner <> :item_uuid and
          uxa_kind = 0 and
          uxa_owner = ux_uuid and
          ux_us_uuid = us_uuid and
          ux_uuid = uu_id and
          uu_kind = :ux_kind and
          us_code <> 'ux$preblanks'
    into tmp_uuid, id, id_visible, text
    do
    begin
      -- Если ранее добавлен, не показываем в списке чтобы повторно не добавить
      if (exists(select * from ux$action
      where ua_owner = :item_uuid and
            ua_uxa_fcode = :id and
            ua_usr_id_field = '')) then
        Continue;

      suspend;
    end

    exit;
  end

  if (list_kind = 'action_form_insert') then
  begin
    for select uxa_uuid, uxa_fcode, uxa_fcode,
               uxa_name || '  - ' || Upper(us_code)
    from ux$alias, ux$code, ux$source
    where uxa_owner <> :item_uuid and
          uxa_kind = 2 and
          uxa_owner = ux_uuid and
          ux_us_uuid = us_uuid and
          us_code <> 'ux$preblanks'
    into ux_uuid, id, id_visible, text
    do
    begin
      -- Если ранее добавлен, не показываем в списке чтобы повторно не добавить
      if (exists(select * from ux$action
      where ua_owner = :item_uuid and
            ua_uxa_fcode = :id and
            ua_usr_id_field = '')) then
        Continue;

      if (exists(select *
      from ux$frame
      where ufr_owner = :ux_uuid and
            ufr_is_mainframe = 1 and
            ufr_is_grid = 1)) then
        continue;

      suspend;
    end

    exit;
  end

  if (list_kind = 'action_command_open') then
  begin
    for select uxa_uuid, uxa_fcode, uxa_fcode,
               uxa_name || '  - ' || Upper(us_code)
    from ux$code, ux$form, ux$command, ux$alias, ux$source
    where ufm_uuid = cmd_uuid and
          ux_uuid = ufm_ux_uuid and
          uxa_owner = ux_uuid and
          ux_us_uuid = us_uuid
    into tmp_uuid, id, id_visible, text
    do
    begin
      -- Если ранее добавлен, не показываем в списке чтобы повторно не добавить
      if (exists(select * from ux$action
      where ua_owner = :item_uuid and
            ua_uxa_fcode = :id and
            ua_usr_id_field = '')) then
        Continue;

      suspend;
    end

    exit;
  end

  if (list_kind = 'action_report_open') then
  begin
    ux_kind = uxf$kind('uxreport', 'id');

    is_preblank = 0;
    if (exists(select * from ux$form, ux$code
               where ufm_uuid = :item_uuid and
                     ufm_ux_uuid = ux_uuid and
                     ux_fcode = 'fm_ux$preblanks')) then
      is_preblank = 1;

    for select uxa_uuid, uxa_fcode,
               uxa_fcode, uxa_name, ux_draft_uuid
    from ux$uniq, ux$code, ux$alias
    where uu_kind = :ux_kind and
          ux_uuid = uu_id and
          uxa_owner = ux_uuid
    into tmp_uuid, id, id_visible, text, ufm_uuid
    do
    begin
      -- Если ранее добавлен, не показываем в списке чтобы повторно не добавить
      if (exists(select * from ux$action
      where ua_owner = :item_uuid and
            ua_uxa_fcode = :id and
            ua_usr_id_field = '')) then
        Continue;

      -- Бланки не показываем
      if (exists(select * from ux$report
                 where urt_uuid = :ufm_uuid and
                       urt_kind = 3)) then
      begin
        -- В обычных формах не показываем бланки
        if (is_preblank = 0) then
          Continue;
      end else
      begin
        -- В спец форме (ux$preblanks не показываем отчеты
        if (is_preblank = 1) then
          Continue;
      end

      suspend;
    end

    exit;
  end

  if (list_kind = 'uxcode_action_uxa_list') then
  begin
    if (uxf$code(item_uuid, 'kind') <> 'uxFrame') then
    begin
      for select uuid_to_char(ua_uuid),
                 ua_uuid,
                 uxa_fcode,
                 ux_name || ' -> ' || uxa_name
      from ux$form fm1, ux$alias, ux$action, ux$form fm2, ux$code
      where fm1.ufm_uuid = :item_uuid and
            fm1.ufm_ux_uuid = uxa_owner and
            uxa_uuid = ua_uxa_uuid and
            ua_owner = fm2.ufm_uuid and
            fm2.ufm_ux_uuid = ux_uuid
      into id, id_uuid, id_visible, text
      do
        suspend;
    end

    exit;
  end


  if (list_kind = 'uxfield_add_fk_source') then
  begin
    for select lower(ut_code), coalesce(ut_name, ''), left(ut_memo, 255)
    from uxp$db_table_list
    into :id, :id_visible, :text
    do
    begin
      if (id_visible <> '') then
        text = id_visible;

      id_visible = id;
      suspend;
    end

    exit;
  end

  if (list_kind = 'uxfield_del_fk_source') then
  begin
    field_name = uxf$code(item_uuid, 'code');
    tmp_source = uxf$owner(item_uuid, 'code');

    for select uix_fk_source, uix_fk_source || iif(uix_virtual = 1, ' (виртуальный)', '')
    from ux$index
    where uix_source = :tmp_source and
          uix_fields = :field_name and
          uix_fk_source <> ''
    into :id, :text
    do
    begin
      id_visible = id;

      suspend;
    end

    exit;
  end

  if (list_kind = 'uxfritem_add_params') then
  begin
    select ufi_owner, ufi_parent_uuid, ufi_kind, uxf$kind(uu_kind, 'code'), ufi_control
    from ux$fritem, ux$uniq
    where ufi_uuid = :item_uuid and
          ufi_owner = uu_id
    into ux_uuid, tmp_uuid, ufi_kind, ux_kind, ufi_control;

    if (ufi_control in ('grid', 'tree')) then
      exit;

    if (ufi_kind in (0, 2)) then
      parent_uuid = tmp_uuid;
    else
      parent_uuid = item_uuid;

    -- если фрейм участвует в командах, не показываем исходящие поля
    if (ux_kind = 'uxframe') then
    begin
      dont_show_fields = 0;
      if (exists(select * from ux$frame, ux$command
      where cmd_uuid = :ux_uuid and
            ufr_owner = cmd_uuid)) then
        dont_show_fields = 1;
  
      if (tmp_uuid is null) then
        tmp_uuid = item_uuid;
  
      for select uf_uuid,
                 uuid_to_char(ufp_uuid),
                 uf_field_name,
                 uf_is_param,
                 iif(uf_field_name = uf_name,
                     us_code || '.' || uf_field_name,
                     us_code || '.' || uf_field_name || '   - ' || uf_name)
      from ux$frparam, ux$field, ux$source
      where ufp_ufr_uuid = :ux_uuid and
            ufp_uf_uuid = uf_uuid and
            uf_owner = us_uuid and
            ufp_editor <> 'current$user' and
            not(:dont_show_fields = 1 and uf_is_param = 0)
      into id_uuid, id, id_visible, is_param, text
      do
      begin
        if (exists(select * from ux$fritem
        where ufi_owner = :ux_uuid and
              ufi_parent_uuid = :parent_uuid and
              ufi_uf_uuid = :id_uuid)) then
          Continue;

        suspend;
      end
    end else
    begin
      for select uf_uuid,
                 uuid_to_char(ufp_uuid),
                 uf_field_name,
                 iif(uf_field_name = uf_name,
                     us_code || '.' || uf_field_name,
                     us_code || '.' || uf_field_name || '   - ' || uf_name)
      from ux$frame, ux$frparam, ux$field, ux$source
      where ufr_owner = :ux_uuid and
            ufp_ufr_uuid = ufr_uuid and
            ufp_uf_uuid = uf_uuid and
            uf_owner = us_uuid and
            ufp_editor <> 'current$user'
      into id_uuid, id, id_visible, text
      do
        suspend;
    end
    exit;
  end

  if (list_kind = 'uxfritem_add_fields') then
  begin
    select ufi_owner, ufi_parent_uuid, ufi_kind, uxf$kind(uu_kind, 'code'), ufi_control
    from ux$fritem, ux$uniq
    where ufi_uuid = :item_uuid and
          ufi_owner = uu_id
    into ux_uuid, tmp_uuid, ufi_kind, ux_kind, ufi_control;

    if (ufi_control in ('grid', 'tree')) then
      exit;

    if (ufi_kind in (0, 2)) then
      parent_uuid = tmp_uuid;
    else
      parent_uuid = item_uuid;

    -- если фрейм участвует в командах, не показываем исходящие поля
    if (ux_kind = 'uxframe') then
    begin
      dont_show_fields = 0;
      if (exists(select * from ux$frame, ux$command
      where cmd_uuid = :ux_uuid and
            ufr_owner = cmd_uuid)) then
        dont_show_fields = 1;
  
      if (tmp_uuid is null) then
        tmp_uuid = item_uuid;
  
      for select uf_uuid,
                 uuid_to_char(ufp_uuid),
                 uf_field_name,
                 iif(uf_field_name = uf_name,
                     us_code || '.' || uf_field_name,
                     us_code || '.' || uf_field_name || '   - ' || uf_name)
      from ux$frparam, ux$field, ux$source
      where ufp_ufr_uuid = :ux_uuid and
            ufp_uf_uuid = uf_uuid and
            uf_owner = us_uuid and
            ufp_editor <> 'current$user' and
            uf_is_param = 0 and
            not(:dont_show_fields = 1 and uf_is_param = 0)
      into id_uuid, id, id_visible, text
      do
      begin
        if (exists(select * from ux$fritem
        where ufi_owner = :ux_uuid and
              ufi_parent_uuid = :parent_uuid and
              ufi_uf_uuid = :id_uuid)) then
          Continue;

        suspend;
      end
    end else
    begin
      for select uf_uuid,
                 uuid_to_char(ufp_uuid),
                 uf_field_name,
                 iif(uf_field_name = uf_name,
                     us_code || '.' || uf_field_name,
                     us_code || '.' || uf_field_name || '   - ' || uf_name)
      from ux$frame, ux$frparam, ux$field, ux$source
      where ufr_owner = :ux_uuid and
            ufp_ufr_uuid = ufr_uuid and
            ufp_uf_uuid = uf_uuid and
            uf_owner = us_uuid and
            uf_is_param = 0 and
            ufp_editor <> 'current$user'
      into id_uuid, id, id_visible, text
      do
        suspend;
    end

    exit;
  end

  if (list_kind = 'FsConfig_TestUsers') then
  begin
    Exit;
  end

  if (list_kind = 'metadata_table_list') then
  begin
    for select us_code, us_name, left(us_memo, 255)
    from uxp$uxsource_list(1, :item_code)
    into id, id_visible, text
    do
    begin
      if (id <> id_visible) then
        id_visible = id || '    - ' || id_visible;
      suspend;
    end

    Exit;
  end

  if (list_kind = 'metadata_procedure_list') then
  begin
    for select us_code, us_name, left(us_memo, 255)
    from uxp$uxsource_list(0, :item_code)
    into id, id_visible, text
    do
    begin
      if (id <> id_visible) then
        id_visible = id || '    - ' || id_visible;
      suspend;
    end

    Exit;
  end

  if (list_kind = 'import_frame') then
  begin
    if (coalesce(item_uuid, '') = '') then
      exit;

    for select ufr_uuid,
               ufr_name || iif(ufr_code <> '' and position(ufr_code, ufr_name) = 0,  ' (' || ufr_code || ')', ''),
               ux_fcode || iif(ux_fcode <> ux_name, '   - ' || ux_name, '')
    from ux$frame, ux$form, ux$code
    where ufr_owner <> :item_uuid and
          ufr_owner = ufm_uuid and
          ux_uuid = ufm_ux_uuid
    order by ufr_owner, ufr_code
    into :id_uuid, :id, :text
    do
    begin
      id_visible = id;
      suspend;
    end

    exit;
  end

  if (list_kind = 'frame_param_editors') then
  begin
    for select editor_id, editor_name, editor_id || '   - ' || editor_memo
    from uxp$uxfrparam_controls('editor')
    into :id, :id_visible, :text
    do
      suspend;

    exit;
  end

  if (list_kind = 'uxcode_button_action_list') then
  begin
    if (uxf$code(item_uuid, 'kind') = 'uxFrame') then
    begin
      select ufr_owner
      from ux$frame
      where ufr_uuid = :item_uuid
      into item_uuid;
    end

    for select ufm_uuid
    from ux$form
    where ufm_uuid = :item_uuid
    union
    select ufr_uuid
    from ux$frame
    where ufr_owner = :item_uuid
    into tmp_uuid
    do
    begin
      for select uuid_to_char(ua_uuid),
                 ua_uuid,
                 uxa_fcode,
                 ua_name
      from ux$action, ux$alias
      where ua_owner = :tmp_uuid and
            ua_uxa_uuid = uxa_uuid and
            (ua_usr_id_field = '' or ua_preblank_kind <> 0)
      into id, id_uuid, id_visible, text
      do
        suspend;
    end

    exit;
  end

  if (list_kind = 'uxcode_find_object') then
  begin
    mm_uuid = uxf$mainmenu();

    for select uuid_to_char(ufm_uuid), ux_us_uuid, ux_name, ufm_ux_uuid, us_code, uxf$kind(uu_kind, 'code')
    from ux$form, ux$code, ux$source, ux$uniq
    where ufm_ux_uuid = ux_uuid and
          ux_us_uuid = us_uuid and
          ux_uuid = uu_id
    into id, id_uuid, id_visible, ux_uuid, text, ux_kind
    do
    begin
      ml_id = null;
      tmp_uuid = null;

      select ua_uuid, ua_name
      from ux$action, ux$alias
      where ua_owner = :mm_uuid and
            ua_uxa_uuid = uxa_uuid and
            uxa_owner = :ux_uuid
      rows 1
      into tmp_uuid, text;

      if (tmp_uuid is not null) then
      begin
        select item_id
        from uxp$actionmenu(:tmp_uuid)
        into ml_id;
      end

      if (ml_id is not null) then
        text = uxf$menupath(ml_id, id_visible);

      rep_kind = null;
      if (ux_kind = 'uxreport') then
      begin
        select urt_kind
        from ux$report
        where urt_uuid = char_to_uuid(:id)
        into rep_kind;

        ux_kind = trim(case rep_kind
          when 0 then 'Отчет: Grid'
          when 1 then 'Отчет: Excel'
          when 2 then 'Отчет: FastRep'
          when 3 then 'Бланк'
        end);

      end else
      if (ux_kind = 'uxcommand') then
        ux_kind = 'command';
      else
        ux_kind = '';

      if (ux_kind <> '') then
        text = text || '  (' || ux_kind || ')';

      suspend;

      if (rep_kind is null) then
      begin
        ux_uuid = char_to_uuid(id);
        tmp_uuid = id_uuid;
  
        for select uuid_to_char(ufr_uuid), ufr_us_uuid, '   ' || ufr_name,
                   iif(us_kind <> -1, us_code, '<sql текст>')
        from ux$frame, ux$source
        where ufr_owner = :ux_uuid and
              ufr_us_uuid = us_uuid and
              ufr_us_uuid <> :tmp_uuid
        into id, id_uuid, id_visible, text
        do
          suspend;
      end
    end

    exit;
  end

  if (list_kind = 'unitframe_list_to_add') then
  begin
    for select unf_uuid, unf_code, unf_name, iif(unf_uni_uuid is null, '', 'Прикреплен')
    from ux$unitframe
    where unf_owner = :item_uuid
    into :id_uuid, :id, :id_visible, text
    do
    begin
      if (id_visible not containing id) then
        id_visible = id || '   / ' || id_visible;

      suspend;
    end

    exit;
  end

  if (list_kind = 'uxsource_find_registered') then
  begin
    for select us_uuid, us_code, us_code, us_name
    from ux$source
    where us_kind <> -1
    into :id_uuid, :id, :id_visible, text
    do
      suspend;

    exit;
  end

  if (list_kind = 'uxsource_table_registered') then
  begin
    for select us_uuid, us_code, us_code, us_name
    from ux$source
    where us_kind <> -1 and
          us_is_table = 1
    into :id_uuid, :id, :id_visible, text
    do
      suspend;

    exit;
  end

  if (list_kind in ('uxaliasaction_menu_to_add', 'uxaliasaction_menu_to_del')) then
  begin
    select ua_owner, uxf$kind(uu_kind, 'code')
    from ux$action, ux$uniq
    where ua_uuid = :item_uuid and
          ua_owner = uu_id
    into tmp_uuid, ux_kind;
  
    if (tmp_uuid is null) then
      exit;
  
    if (ux_kind = 'uxframe') then
    begin
      select ufm_ux_uuid
      from ux$frame, ux$form
      where ufr_uuid = :tmp_uuid and
            ufr_owner = ufm_uuid
      into ux_uuid;
    end else
    begin
      select ufm_ux_uuid
      from ux$form
      where ufm_uuid = :tmp_uuid
      into ux_uuid;
    end
  
    if (ux_uuid is null) then
      exit;

    for select uxa_uuid, uuid_to_char(uxa_uuid), uxa_fcode, uxa_name
    from ux$alias
    where uxa_owner = :ux_uuid
    into id_uuid, id, id_visible, text
    do
    begin
      -- Если ранее добавлен, не показываем в списке чтобы повторно не добавить
      if (exists(select * from ux$aliasaction
      where uaa_ua_uuid = :item_uuid and
            uaa_uxa_uuid = :id_uuid)) then
      begin
        if (list_kind = 'uxaliasaction_menu_to_add') then
          Continue;
      end else
      begin
        if (list_kind = 'uxaliasaction_menu_to_del') then
          Continue;
      end

      suspend;
    end

    exit;
  end

  if (list_kind = 'uxaliasaction_menu_to_sync') then
  begin
    select uxa_owner
    from ux$alias
    where uxa_uuid = :item_uuid
    into ux_uuid;
  
    if (ux_uuid is null) then
      exit;
  
    for select uxa_uuid, uuid_to_char(uxa_uuid), uxa_fcode, uxa_name
    from ux$alias
    where uxa_owner = :ux_uuid
    into id_uuid, id, id_visible, text
    do
    begin
      if (id_uuid = item_uuid) then
        Continue;

      suspend;
    end

    exit;
  end

  if (list_kind = 'uxfritem_add_grid') then
  begin
    id = '';
    id_visible = 'new';
    text = 'Создать новый набор данных';
    id_uuid = null;
    suspend;

    for select ugr_uuid, ugr_name, ugr_memo
    from ux$grid
    where ugr_ufr_uuid = :item_uuid
    into id_uuid, id_visible, text
    do
    begin
      if (exists(select * from ux$fritem
      where ufi_uuid = :id_uuid)) then
        Continue;

      id = uuid_to_char(id_uuid);
      suspend;
    end

    exit;
  end


  if (list_kind = 'uxgrid_list_for_edit') then
  begin
    id = '';

    select count(*)
    from ux$frame
    where ufr_owner = :item_uuid
    into tmp_count;

    for select ufr_uuid, 'Фрейм: ' || ufr_name
    from ux$frame
    where ufr_uuid = :item_uuid or ufr_owner = :item_uuid
    into tmp_uuid, text
    do
    begin
      for select ugr_uuid, ugr_name, ugr_memo
      from ux$grid
      where ugr_ufr_uuid = :tmp_uuid
      into :id_uuid, :id_visible, tmp_text
      do
      begin
        if (tmp_count <= 1) then
          text = tmp_text;

        id = uuid_to_char(id_uuid);

        suspend;
      end
    end

    exit;
  end

  if (list_kind = 'select_editor_enum') then
  begin
    id_uuid = null;
    for select erl_code, erl_code, erl_name
    from ux$ereflist
    where erl_kind = 0
    into :id, id_visible, :text
    do
      suspend;

    exit;
  end

  if (list_kind = 'select_editor_reflist') then
  begin
    id_uuid = null;
    for select erl_code, erl_code, erl_name
    from ux$ereflist
    where erl_kind = 1 and
          erl_code <> 'None'
    into :id, id_visible, :text
    do
      suspend;

    exit;
  end

  exception e_message f_format('Undefined params: list_kind: %1, item_code: %2', list_kind,  item_code);
end
