CREATE PROCEDURE AAREP_CLIENT_QUESTIONNAIRE(
    KC_ID ID_TYPE,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    C_NAME STR_100,
    C_INN D_STR_20,
    C_DTBIRTH DATE,
    C_NAME_ENG STR_100,
    COMPANY_NAME STR_100,
    AN01_ID INTEGER,
    AN02_ID INTEGER,
    AN03_ID INTEGER,
    AN04_ID INTEGER,
    AN05_ID INTEGER,
    AN06_ID INTEGER,
    AN07_ID INTEGER,
    AN08_ID INTEGER,
    AN09_ID INTEGER,
    AN10_ID INTEGER,
    AN11_ID INTEGER,
    AN12_ID INTEGER,
    AN13_ID INTEGER,
    AN01_OTHER STR_255,
    AN02_OTHER STR_255,
    AN03_OTHER STR_255,
    AN04_OTHER STR_255,
    AN05_OTHER STR_255,
    AN06_OTHER STR_255,
    AN07_OTHER STR_255,
    AN08_OTHER STR_255,
    AN09_OTHER STR_255,
    AN10_OTHER STR_255,
    AN11_OTHER STR_255,
    AN12_OTHER STR_255,
    AN13_OTHER STR_255,
    AN_DATE D_DATE,
    QR_CODE D_STR_LONG)
AS
declare variable MA_ID D_INTEGER;
declare variable QU_ID D_INTEGER;
declare variable AN_ID D_INTEGER;
declare variable AN_OTHER STR_255;
declare variable EDL_ID D_ID;
declare variable K_ID D_ID;
declare variable C_ID D_ID;
declare variable Z_DATE date;
begin
  if (kc_id > 0) then
  begin
    select
      k_from_ma_app_id,
      k_id,
      c_id,
      c_name_fio,
      c_birth_date,
      c_name_eng,
      c_inn
    from knclient
      inner join clients
        on c_id = kc_from_c_id
      inner join kontract
        on k_id = kc_from_k_id
    where kc_id = :kc_id
    into
      :ma_id,
      :k_id,
      :c_id,
      :c_name,
      :c_dtbirth,
      :c_name_eng,
      :c_inn;
  end else
  begin
    select
      k_from_ma_app_id,
      k_id,
      c_id,
      c_name_fio,
      c_birth_date,
      c_name_eng,
      iif((z_date_begin is not null) and (z_is_ready = 1) and (z_source_z_id > 0),  z_date_begin, null)
    from zalog
      inner join clients
        on c_id = z_from_c_id
      inner join kontract
        on k_id = z_from_k_id
    where z_id = abs(:kc_id)
    into
      :ma_id,
      :k_id,
      :c_id,
      :c_name,
      :c_dtbirth,
      :c_name_eng,
      :z_date;
  end
  if (c_id is null) then exit;

  an_date = (select cast(client_sign_datetime as date)
    from hp_edoc_get_status('loan.application', 'miniapp', :ma_id));

  if ((z_date is not null) and not exists(
    select *
    from oksummary
      inner join knclient
        on kc_from_k_id = oks_from_k_id
          and kc_from_c_id = :c_id
          and kc_amount_approved > 0
    where oks_date_begin between '01.01.1900' and :z_date
     and oks_date_end between :z_date and '31.12.9999'
     and oks_close_state = 0))
  then an_date = z_date;

  an01_id = 0;
  an02_id = 0;
  an03_id = 0;
  an04_id = 0;
  an05_id = 0;
  an06_id = 0;
  an07_id = 0;
  an08_id = 0;
  an09_id = 0;
  an10_id = 0;
  an11_id = 0;
  an12_id = 0;
  an13_id = 0;
  an01_other = '';
  an02_other = '';
  an03_other = '';
  an04_other = '';
  an05_other = '';
  an06_other = '';
  an07_other = '';
  an08_other = '';
  an09_other = '';
  an10_other = '';
  an11_other = '';
  an12_other = '';
  an13_other = '';

  for select fqa_qu_id, fqa_an_id,
             coalesce(fqa_an_other, '')
      from fp_question_anketa
      where fqa_from_k_id = :k_id
        and fqa_from_c_id = :c_id
      into :qu_id, :an_id,
           :an_other
  do
  begin
    if (qu_id = 1) then
    begin
      an01_id = an_id;
      an01_other = an_other;
    end
    else
    if (qu_id = 2) then
    begin
      an02_id = an_id;
      an02_other = an_other;
    end
    else
    if (qu_id = 3) then
    begin
      an03_id = an_id;
      an03_other = an_other;
    end
    else
    if (qu_id = 4) then
    begin
      an04_id = an_id;
      an04_other = an_other;
    end
    else
    if (qu_id = 5) then
    begin
      an05_id = an_id;
      an05_other = an_other;
    end
    else
    if (qu_id = 6) then
    begin
      an06_id = an_id;
      an06_other = an_other;
    end
    else
    if (qu_id = 7) then
    begin
      an07_id = an_id;
      an07_other = an_other;
    end
    else
    if (qu_id = 8) then
    begin
      an08_id = an_id;
      an08_other = an_other;
    end
    else
    if (qu_id = 9) then
    begin
      an09_id = an_id;
      an09_other = an_other;
    end
    else
    if (qu_id = 10) then
    begin
      an10_id = an_id;
      an10_other = an_other;
    end
    else
    if (qu_id = 11) then
    begin
      an11_id = an_id;
      an11_other = an_other;
    end
    else
    if (qu_id = 12) then
    begin
      an12_id = an_id;
      an12_other = an_other;
    end
    else
    if (qu_id = 13) then
    begin
      an13_id = an_id;
      an13_other = an_other;
    end
  end

  -- QR-код электронной подписи клиента. Текст пустой если не подписан
  if (coalesce(add_params, '') <> '') then
  begin
    edl_id = f_json_str_get_value(add_params, 'edl_id', null);
    qr_code = f_edoc_qr_by_edl(edl_id, c_id);
  end

  company_name = f_company_name('short','local');
  suspend;
end
