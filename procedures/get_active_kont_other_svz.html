CREATE PROCEDURE GET_ACTIVE_KONT_OTHER_SVZ(
    C_ID ID_TYPE)
RETURNS (
    REG_NAME VARCHAR(100),
    KF_NAME VARCHAR(100),
    KF_OFFICER VARCHAR(100),
    KF_OFFICER_ID DECIMAL(18,0),
    REG_ID INTEGER)
AS
begin
  for select kal_new_kf_name, reg_code || ' ' || reg_name, c_name_fio, kal_new_kf_officer, reg_unicode
      from oksummary, knclient, kontract, knaktlist, region, clients
      where knaktlist.is_delete = 0 and
            kal_date_end >= '31.12.2999' and
            kal_from_kf_id = k_from_kf_id and
            kal_new_reg_id = reg_unicode and
            kal_new_kf_officer = c_id and
            k_status >= 1

            and
            oks_from_k_id = k_id and
            oks_balance > 0 and
            oks_date_begin <= current_date and
            oks_date_end >= current_date and
            oks_close_state = 0

            and
            kc_from_k_id = k_id and
            kc_from_c_id = :c_id and
            kc_amount_approved > 0
      into :kf_name, :reg_name, kf_officer, kf_officer_id, reg_id
  do
  begin
    suspend;
  end
end
