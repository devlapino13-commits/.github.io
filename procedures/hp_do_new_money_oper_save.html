CREATE PROCEDURE HP_DO_NEW_MONEY_OPER_SAVE(
    OPER_DATE DATE,
    REG_ID INTEGER,
    DOC_KIND INTEGER,
    OPER_SUBTYPE INTEGER,
    IS_ZALOG INTEGER,
    DOC_NO INTEGER,
    DOC_NO_STR D_STR_CODE,
    MEMO D_STR_2048,
    KONTRAGENT_TYPE INTEGER,
    CLIENT_ID ID_TYPE,
    BANK_ACC_NO_CLIENT D_STR_CODE,
    OFFICE_ID ID_TYPE,
    BANK_ACC_NO D_STR_CODE,
    DT_ACC D_STR_CODE,
    KT_ACC D_STR_CODE,
    CURR_ID D_STR_10,
    CURR_RATE CURR_TYPE,
    AMOUNT_VAL CURR_TYPE,
    PC_CODE D_STR_CODE,
    KP_CODE D_STR_CODE,
    OB_DETAIL D_STR_LONG,
    SAC_ID INTEGER,
    KONT_ID ID_TYPE,
    KONTRACT_NO D_STR_CODE,
    OPERATION_TYPE INTEGER,
    ACC_ID_DT D_ID,
    ACC_ID_KT D_ID,
    UN_ID ID_TYPE)
RETURNS (
    OL_ID ID_TYPE,
    OL_ID_STR D_STR_255,
    RES_TEXT D_STR_LONG)
AS
declare variable sac_month_minus curr_type;
declare variable sac_month_saldo curr_type;
declare variable amount curr_type;
declare variable ob_client_dt id_type;
declare variable ob_client_kt id_type;
declare variable sac_sign integer;
declare variable reg_salary integer;
declare variable ok_id id_type;
declare variable db_id d_id;
declare variable need_kn_operbook d_smallint;
declare variable kob_id d_id;
declare variable operbook_kind d_smallint;
begin
  -- Процедура сохранения денежной операции
  ol_id = 0;
  res_text = '';
  ol_id_str = '';

  db_id = f_get_db_id();

  if (is_zalog is null) then
    is_zalog = 0;

  if (coalesce(curr_rate, 0) = 0) then
    curr_rate = 1;
  amount = round(amount_val * curr_rate, 2);

  if (client_id = 0) then
    client_id = null;
  if (office_id = 0) then
    office_id = null;

  if (res_text = '' and
      oper_date is null) then
    res_text = 'Дата не указана';
  if (res_text = '' and
      coalesce(reg_id, 0) = 0) then
    res_text = 'Офис не указан';
  if (res_text = '' and
      coalesce(dt_acc, '') = '') then
    res_text = 'Дебет счет не указан';
  if (res_text = '' and
      coalesce(kt_acc, '') = '') then
    res_text = 'Кредит счет не указан';
  if (res_text = '' and
      coalesce(curr_id, '') = '') then
    res_text = 'Валюта не указана';
  if (res_text = '' and
      coalesce(doc_kind, -1) = -1) then
    res_text = 'Тип документа не указан';
  if (res_text = '' and
      coalesce(amount_val, 0) = 0) then
    res_text = 'Сумма не указана';
  if (res_text = '' and
      coalesce(memo, '') = '') then
    res_text = 'Назначение платежа не указано';
  if (res_text = '' and
      coalesce(memo, '') = '') then
    res_text = 'Назначение платежа не указано';
  if (res_text = '' and
      is_zalog = 1 and
      coalesce(kont_id, 0) = 0) then
    res_text = 'Договор займа не указан';

  need_kn_operbook = 0;

  if (db_id = 81001 and
      oper_date >= '31.12.2018' and
      f_get_const('operbook_kind') <> '0') then
  begin
    if (res_text = '' and
        (coalesce(acc_id_dt, 0) = 0 and
        coalesce(acc_id_kt, 0) <> 0)) then
      res_text = 'Дебет счет по ЕПС не указан';

    if (res_text = '' and
        (coalesce(acc_id_dt, 0) <> 0 and
        coalesce(acc_id_kt, 0) = 0)) then
      res_text = 'Кредит счет по ЕПС не указан';

    if (oper_date >= '01.06.2019') then
    begin
      res_text = '';
      if (coalesce(acc_id_dt, 0) = 0 or coalesce(acc_id_kt, 0) = 0) then
      begin
        acc_id_dt = 0;
        acc_id_kt = 0;
      end else need_kn_operbook = 1;
    end
  end

  if (oper_date >= '01.06.2019' and coalesce(acc_id_dt, 0) <> 0 and coalesce(acc_id_kt, 0) <> 0) then
    need_kn_operbook = 1;

  if (res_text = '' and
      oper_subtype in (1, 2) and
      coalesce(kontract_no, '') <> '') then
  begin
    select error_text, ok_id
    from hp_operkontract_ins(-1, -1, :kontract_no, :operation_type, :amount)
    into :res_text, ok_id;
  end

  if (res_text <> '') then
  begin
    suspend;
    exit;
  end

  sac_month_minus = -1;
  sac_month_saldo = -1;

  if (coalesce(sac_id, 0) > 0) then
  begin
    select sac_id
    from sa_config
    where sac_is_credit = 9
    into sac_month_minus;

    select sac_id
    from sa_config
    where sac_is_credit = 10
    into sac_month_saldo;
  end

  -- Создаем запись для денежной операции
  insert into operlist (ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                        ol_status, is_delete, ol_user_id, ol_document_client, ol_acc_no_client, ol_is_mo_modified,
                        ol_document_office, ol_acc_no_office, ol_oper_subtype, ol_is_client)
  values (:reg_id, 2, -- денежная операция
          :oper_date, :doc_no, :doc_no_str, :doc_kind, :memo, 0, -- ol_status
          0, -- is_delete
          :un_id, :client_id, iif(:client_id is not null, :bank_acc_no_client, null), 0, -- ol_is_mo_modified
          :office_id, iif(:office_id is not null, :bank_acc_no, null), :oper_subtype, iif(:kontragent_type = 1, 1, 0))
  returning ol_id
  into :ol_id;

  if (ol_id is null) then
    exception e_default 'Operlist save error';

  ol_id_str = ol_id || '-' || doc_kind;

  if (ok_id is not null) then
  begin
    select error_text
    from hp_operkontract_ins(:ok_id, :ol_id, '', :operation_type, 0)
    into :res_text;

    res_text = '';
  end

  ob_client_dt = null;
  ob_client_kt = null;

  sac_sign = 0;
  if (client_id is not null) then
  begin
    if (oper_subtype in (2, 7, 8)) then
    begin
      sac_sign = 1;
      ob_client_dt = client_id;
    end
    if (oper_subtype in (1, 6)) then
    begin
      sac_sign = -1;
      ob_client_kt = client_id;
    end

    if (oper_subtype in (3, 4, 5, 9)) then
    begin
      sac_sign = 0;
      ob_client_dt = client_id;
      ob_client_kt = client_id;
    end
  end

  insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                        ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                        ob_from_sac_id, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
  values (:ol_id, -1, :curr_id, :curr_rate, :ob_client_dt, :dt_acc, :amount_val, :amount, :pc_code, :kp_code, null,
          :kt_acc, :ob_client_kt, :ob_detail, :sac_id, :oper_date, 1, :reg_id);


  if (need_kn_operbook = 1) then
  begin
    select kob_id
    from hp_do_kn_operbook_memorder(:ol_id, null, null, null, :amount, :amount_val,
        :amount, :amount, :ob_detail, :acc_id_dt, :acc_id_kt, :oper_date, :curr_id)
    into :kob_id;

    if (ob_client_dt is not null or ob_client_kt is not null) then
    begin
      update kn_operbook
      set kob_client_dt = :ob_client_dt,
          kob_client_kt = :ob_client_kt
      where (kob_id = :kob_id);
    end
  end

  if (client_id is not null and
      sac_sign <> 0 and
      sac_id > 0) then
  begin
    reg_salary = null;

    select ps_from_reg_id
    from hr_pstatus
    where ps_from_c_id = :client_id and
          ps_date_begin <= :oper_date and
          ps_date_end >= :oper_date and
          ps_status = 1
    order by ps_date_begin desc, ps_date_end_in_order desc, ps_is_working desc
    rows 1
    into reg_salary;

    if (reg_salary is null) then
    begin
      reg_salary = f_hq_id(db_id);
    end

    insert into opersalary (os_from_ol_id, os_from_reg_id, os_from_c_id, os_from_sac_id, os_value)
    values (:ol_id, :reg_salary, :client_id, :sac_id, :amount);

    if (coalesce(sac_month_minus, -1) <> -1) then
    begin
      insert into opersalary (os_from_ol_id, os_from_reg_id, os_from_c_id, os_from_sac_id, os_value)
      values (:ol_id, :reg_salary, :client_id, :sac_month_minus, :amount);
    end

    if (coalesce(sac_month_saldo, -1) <> -1) then
    begin
      insert into opersalary (os_from_ol_id, os_from_reg_id, os_from_c_id, os_from_sac_id, os_value)
      values (:ol_id, :reg_salary, :client_id, :sac_month_saldo, :amount);
    end
  end

  if (client_id is not null and
      sac_sign <> 0 and
      sac_id < 0) then
  begin
    insert into servicecalc (ssc_from_ol_id, ssc_id, ssc_from_sd_id, ssc_date, ssc_amount_dogovor, ssc_other_plus,
                             ssc_tax_client_sfond, ssc_tax_client_pdnal, ssc_tax_client_other, ssc_tax_company_sfond,
                             ssc_client_saldo, ssc_total_days, ssc_patent_days, ssc_spolis_days, ssc_kind,
                             ssc_disb_amount, ssc_zp_koef)
    values (:ol_id, 1, -:sac_id, :oper_date, 0, 0, 0, 0, 0, 0, :amount * cast(:sac_sign as curr_type) * -1.0000, -- ssc_client_saldo
            0, 0, 0, 1, :amount * cast(:sac_sign as curr_type), -- ssc_disb_amount
            1);
  end

  if (coalesce(is_zalog, 0) = 1 and
      coalesce(kont_id, 0) > 0) then
  begin
    insert into operkredzalog (okz_from_ol_id, okz_number, okz_from_k_id, okz_summa_plus, okz_summa_minus, okz_is_money,
                               okz_account)
    values (:ol_id, null, :kont_id, iif(:oper_subtype in (1, 6), :amount, 0.0), -- okz_summa_plus
            iif(:oper_subtype not in (1, 6), :amount, 0.0), -- okz_summa_minus
            1, -- okz_is_money
            iif(:oper_subtype in (1, 6), :kt_acc, :dt_acc)); -- okz_account
  end

  suspend;
end
