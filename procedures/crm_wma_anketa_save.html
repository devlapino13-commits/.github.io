CREATE PROCEDURE CRM_WMA_ANKETA_SAVE(
    WMA_ID D_INTEGER,
    CLIENT_LANGUAGE STR_255,
    GENDER D_SMALLINT,
    FIO STR_100,
    PHONE STR_100,
    BIRTH_DATE D_DATE,
    PHONE_ADD STR_100,
    CITIZENSHIP STR_255,
    CITY_ID ID_TYPE,
    CITY STR_100,
    DOCLIST STR_255,
    WORK_STATUS STR_255,
    LOAN_PURPOSE STR_255,
    LOAN_PERIOD D_SMALLINT,
    LOAN_AMOUNT D_INTEGER,
    MEETING_POINT STR_255,
    MEETING_ADDRESS STR_255,
    MEETING_DATE D_DATE,
    OFFICE_ID D_INTEGER,
    INFO_SOURCE STR_255,
    INFO_SOURCE_TEXT STR_255,
    ANKETA_MEMO D_STR_8100,
    CLOSE_REASON STR_255,
    REFUSAL_REASON STR_255,
    C_ID D_ID,
    INN D_STR_20 = null)
RETURNS (
    WMA_ID_OUTPUT D_INTEGER,
    END_CODE STR_20)
AS
declare variable STATUS smallint;
declare variable WMAT_ID integer;
declare variable ERROR_TEXT D_STR_255;
begin
    wma_id_output = wma_id;

    if (city_id = 0) then
      city_id = null;
    if (office_id = 0) then
      office_id = null;
    close_reason = coalesce(close_reason, '');
    status = 0;

    if (close_reason <> '') then -- используется только для негативных статусов
     begin
         status = -1;
         update or insert into cc_website_miniapp_close (wmc_from_wma_id, wmc_memo,
                                                         wmc_closed_rd_id, wmc_refuse_id)
         values (:wma_id, :anketa_memo, f_get_rd_id(:close_reason), f_get_rd_id(:refusal_reason))
         matching (wmc_from_wma_id);
     end

    insert into cc_website_miniapp_tmp (wmat_from_wma_id, wmat_language, wmat_gender, wmat_name,wmat_phone_add,
                                        wmat_citizenship, wmat_city_id, wmat_city, wmat_birthday, wmat_doclist,
                                        wmat_work_status, wmat_loan_period, wmat_loan_amount, wmat_meeting_point,
                                        wmat_meeting_address, wmat_meeting_date, wmat_office, wmat_anketa_memo,
                                        wmat_info_source, wmat_info_source_text, wmat_loan_purpose, wmat_inn)
    values (:wma_id, f_get_rd_id(:client_language), :gender, :fio, :phone_add, f_get_rd_id(:citizenship), :city_id,
          :city, :birth_date, :doclist, f_get_rd_id(:work_status), :loan_period, :loan_amount, f_get_rd_id(:meeting_point),
          :meeting_address, :meeting_date, :office_id, :anketa_memo, f_get_rd_id(:info_source),
          :info_source_text, f_get_rd_id(:loan_purpose), :inn)
    returning wmat_id
    into :wmat_id;

    if ((office_id is not null) and
      (status = 0)) then
    status = 2;

    update cc_website_miniapp t
    set wma_blocked_un_id = null,
        wma_cant_call_qty = 0,
        wma_last_wmat_id = :wmat_id,
        wma_status = :status,
        wma_send_to_office_datetime = iif(:office_id is not null, current_timestamp, null),
        wma_phone_no = :phone
    where (wma_id = :wma_id);

  if (status = 2) then
  begin
    if (:c_id = -1) then -- если клиент не найден и требуется создать нового
    begin
      select c_id
      from cc_do_add_client(0, 1000000000000, :fio, :gender, :birth_date, f_get_rd_id(:client_language), f_get_rd_id(:citizenship), :phone, :phone_add, :inn)
      into :c_id;
    end
    else
    begin
      -- Сохранение номера телефона
      if (coalesce(phone, '') <> '') then
      begin
        begin
          if (f_cont_info('clients', :c_id, 'cell_main', 'ok') <> :phone) then
            execute procedure hp_continfo_update(:c_id, 'contactinfo.cell_main', :phone, 0, 1000000000000);
          when any do
            error_text = 'не удается сохранить основной номер телефона';
        end
      end

      if (coalesce(phone_add, '') <> '') then
      begin
        begin
            if (f_cont_info('clients', :c_id, 'cell_add', 'last') <> :phone_add) then
                execute procedure hp_continfo_update(:c_id, 'contactinfo.cell_add', :phone_add, 0, 1000000000000);
            when any do
                error_text = 'не удается сохранить дополнительный номер телефона';
        end
      end
    end

    update cc_website_miniapp t
    set wma_from_c_id = :c_id
    where (t.wma_id = :wma_id);

  end

  if (wma_id_output is not null) then end_code = 'Успешно';
  else end_code = 'Ошибка';

  suspend;

end
