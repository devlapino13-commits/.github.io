CREATE PROCEDURE AAREP_WRITEOFF_CL_REPORT(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    K_ID D_ID,
    REG_ID INTEGER,
    REP_OFF_END_OS DECIMAL(18,4),
    REP_OFF_END_PERC DECIMAL(18,4),
    REP_OFF_END_OTHER DECIMAL(18,4),
    REP_WRITEOFF_DATE DATE,
    REP_LASTPAY_DATE DATE,
    K_DISB_DATE DATE,
    K_KONTRACT_NO VARCHAR(20),
    KF_WRITEOFF_NAME VARCHAR(100),
    KF_LAST_NAME VARCHAR(100),
    AA_CURR_OFFICE_NAME VARCHAR(100),
    AA_CURR_BRANCH_NAME VARCHAR(100),
    AA_CURR_LEVEL2_NAME VARCHAR(100),
    C_ID D_ID,
    C_NAME_FIO VARCHAR(100),
    C_BIRTH_DATE DATE,
    EUROPLAT_ACC_NO VARCHAR(10),
    C_PLASTCARD_NO VARCHAR(20),
    C_FACT_ADDRESS D_STR_255,
    C_FULL_ADDRESS VARCHAR(255),
    C_PINCODE D_STR_20,
    C_INN D_STR_20,
    PR_NAME D_STR_CODE)
AS
declare variable K_FROM_KF_ID decimal(18,0);
declare variable ADC_ADDRESS_TYPE integer;
declare variable ADL_ADDRESS_STR D_STR_255;
declare variable OKS_CLOSE_STATE D_INTEGER;
declare variable K_AMOUNT_APPROVED D_CURRENCY;
declare variable C_DOLYA D_CURRENCY;
begin

  for select distinct oko_from_k_id, k_last_reg_id, k_amount_approved
      from operkroff, kontract
      where k_id = oko_from_k_id
      into :k_id, :reg_id, :k_amount_approved
  do
  begin
    for select cl.c_id, cl.c_name_fio, cl.c_birth_date, kc.kc_from_k_id, cl.c_pincode,
               (kc.kc_amount_approved / :k_amount_approved) , cl.c_inn
        from knclient kc
        inner join clients cl on cl.c_id = kc.kc_from_c_id
        where kc.kc_amount_approved > 0 and
              kc.kc_from_k_id = :k_id
        into :c_id, c_name_fio, c_birth_date, k_id, c_pincode, c_dolya , c_inn
    do
    begin
      rep_off_end_os = 0;
      rep_off_end_perc = 0;
      rep_off_end_other = 0;
      if (exists(select *
                 from region_list(:reg_id, 1, -1, 0)
                 where id_list = :rep_region_id)) then
      begin
        for select oks.oks_out_os, oks.oks_out_perc, oks.oks_out_other
            from oksummary oks
            where oks.oks_from_k_id = :k_id and
                  :rep_date_end between oks_date_begin and oks_date_end
            into :rep_off_end_os, :rep_off_end_perc, :rep_off_end_other

        do
        begin
          if (rep_off_end_os is null) then
            rep_off_end_os = 0;
          if (rep_off_end_perc is null) then
            rep_off_end_perc = 0;
          if (rep_off_end_other is null) then
            rep_off_end_other = 0;

          if ((rep_off_end_os <> 0) or (rep_off_end_perc <> 0) or (rep_off_end_other <> 0)) then
          begin
            select min(ol_date_oper_day), max(ol_date_oper_day)
            from operkroff, operlist
            where oko_from_k_id = :k_id and
                  oko_from_ol_id = ol_id and
                  operlist.is_delete = 0
            into :rep_writeoff_date, :rep_lastpay_date;

            select k_disb_date, k_kontract_no, k_from_kf_id, pr.pr_name
            from kontract
            inner join product pr on pr.pr_id = k_from_pr_id
            where k_id = :k_id
            into :k_disb_date, k_kontract_no, k_from_kf_id, pr_name;

            select kal_new_kf_name
            from knaktlist
            where kal_from_kf_id = :k_from_kf_id and
                  kal_date <= :rep_writeoff_date and
                  kal_date_end >= :rep_writeoff_date and
                  knaktlist.is_delete = 0
            into :kf_writeoff_name;

            select kal_new_kf_name
            from knaktlist
            where kal_from_kf_id = :k_from_kf_id and
                  kal_date <= :rep_date_end and
                  kal_date_end >= :rep_date_end and
                  knaktlist.is_delete = 0
            into :kf_last_name;

            select aa_curr_office_name, aa_curr_branch_name, aa_curr_level3_name
            from aarep_get_office_categ(:reg_id, :rep_date_end)
            into :aa_curr_office_name, :aa_curr_branch_name, aa_curr_level2_name;

            c_full_address = '';
            c_fact_address = '';

            for select adc_address_type, adl.adl_address_str
                from addrlist adl
                inner join addrclient adcl on adl.adl_id = adcl.adc_from_adl_id
                where adcl.adc_from_c_id = :c_id and
                      adc_address_type in (1, 2)
                into :adc_address_type, :adl_address_str
            do
            begin
              if (adc_address_type = 1) then
                c_full_address = adl_address_str;
              if (adc_address_type = 2) then
                c_fact_address = adl_address_str;
            end

            c_plastcard_no = '';
            if (coalesce(c_id, 0) <> 0) then
            begin
              select cont_value
              from hp_contdata_get_cell_main(0, :c_id)
              into :c_plastcard_no;
            end

            europlat_acc_no = '';

            select uniq_acc_str
            from hp_show_clientuniq_acc_no(:c_id)
            into :europlat_acc_no; -- № лицевого счета

            rep_off_end_os = rep_off_end_os * c_dolya;
            rep_off_end_perc = rep_off_end_perc * c_dolya;
            rep_off_end_other = rep_off_end_other * c_dolya;

            suspend;
          end
        end
      end
    end
  end
end
