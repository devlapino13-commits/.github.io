CREATE PROCEDURE AAREP_INFO_LOAN_SEND(
    REP_DATE D_DATE)
RETURNS (
    ROW_NUM D_INTEGER,
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    KONTRACTS_LIST D_STR_100,
    MSG_MAIL_TO D_STR_20,
    MSG_SEND_STATUS_STR D_STR_CODE,
    MSG_DATETIME D_DATETIME)
AS
declare variable edl_id d_id;
begin
  -- отчет об отправке уведомлений о задолженности
  -- rep_date - дата, на которую сформированы остатки задолженности

  row_num = 0;

  for select c_id,
             c_name_fio,
             edl_id
      from edoc_list
      inner join info_loan_client on ilc_id = edl_object_id
      inner join clients on c_id = ilc_from_c_id
      where edl_status = 0 and
            edl_object_kind = f_objkind('info_loan_client', 'id') and
            ilc_date = :rep_date
      order by c_name_fio
      into c_id,
           c_name_fio,
           edl_id
  do
  begin
    row_num = row_num + 1;

    select list(k_kontract_no, ', ')
    from kontract
    inner join info_loan_debt on ild_from_k_id = k_id
    inner join info_loan_client on ild_from_ilc_id = ilc_id
    where ilc_from_c_id = :c_id
    into kontracts_list;

    select msg_mail_to,
           msg_datetime,
           sds_name
    from fsrobot_msgbox
    inner join sms_delivery_status on sds_id = msg_send_status
    inner join edoc_msg on edm_from_msg_id = msg_id
    inner join edoc_sign on eds_id = edm_from_eds_id
    where eds_from_edl_id = :edl_id
    into msg_mail_to,
         msg_datetime,
         msg_send_status_str;

    suspend;
  end
end
