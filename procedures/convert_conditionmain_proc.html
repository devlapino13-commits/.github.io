CREATE PROCEDURE CONVERT_CONDITIONMAIN_PROC
RETURNS (
    RES_SCRIPT D_BLOB_TEXT)
AS
declare variable tmp$id type of column conditionmain.cdm_id;
declare variable tmp$sql d_blob_text collate unicode_ci;
declare variable tmp$memo type of column conditionmain.cdm_memo;
declare variable tmp$kind_str type of column conditionmain.cdm_object_kind;
declare variable tmp$db_id d_integer;
declare variable tmp$name d_str_20;
declare variable tmp$proc_name d_str_255;
declare variable tmp$i d_id;
declare variable tmp$j d_integer;
declare variable tmp$un_id d_id;
begin
  RES_SCRIPT = '';
  TMP$DB_ID = F_GET_DB_ID();

  if (:TMP$DB_ID = 81001) then      -- База РФ
    TMP$NAME = 'CM_RU_';
  else if (:TMP$db_id in (39801, 41701)) then -- База КГ
    TMP$NAME = 'CM_KG_';
  else if (:TMP$DB_ID = 94401) then -- База АЗ
    TMP$NAME = 'CM_AZ_';
  else
    TMP$NAME = null;

  if (:TMP$NAME is not null) then
    for
      select
          cm.CDM_ID
          , cm.CDM_SQL_TMP
          , cm.CDM_MEMO
          , cm.CDM_OBJECT_KIND,
          cm.cdm_from_un_id
        from
          ConditionMain as cm
        where
              (cm.cdm_sql_tmp_check = 1)
        into
            :TMP$ID
          , :TMP$SQL
          , :TMP$MEMO
          , :TMP$KIND_STR
          , :TMP$UN_ID
      do
      begin
        if (TMP$UN_ID is null) then
        begin
          if (current_user <> 'SISLAMOV') then
            Continue;
        end else
        begin
          if (not exists(select * from username where un_id = :TMP$UN_ID and un_name = current_user)) then
            Continue;
        end

        TMP$J = position(ascii_char(13) || ascii_char(10) || 'begin' || ascii_char(13) || ascii_char(10), :TMP$SQL);
        TMP$I = position(ascii_char(13) || ascii_char(10) || 'declare ', :TMP$SQL);
        if ((:TMP$J = 0) or ((:TMP$I > 0) and (:TMP$I < :TMP$J))) then
          TMP$J = :TMP$I;
        TMP$SQL = right(:TMP$SQL, char_length(:TMP$SQL) - (:TMP$J - 1));
        TMP$SQL = replace(TMP$SQL, 'declare variable un_id d_id;', '');
        TMP$SQL = replace(TMP$SQL, 'declare variable un_id id_type;', '');
        TMP$SQL = replace(TMP$SQL, 'declare un_id d_id;', '');
        TMP$SQL = replace(TMP$SQL, 'declare un_id id_type;', '');

        TMP$PROC_NAME = :TMP$NAME || LPAD(:TMP$ID, 4, '0') || '_SQL';
        TMP$SQL = BLOB_APPEND('create or alter procedure ' || :TMP$PROC_NAME || ' ('  || ascii_char(13) || ascii_char(10) ||
                  IIF(TMP$KIND_STR='kontract', '    k_id d_id not null,'  || ascii_char(13) || ascii_char(10), '') ||
                  IIF(TMP$KIND_STR='miniapp',  '    ma_id d_id not null,' || ascii_char(13) || ascii_char(10), '') ||
                  '    check_kind d_str_code not null,'                   || ascii_char(13) || ascii_char(10) ||
                  '    un_id d_id not null)'                              || ascii_char(13) || ascii_char(10) ||
                  '  returns ('                                           || ascii_char(13) || ascii_char(10) ||
                  '    res_info_text d_str_100,'                          || ascii_char(13) || ascii_char(10) ||
                  '    res_error_text d_str_255,'                         || ascii_char(13) || ascii_char(10) ||
                  '    res_error_is_critical d_integer)'                  || ascii_char(13) || ascii_char(10) ||
                  'as',
                  TMP$SQL);
        execute statement :TMP$SQL;

        RES_SCRIPT = BLOB_APPEND(RES_SCRIPT, 'SET TERM ^ ;');
        RES_SCRIPT = BLOB_APPEND(RES_SCRIPT, '

');
        RES_SCRIPT = BLOB_APPEND(RES_SCRIPT, TMP$SQL);
        RES_SCRIPT = BLOB_APPEND(RES_SCRIPT, '^');
        RES_SCRIPT = BLOB_APPEND(RES_SCRIPT, '

');
        RES_SCRIPT = BLOB_APPEND(RES_SCRIPT, 'SET TERM ; ^');
        RES_SCRIPT = BLOB_APPEND(RES_SCRIPT, '

');

        TMP$PROC_NAME = 'COMMENT ON PROCEDURE ' || :TMP$PROC_NAME || ' IS ''' || tmp$memo || ''';';

        RES_SCRIPT = BLOB_APPEND(RES_SCRIPT, TMP$PROC_NAME);
        RES_SCRIPT = BLOB_APPEND(RES_SCRIPT, '

');
      end

  suspend;
end
