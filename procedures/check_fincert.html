CREATE PROCEDURE CHECK_FINCERT(
    MA_ID D_ID,
    K_ID D_ID,
    TAG D_STR_CODE)
RETURNS (
    RES D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable un_id d_id;
declare variable c_id d_id;
declare variable fcl_id d_id;
declare variable res_type d_str_20;
declare variable c_name_fio d_str_100;
declare variable c_birth_date d_date;
declare variable c_full_address d_str_255;
declare variable c_from_reg_id d_integer;
declare variable email_list d_str_255;
declare variable c_pass_no d_str_code;
declare variable asp_status d_str_code;
declare variable res_code d_str_code;
declare variable officer_un d_id;
begin
-- RFCRU-3955 автоматический отказ при совпадении со списками финцерт - номер паспорта, инн
  res = 0;
  res_text = '';

  if (k_id is not null) then
  begin
    select kc.kc_from_c_id, k_from_ma_app_id
    from kontract k
    inner join knclient kc on kc.kc_from_k_id = k.k_id
    where k.k_id = :k_id
    and kc.kc_amount_approved > 0
    into :c_id, ma_id;
  end
  else
  begin
    select mac.mac_from_c_id
    from miniappclient mac
    where mac.mac_from_ma_id = :ma_id
    and mac.mac_amount > 0
    into :c_id;
  end

  select un.un_id
  from miniapp ma
  inner join knaktlist kal on kal.kal_from_kf_id = ma.ma_from_kf_id
  inner join username un on kal.kal_new_kf_officer = un.un_from_c_id
  where ma_id = :ma_id
  and ma_date_end = '31.12.9999'
  and kal.is_delete = 0
  and current_date between kal_date and kal_date_end
  into :officer_un;

  insert into fincert_check_log (fcl_object_kind, fcl_object_id, fcl_datetime, fcl_tag)
  values (11, :ma_id, current_timestamp, f_enum('bp_check_kind', :tag, 'id'))
  returning fcl_id
  into fcl_id;

  for select res_type, res_code
  from hp_fincert_get_result(:c_id, null)
  where res = 1
  rows 1
  into res_type, res_code
  do
  begin
    res = 1;

    update fincert_check_log
    set fcl_fid = f_enum('fincert_file', :res_code, 'id')
    where fcl_id = :fcl_id;

    un_id = (select un_id
              from username
              where un_name = rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN'));
    if (un_id is null) then
        un_id = 1000000000000;

    select c_name_fio,
           c_pass_no,
           c_birth_date,
           adl_address_str,
           c_birth_date,
           c_from_reg_id
    from clients c, addrclient, addrlist
    where c.c_id = :c_id and
      c_type = 0 and
      adc_from_c_id = c_id and
      adc_address_type = 1 and
      adc_from_adl_id = adl_id
    into  c_name_fio,
          c_pass_no,
          c_birth_date,
          c_full_address,
          c_birth_date,
          c_from_reg_id;

    execute procedure hp_fincert_reject(:ma_id, :k_id);

    --смс клиенту с уведомлением об отказе - эл док
    asp_status = f_edoc_status('asp.agreement', :c_id, current_date);
    if (asp_status <> 'signed') then
      asp_status = f_edoc_status('asp.agreement_oep', :c_id, current_date);

    if (asp_status <> 'signed') then
      res_text = 'Нет подписанного договора АСП. ';
    else
    begin
      select res_text
      from hp_edoc_ins('loan.fincert.reject', f_objkind('miniapp', 'id'), :ma_id, 1000000000000)
      into res_text;
    end

    if (res_type = 'Номер телефона') then
    begin
      res_text = res_text || res_type || ' клиента ' || c_name_fio || ' в списках ФинЦЕРТ. Кредитование с данным номером запрещено';

      --список отказников
      if (not exists(
        select 1
        from moratory
        where (m_from_c_id = :c_id or
        (m_client_fio = :c_name_fio and
        m_client_birth_date = :c_birth_date)) and
        m_is_active = 1 and
        m_is_black = 1 and
        m_date_begin = current_date rows 1)) then
      begin
        insert into moratory (m_id, m_number, m_from_c_id, m_last_k_id, m_is_black, m_date_begin, m_date_end, m_month,
                              m_is_active, m_comment_group, m_comment_officer, m_comment_manager, m_client_fio,
                              m_client_pass_no, m_client_address, m_client_birth_date, m_our_clients, m_from_reg_id,
                              m_no, m_datetime, m_datetime_edit, m_user_create, m_user_edit, m_db_code, m_original_id)
        values (-255, 1, :c_id, null, 1, current_date, '31.12.9999', 0, 1, null, 'В заявке отказано в связи с совпадением со списком финЦЕРТ',
                null, :c_name_fio, :c_pass_no, :c_full_address, :c_birth_date, 1, :c_from_reg_id, null, current_timestamp,
                current_timestamp, null, null, null, null);
      end
    end
    else
    begin
      if (not exists(    --стоп-лист
        select 1
        from moratory
        where (m_from_c_id = :c_id or
        (m_client_fio = :c_name_fio and
        m_client_birth_date = :c_birth_date)) and
        m_is_active = 1 and
        m_is_black = 2 rows 1)) then
      begin
        insert into moratory (m_id, m_number, m_from_c_id, m_last_k_id, m_is_black, m_date_begin, m_date_end, m_month,
                              m_is_active, m_comment_group, m_comment_officer, m_comment_manager, m_client_fio,
                              m_client_pass_no, m_client_address, m_client_birth_date, m_our_clients, m_from_reg_id,
                              m_no, m_datetime, m_datetime_edit, m_user_create, m_user_edit, m_db_code, m_original_id)
        values (-255, 1, :c_id, null, 2, current_date, '31.12.9999', 0, 1, null, 'В заявке отказано в связи с совпадением со списком финЦЕРТ',
                null, :c_name_fio, :c_pass_no, :c_full_address, :c_birth_date, 1, :c_from_reg_id, null, current_timestamp,
                current_timestamp, null, null, null, null);
      end

      res_text = res_text || res_type || ' клиента ' || c_name_fio || ' в списках ФинЦЕРТ. Кредитование запрещено';
    end

    --пуш випсу
    if (officer_un is not null) then
          f_send_push('username', :officer_un, 'Уведомление об отказе', :res_text, 'check_fincert_new', 7, 1000000000000);

    --письмо комплайенс  RFCRU-4215
    email_list = ' fincert_client_match@molbulak.com';
    execute procedure hp_send_quick_message('email', email_list, 'Совпадение клиента со списком ФинЦерт',
      'Клиент ' || :c_name_fio ||
      ', дата рождения: ' ||  f_datetostr(c_birth_date) ||
      ', тип проверки: ' || res_type ||
      ', проверено: ' ||  f_datetostr(current_date),
       '', null, :un_id);
  end
  suspend;
end
