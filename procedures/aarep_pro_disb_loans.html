CREATE PROCEDURE AAREP_PRO_DISB_LOANS(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REP_BUS_NAME VARCHAR(255),
    REP_DISB_AMOUNT DECIMAL(18,4),
    REP_DISB_COUNT INTEGER,
    REP_DISB_1 DECIMAL(18,4),
    REP_DISB_2 DECIMAL(18,4),
    REP_DISB_3 DECIMAL(18,4),
    REP_DISB_4 DECIMAL(18,4),
    REP_DISB_5 DECIMAL(18,4),
    REP_DISB_6 DECIMAL(18,4),
    REP_DISB_7 DECIMAL(18,4),
    REP_DISB_8 DECIMAL(18,4),
    REP_PERC_ANNUAL_KOEF DECIMAL(18,7),
    REP_PERC_EFFECT_KOEF DECIMAL(18,7))
AS
declare variable bus_subtype integer;
declare variable k_perc_annu decimal(18,7);
declare variable p_perc_eff decimal(18,7);
declare variable rep_annual_perc decimal(18,4);
declare variable rep_effect_perc decimal(18,4);
begin
  for select rep_bus_name, sum(rep_disb_amount), sum(rep_disb_count), sum(rep_disb_1), sum(rep_disb_2), sum(rep_disb_3),
             sum(rep_disb_4), sum(rep_disb_5), sum(rep_disb_6), sum(rep_disb_7), sum(rep_disb_8), sum(rep_annual_perc),
             sum(rep_effect_perc)
      from aarep_pro_disb_loans_tmp(:rep_region_id, :rep_date_begin, :rep_date_end)
      group by rep_bus_name
      into :rep_bus_name, :rep_disb_amount, :rep_disb_count, :rep_disb_1, :rep_disb_2, :rep_disb_3, :rep_disb_4,
           :rep_disb_5, :rep_disb_6, :rep_disb_7, :rep_disb_8, :rep_annual_perc, :rep_effect_perc
  do
  begin
    if (rep_disb_amount <> 0) then
    begin
      rep_perc_annual_koef = cast(rep_annual_perc / rep_disb_amount as decimal(18,4));
      rep_perc_effect_koef = cast(rep_effect_perc / rep_disb_amount as decimal(18,4));
      rep_perc_annual_koef = rep_perc_annual_koef * 100.0;
      rep_perc_effect_koef = rep_perc_effect_koef * 100.0;
    end
    else
    begin
      rep_perc_annual_koef = 0;
      rep_perc_effect_koef = 0;
    end
    suspend;
  end
end
