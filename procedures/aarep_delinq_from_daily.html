CREATE PROCEDURE AAREP_DELINQ_FROM_DAILY(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    KF_OFFICER_NAME VARCHAR(100),
    KF_REGION_NAME VARCHAR(100),
    KF_NAME VARCHAR(100),
    CL_CASHIER_NAME VARCHAR(100),
    REP_DELIQ_OS CURR_TYPE,
    REP_DELIQ_PERC CURR_TYPE,
    REP_CALC_PERC CURR_TYPE,
    REP_ACTIVE_BALANCE CURR_TYPE,
    REP_DELIQ_BEGIN_DATE DATE,
    REP_DELIQ_DAYS_KOLICH INTEGER,
    AAREP_IS_RESTRUCTED INTEGER)
AS
declare variable REG_ID_PRIOR integer;
declare variable OKS_KC_ID ID_TYPE;
declare variable REG_ID integer;
declare variable K_ID ID_TYPE;
declare variable CASHIER_ID ID_TYPE;
declare variable OKS_KC_ID_PRIOR ID_TYPE;
declare variable K_FROM_KF_ID ID_TYPE;
begin
  reg_id_prior = 0;
  oks_kc_id_prior = 0;
  for select oks_kc_id, oks_region_id, k_id, k_clients_cashier_id, oks_balance, oks_delinq_date, k_from_kf_id
      from oksummary, kontract, region_list(:rep_region_id, 1, 3, 1)
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_region_id = id_list and
            oks_kc_id <> 1000000000003 and
            oks_from_k_id = k_id and
            oks_delinq_date <= :rep_date_end
            and oks_close_state = 0 
      order by oks_region_id, oks_kc_id
      into oks_kc_id, reg_id, k_id, cashier_id, rep_active_balance, rep_deliq_begin_date, k_from_kf_id
  do
  begin
    if (reg_id_prior <> reg_id) then
    begin
      reg_id_prior = reg_id;
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :reg_id
      into kf_region_name;
    end
    if (oks_kc_id_prior <> oks_kc_id) then
    begin
      oks_kc_id_prior = oks_kc_id;
      select c_name_fio
      from clients
      where c_id = :oks_kc_id
      into kf_officer_name;
    end
    rep_deliq_days_kolich = rep_date_end - rep_deliq_begin_date;
    aarep_is_restructed = 0;
    if (exists(select *
               from operrest, operlist
               where or_from_k_id = :k_id and
                     or_from_ol_id = ol_id and
                     operlist.is_delete = 0 and
                     ol_date <= :rep_date_end and
                     or_date_end >= :rep_date_end)) then
      aarep_is_restructed = 1;

    rep_deliq_os = null;
    rep_deliq_perc = null;
    rep_calc_perc = null;
    select sum(iif(sh_date <= :rep_date_end, zad_os, 0)), sum(iif(sh_date <= :rep_date_end, zad_perc, 0)),
           sum(sh_calc_perc)
    from aarep_make_zad_list(:k_id, :rep_date_end, 1)
    where sh_date is not null
    into rep_deliq_os, rep_deliq_perc, rep_calc_perc;
    rep_deliq_os = coalesce(rep_deliq_os, 0);
    rep_deliq_perc = coalesce(rep_deliq_perc, 0);
    rep_calc_perc = coalesce(rep_calc_perc, 0);

    select kal_new_kf_name
    from knaktlist
    where kal_from_kf_id = :k_from_kf_id and
          :rep_date_end between kal_date and kal_date_end and
          knaktlist.is_delete = 0
    into kf_name;

    select c_name_fio
    from clients
    where c_id = :cashier_id
    into cl_cashier_name;

    suspend;
  end
end
