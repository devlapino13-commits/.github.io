CREATE PROCEDURE HP_RETURN_CONSULT_OPER_SAVE(
    UN_ID D_ID,
    REG_ID D_INTEGER,
    OPER_DATE D_DATE,
    K_ID D_ID,
    C_ID D_ID)
RETURNS (
    ERR_MSG D_STR_100,
    OL_ID D_ID)
AS
declare variable reg_id_tmp d_integer;
declare variable k_disb_date d_date;
declare variable kontract_no d_str_20;
declare variable sum_ovp d_currency;
declare variable arc_id d_integer;
declare variable status d_smallint;
begin

  ol_id = 0;

  select arc.arc_id, arci.arci_consult_office_id, k.k_kontract_no, arci.arci_amount, k.k_disb_date, arc.arc_status
  from app_return_consult arc
  inner join app_return_consult_info arci on arc.arc_id = arci.arci_id
  inner join kontract k on k.k_id = arc.arc_from_k_id
  where arc.arc_from_c_id = :c_id and
    arc.arc_from_k_id = :k_id and
    arc.arc_status <> -2
  into :arc_id, :reg_id_tmp, :kontract_no, :sum_ovp, :k_disb_date, :status;

  select err_msg
  from hp_return_consult_oper_check(:status)
  into :err_msg;

  if (:err_msg <> '') then
    begin
        suspend;
        exit;
    end

  select err_msg, coalesce(new_ol_id, 0)
  from hp_save_return_money_oper('consultation', :un_id, :reg_id, :reg_id_tmp, :k_disb_date, :kontract_no, :oper_date, :c_id, :sum_ovp, :k_id)
  into err_msg, ol_id;

  if (err_msg = '' and ol_id > 0) then
  begin
    update app_return_consult
    set arc_status = f_enum('app_return_consult', 'done', 'id'),
        arc_status_un_id = :un_id,
        arc_status_datetime = current_timestamp
    where (arc_id = :arc_id);
  end

  suspend;
end
