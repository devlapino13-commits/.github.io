CREATE PROCEDURE HP_COMPET_KG_2021_SUPERP(
    C_ID D_ID,
    REG_ID D_ID,
    REP_PERIOD D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    IN_SORTING_ORDER D_SMALLINT = 0)
RETURNS (
    SUPER_POKAZATEL D_PERCENT,
    SUPER_POKAZATEL_TXT D_STR_255,
    ROWS_DESC D_SMALLINT)
AS
declare variable lo_disb_sum_ipf_per d_currency;
declare variable lo_portf_ipf_curr d_currency;
declare variable lo_portf_ipf_prv d_currency;
declare variable lo_portf_ipf_diff d_currency;
declare variable lo_portf_ipf_prin d_currency;
declare variable last_day_prv_month d_date;
declare variable sp_type d_smallint;
declare variable lo_par30_perc_begin decimal(15,4);
declare variable lo_par30_perc_end decimal(15,4);
declare variable lo_par30_perc decimal(15,4);
declare variable reg_par30_begin decimal(15,4);

declare variable reg_balance_end decimal(15,4);

declare variable lo_portf_prv decimal(15,2);

declare variable lo_portf_diff decimal(15,2);
declare variable kol_disb decimal(15,2);

declare variable kc_amount_approved decimal(15,2);

    inner join product pr on pr.pr_id = kn.k_from_pr_id and
          pr.pr_code containing '.IPF.'
    where :rep_date_end between oks.oks_date_begin and oks.oks_date_end and
          oks_close_state = 0 and
          ((:sp_type = 0 and
          oks.oks_kc_id = :c_id) or
          (:sp_type = 1 and
          oks.oks_region_id = :reg_id))

          (:sp_type = 1 and

          ks_state = 1 and

          ks.ks_kc_id = :c_id) or
          (:sp_type = 1 and
          ks.ks_region_id = :reg_id))

    from oksummary oks
    inner join kontract kn on kn.k_id = oks.oks_from_k_id

          pr.pr_code containing '.IPF.'

          oks.oks_kc_id = :c_id) or
          (:sp_type = 1 and
          oks.oks_region_id = :reg_id))

          pr.pr_code containing '.IPF.'

    inner join knaktlist kal on kf.kf_id = kal.kal_from_kf_id and

          kal.kal_date between :rep_date_begin and :rep_date_end and
          kal.kal_new_kf_officer <> kal.kal_prior_lo_id

          ka.ka_id <> 0 and
          ka.is_delete = 0

          oks_close_state = 0

    begin
      select coalesce(sum(iif(:rep_date_end between oks_date_begin and oks_date_end and oks_close_state = 0 and oks_is_delinq = 0, oks_balance, 0)), 0),
             coalesce(sum(iif(:last_day_prv_month between oks_date_begin and oks_date_end and oks_close_state = 0 and oks_is_delinq = 0, oks_balance, 0)), 0)
      from kontract
      inner join product pr on pr_id = k_from_pr_id and
            pr_code not containing '.IPF.' and
            pr.pr_id > 0
      inner join oksummary on oks_from_k_id = k_id
      where k_fact_disb_reg_id = :reg_id

    inner join product pr on pr.pr_id = k.k_from_pr_id and

          ks.ks_state = 1 and

             coalesce(sum(iif(:last_day_prv_month between oks_date_begin and oks_date_end and oks_close_state = 0 and oks_is_delinq = 0, oks_balance, 0)), 0)
      from kontract
      inner join oksummary on oks_from_k_id = k_id

      super_pokazatel = lo_portf_ipf_curr - lo_portf_ipf_prv;
    end
  end
  else
  if (rep_period = 202208) then
  begin
    lo_portf_ipf_prv = 0;
    lo_portf_ipf_curr = 0;
    super_pokazatel_txt = 'Супер-показатель за август: Прирост по ИПФ продуктам без просрочек';
    if (:sp_type = 0) then
    begin
      select coalesce(sum(iif(:rep_date_end between oks_date_begin and oks_date_end and oks_close_state = 0 and oks_is_delinq = 0, oks_balance, 0)), 0),
             coalesce(sum(iif(:last_day_prv_month between oks_date_begin and oks_date_end and oks_close_state = 0 and oks_is_delinq = 0, oks_balance, 0)), 0)
      from kontract
      inner join oksummary on oks_from_k_id = k_id
      inner join product pr on pr.pr_id = k_from_pr_id and
            pr.pr_id > 0 and
            pr.pr_code containing '.ipf.'
      where k_fact_disb_lo_id = :c_id

      select coalesce(sum(iif(:rep_date_end between oks_date_begin and oks_date_end and oks_close_state = 0 and oks_is_delinq = 0, oks_balance, 0)), 0),

      inner join oksummary on oks_from_k_id = k_id

    kol_disb_tmp = coalesce(kol_disb_ipf, 0) + coalesce(kol_disb_classic, 0);

          k.k_fact_disb_lo_id = :c_id

    select count(kn.k_id) kol_disb_tmp
    from kontstate ks
    inner join kontract kn on kn.k_id = ks.ks_from_k_id
    inner join product pr on pr.pr_id = kn.k_from_pr_id and

    lo_balance_end = 0;
    lo_par30_begin = 0;
    lo_par30_end = 0;
    for select coalesce(oks_balance, 0), coalesce(iif(:rep_date_begin - oks_delinq_date > 30, oks_balance, 0), 0),

              oks_balance > 0 and
              oks_close_state = 0 and
              oks_from_k_id = k_id and
              oks_kc_id = :c_id and
              oks_kc_id = k_fact_disb_lo_id
        into k_balance_begin, k_balance_par30_begin, kf_id
    do
    begin
/*      if (exists(select *
                 from knaktlist kal
                 where kal.kal_new_kf_officer <> kal.kal_prior_lo_id and
                       kal.kal_prior_lo_id = :c_id and
                       kal.kal_from_kf_id = :kf_id)) then

      if (exists(select *
                 from knaktlist kal

                       kal.kal_from_kf_id = :kf_id)) then
        continue;  */
      :lo_balance_begin = :lo_balance_begin + k_balance_begin;
      :lo_par30_begin = :lo_par30_begin + k_balance_par30_begin;

    end

    :lo_par30_perc_begin = iif(:lo_balance_begin <> 0, (:lo_par30_begin / :lo_balance_begin) * 100, 0);

    for select coalesce(oks_balance, 0), coalesce(iif(:rep_date_end - oks_delinq_date > 30, oks_balance, 0), 0),
               k_from_kf_id
        from oksummary, kontract
        where :rep_date_end between oks_date_begin and oks_date_end and
              oks_balance > 0 and
              oks_close_state = 0 and
              oks_from_k_id = k_id and
              oks_kc_id = :c_id  and
              oks_kc_id = k_fact_disb_lo_id
        into k_balance_end, k_balance_par30_end, kf_id
    do
    begin
    /*  if (exists(select *

                       kal.kal_prior_lo_id = :c_id and
                       kal.kal_from_kf_id = :kf_id)) then
        continue;
      if (exists(select *
                 from knaktlist kal
                 where kal.kal_new_kf_officer <> kal.kal_prior_lo_id and
                       kal.kal_new_kf_officer = :c_id and
                       kal.kal_from_kf_id = :kf_id)) then
        continue;   */
      :lo_balance_end = :lo_balance_end + k_balance_end;
      :lo_par30_end = :lo_par30_end + k_balance_par30_end;

    end
    :lo_par30_perc_end = iif(:lo_balance_end <> 0, (:lo_par30_end / :lo_balance_end) * 100, 0);
    :lo_par30_perc = :lo_par30_perc_end - :lo_par30_perc_begin;
    super_pokazatel = coalesce(:lo_par30_perc, 0);
  end
  else
  if (rep_period = 202108) then
  begin
    super_pokazatel_txt = 'Супер-показатель за август: Снижение ПАР30 в %';

    lo_par30_perc_begin = 0;
    lo_par30_perc_end = 0;
    lo_par30_perc = 0;

    begin
      reg_par30_begin = 0;
      par30_perc_end = 0;
      k_balance_par30_begin = 0;
      k_balance_par30_end = 0;
      par30_perc = 0;
      par30_perc_begin = 0;
      k_balance_end = 0;
      k_balance_begin = 0;
      reg_balance_end = 0;
      reg_balance_begin = 0;

      select sum(oks_balance), sum(iif(:rep_date_begin - oks_delinq_date > 30, oks_balance, 0))
      from oksummary, region_list(:reg_id, 1, 3, 1), kontract, product

            :rep_date_begin between oks_date_begin and oks_date_end and

      where cast(oks_region_id as integer) = id_list and
            :rep_date_end between oks_date_begin and oks_date_end and
            oks_from_k_id = k_id and
            k_from_pr_id = pr_id and
            oks_balance > 0 and
            oks_close_state = 0
      into reg_balance_end, reg_par30_end;

                oks_close_state = 0 and

                oks_kc_id = :c_id
          into k_balance_begin, k_balance_par30_begin, kf_id

      begin
        if (exists(select *
                   from knaktlist kal
                   where kal.kal_new_kf_officer <> kal.kal_prior_lo_id and
                         kal.kal_prior_lo_id = :c_id and
                         kal.kal_from_kf_id = :kf_id)) then
          continue;
        if (exists(select *
                   from knaktlist kal
                   where kal.kal_new_kf_officer <> kal.kal_prior_lo_id and
                         kal.kal_new_kf_officer = :c_id and
                         kal.kal_from_kf_id = :kf_id)) then

        :lo_par30_begin = :lo_par30_begin + k_balance_par30_begin;

      for select coalesce(oks_balance, 0), coalesce(iif(:rep_date_end - oks_delinq_date > 30, oks_balance, 0), 0),
                 k_from_kf_id
          from oksummary, kontract
          where :rep_date_end between oks_date_begin and oks_date_end and
                oks_balance > 0 and
                oks_close_state = 0 and
                oks_from_k_id = k_id and
                oks_kc_id = :c_id
          into k_balance_end, k_balance_par30_end, kf_id
      do
      begin
        if (exists(select *
                   from knaktlist kal
                   where kal.kal_new_kf_officer <> kal.kal_prior_lo_id and
                         kal.kal_prior_lo_id = :c_id and
                         kal.kal_from_kf_id = :kf_id)) then
          continue;
        if (exists(select *
                   from knaktlist kal
                   where kal.kal_new_kf_officer <> kal.kal_prior_lo_id and
                         kal.kal_new_kf_officer = :c_id and
                         kal.kal_from_kf_id = :kf_id)) then
          continue;
        :lo_balance_end = :lo_balance_end + k_balance_end;
        :lo_par30_end = :lo_par30_end + k_balance_par30_end;

      :lo_par30_perc_end = iif(:lo_balance_end <> 0, (:lo_par30_end / :lo_balance_end) * 100, 0);
      --  :lo_par30_perc_end = (:lo_par30_end / :lo_balance_end) * 100;
      :lo_par30_perc = :lo_par30_perc_end - :lo_par30_perc_begin;

      super_pokazatel = coalesce(:lo_par30_perc, 0);
    end
  end
  if (:rep_period = 202109) then

    super_pokazatel_txt = 'Супер-показатель за сентябрь: Выдачи в штуках суммарно по ИПФ и Классике';

      from kontstate ks

            k.k_fact_disb_lo_id = :c_id
      where ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and -- говорит о том что кредит активный
            ks.ks_date_begin between :rep_date_begin and :rep_date_end
      into kol_disb_tmp;
      super_pokazatel = coalesce(:kol_disb_tmp, 0);
    end
