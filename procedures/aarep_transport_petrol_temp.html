CREATE PROCEDURE AAREP_TRANSPORT_PETROL_TEMP(
    REP_REGION_ID INTEGER,
    TRANS_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    PED_ID INTEGER,
    PED_DATE DATE,
    PED_REGION_PLACE_ID INTEGER,
    PED_REGION_BALANCE_ID INTEGER,
    PED_FROM_C_ID DECIMAL(18,0),
    PED_EXPENSE_SUMMER DECIMAL(18,4),
    PED_EXPENSE_WINTER DECIMAL(18,4),
    PED_EXPENSE_SUMMER_MOUNT DECIMAL(18,4),
    PED_EXPENSE_WINTER_MOUNT DECIMAL(18,4),
    PED_ACC_BALANCE VARCHAR(20),
    PED_ACC_NUMBER INTEGER,
    PED_IS_ACTIVE INTEGER,
    PET_ID INTEGER,
    PET_NAME VARCHAR(255),
    PET_LABEL VARCHAR(50),
    PET_IS_RENT INTEGER,
    PET_NORMA_YEDIZ INTEGER,
    PET_NORMA_KM INTEGER,
    PED_CLIENT_FIO VARCHAR(100),
    PED_PLACE_NAME VARCHAR(100),
    PED_SALDO_LITER DECIMAL(18,4),
    PED_SALDO_SUMMA DECIMAL(18,4),
    PED_FILL_LITER DECIMAL(18,4),
    PED_FILL_LITER_TALON DECIMAL(18,4),
    PED_FILL_SUMMA DECIMAL(18,4),
    PET_NORMA_NAME VARCHAR(20),
    PED_WRITEOFF_LITER DECIMAL(18,4),
    PED_WRITEOFF_SUMMA DECIMAL(18,4),
    PED_WRITEOFF_KM DECIMAL(18,4),
    PED_SALDO_END_LITER DECIMAL(18,4),
    PED_SALDO_END_SUMMA DECIMAL(18,4))
AS
declare variable reg integer;
declare variable st_staff integer;
declare variable st_client integer;
declare variable db_id integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  :db_id = f_get_db_id();

  if (rep_region_id is null) then
    rep_region_id = 1000000;
  if (rep_region_id = 0) then
    rep_region_id = 1000000;
  if (trans_id is null) then
    trans_id = 0;
  for select pet_id, pet_name, pet_label, pet_is_rent, pet_norma_yediz, pet_norma_km
      from pettrans
      where ((pet_id = :trans_id) or
            (:trans_id = 0))
      into :pet_id, :pet_name, :pet_label, :pet_is_rent, :pet_norma_yediz, :pet_norma_km
  do
  begin
    for select ped_id, ped_date, ped_region_place_id, ped_from_c_id, ped_expense_summer, ped_expense_winter,
               ped_expense_summer_mount, ped_expense_winter_mount, ped_acc_balance, ped_acc_number, ped_is_active,
               c_name_fio, reg_code || ' ' || reg_name
        from petdata, clients, region
        where ped_from_pet_id = :pet_id and
              ped_date <= :rep_date_end and
              c_id = ped_from_c_id and
              reg_unicode = ped_region_place_id
        order by ped_date desc
        rows 1
        into :ped_id, :ped_date, :ped_region_place_id, :ped_from_c_id, :ped_expense_summer, :ped_expense_winter,
             :ped_expense_summer_mount, :ped_expense_winter_mount, :ped_acc_balance, :ped_acc_number, :ped_is_active,
             :ped_client_fio, :ped_place_name
    do
    begin
      reg = null;
      select id_list
      from region_orig_list(:rep_region_id, 1, -1, 1)
      where id_list = :ped_region_place_id
      into :reg;
      if (reg is not null) then
      begin

        if (pet_norma_yediz = 0) then
          pet_norma_name = 'км';
        if (pet_norma_yediz = 1) then
          pet_norma_name = 'час';
        if (pet_norma_yediz = 2) then
          pet_norma_name = 'миль';

        ped_saldo_liter = 0;
        ped_saldo_summa = 0;
        ped_fill_liter = 0;
        ped_fill_liter_talon = 0;
        ped_fill_summa = 0;
        ped_writeoff_liter = 0;
        ped_writeoff_summa = 0;
        ped_writeoff_km = 0;
        ped_saldo_end_liter = 0;
        ped_saldo_end_summa = 0;

        if (db_id <> 94401) then
        begin
          for select reg_head_office, sum(peo_trans_liter), sum(peo_trans_summa)
              from petopers, operlist, region
              where peo_from_ol_id = ol_id and
                    operlist.is_delete = 0 and
                    ol_date < :rep_date_begin and
                    peo_from_pet_id = :pet_id and
                    ol_from_reg_id = reg_unicode
              group by reg_head_office
              into :ped_region_balance_id, :ped_saldo_liter, :ped_saldo_summa
          do
          begin
            if (ped_saldo_liter is null) then
              ped_saldo_liter = 0;
            if (ped_saldo_summa is null) then
              ped_saldo_summa = 0;

            suspend;
          end

          ped_saldo_liter = 0;
          ped_saldo_summa = 0;
          ped_fill_liter = 0;
          ped_fill_liter_talon = 0;
          ped_fill_summa = 0;
          ped_writeoff_liter = 0;
          ped_writeoff_summa = 0;
          ped_writeoff_km = 0;
          ped_saldo_end_liter = 0;
          ped_saldo_end_summa = 0;

          for select reg_head_office, sum(peo_trans_liter), sum(peo_talon_liter), sum(peo_trans_summa)
              from petopers, operlist, region
              where peo_from_ol_id = ol_id and
                    operlist.is_delete = 0 and
                    ol_date_oper_day between :rep_date_begin and :rep_date_end and
                    peo_from_pet_id = :pet_id and
                    peo_oper_type <> 7 and
                    reg_unicode = cast(ol_from_reg_id as integer)
              group by reg_head_office
              into :ped_region_balance_id, :ped_fill_liter, :ped_fill_liter_talon, :ped_fill_summa
          do
          begin
            if (ped_fill_liter is null) then
              ped_fill_liter = 0;
            if (ped_fill_liter_talon is null) then
              ped_fill_liter_talon = 0;
            if (ped_fill_summa is null) then
              ped_fill_summa = 0;
            ped_fill_liter = ped_fill_liter - ped_fill_liter_talon;
            suspend;
          end

          ped_saldo_liter = 0;
          ped_saldo_summa = 0;
          ped_fill_liter = 0;
          ped_fill_liter_talon = 0;
          ped_fill_summa = 0;
          ped_writeoff_liter = 0;
          ped_writeoff_summa = 0;
          ped_writeoff_km = 0;
          ped_saldo_end_liter = 0;
          ped_saldo_end_summa = 0;

          for select reg_head_office, sum(-1 * peo_trans_liter - peo_talon_liter), sum(-1 * peo_trans_summa),
                     sum(peo_probeg_fact_km)
              from petopers, operlist, region
              where peo_from_ol_id = ol_id and
                    operlist.is_delete = 0 and
                    ol_date_oper_day between :rep_date_begin and :rep_date_end and
                    peo_from_pet_id = :pet_id and
                    peo_oper_type = 7 and
                    cast(ol_from_reg_id as integer) = reg_unicode
              group by reg_head_office
              into :ped_region_balance_id, :ped_writeoff_liter, :ped_writeoff_summa, :ped_writeoff_km
          do
          begin
            if (ped_writeoff_liter is null) then
              ped_writeoff_liter = 0;
            if (ped_writeoff_summa is null) then
              ped_writeoff_summa = 0;
            if (ped_writeoff_km is null) then
              ped_writeoff_km = 0;
            suspend;
          end
        end
        else
        begin
          for select reg_unicode, sum(peo_trans_liter), sum(peo_trans_summa)
              from petopers, operlist, region
              where peo_from_ol_id = ol_id and
                    operlist.is_delete = 0 and
                    ol_date < :rep_date_begin and
                    peo_from_pet_id = :pet_id and
                    ol_from_reg_id = reg_unicode
              group by reg_unicode
              into :ped_region_balance_id, :ped_saldo_liter, :ped_saldo_summa
          do
          begin
            if (ped_saldo_liter is null) then
              ped_saldo_liter = 0;
            if (ped_saldo_summa is null) then
              ped_saldo_summa = 0;

            suspend;
          end

          ped_saldo_liter = 0;
          ped_saldo_summa = 0;
          ped_fill_liter = 0;
          ped_fill_liter_talon = 0;
          ped_fill_summa = 0;
          ped_writeoff_liter = 0;
          ped_writeoff_summa = 0;
          ped_writeoff_km = 0;
          ped_saldo_end_liter = 0;
          ped_saldo_end_summa = 0;

          for select reg_unicode, sum(peo_trans_liter), sum(peo_talon_liter), sum(peo_trans_summa)
              from petopers, operlist, region
              where peo_from_ol_id = ol_id and
                    operlist.is_delete = 0 and
                    ol_date_oper_day between :rep_date_begin and :rep_date_end and
                    peo_from_pet_id = :pet_id and
                    peo_oper_type <> 7 and
                    reg_unicode = cast(ol_from_reg_id as integer)
              group by reg_unicode
              into :ped_region_balance_id, :ped_fill_liter, :ped_fill_liter_talon, :ped_fill_summa
          do
          begin
            if (ped_fill_liter is null) then
              ped_fill_liter = 0;
            if (ped_fill_liter_talon is null) then
              ped_fill_liter_talon = 0;
            if (ped_fill_summa is null) then
              ped_fill_summa = 0;
            ped_fill_liter = ped_fill_liter - ped_fill_liter_talon;
            suspend;
          end

          ped_saldo_liter = 0;
          ped_saldo_summa = 0;
          ped_fill_liter = 0;
          ped_fill_liter_talon = 0;
          ped_fill_summa = 0;
          ped_writeoff_liter = 0;
          ped_writeoff_summa = 0;
          ped_writeoff_km = 0;
          ped_saldo_end_liter = 0;
          ped_saldo_end_summa = 0;

          for select reg_unicode, sum(-1 * peo_trans_liter - peo_talon_liter), sum(-1 * peo_trans_summa),
                     sum(peo_probeg_fact_km)
              from petopers, operlist, region
              where peo_from_ol_id = ol_id and
                    operlist.is_delete = 0 and
                    ol_date_oper_day between :rep_date_begin and :rep_date_end and
                    peo_from_pet_id = :pet_id and
                    peo_oper_type = 7 and
                    cast(ol_from_reg_id as integer) = reg_unicode
              group by reg_unicode
              into :ped_region_balance_id, :ped_writeoff_liter, :ped_writeoff_summa, :ped_writeoff_km
          do
          begin
            if (ped_writeoff_liter is null) then
              ped_writeoff_liter = 0;
            if (ped_writeoff_summa is null) then
              ped_writeoff_summa = 0;
            if (ped_writeoff_km is null) then
              ped_writeoff_km = 0;
            suspend;
          end
        end
        ped_saldo_end_liter = :ped_saldo_liter + :ped_fill_liter + :ped_fill_liter_talon - :ped_writeoff_liter;
        ped_saldo_end_summa = :ped_saldo_summa + :ped_fill_summa - :ped_writeoff_summa;
      end
    end
  end
end
