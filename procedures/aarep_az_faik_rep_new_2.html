CREATE PROCEDURE AAREP_AZ_FAIK_REP_NEW_2(
    REP_DATE D_DATE)
RETURNS (
    BALANCE_ACCOUNT_OS VARCHAR(100),
    REG_NAME VARCHAR(100),
    K_KONTRACT_NO VARCHAR(20),
    KAL_NEW_KF_NAME VARCHAR(100),
    DISB_KS VARCHAR(100),
    CURR_KS VARCHAR(100),
    C_NAME_FIO VARCHAR(100),
    K_FACT_DISB_DATE D_DATE,
    K_KREDLINE_END_PRIMARY DATE,
    K_AMOUNT_APPROVED DECIMAL(18,4),
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    K_PERIOD DECIMAL(15,2),
    K_PERCENT DECIMAL(18,2),
    KD_PERCENT DECIMAL(18,2),
    KC_LOAN_TARGET D_STR_255,
    K_PERCENT_EFFECT DECIMAL(18,2),
    K_PERCENT_AFTER DECIMAL(18,2),
    PR_NAME VARCHAR(50),
    PR_LOAN_TYPE VARCHAR(100),
    BUSUBTYPE VARCHAR(100),
    ZAD_OS CURR_TYPE,
    ZAD_PERC CURR_TYPE,
    LK_PERCENT PERC_TYPE,
    LLC_IND_PERCENT PERC_TYPE,
    RPPU_BALANCE CURR_TYPE,
    RPPU_CALCPERC CURR_TYPE,
    RPPU_ALL_SUM CURR_TYPE,
    LLC_NAME D_STR_100,
    IS_RESTRUCT_STR D_STR_10,
    RESTRUCT_KOL INTEGER,
    REP_ACTIVE_BALANCE_NEW D_CURRENCY,
    LK_PERCENT_PERT STR_100,
    RESTRUCT_DATE DATE,
    K_DATE_END D_DATE,
    KONT_STATUS STR_100,
    OKS_DELINQ_DATE_KOL D_INTEGER,
    REP_DELINQ_BEGIN D_DATE,
    DELINQ_OS CURR_TYPE,
    DELINQ_PERC CURR_TYPE,
    FINE CURR_TYPE,
    ZAL_STR STR_10,
    Z_SUBTYPE_TEXT VARCHAR(50),
    COST_MARKET CURR_TYPE,
    POR_QTY_STR D_STR_100,
    REP_WRITEOFF_DATE DATE,
    OKO_SH_OS CURR_TYPE,
    OKO_SH_PERC CURR_TYPE,
    OKF_ZAD_OS CURR_TYPE,
    OKF_PERC CURR_TYPE,
    OKF_ZAD_TOTAL CURR_TYPE,
    LAM_DATE_RESULT DATE,
    LAM_DATE_ACTIVATE DATE,
    LAM_COST_ALL CURR_TYPE)
AS
declare variable kont_id id_type;
declare variable v_zal_subtype d_str_100;
declare variable disb_ol_id id_type;
declare variable sh_date date;
declare variable zad_os_tmp curr_type;
declare variable sh_calc_perc_tmp curr_type;
declare variable sh_calc_fine_tmp curr_type;
declare variable is_restruct smallint;
declare variable max_date_graph d_date;
declare variable max_date_rest d_date;
declare variable por_qty integer;
declare variable restruct_date_last d_date;
declare variable db_id d_integer;
declare variable zal_subtype d_str_100;
declare variable kd_percent_str d_str_20;
declare variable z_type integer;
declare variable c_id id_type;
declare variable v_cost_market curr_type;
declare variable lam_datetime timestamp;
begin
  DB_ID = F_GET_DB_ID();

  --расматриваем все активне займы и займы на внебалансе
  for select (OKS_FROM_K_ID),
                            case OKS_CLOSE_STATE
                              when 1 then 'спец. учёт'
                              when 0 then 'активный'
                              when 2 then 'закрытый'
                              else 'ошибка'
                            end
      from OKSUMMARY, KONTRACT
      where OKS_DATE_BEGIN between '01.01.1900' and :REP_DATE and
            OKS_DATE_END between(:REP_DATE) and '31.12.9999' and
            OKS_FROM_K_ID = K_ID and
            OKS_CLOSE_STATE in (0, 1)
            --and OKS_FROM_K_ID = 101010003863700101
      into :KONT_ID, :KONT_STATUS
  do
  begin
    for select KC.KC_FROM_C_ID, KONT.K_KONTRACT_NO, -- № контракта
               KAL.KAL_NEW_KF_NAME, -- № кр. Папки
               CURR_KS.C_NAME_FIO, -- Текущий КС
               KONT.K_AMOUNT_APPROVED, -- Сумма выдачи
               KONT.K_PERCENT, -- % ставка кредита
               KONT.K_PERCENT_EFFECT, -- FIFD (ЭПС) при выдаче
               PROD.PR_NAME, -- кр. продукт
               iif(PR_LOAN_TYPE = 1, 'Индивидуальный', 'Групповой'), KONT.K_PERIOD -- тип кред. продукта  */
        from KONTRACT KONT
        inner join KNCLIENT KC on KC.KC_FROM_K_ID = KONT.K_ID
        inner join KNAKTLIST KAL on KAL.KAL_FROM_KF_ID = KONT.K_FROM_KF_ID
        left join CLIENTS CURR_KS on CURR_KS.C_ID = KAL.KAL_NEW_KF_OFFICER
        inner join PRODUCT PROD on PROD.PR_ID = KONT.K_FROM_PR_ID
        where :REP_DATE between KAL.KAL_DATE and KAL.KAL_DATE_END and
              KAL.IS_DELETE = 0 and
              KONT.K_ID = :KONT_ID and
              KC.KC_AMOUNT_APPROVED > 0
        into :C_ID, :K_KONTRACT_NO, :KAL_NEW_KF_NAME, :CURR_KS, :K_AMOUNT_APPROVED, :K_PERCENT, :K_PERCENT_EFFECT,
             :PR_NAME, :PR_LOAN_TYPE, :K_PERIOD
    do
    begin

      select CLNT.C_NAME_FIO, -- ФИО клиента
             REFD.RD_NAME, -- Цель кредита
             BST.BS_NAME, --Тип бизнеса
             KNCL.KC_AMOUNT_APPROVED
      from KNCLIENT KNCL
      inner join CLIENTS CLNT on CLNT.C_ID = KNCL.KC_FROM_C_ID
      left join REFDATA REFD on REFD.RD_ID = KNCL.KC_LOAN_TARGET
      left join BUSSUBTYPE BST on BST.BS_ID = KC_BUSUNESS_SUBTYPE
      where KNCL.KC_FROM_K_ID = :KONT_ID and
            KNCL.KC_FROM_C_ID = :C_ID and
            KNCL.KC_AMOUNT_APPROVED > 0
      into :C_NAME_FIO, :KC_LOAN_TARGET, :BUSUBTYPE, :KC_AMOUNT_APPROVED;

      K_PERCENT_AFTER = 0; -- Текущая Эффективная % ставка по кредиту
      select round(KD_PERCENT_PER_DAY * 365, 2) K_PERCENT
      from KONTDATA KD
      where KD.KD_OL_IS_DELETE = 0 and
            KD.KD_FROM_K_ID = :KONT_ID and
            :REP_DATE between KD_OL_DATE and KD.KD_DATE_END
      into K_PERCENT_AFTER;

      select DISB_KS.C_NAME_FIO, -- Выдавший КС
             KS.KS_DATE_BEGIN, -- Дата выдачи
             KS.KS_FROM_OL_ID, REG.REG_NAME || ' ' || REG.REG_CODE -- Филиал (Офис) -- Офис выдачи кредита
      from KONTSTATE KS
      left join CLIENTS DISB_KS on DISB_KS.C_ID = KS.KS_KC_ID
      inner join REGION REG on REG.REG_UNICODE = KS.KS_REGION_ID
      where KS.KS_OL_IS_DELETE = 0 and
            KS.KS_STATE = 1 and
            KS.KS_FROM_K_ID = :KONT_ID
      into :DISB_KS, :K_FACT_DISB_DATE, :DISB_OL_ID, :REG_NAME;

      OKS_DELINQ_DATE_KOL = 0;

      select KD.KD_PERCENT --Текущая% ставка кредита
      from KONTDATA KD
      where KD.KD_FROM_K_ID = :KONT_ID and
            KD.KD_OL_DATE <= :REP_DATE
      order by KD_OL_DATE desc rows 1
      into :KD_PERCENT;

      if (KONT_STATUS = 'Активный') then
      begin
        select :REP_DATE - OKS.OKS_DELINQ_DATE -- Кол-во проср. Дней
        from OKSUMMARY OKS
        where OKS.OKS_FROM_K_ID = :KONT_ID and
              :REP_DATE between OKS.OKS_DATE_BEGIN and OKS.OKS_DATE_END and
              OKS_CLOSE_STATE = 0
        into :OKS_DELINQ_DATE_KOL;
      end

      REP_DELINQ_BEGIN = null;
      select OK_SH_DATE --Дата нач. просрочки
      from OPERKRED
      where OK_FROM_K_ID = :KONT_ID and
            OK_OL_IS_ACTIVE = 1 and
            OK_TYPE < 7 and
            OK_OL_DATE <= :REP_DATE and
            OK_SH_DATE <= :REP_DATE and
            OK_FROM_C_ID = 0
      group by OK_FROM_K_ID, OK_SH_DATE
      having sum(OK_OS + OK_SH_OS + OK_PERC + OK_SH_PERC) > 1
      order by OK_FROM_K_ID, OK_SH_DATE rows 1
      into :REP_DELINQ_BEGIN;

      if (OKS_DELINQ_DATE_KOL < 0) then
        OKS_DELINQ_DATE_KOL = 0;

      ZAD_OS = 0; -- Задолж. по ОС
      ZAD_PERC = 0; -- Накопленные  %
      select sum(OKO_SH_OS + OKO_OS), sum(OKO_SH_PERC + OKO_PERC)
      from OPERKROFF, OPERLIST
      where OKO_FROM_OL_ID = OL_ID and
            OKO_FROM_K_ID = :KONT_ID and
            OKO_FROM_C_ID = :C_ID and
            OL_DATE < :REP_DATE and
            OPERLIST.IS_DELETE = 0
      into :ZAD_OS, :ZAD_PERC;

      -- Балансовый счёт по ОС
      BALANCE_ACCOUNT_OS = 0;
      select OB.OB_SHORT_ACC_DT
      from OPERBOOK OB
      where OB.OB_FROM_OL_ID = :DISB_OL_ID rows 1
      into :BALANCE_ACCOUNT_OS;

      -- Дата закрытия кредита по первоначальному графику погашений
      K_KREDLINE_END_PRIMARY = null;
      select max(SH.SH_DATE)
      from SCHEDULE SH
      where SH.SH_FROM_K_ID = :KONT_ID
      into :K_KREDLINE_END_PRIMARY;
      ------------------------------------------------------------------------
      --получаем сведения о залоге
      ZAL_SUBTYPE = null;
      COST_MARKET = 0;
      Z_SUBTYPE_TEXT = 'Не указан';
      ZAL_STR = 'Нет';
      V_COST_MARKET = 0;
      Z_TYPE = 0;
      for select ZAL.Z_SUBTYPE, -- Вид Залога
                 ZAL.Z_COST_MARKET, -- Сумма Залога (Рыночная стоимость)
                 ZAL.Z_TYPE
          from ZALOG ZAL
          where ZAL.Z_FROM_K_ID = :KONT_ID
                --zal.z_type = 1
          into :V_ZAL_SUBTYPE, :V_COST_MARKET, :Z_TYPE
      do
      begin
        if (char_length(:ZAL_SUBTYPE || :V_ZAL_SUBTYPE || ',') <= 255) then
          :ZAL_SUBTYPE = :ZAL_SUBTYPE || :V_ZAL_SUBTYPE || ',';

        COST_MARKET = COST_MARKET + V_COST_MARKET;
      end

      if (:ZAL_SUBTYPE like '%,%') then
        :ZAL_SUBTYPE = substring(:ZAL_SUBTYPE from 1 for char_length(:ZAL_SUBTYPE) - 1);

      if (V_COST_MARKET > 0) then
        ZAL_STR = 'Да';

      --Суб.тип залога

      select ZST.ZS_NAME
      from ZALOGSUBTYPE ZST
      where ZST.ZS_FROM_ZT_ID = :Z_TYPE and
            ZST.ZS_ID = :V_ZAL_SUBTYPE
      into Z_SUBTYPE_TEXT;

      --------------------------------------------------------------------
      -- Число поручителей
      POR_QTY = 0;
      POR_QTY_STR = null;

      select count(*)
      from ZALOG POR
      where POR.Z_FROM_K_ID = :KONT_ID and
            POR.Z_TYPE = 0
      into :POR_QTY;

      if (POR_QTY = 0) then
        POR_QTY_STR = 'нет';
      else
        POR_QTY_STR = :POR_QTY;
      --------------------------------------------------------------------
      -- получаем данные по РППУ
      LK_PERCENT = 0;
      RPPU_BALANCE = 0;
      RPPU_CALCPERC = 0;
      LLC_NAME = '';
      LLC_IND_PERCENT = 0;

      select LK.LK_PERCENT, LK_BALANCE, LK_CALCPERC, CL.CS_NAME
      from LLC_MAIN LM, LLC_KONT LK, CLASSIFIC CL
      where LM_IS_SIGNED = 1 and
            LM_ID = LK_FROM_LM_ID and
            LK.LK_FROM_K_ID = :KONT_ID and
            :REP_DATE between LK_DATE and LK_DATE_END and
            LK_STATUS = CS_ID
      into LK_PERCENT, RPPU_BALANCE, RPPU_CALCPERC, LLC_NAME;

      LLC_IND_PERCENT = :LK_PERCENT;

      RPPU_ALL_SUM = RPPU_BALANCE + RPPU_CALCPERC;

      RPPU_BALANCE = cast(RPPU_BALANCE as double precision) * LK_PERCENT / 100;
      RPPU_CALCPERC = cast(RPPU_CALCPERC as double precision) * LK_PERCENT / 100;
      -----------------------------------------------------------------------
      --получаем сведения о реструктуризации
      IS_RESTRUCT = 0;
      RESTRUCT_DATE = null;
      RESTRUCT_KOL = 0;
      RESTRUCT_DATE_LAST = null;

      select IS_RESTRUCT, RESTRUCT_DATE, RESTRUCT_KOL, RESTRUCT_DATE_LAST
      from AAREP_GET_RESTRUCTURING(:KONT_ID, :REP_DATE, :DB_ID)
      into :IS_RESTRUCT, :RESTRUCT_DATE, :RESTRUCT_KOL, :RESTRUCT_DATE_LAST;

      IS_RESTRUCT_STR = 'Нет';
      if (IS_RESTRUCT = 1) then
        IS_RESTRUCT_STR = 'Да';

      --    баланс на дату реструктуризации
      REP_ACTIVE_BALANCE_NEW = null;
      select OKS_BALANCE
      from OKSUMMARY
      where OKS_FROM_K_ID = :KONT_ID and
            :RESTRUCT_DATE between OKS_DATE_BEGIN and OKS_DATE_END and
            OKS_CLOSE_STATE = 0
      into REP_ACTIVE_BALANCE_NEW;
      if (REP_ACTIVE_BALANCE_NEW is null) then
        REP_ACTIVE_BALANCE_NEW = 0;

      -- Новый % после реструктуризации
      if (:K_PERCENT = :KD_PERCENT) then
        LK_PERCENT_PERT = 'Не менялось';
      else
      begin
        LK_PERCENT_PERT = :K_PERCENT;
      end

      select max(iif(OK_TYPE = 1, OK_SH_DATE, cast('01.01.1900' as date))),
             max(iif(OK_TYPE = 2, OK_SH_DATE, cast('01.01.1900' as date)))
      from OPERKRED
      where OK_OL_IS_ACTIVE = 1 and
            OK_OL_DATE <= :REP_DATE and
            OK_FROM_K_ID = :KONT_ID and
            OK_TYPE in (1, 2) and
            OK_FROM_C_ID = 0
      into MAX_DATE_GRAPH, MAX_DATE_REST;

      if (MAX_DATE_GRAPH < MAX_DATE_REST) then
      begin
        K_DATE_END = MAX_DATE_REST; --Дата закрытия после реструктуризации
      end
      else
      begin
        K_DATE_END = MAX_DATE_GRAPH;
      end

      --------------------------------------------------------------------
      --сведения о внебалансе
      OKF_ZAD_OS = 0;
      OKF_PERC = 0;
      OKO_SH_OS = 0;
      OKO_SH_PERC = 0;

      select sum(OKF.OKO_OS + OKF.OKO_SH_OS), sum(OKF.OKO_PERC + OKF.OKO_SH_PERC)
      from OPERKROFF OKF
      where OKF.OKO_FROM_K_ID = :KONT_ID and
            OKF.OKO_OL_IS_ACTIVE = 1 and
            OKF.OKO_OL_DATE <= :REP_DATE
      into :OKF_ZAD_OS, -- Остаток ОС на внебалансе
           :OKF_PERC; -- Остаток  % на внебалансе

      OKF_ZAD_OS = coalesce(OKF_ZAD_OS, 0);
      OKF_PERC = coalesce(OKF_PERC, 0);

      OKF_ZAD_TOTAL = OKF_ZAD_OS + OKF_PERC;

      REP_WRITEOFF_DATE = null;
      select min(OL_DATE_OPER_DAY) -- Дата списания на внебаланс
      from OPERKROFF, OPERLIST
      where OKO_FROM_K_ID = :KONT_ID and
            OKO_FROM_OL_ID = OL_ID and
            OPERLIST.IS_DELETE = 0
      into :REP_WRITEOFF_DATE;

      select sum(OKO_SH_OS), sum(OKO_SH_PERC)
      from OPERKROFF, OPERLIST
      where OKO_FROM_OL_ID = OL_ID and
            OKO_FROM_K_ID = :KONT_ID and
            OKO_FROM_C_ID = :C_ID and
            OKO_TYPE = 1 and
            OPERLIST.IS_DELETE = 0
      into :OKO_SH_OS, -- Сумма списания на внебаланс по ОС
           :OKO_SH_PERC; -- Сумма списания на внебаланс по %
      -------------------------------------------------------------------------
      -- получаем данные по судам

      select LAW.LAM_DATETIME, LAW.LAM_DATE_RESULT, LAW.LAM_DATE_ACTIVATE, LAW.LAM_COST_ALL
      from LAWKREDIT LAK
      inner join LAWMAIN LAW on LAW.LAM_ID = LAK.LAK_FROM_LAM_ID
      inner join LAWLIST LAL on LAL.LAL_ID = LAW.LAM_FROM_LAL_ID
      where LAK.LAK_FROM_C_ID = :KONT_ID and
            LAK.LAK_FROM_K_ID = :C_ID
      order by LAW.LAM_DATETIME desc rows 1
      into LAM_DATETIME, LAM_DATE_RESULT, LAM_DATE_ACTIVATE, LAM_COST_ALL;
      suspend;
    end
  end
end
