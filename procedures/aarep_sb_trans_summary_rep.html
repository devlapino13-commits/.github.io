CREATE PROCEDURE AAREP_SB_TRANS_SUMMARY_REP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_SB_KIND INTEGER)
RETURNS (
    SBA_FROM_SBL_ID INTEGER,
    SBL_NAME VARCHAR(255),
    SBL_TYPE_STR VARCHAR(20),
    REP_REPAY_OS DECIMAL(18,4),
    REP_REPAY_PERC DECIMAL(18,4),
    REP_REPAY_FINE DECIMAL(18,4),
    REP_BEGIN_BALANCE DECIMAL(18,4),
    REP_BEGIN_OS DECIMAL(18,4),
    REP_BEGIN_PERC DECIMAL(18,4),
    REP_END_BALANCE DECIMAL(18,4),
    REP_END_OS DECIMAL(18,4),
    REP_END_PERC DECIMAL(18,4),
    REP_KOL_BEGIN INTEGER,
    REP_KOL_REPAY INTEGER,
    REP_KOL_END INTEGER,
    REP_KOL_TO_SB INTEGER,
    REP_KOL_FROM_SB INTEGER,
    REP_KOL_NORMAL INTEGER,
    REP_KOL_DELINQ INTEGER,
    REP_KOL_CLOSED INTEGER)
AS
begin
  for select sbl_type_str, sba_from_sbl_id, sbl_name, sum(rep_repay_os), sum(rep_repay_perc), sum(rep_repay_fine),
             sum(rep_begin_balance), sum(rep_begin_os), sum(rep_begin_perc), sum(rep_end_balance), sum(rep_end_os),
             sum(rep_end_perc), sum(rep_kol_begin), sum(rep_kol_repay), sum(rep_kol_end), sum(rep_kol_to_sb),
             sum(rep_kol_from_sb), sum(rep_kol_normal), sum(rep_kol_delinq), sum(rep_kol_closed)
      from aarep_sb_transaction_rep(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_sb_kind)
      group by sbl_type_str, sba_from_sbl_id, sbl_name
      into sbl_type_str, sba_from_sbl_id, sbl_name, rep_repay_os, rep_repay_perc, rep_repay_fine, rep_begin_balance,
           rep_begin_os, rep_begin_perc, rep_end_balance, rep_end_os, rep_end_perc, rep_kol_begin, rep_kol_repay,
           rep_kol_end, rep_kol_to_sb, rep_kol_from_sb, rep_kol_normal, rep_kol_delinq, rep_kol_closed
  do
  begin
    suspend;
  end
end
