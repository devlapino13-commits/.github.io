CREATE PROCEDURE HP_EDOC_MSG_SEND_DRAFT(
    EDL_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable edl_status d_smallint;
declare variable email_text d_str_8100;
declare variable c_id d_id;
declare variable mail_tmp d_str_8100;
declare variable doc_name d_str_255;
declare variable edl_object_kind d_integer;
declare variable edl_object_id d_id;
declare variable mail_to d_str_long;
declare variable msg_type d_integer;
declare variable edc_state d_integer;
declare variable edc_name d_str_255;
declare variable eds_id d_id;
declare variable msg_id d_id;
declare variable esc_message d_str_long;
declare variable msg_subject d_str_long;
declare variable un_id d_id;
declare variable reg_id d_integer;
declare variable reg_email_operator d_str_100;
declare variable cont_info d_str_100;
declare variable c_name d_str_100;
declare variable edc_email_on_draft d_str_255;
declare variable edl_blank_date d_date;
begin
  -- Процедура используется для отправки сообщений в режиме теста (eDoc_Cfg.EDC_State in (1, 2))

  res_text = '';

  select edl_status, edc_state, edc_name, edl_object_kind, edl_object_id,
         edl_create_un_id, edc_email_on_draft, edl_blank_date, edl_from_reg_id
  from edoc_list, edoc_cfg
  where edl_id = :edl_id and
        edl_from_edc_id = edc_id
  into edl_status, edc_state, edc_name, edl_object_kind, edl_object_id,
       un_id, edc_email_on_draft, edl_blank_date, reg_id;

  if (edc_state = 0) then res_text = 'Проект не актуален';
  if (res_text = '' and edc_state = 3) then res_text = 'Процедура должна вызываться только в режиме тестирования';
  if (res_text = '' and edl_status = -1) then res_text = 'Данные устарели';
  if (res_text = '' and edl_status = 1) then res_text = 'Документы уже подписаны';

  if (res_text = '') then
  begin
    mail_to = '';
    msg_type = 4;

    msg_subject = 'Электронный документ - Тестовый режим';

    -- получаем офис и номер документа
    select iif(:reg_id is null, reg_id, :reg_id), doc_name
    from hp_edoc_object_info(:edl_object_kind, :edl_object_id, :un_id, :edl_blank_date)
    into :reg_id, doc_name;

    if (edc_state = 1)
    then mail_to = edc_email_on_draft; -- получатели тестеры
    else
    begin
      -- получатели: менеджер и КОБД

      select reg_email_manager
      from region
      where reg_unicode = :reg_id
      into mail_to;
      mail_to = trim(coalesce(mail_to, ''));

      select reg_email_operator
      from region
      where reg_unicode = :reg_id
      into reg_email_operator;
      reg_email_operator = trim(coalesce(reg_email_operator, ''));

      if (mail_to = '')
      then mail_to = reg_email_operator;
      else
        if (reg_email_operator <> '')
        then mail_to = mail_to || '; ' || reg_email_operator;

      if (mail_to = '') then
        res_text = 'Email менеджера офиса не заполнен';
    end
  end

  email_text = '';
  mail_tmp = '';

  if (res_text = '') then
  begin
    -- создаем сообщение
    insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id,
                                msg_command_kind, msg_object_kind, msg_object_id)
    values (:msg_type, :mail_to, :msg_subject, :email_text, -1, :un_id, '', null, null)
    returning msg_id
    into msg_id;

    -- собираем текст СМС по всем подписантам
    for select eds_id, eds_from_c_id
    from edoc_sign
    where eds_from_edl_id = :edl_id and
          eds_from_c_id is not null
    into eds_id, c_id
    do
    begin
      esc_message = '';
      select cont_info, esc_message, res_text
      from hp_edoc_sms_text(:eds_id, :msg_id)
      into cont_info, esc_message, res_text;

      if (res_text <> '') then
      begin
        suspend;
        exit;
      end

      c_name = '<Без ФИО>';
      select c_name_fio
      from clients
      where c_id = :c_id
      into c_name;
  
      if (mail_tmp <> '') then
        mail_tmp = mail_tmp || '<br><br>';
  
      mail_tmp = mail_tmp ||
                'Получатель: <b>' || c_name || '</b><br>' ||
                'Телефон: <b>' || cont_info || '</b><br>' ||
                'Сообщение: <br>' || esc_message || '<br>';
    end

    email_text = 'Электронный документ: ' || edc_name;

    if (doc_name <> '') then
      email_text = email_text || ', № ' || doc_name;
  
    email_text = email_text || '<br>' ||
                              'Тестируемый проект: ' || edc_name ||'<br>' ||
                              'Id электронного документа: ' || edl_id || '<br>' ||
                              'В боевом режиме будут отправлены нижеуказанные СМС.<br><br>';

    email_text = email_text || mail_tmp || '<br>';
  
    -- обновляем текст email для отправки получателям (тестерам)
    update fsrobot_msgbox
    set msg_body = :email_text
    where msg_id = :msg_id;
  end

  Suspend;
end
