CREATE PROCEDURE AAREP_RESTRUCT(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    RES_NAME_FIO D_STR_100,
    RES_KONTRACT_NO D_STR_20,
    RES_DISB_DATE D_DATE,
    RES_EXPIRE_DATE D_DATE,
    RES_KREDLINE_END D_DATE,
    RES_RESTRUCT_DATE D_DATE,
    RES_REG_NAME D_STR_100,
    RES_AMOUNT CURR_TYPE,
    RES_PERCENT PERC_TYPE,
    RES_SUM_DEBT1 CURR_TYPE,
    RES_SUM_DEBT2 CURR_TYPE,
    RES_SUM_DEBT3 CURR_TYPE,
    RES_SUM_DEBT4 CURR_TYPE,
    RES_REASON D_STR_255,
    RES_EXT_PERIOD D_INTEGER,
    RES_GRACE_PERIOD D_INTEGER,
    RES_FORMAT D_STR_20,
    RES_PR_NAME D_STR_100,
    BUSTYPE_NAME D_STR_100,
    BUSSUBTYPE_NAME D_STR_100,
    LOAN_TARGET_NAME D_STR_100,
    KS_STATE D_STR_100,
    ZAD_OS D_CURRENCY)
AS
declare variable k_id id_type;
declare variable k_amount_applicate curr_type;
declare variable koef perc_type;
declare variable cml_id d_id;
declare variable cmc_id d_integer;
declare variable cml_add_params d_str_long;
declare variable sum_debt3 curr_type;
declare variable sum_debt4 curr_type;
declare variable grace_period d_integer;
declare variable num_count d_integer;
declare variable ol_user_id id_type;
declare variable or_from_rd_id d_integer;
declare variable kc_id decimal(15,2);
begin
  for select k_id, k_kontract_no, c_name_fio, k_disb_date, k_expire_date, k_kredline_end, or_ol_date, reg_name,
             k_amount_applicate, kc_amount_approved, k_percent, rd_name, pr_name, ol_user_id, or_from_rd_id, kc_id
      from operrest
      inner join operlist on ol_id = or_from_ol_id
      inner join region_list(:rep_region_id, 1, 3, 1) on id_list = or_from_reg_id
      inner join region on reg_unicode = id_list
      inner join kontract on k_id = or_from_k_id
      inner join knclient on kc_from_k_id = k_id and
            kc_amount_approved > 0
      inner join clients on c_id = kc_from_c_id
      inner join product on pr_id = k_from_pr_id
      left join refdata on rd_id = or_from_rd_id and
      (
      (f_get_db_id() = 39801 and ol_memo containing 'Реструктуризация') --RFCKZ-595
   or (f_get_db_id() <> 39801 and rd_memo = 'Реструктуризация')
      )
      where or_ol_date between :rep_date_begin and :rep_date_end and
            or_ol_is_active = 1
      into :k_id, :res_kontract_no, :res_name_fio, :res_disb_date, :res_expire_date, :res_kredline_end,
           :res_restruct_date, :res_reg_name, :k_amount_applicate, :res_amount, :res_percent, :res_reason, :res_pr_name,
           :ol_user_id, :or_from_rd_id, :kc_id
  do
  begin
    bustype_name = '';
    bussubtype_name = '';
    loan_target_name = '';

    select bt.bt_name, bst.bs_name, rd.rd_name
    from knclient kncl
    left join bussubtype bst on bst.bs_id = kncl.kc_busuness_subtype
    left join bustype bt on bt.bt_id = kncl.kc_business_type
    left join refdata rd on rd.rd_id = kncl.kc_loan_target
    where kncl.kc_id = :kc_id
    into :bustype_name, :bussubtype_name, :loan_target_name;

    ks_state = '';

    select (case ks.ks_state
              when 1 then 'Активный'
              when 2 then 'Закрытый'
              else 'Неопределено'
            end)
    from kontstate ks
    where ks.ks_from_k_id = :k_id and
          ks.ks_ol_is_delete = 0 and
          :rep_date_end between ks.ks_date_begin and ks.ks_date_end
    into :ks_state;

    if (not((or_from_rd_id = 0 and
        ol_user_id = 1000000000000) or (or_from_rd_id > 0 and
        res_reason is not null))) then
      continue;

    koef = res_amount / k_amount_applicate;
    select sum(zad_os + sh_calc_perc + round(0.02 * sh_calc_perc, 0)), sum(sh_calc_perc), sum(zad_os)
    from aarep_make_zad_list(:k_id, :rep_date_end, 1)
    into :res_sum_debt1, :res_sum_debt2, :zad_os;
    res_sum_debt1 = round(koef * res_sum_debt1, 0);
    res_sum_debt2 = round(koef * res_sum_debt2, 0);
    zad_os = round(koef * zad_os, 0);

    select cml_id, cml_from_cmc_id, cml_add_params
    from com_list cl
    where cml_object_kind = 9 and
          cml_object_id = :k_id and
          cl.cml_date = :res_restruct_date
    order by cml_id desc
    rows 1
    into :cml_id, :cmc_id, :cml_add_params;
    if (row_count = 1) then
    begin
      res_format = 'Онлайн КК';
      --res_reason = (select cmc_name from com_cfg where cmc_id = :cmc_id);
      res_ext_period = f_json_str_get_value(cml_add_params, 'add_month_count', '');
      res_grace_period = f_json_str_get_value(cml_add_params, 'add_month_kind', '');
    end
    else
    begin
      res_format = 'Обычный КК';
      res_ext_period = (12 * extract(year from res_kredline_end) + extract(month from res_kredline_end)) - (12 * extract(year from res_expire_date) + extract(month from res_expire_date));
      res_grace_period = 0;
    end

    res_sum_debt3 = 0;
    res_sum_debt4 = 0;
    res_grace_period = 0;
    num_count = 0;
    for select zad_os, zad_perc, iif(sh_repay_type = 1, 1, 0)
        from aarep_make_zad_list(:k_id, :res_restruct_date - 1, 1)
        order by sh_date
        into :sum_debt3, :sum_debt4, :grace_period
    do
    begin
      num_count = num_count + 1;
      res_sum_debt3 = res_sum_debt3 + sum_debt3;
      if (num_count <= res_ext_period) then
        res_sum_debt4 = res_sum_debt4 + sum_debt4;
      res_grace_period = res_grace_period + grace_period;
    end
    res_sum_debt3 = round(koef * res_sum_debt3, 0);
    res_sum_debt4 = round(koef * res_sum_debt4, 0);

    suspend;
  end
end
