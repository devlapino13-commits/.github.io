CREATE PROCEDURE HP_EDOC_MANUAL_SIGN_REQUEST(
    EDL_ID D_ID NOT NULL,
    KIND D_SMALLINT NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable edb_id d_id;
declare variable edf_print_name d_str_100;
declare variable edb_blob d_blob_binary;
declare variable bls_id d_id;
declare variable edl_status d_smallint;
declare variable edc_code d_str_code;
declare variable edc_name d_str_100;
declare variable db_id d_integer;
begin
  -- Процедура создает запрос на ручное подписание
  -- Kind:
  --      0 - из-за технических проблем, некоторые клиенты не могут подписать через СМС
  --      1 - Изначально требуется дополнительно ручное подписание. Вызывается сразу при создании эл.документа и бланков
  --      2 - Отложенное подписание (Для этого все бланки должны быть готовы, подписаны через СМС)

  db_id = f_get_db_id();
  if (kind not in (0, 1, 2)) then
    exception e_default f_format('Incorrect param value: kind = %1, expected in (0, 1, 2)', kind);

  res_is_error = 1;
  res_text = '';

  select edl_status, edc_code, edc_name
  from edoc_list, edoc_cfg
  where edl_id = :edl_id and
        edl_from_edc_id = edc_id
  into edl_status, edc_code, edc_name;

  if (kind <> 1) then
  begin
    if (res_text = '' and
        edl_status = -1) then
      res_text = 'Документ аннулирован, запрос отменяется';

    if (res_text = '' and
        edl_status = 1) then
      res_text = 'Документ уже подписан, запрос отменяется';

    if (res_text = '' and
        not exists(select *
                   from edoc_blank
                   where edb_from_edl_id = :edl_id)) then
      res_text = 'Бланки еще не созданы';

    if (res_text = '') then
    begin
      if (exists(select *
                 from edoc_blank
                 where edb_from_edl_id = :edl_id and
                       (edb_from_bls_id is null and
                       edb_blob is null))) then
        res_text = 'Не все бланки еще приготовлены';
    end

    if ((res_text = '' and edc_code = 'asp.agreement') or
        db_id = 81001) then -- Ручное подписание отключено в РФ, RFCRU-696
    begin
      if (kind = 0) then
        res_text = f_format('%1 не может быть подписан вручную в виде исключения.', edc_name);

      if (kind = 2 and f_get_db_id() not in (39801, 41701)) then
        res_text = f_format('%1  - запрещается отложенное ручное подписание', edc_name);
    end
  end

  if (res_text = '' and
      kind = 2) then
  begin
    if (exists(select *
               from edoc_sign
               where eds_status = 0 and
                     eds_from_c_id is not null)) then
      res_text = 'Для отложенной ручной подписи все участники сначала должны подписать онлайн.';
  end

  if (res_text = '' and
      kind = 0) then -- если технически невозможно подписать онлайн, переносим бланки на BLS, чтобы распечатать их
  begin
    -- переносим блоб в BLS
    for select edb_id, edf_print_name, edb_blob
        from edoc_blank, edoc_frx
        where edb_from_edl_id = :edl_id and
              edb_from_edf_id = edf_id and
              edb_blob is not null and
              edb_from_bls_id is null
        into edb_id, edf_print_name, edb_blob
    do
    begin
      execute procedure hp_insert_into_blobstore ('edoc_' || :edb_id || '.pdf', :edf_print_name, :edb_blob, 1000000000000, 2,
              0)
      returning_values :bls_id;

      update edoc_blank
      set edb_from_bls_id = :bls_id,
          edb_blob = null,
          edb_need_approve = 0
      where edb_id = :edb_id;
    end
  end

  if (res_text = '') then
  begin
    for select edb_id
        from edoc_blank, edoc_frx
        where edb_from_edl_id = :edl_id and
              edb_from_edf_id = edf_id and
              edf_need_qr = 1
        into edb_id
    do
    begin
      update or insert into edoc_blank_scan (edbs_from_edb_id, edbs_kind, edbs_from_un_id, edbs_status,
                                             edbs_status_un_id)
      values (:edb_id, :kind, :un_id, 0, :un_id)
      matching (edbs_from_edb_id);
    end
  end

  if (res_text = '') then
  begin
    -- добавляем подписанта от имени контролера (менеджер или андерайтер)
    if (not exists(select *
                   from edoc_sign
                   where eds_from_edl_id = :edl_id and
                         eds_from_c_id is null)) then
    begin
      insert into edoc_sign (eds_from_edl_id, eds_from_c_id)
      values (:edl_id, null);
    end
    else
    begin
      update edoc_sign
      set eds_status = 0,
          eds_enum_edoc_sign_kind = f_enum('edoc_sign_kind', 'none', 'id')
      where eds_from_edl_id = :edl_id and
            eds_from_c_id is null;
    end
  end

  if (res_text = '') then
  begin
    res_is_error = 0;
    res_text = 'Запрос на ручное подписание создан. Распечатайте бланки и получите подпись клиента';
  end

  suspend;
end
