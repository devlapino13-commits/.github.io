CREATE PROCEDURE HP_ACTIONLIST_MEMO_INFO(
    ACL_ID ID_TYPE,
    IS_HTML INTEGER = 0)
RETURNS (
    INF_MEMO D_BLOB_TEXT,
    ACM_MEMO D_BLOB_TEXT,
    SFT_MEMO D_BLOB_TEXT,
    DT_VER TIMESTAMP,
    SUM_PAYMENT CURR_TYPE,
    SUM_DEBT CURR_TYPE,
    DAYS_DELAY STR_255,
    FSSP_DATA STR_255,
    EXTB_DATA STR_255,
    FMS_DATA STR_255,
    EFRSB_DATA STR_255,
    ACL_COMMENT STR_255,
    CR_LEVEL INTEGER,
    CL_ID ID_TYPE)
AS
declare variable c_id d_id;
declare variable col_info str_10;
declare variable json_key str_255;
declare variable json_value str_255;
declare variable no_data str_255;
declare variable has_info d_smallint;
declare variable extinfo_memo d_str_8100;
declare variable eic_id d_integer;
declare variable eic_name d_str_255;
declare variable eic_info_actual_days d_integer;
declare variable db_id d_integer;
declare variable fincert_date d_date;
declare variable fincert_status d_smallint;
declare variable acl_datetime d_datetime;
declare variable ff_create_datetime d_datetime;
declare variable res_type d_str_20;
begin
  db_id = f_get_db_id();
  no_data = 'нет данных';
  inf_memo = '';
  acm_memo = '';
  sum_payment = 0;
  sum_debt = 0;
  days_delay = no_data;
  fssp_data = no_data;
  extb_data = no_data;
  fms_data = no_data;
  efrsb_data = no_data;
  acl_comment = '';
  cr_level = -1;
  dt_ver = null;

  select acl_object_id, acm_memo, acl_datetime
  from actionlist
  left join actionmemo on acm_from_acl_id = acl_id
  where acl_id = :acl_id and
        acl_object_kind = 0
  into :c_id, :acm_memo, :acl_datetime;

  if (coalesce(acm_memo, '') = '')
  then has_info = 0;
  else has_info = 1;

  if (acm_memo is not null) then
  begin
    for select json_key, json_value
        from fm_json_parse(:acm_memo, 0, 'verify')
        into :json_key, :json_value
    do
    begin
      if (json_key = 'dtver') then
        dt_ver = cast(json_value as timestamp);
      else
      if (json_key = 'sumpayment') then
        sum_payment = cast(json_value as curr_type);
      else
      if (json_key = 'sumdebt') then
        sum_debt = cast(json_value as curr_type);
      else
      if (json_key = 'daysdelay') then
        days_delay = json_value;
      else
      if (json_key = 'fsspdata') then
        fssp_data = json_value;
      else
      if (json_key = 'extbdata') then
        extb_data = json_value;
      else
      if (json_key = 'fmsdata') then
        fms_data = json_value;
      else
      if (json_key = 'efrsbdata') then
        efrsb_data = json_value;
      else
      if (json_key = 'comment') then
        acl_comment = json_value;
      else
      if (json_key = 'level') then
        cr_level = cast(json_value as int);
    end
  end
  if (is_html = 1) then
  begin
    if (has_info = 1) then
    begin
      col_info = case :cr_level
                   when 1 then 'red'
                   when 0 then 'blue'
                   else 'green'
                 end;
      acm_memo = '<font color="black">' || 'Дата проверки: ' || '</font><br>' || '<font color="' || col_info || '">' || ' ' || f_datetostr(dt_ver) || '</font><br><br>' || '<font color="black">' || 'Сумма ежемесячных платежей' || '</font><br>' || '<font color="black">' || 'по займам/кредитам согласно' || '</font><br>' || '<font color="black">' || 'внешних источников: ' || '</font><br>' || '<font color="' || col_info || '">' || ' ' || sum_payment || '</font><br><br>' || '<font color="black">' || 'Сумма текущей задолженности' || '</font><br>' || '<font color="black">' || 'задолженности: ' || '</font><br>' || '<font color="' || col_info || '">' || ' ' || sum_debt || '</font><br><br>' || '<font color="black">' || 'Дни просрочки: ' || '</font><br>' || '<font color="' || col_info || '">' || ' ' || days_delay || '</font><br><br>' || '<font color="black">' || 'Данные ФССП: ' || '</font><br>' || '<font color="' || col_info || '">' || ' ' || fssp_data || '</font><br><br>' || '<font color="black">' || 'Данные ВБ: ' || '</font><br>' || '<font color="' || col_info || '">' || ' ' || extb_data || '</font><br><br>' || '<font color="black">' || 'Данные ФМС: ' || '</font><br>' || '<font color="' || col_info || '">' || ' ' || fms_data || '</font><br><br>' || '<font color="black">' || 'Данные ЕФРСБ: ' || '</font><br>' || '<font color="' || col_info || '">' || ' ' || efrsb_data || '</font><br><br>' || '<font color="black">' || 'Комментарий проверяющего: ' || '</font><br>' || '<font color="' || col_info || '">' || ' ' || iif(acl_comment = '', 'негативная информация не найдена', acl_comment) || '</font>';
    end else acm_memo = '';

    extinfo_memo = '';

    if (db_id = 81001) then
    begin
      for select res, res_type, ff_create_datetime, block_date
      from hp_fincert_get_result(:c_id, :acl_datetime)
      into fincert_status, res_type, ff_create_datetime, fincert_date
      do
      begin
        extinfo_memo = extinfo_memo ||
                 f_format('<span style="color: %1"><b>Стоп-лист ЦБ (%2)</b></span><br>', iif(fincert_status = 1, 'red', 'blue'), res_type) ||
                 f_format('<font color="black">Время проверки <b>%1</b></font><br>', f_datetimetostr(acl_datetime)) ||
                 f_format('<font color="black">Время файла <b>%1</b></font><br>', f_datetimetostr(ff_create_datetime)) ||
                 f_format('<font color="black">%1: </font><span style="color: %2"><b>%3</span></b><br>', 'Результат',
                          iif(fincert_status = 1, 'red', 'green'),
                          iif(fincert_status = 1, 'Найден в списке', 'Не найден') ) ||
                 f_format('<font color="black">%1: </font><font color="black"><b>%2</font></b><br>', 'Дата внесения в Стоп-лист ЦБ', f_datetostr(fincert_date)) ||
                 '<br><br><br>';
      end
    end
    --------------------------------------------------------------

    for select eic_id, eic_name, eic_info_actual_days
    from extinfocfg
    into eic_id, eic_name, eic_info_actual_days
    do
    begin
      for select eil_id, eil_from_c_id, eil_from_eic_id, eil_date, eil_object_kind, eil_object_id, eil_ext_id, eil_is_json,
             eil_from_bls_id, eil_status, eil_create_datetime, eil_status_datetime, eil_kk_memo_id, eil_kk_un_id,
             eil_kk_result, eil_kk_datetime,
             mml_memo
      from extinfolist, memolist
      where eil_from_c_id = :c_id and
            eil_from_eic_id = :eic_id and
            mml_id = eil_kk_memo_id
      order by eil_date desc, eil_create_datetime desc, eil_id desc
      rows 1
      as cursor c
      do
      begin
        if (coalesce(c.eil_kk_result, 0) < 0)
        then col_info = 'red';
        else
          if (coalesce(c.eil_kk_result, 0) = 0)
          then col_info = 'black';
          else col_info = 'green';

          extinfo_memo = extinfo_memo ||
                         '<font color="blue"><b>' || eic_name || '</b></font><br>' ||
                         '<font color="black">Инфо от <b>' || f_datetostr(c.eil_date) || '</b> г.' || '</font>' ||
                         trim(iif(datediff(day, c.eil_date, current_date) > eic_info_actual_days, ', <font color="red"> устарел </font><br>', '<br>')) ||
                         '<font color="black">' || 'Результат: ' || '</font>' ||
                         '<font color="' || col_info || '"><b>' ||
                         trim(case coalesce(c.eil_kk_result, 0)
                         when -1 then 'Негативный'
                         when 0 then 'Нет оценки'
                         when 1 then 'Положительный'
                         when 2 then 'Ок, без проверки'
                         end) || '</font></b><br>' ||
                         '<font color="black">' || 'Комментарий:<br><b>' || c.mml_memo || '</b></font><br><br>';
      end
    end
    if (extinfo_memo <> '') then
    begin
      if (acm_memo <> '' and has_info = 1)
      then acm_memo = BLOB_APPEND(acm_memo, '<br><br>', extinfo_memo);
      else acm_memo = extinfo_memo;
    end
  end

  cl_id = c_id;
  sft_memo = coalesce(sft_memo, '');
  suspend;
end
