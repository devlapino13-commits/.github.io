CREATE PROCEDURE FM_HTML_CLIENT_INFO_FT_ANKETA(
    C_ID D_ID,
    UN_ID D_ID,
    FAK_CODE D_STR_20 = 'client')
RETURNS (
    RES_TEXT D_STR_8100)
AS
declare variable EDL_STATUS_STR D_STR_CODE;
declare variable DATE_TMP date;
declare variable LANG D_STR_LANG;
declare variable EDOC_STATE D_STR_CODE;
declare variable EDC_ID D_INTEGER;
declare variable OBJ_KIND D_INTEGER;
declare variable EDL_ID D_ID;
declare variable EDL_STATUS D_SMALLINT;
declare variable EDS_ID D_ID;
declare variable FT_VALID D_SMALLINT;
declare variable NEED_SHOW_FT_LINK D_SMALLINT;
declare variable SELFEMPLOYED_START_DATE D_DATE;
declare variable EDL_BLANK_DATE D_DATE;
declare variable IS_CLIENT D_SMALLINT;
declare variable STANDART_METHOD D_SMALLINT;
declare variable EDL_ID_ASP D_ID;
declare variable EDL_BLANK_DATE_ASP D_DATE;
begin
  -- Формирует ссылку на создание Анкеты-опросника для fm_html_client_info
  if ((:fak_code is null) or (coalesce(:fak_code, '') = '')) then
    fak_code = 'client';
  res_text = '';
  edoc_state = f_edoc_state2('ft.anketa', current_date, un_id);
  obj_kind = f_objkind('ft_anketa_list', 'id');

  --Процедура работает только в РФ, т.к. в кг нет edoc_cfg.edc_code = 'ft.anketa'
  if (edoc_state <> 'inactive') then
  begin
    --RFCRU-2543

    selfemployed_start_date = cast(f_get_const('selfemployed_start_date') as date);
    need_show_ft_link = 0;

    select edl.edl_blank_date
    from edoc_list edl
    inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id
    where eds.eds_from_c_id = :c_id and
          edl.edl_object_kind = :obj_kind and
          edl.edl_from_edc_id = f_edoc_kind('ft.anketa', 'id') and
          edl.edl_status = 1
    order by edl.edl_id desc
    rows 1
    into edl_blank_date;

    if (selfemployed_start_date <> '31.12.9999' and
        coalesce(edl_blank_date, '01.01.1900') < selfemployed_start_date and
        exists(select *
               from miniappclient mac
               inner join miniapp ma on ma.ma_id = mac.mac_from_ma_id and
                     ma.ma_is_active = 1 and
                     ma.ma_date_end >= '31.12.9999'
               inner join product pr on pr.pr_id = ma.ma_from_pr_id and
                     pr.pr_code containing '.selfemployed.'
               where mac.mac_from_c_id = :c_id and
                     not exists(select *
                                from kontract kt
                                inner join kontstate ks on ks.ks_from_k_id = kt.k_id and
                                      ks.ks_state = 1 and
                                      ks.ks_ol_is_delete = 0 and
                                      ks.ks_date_begin <= current_date
                                where kt.k_from_ma_app_id = ma.ma_id)) and f_tmp_const('use_perstemp') <> 1) then
      need_show_ft_link = 1;

    edl_id_asp = null;
    edl_blank_date_asp = null;
    select edl_id, edl_blank_date
    from edoc_list edl, edoc_cfg edc
    where edl_object_kind = f_objkind('clients', 'id') and
          edl.edl_from_edc_id = edc.edc_id and
          edl_object_id = :c_id and
          edc.edc_code = 'asp.agreement' and
          edl_status = 1 and -- подписанный
          current_date between edl_blank_date and coalesce(edl_date_end, current_date)
    order by edl_id desc
    rows 1
    into edl_id_asp, edl_blank_date_asp;

    lang = f_get_lang_code(un_id, null);
    edc_id = f_edoc_kind('ft.anketa', 'id');

    res_text = '<h2>' || f_lang(lang, 'Анкета-опросник') || '</h2>';

    ft_valid = f_ft_valid(c_id, current_date);

    if (f_asp_valid(c_id, current_date) = 0) then
      res_text = res_text || f_lang(lang, 'Требуется подписание договора АСП с клиентом.');
    else
    begin
      -- Здесь стандартно статус не получишь, т.к. у нас есть клиент, а нужна анкета (ft_anketa_list)
      --Кредитоваться клиент должен после подписания договора АСП. Тогда все условия выполняются. Здесь клиент после подписания АСП не кредитовался.
      --Если он прокредитуется после подписания АСП, то да, можем анкету залогодателя не подписывать
      --Система при добавлении залогодателя, проверяет следующие моменты:
      --1. Получало ли данное лицо заем?
      --2. Есть ли у него подписанный договор АСП
      --3. Есть ли у него подписанная анкета опросник.
      is_client = 0;
      if (exists(select *
                 from kontstate ks
                 inner join knclient knc on knc.kc_from_k_id = ks.ks_from_k_id
                 where knc.kc_from_c_id = :c_id and
                       knc.kc_amount_approved > 0 and
                       ks.ks_ol_is_delete = 0 and
                       ks.ks_state = 1 and
                       :edl_blank_date_asp between ks.ks_date_begin and ks.ks_date_end)) then
        is_client = 1;

      standart_method = 0;
      if ((is_client = 1) or (:fak_code = 'client')) then
        standart_method = 1;

      edl_id = null;

      select edl_id, eds_id, edl_status, edl_blank_date
      from edoc_list edl
      inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id
      inner join ft_anketa_list fal on fal.fal_id = edl.edl_object_id
      inner join ft_anketa_kind fak on fak.fak_id = fal.fal_from_fak_id
      where edl.edl_object_kind = :obj_kind and
            fal_from_c_id = :c_id and
            eds_from_c_id = :c_id and
            edl_from_edc_id = :edc_id and
            current_date between edl_blank_date and coalesce(edl_date_end, current_date) and
            iif(:standart_method = 1, 1 = 1, fak.fak_code = :fak_code)
      order by edl_status desc, edl_id desc
      rows 1
      into edl_id, eds_id, edl_status, date_tmp;

      edl_status_str = 'empty'; -- нет действующего эл.документа
      if (edl_status = 1) then
        edl_status_str = 'signed';
      if (edl_status = 0) then
        edl_status_str = 'unsigned';

      if (edl_status_str = 'signed' and
          need_show_ft_link = 0) then
      begin
        res_text = res_text || f_format(f_lang(lang, 'Анкета-опросник подписана: %1'), f_datetostr(date_tmp)) || '<br>';
      end
      else
      begin
        if (edl_status_str = 'unsigned') then
        begin
          res_text = res_text || f_lang(lang, 'Анкета-опросник отправлена, но еще не подписана.') || '<br>';

          res_text = res_text || f_uri_sql(f_lang(lang, 'Отправить повторно'), 'edoc_msg_ins', 'edoc_list', edl_id, 'none', f_lang(lang, 'Повторно отправить Анкету-опросник через СМС клиенту?'), '', null, null, null, null, 'dialog', :eds_id);
        end
        else
        if (ft_valid = 1 and
            need_show_ft_link = 0) then
        begin
          res_text = res_text || f_lang(lang, 'Подписание Анкеты-опросника не требуется.') || '<br>';
        end
        else -- нет Анкеты-опросника
        begin
          if (f_meet_valid(c_id, 120) = 1 or (select is_oivo from hp_check_oivo_assignment(:c_id, :un_id)) = 1) then
            res_text = res_text || f_uri_open(f_lang(lang, 'Заполнить анкету-опросник '), 'ft_anketa', 'client', c_id, f_lang(lang, 'Анкета-опросник'), '&fak_code=' || :fak_code, un_id);
          else
            res_text = res_text || f_lang(lang, 'Требуется верификация клиента по QR-коду.');
        end
      end
    end
  end

  if (res_text <> '') then
    res_text = res_text || '<br>';
  suspend;
end
