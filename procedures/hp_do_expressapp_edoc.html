CREATE PROCEDURE HP_DO_EXPRESSAPP_EDOC(
    EXA_ID D_INTEGER,
    EDOC_CODE D_STR_CODE,
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    UN_ID D_ID)
RETURNS (
    WAIT_STATUS D_SMALLINT)
AS
declare variable EDC_ID D_INTEGER;
declare variable EDL_ID D_ID;
declare variable RES_TEXT D_STR_255;
declare variable EDL_STATUS D_SMALLINT;
declare variable EXA_LO_ID D_ID;
declare variable EXA_FROM_C_ID D_ID;
declare variable EXA_AMOUNT CURR_TYPE;
declare variable CLI_NAME D_STR_100;
declare variable SUBJECT D_STR_255;
declare variable body D_STR_LONG;
declare variable DB_ID D_ID;
declare variable EXA_DATE D_DATE;
declare variable EXA_DISB_DATE D_DATE;
declare variable EXA_FROM_PR_ID D_INTEGER;
begin
  -- Процедура отвечает за создание и проверку электронного документа для текущего этапа
  -- Перед созданием нужно проверять, возможно эл.документ ранее создан, например вручную через специалиста
  db_id = f_get_db_id();
  wait_status = 0;

  --Нижеперечисленные edoc_code относятся к товарам в рассрочку, поэтому возвращаем wait_status = 0 и завершаем работу
  if (db_id in (39801, 41701) and edoc_code in ('loan.application_autosign', 'loan.agreement_autosign', 'mc_purchase.agreement_autosign')) then
  begin
    suspend;
    exit;
  end 

  exa_date = null;
  exa_disb_date = null;
  exa_from_pr_id = null;
  select exa.exa_date, exa.exa_disb_date, exa_from_pr_id
  from expressapp exa
  where exa.exa_id = :exa_id
  into exa_date, exa_disb_date, exa_from_pr_id;
  exa_from_pr_id = coalesce(exa_from_pr_id, 0);

  -- для проекта ТМС не нужно консультационные услуги
  if (db_id in (39801, 41701) and
  edoc_code in ('consulting.agreement') and
  exists(select * from product
  where pr_id = :exa_from_pr_id
  and pr_code containing '.tms.')) then
  begin
    suspend;
    exit;
  end

  obj_kind = f_objkind(obj_kind, 'id');

  edl_id = null;
  edl_status = null;
  select edl_id, edl_status
  from edoc_list edl
  inner join edoc_cfg edc on edc.edc_id = edl.edl_from_edc_id
  where edl_object_kind = :obj_kind and
        edl_object_id = :obj_id and
        edc.edc_code = :edoc_code and
        iif(:db_id in (39801, 41701), edl_blank_date between :exa_date and :exa_disb_date, true) and
        iif(:db_id in (39801, 41701) and edc.edc_code = 'loan.express', true, edl_status = 1) and
        iif(:db_id in (39801, 41701) and edc.edc_code = 'consulting.agreement', edl_date_end is null, true)
  order by edl_id desc
  rows 1
  into edl_id, edl_status;

  if (edl_id is not null) then
  begin
    if (edl_status = 0) then
      wait_status = 2; -- до этого был 1

    if (edl_status = -1) then
      wait_status = -1;
  end else
  begin
    res_text = '';

    if (edoc_code = 'loan.agreement') then
    begin
      select edl_id, res_text
      from hp_edoc_ins_loan_agreement(:obj_id, :un_id)
      into :edl_id, :res_text;
    end
    else if (edoc_code = 'loan.application' and :db_id in (39801, 41701)) then
    begin
      select edl_id, res_text
      from HP_EDOC_INS_LOAN_APPLICATION (:obj_id, :un_id)
      into edl_id, res_text;
    end
    else
    begin
      select edl_id, res_text
      from hp_edoc_insert(:edoc_code, :obj_kind, :obj_id, :un_id)
      into edl_id, res_text;
    end

    if (edl_id is not null and res_text = '')
    then wait_status = 2; -- до этого был 1
    else wait_status = -1;

    if (wait_status = 2) then -- отправляем пуш уведомление ВИПСу
    begin
      select exa_lo_id, exa_from_c_id, exa_amount
      from expressapp
      where exa_id = :exa_id
      into exa_lo_id, exa_from_c_id, exa_amount;

      select c_name_fio from clients
      where c_id = :exa_from_c_id
      into cli_name;

      subject = 'Предодобрен заем ' || iif(db_id = 81001,trim('"Доверие"'), trim('"Индивидуальный Ишеним"')) || ' ' || cli_name;
      body = 'Вашему клиенту '|| cli_name || ' предодобрен заем по продукту ' || iif(db_id = 81001, trim('"Доверие"'), trim('"Индивидуальный стандарт"')) || ' на сумму '|| round(exa_amount,0)|| ' ' || iif(db_id = 81001,trim('руб.'), trim('сом'));

      execute procedure hp_send_push_to_staff(:exa_lo_id, :subject, :body, :un_id, '', 7);
    end
  end

  if (edl_id is not null) then
  begin
    update expressapp
    set exa_from_edl_id = :edl_id
    where exa_id = :exa_id and
          coalesce(exa_from_edl_id, 0) <> :edl_id;

    -- В тестовой базе всегда подписываем eDoc автоматически
    if (not exists(select * from hp_db_is_actual)) then
    begin
      execute procedure hp_edoc_sign_draft(:edl_id);
      wait_status = 0;
    end
  end

  suspend;
end
