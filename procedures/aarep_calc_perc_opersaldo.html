CREATE PROCEDURE AAREP_CALC_PERC_OPERSALDO(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    OB_OBOROT_DT DECIMAL(18,4),
    OB_OBOROT_KT DECIMAL(18,4),
    AAREP_OFFICE_ID INTEGER,
    AAREP_ACCOUNT VARCHAR(20),
    OB_SALDO_END DECIMAL(18,4),
    OB_SALDO_BEGIN DECIMAL(18,4),
    AAREP_ACCOUNT_TYPE INTEGER)
AS
declare variable ob_end_dt decimal(15,2);
declare variable ob_end_kt decimal(18,4);
begin
  for select sum(ob_end_dt), sum(ob_end_kt), sum(ob_oborot_dt), sum(ob_oborot_kt), aarep_office_id, aarep_account,
             aarep_account_type
      from aarep_oborsal_calc_perc(:report_region_id, :report_date_begin, :report_date_end)
      group by aarep_office_id, aarep_account, aarep_account_type
      into :ob_end_dt, :ob_end_kt, :ob_oborot_dt, :ob_oborot_kt, :aarep_office_id, :aarep_account, :aarep_account_type
  do
  begin
    ob_saldo_end = :ob_end_dt - :ob_end_kt;
    ob_saldo_begin = ob_saldo_end - (:ob_oborot_dt - :ob_oborot_kt);
    suspend;
  end
end
