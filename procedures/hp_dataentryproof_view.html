CREATE PROCEDURE HP_DATAENTRYPROOF_VIEW(
    ACL_ID D_ID,
    C_ID D_ID)
RETURNS (
    KC_ID D_ID,
    K_KONTRACT_NO D_STR_20,
    C_NAME_FIO D_STR_100,
    KI_ID D_INTEGER,
    KI_FROM_ACL_ID D_ID,
    KI_FROM_UN_ID D_ID,
    KI_FROM_K_ID D_ID,
    KI_FROM_C_ID D_ID,
    KI_DATETIME D_DATETIME,
    KI_VERIFY_INCOME D_CURRENCY,
    BCT_CODELST D_STR_8100,
    KI_IS_MANUAL D_BOOL,
    KI_SURVEY D_STR_8100,
    KI_SURVEY_PREV D_STR_8100)
AS
declare variable dt_beg d_date;
declare variable dt_end d_date;
begin
  dt_beg = current_date - extract(day from current_date) + 1;
  dt_end = dateadd(month, 1, dt_beg);
  for
    select
      kc_id,
      k_kontract_no,
      c_name_fio,
      ki_id,
      acl_id,
      ki_from_un_id,
      k_id,
      c_id,
      ki_datetime,
      ki_verify_income,
      coalesce(ki_is_manual, 0),
      ki_survey
    from actionlist
      inner join kontract
        on k_id = acl_object_id
      inner join knclient
        on kc_from_k_id = k_id
          and kc_from_c_id = :c_id
      inner join clients
        on c_id = kc_from_c_id
      left join kk_income
        on ki_from_acl_id = acl_id
          and ki_from_c_id = c_id
    where acl_id = :acl_id
      and acl_object_kind = 9
  into
    :kc_id,
    :k_kontract_no,
    :c_name_fio,
    :ki_id,
    :ki_from_acl_id,
    :ki_from_un_id,
    :ki_from_k_id,
    :ki_from_c_id,
    :ki_datetime,
    :ki_verify_income,
    :ki_is_manual,
    :ki_survey
  do
  begin
    bct_codelst = (
      select
        list(bct_code, ',')
      from kk_income_cib
        inner join bki_catalog
          on bct_id = kic_from_bct_id
      where kic_from_ki_id = :ki_id);
    ki_survey_prev = (
       select first 1
         ki_survey
       from kk_income
       where ki_from_c_id = :c_id
        and ki_from_acl_id <> :acl_id
        and ki_is_manual = 0
        and ki_datetime >= :dt_beg and ki_datetime < :dt_end
       order by ki_id desc);

    suspend;
  end
end
