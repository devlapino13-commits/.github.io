CREATE PROCEDURE AAREP_UNSENDED_POCKET(
    REP_REGION_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    KONTRACT_NO D_STR_20,
    POCKET_ID D_ID,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE,
    ERR_TEXT D_STR_255)
AS
begin
  exit;
  for select distinct k.k_kontract_no, mp.mp_id, mp.mp_date_begin, mp.mp_date_end, mlc.lc_error_code
      from mkr_err_log_credits mlc
      inner join mkr_poket mp on mlc.lc_from_poket_id = mp.mp_id
      inner join mkr_credit mc on mlc.lc_k_id = mc.mc_from_k_id
      inner join kontract k on mc.mc_from_k_id = k.k_id
      where mp.mp_is_sended = -2 and
            :rep_date between mp.mp_date_begin and mp.mp_date_end
      into :kontract_no, :pocket_id, :date_begin, :date_end, :err_text
      do
      begin
        suspend;
      end
end
