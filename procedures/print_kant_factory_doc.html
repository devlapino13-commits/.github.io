CREATE PROCEDURE PRINT_KANT_FACTORY_DOC(
    K_ID D_ID,
    INFO_TYPE D_STR_CODE)
RETURNS (
    K_DISB_DATE D_DATE,
    C_NAME_FIO D_STR_100,
    K_KONTRACT_NO D_STR_20,
    SUM_TOTAL_PRICE DECIMAL(18,2),
    K_AGENT_FIO D_STR_100,
    NAME D_STR_LONG,
    QTY D_INTEGER,
    MEASURE D_STR_20,
    UNIT_PRICE DECIMAL(18,2),
    TOTAL_PRICE DECIMAL(18,2),
    GUARANTEE_NO D_STR_100)
AS
declare variable esb_blob_txt d_blob_text;
declare variable unit_type d_integer;
declare variable good_qty d_integer;
declare variable good_list d_blob_text;
declare variable i d_integer;
declare variable params d_str_long;
declare variable trans_id d_str_20;
begin
  --info_type = 'common_info'  - Общая информация
  --info_type = 'goods_info' - Информация по товарам
  select kt.k_disb_date, cl.c_name_fio, kt.k_kontract_no, esb.esb_blob_txt, kt.k_amount_approved, esw.esw_trans_id
  from kontract kt
  inner join extminiapp ema on ema.ema_from_tkt_id = kt.k_from_tkt_id
  inner join clients cl on cl.c_id = ema.ema_from_c_id
  inner join extsysweb esw on esw.esw_id = ema.ema_from_esw_id
  inner join extsysblob esb on esb.esb_from_esw_id = esw.esw_id
  where kt.k_id = :k_id and
        esw.esw_command = 'app_save' and
        esb.esb_type = 'json' and
        esw.esw_project = 'kant_factory'
  into k_disb_date, c_name_fio, k_kontract_no, esb_blob_txt, sum_total_price, trans_id;

  select cl.c_name_fio
  from kontract kt
  inner join extminiapp ema on ema.ema_from_tkt_id = kt.k_from_tkt_id
  inner join extsysblob esb on esb.esb_from_esw_id = ema.ema_from_esw_id
  inner join clientshops cs on cs.cs_id = f_json_str_get_value(esb.esb_blob_txt, 'seller.shop_id', -1)
  inner join clients cl on cl.c_id = cs.cs_from_c_id
  where kt.k_id = :k_id and
        cs.cs_is_delete = 0
  into k_agent_fio;

  select esw.esw_full_source
  from extsysweb esw
  where esw.esw_trans_id = :trans_id and
        esw.esw_command = 'loan_disb' and
        esw.esw_project = 'kant_factory'
  order by esw.esw_datetime desc
  rows 1
  into params;

  guarantee_no = (select field_value
                  from hp_extsys_parcer(:params, 'act_number'));

  if (info_type = 'common_info') then
  begin
    suspend;
    exit;
  end

  if (info_type = 'goods_info') then
  begin
    i = 0;
    good_qty = f_json_blob_get_count(:esb_blob_txt, 'good_list');
    good_list = f_json_blob_get_value(:esb_blob_txt, 'good_list', '');
    while (i < good_qty and good_list <> '') do
    begin
      name = f_json_blob_get_value(:esb_blob_txt, 'good_list['|| cast(i as varchar(20)) || '].' || 'name', '');
      qty = f_json_blob_get_value(:esb_blob_txt, 'good_list['|| cast(i as varchar(20)) || '].' || 'qty', 0);
      unit_type = f_json_blob_get_value(:esb_blob_txt, 'good_list['|| cast(i as varchar(20)) || '].' || 'unit_type', -1);
      unit_price = f_json_blob_get_value(:esb_blob_txt, 'good_list['|| cast(i as varchar(20)) || '].' || 'unit_price', 0);
      total_price = f_json_blob_get_value(:esb_blob_txt, 'good_list['|| cast(i as varchar(20)) || '].' || 'total_price', 0);

      measure = '';
      select yl.yl_code
      from yedizlist yl
      where yl.yl_id = :unit_type
      into measure;

      i = i + 1;
      suspend;
    end
  end
end
