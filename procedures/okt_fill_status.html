CREATE PROCEDURE OKT_FILL_STATUS(
    ACL_ID ID_TYPE,
    STATUS_CODE STR_100)
RETURNS (
    RES SMALLINT)
AS
declare variable rd_id integer;
declare variable project_code str_255;
begin
  if (coalesce(acl_id, 0) = 0) then
    exit;
  if ((trim(coalesce(status_code, ''))) = '') then
    exit;

  select rd.rd_id
  from refdata rd
  where rd.rd_code = :status_code
  into :rd_id;

  select rd.rd_code
  from actionorder ao
  inner join refdata rd on ao.aco_from_rd_id = rd.rd_id
  where ao.aco_id = :acl_id
  into :project_code;

--  испольняю update, только если проект относится к ОКТЕЛЛ
  if (:project_code in (select rd.rd_code
                        from refdata rd
                        where rd.rd_code starting('OKTELL.Projects') and
                              rd.rd_parent_id is not null)) then
  begin
    update actionlist al
    set al.acl_result_rd_id = :rd_id,
        al.acl_status = -2
    where al.acl_id = :acl_id;

    suspend;
  end
end
