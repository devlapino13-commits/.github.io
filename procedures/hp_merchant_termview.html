CREATE PROCEDURE HP_MERCHANT_TERMVIEW(
    TTYPE D_SMALLINT = 1,
    MODE D_SMALLINT = 2)
RETURNS (
    TRM_ID D_INTEGER,
    TRM_TYPE D_SMALLINT,
    TRM_IPC_ID D_STR_20,
    TRM_FROM_ML_ID D_ID,
    TRM_INVENT_STR D_STR_CODE,
    TRM_SERIAL_NO D_STR_255,
    TRM_SIMCARD_NO D_STR_20,
    ML_NAME D_STR_255,
    TRM_IN_USE D_BOOL,
    TRM_ACTUAL D_BOOL,
    TRM_DATE_ANNUL D_DATETIME)
AS
begin
  for
    select
      trm_id,
      trm_type,
      trm_ipc_id,
      trm_from_ml_id,
      trm_invent_str,
      trm_serial_no,
      trm_simcard_no,
      ml_name,
      trm_actual,
      trm_date_annul
    from ipc_terminals
      inner join matlist
        on ml_id = trm_from_ml_id
    where trm_type = :ttype
      and trm_actual = iif(:mode = 2, trm_actual, :mode)
  into
    :trm_id,
    :trm_type,
    :trm_ipc_id,
    :trm_from_ml_id,
    :trm_invent_str,
    :trm_serial_no,
    :trm_simcard_no,
    :ml_name,
    :trm_actual,
    :trm_date_annul
  do
  begin
    if (exists(
      select *
      from ipc_terminal_ownership
      where ito_trm_id = :trm_id))
    then trm_in_use = 1;
      else trm_in_use = 0;
    suspend;
  end
end
