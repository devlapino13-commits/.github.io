CREATE PROCEDURE AAREP_WORKDOCLIST_TMP(
    UN_ID DECIMAL(18,0),
    FIND_TXT VARCHAR(255),
    INCLUDE_ANNUL INTEGER)
RETURNS (
    WDL_ID INTEGER,
    WDL_HAS_NEW INTEGER,
    WDK_COUNT INTEGER,
    WDL_WORD_COUNT INTEGER)
AS
declare variable c_id decimal(18,0);
declare variable ps_date_begin date;
declare variable jl_id integer;
declare variable wdl_is_open_access integer;
declare variable wdl_from_un_id decimal(18,0);
declare variable wdl_ready_user decimal(18,0);
declare variable wdl_lock_user decimal(18,0);
declare variable is_has_access integer;
declare variable k integer;
declare variable txt_tmp varchar(100);
declare variable is_vip integer;
declare variable has_access integer;
begin
  select un_from_c_id
  from username
  where un_id = :un_id
  into :c_id;

  is_vip = 0;
  if (exists(select *
             from workdocvipusers
             where wdvip_from_c_id = :c_id)) then
    is_vip = 1;

  jl_id = 0;
  has_access = 0;
  select ps_date_begin, ps_from_jl_id
  from hr_pstatus
  where ps_from_c_id = :c_id and
        ps_date_begin <= current_date and
        ps_date_end >= current_date and
        ps_is_working = 1
  order by ps_date_begin
  rows 1
  into :ps_date_begin, :jl_id;
  if (jl_id is null) then
    jl_id = 0;
  if (jl_id <> 0) then
    has_access = 1;
  if (has_access = 0) then
  begin
    if (exists(select *
               from servicedog
               where sd_from_c_id = :c_id and
                     sd_date_begin <= current_date and
                     sd_date_end >= current_date)) then
      has_access = 1;
    if ((has_access = 0) and
        (c_id in (1090100224546, 1010100105262, 1090101324764, 1090101324762, 1090100299889, 1090100880978,
        1090101324765, 1090101004395, 1090101324763, 1090100237638, 1090101324766, 1090101324769, 1090100300021,
        1090101324768, 1090101324767, 1090100299963, 1090101326452, 1090101326456, 1090100532672, 1090101471274,1090101542739))) then
      has_access = 1;
  end
  if (has_access = 1) then
  begin
    find_txt = trim(:find_txt);
    if (:find_txt <> '') then
    begin
      while (:find_txt <> '') do
      begin
        wdl_word_count = 1;
        k = position(' ', :find_txt) - 1;
        if (k >= 0) then
        begin
          txt_tmp = left(:find_txt, k);
          find_txt = trim(substring(:find_txt from k + 1 for 255));
        end
        else
        begin
          if (find_txt <> '') then
          begin
            txt_tmp = :find_txt;
            find_txt = '';
          end
        end
        for with tab
            as (select distinct wdl_id, wdl_is_open_access, wdl_from_un_id, wdl_ready_user, wdl_lock_user, wdk_count
                from workdoclist, workdocblob, workdocwkol, workdocword
                where wdb_from_wdl_id = wdl_id and
                      wdk_from_wdb_id = wdb_id and
                      wdk_from_wdw_id = wdw_id and
                      wdw_inactive = 0 and
                      ((:include_annul = 0 and
                      wdl_stop_date is null) or
                      (:include_annul = 1)) and
                      wdw_word starting :txt_tmp
                union
                select distinct wdl_id, wdl_is_open_access, wdl_from_un_id, wdl_ready_user, wdl_lock_user,
                                1 as wdk_count
                from workdoclist
                where ((:include_annul = 0 and
                      wdl_stop_date is null) or
                      (:include_annul = 1)) and
                      wdl_name containing :txt_tmp)
            select wdl_id, wdl_is_open_access, wdl_from_un_id, wdl_ready_user, wdl_lock_user, max(wdk_count)
            from tab
            group by wdl_id, wdl_is_open_access, wdl_from_un_id, wdl_ready_user, wdl_lock_user
            into :wdl_id, :wdl_is_open_access, :wdl_from_un_id, :wdl_ready_user, :wdl_lock_user, :wdk_count
        do
        begin
          if (is_vip = 0) then
          begin
            is_has_access = 0;
            if ((wdl_is_open_access = 1) or (wdl_from_un_id = :un_id) or (wdl_ready_user = :un_id) or (wdl_lock_user = :un_id)) then
              is_has_access = 1;

            if (is_has_access = 0) then
            begin
              if (exists(select *
                         from workdoccl
                         where wdc_from_wdl_id = :wdl_id and
                               wdc_from_c_id = :c_id)) then
                is_has_access = 1;
            end
            if (is_has_access = 0) then
            begin
              if (exists(select *
                         from workdocjl
                         where wdu_from_wdl_id = :wdl_id and
                               wdu_from_jl_id = :jl_id)) then
                is_has_access = 1;
            end
          end
          else
            is_has_access = 1;
          if (is_has_access = 1) then
          begin
            if (exists(select *
                       from workdocstat
                       where wds_from_wdl_id = :wdl_id and
                             wds_from_un_id = :un_id)) then
              wdl_has_new = 0;
            else
              wdl_has_new = 1;
            suspend;
            wdl_word_count = 0;
          end
        end
      end
    end
    else
    begin
      wdl_word_count = 0;
      wdk_count = 0;
      for select distinct wdl_id, wdl_is_open_access, wdl_from_un_id, wdl_ready_user, wdl_lock_user
          from workdoclist
          where ((:include_annul = 0 and
                wdl_stop_date is null) or
                (:include_annul = 1))
          into :wdl_id, :wdl_is_open_access, :wdl_from_un_id, :wdl_ready_user, :wdl_lock_user
      do
      begin
        if (is_vip = 0) then
        begin
          is_has_access = 0;
          if ((wdl_is_open_access = 1) or (wdl_from_un_id = :un_id) or (wdl_ready_user = :un_id) or (wdl_lock_user = :un_id)) then
            is_has_access = 1;

          if (is_has_access = 0) then
          begin
            if (exists(select *
                       from workdoccl
                       where wdc_from_wdl_id = :wdl_id and
                             wdc_from_c_id = :c_id)) then
              is_has_access = 1;
          end
          if (is_has_access = 0) then
          begin
            if (exists(select *
                       from workdocjl
                       where wdu_from_wdl_id = :wdl_id and
                             wdu_from_jl_id = :jl_id)) then
              is_has_access = 1;
          end
        end
        else
          is_has_access = 1;
        if (is_has_access = 1) then
        begin
          if (exists(select *
                     from workdocstat
                     where wds_from_wdl_id = :wdl_id and
                           wds_from_un_id = :un_id)) then
            wdl_has_new = 0;
          else
            wdl_has_new = 1;
          suspend;
        end
      end

    end
  end
end
