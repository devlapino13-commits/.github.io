CREATE PROCEDURE EPASS_RESP_SAVE(
    ERS_FROM_ER_ID D_INTEGER,
    ERS_STATUS D_INTEGER,
    ERS_DOCUMENT D_BLOB_BINARY,
    ERS_JSON D_STR_LONG)
AS
declare variable bls_id d_id;
declare variable file_name d_str_100;
declare variable caption d_str_100;
declare variable un_id d_id;
begin
  -- если успешно, сохраняем ответ
  if (ers_status = 0) then
  begin
    select er.er_from_un_id
    from epass_req er
    where er.er_id = :ers_from_er_id
    into :un_id;

    caption = 'epass_tunduk';
    file_name = 'tunduk_' || :ers_from_er_id || '.pdf';
    -- сохранение pdf в blob
    execute procedure hp_insert_into_blobstore (:file_name, :caption, :ers_document, :un_id, 4, 3)
    returning_values :bls_id;

    insert into epass_response (erp_from_er_id, erp_pdf_blob, erp_json)
    values (:ers_from_er_id, :bls_id, :ers_json);

    update epass_req er
    set er.er_status = 2
    where er.er_id = :ers_from_er_id;
  end
  else
  begin
    -- в остальных случаях сохраняем статус ответа
    insert into epass_bad_response (ebr_from_er_id, ebr_code)
    values (:ers_from_er_id, :ers_status);

    update epass_req er
    set er.er_status = -1
    where er.er_id = :ers_from_er_id;
  end
end
