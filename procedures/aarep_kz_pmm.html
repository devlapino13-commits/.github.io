CREATE PROCEDURE AAREP_KZ_PMM(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    BALANCE_BEGIN D_CURRENCY,
    BALANCE_END D_CURRENCY,
    DISB_SUM D_CURRENCY,
    REPAY_OS D_CURRENCY,
    DATE_STR D_STR_255,
    BALANCE_PERCENT D_CURRENCY,
    BALANCE_PERC_EFFECT D_CURRENCY,
    PERC_NOMINAL D_CURRENCY,
    PERC_EFFECT D_PERCENT)
AS
declare variable REG_ID D_INTEGER;
declare variable END_LAST_DATE D_DATE;
declare variable OKS_BALANCE D_CURRENCY;
declare variable K_AMOUNT_APPROVED D_CURRENCY;
declare variable OK_OS D_CURRENCY;
declare variable OKS_BALANCE_PERC D_CURRENCY;
declare variable OKS_BALANCE_PERC_EFF D_CURRENCY;
begin
  date_str = 'Отчетный период: по состоянию на '||f_datetostr(:rep_date_end);

  end_last_date = rep_date_begin - 1;
  balance_begin = 0;
  balance_end = 0;
  disb_sum = 0;
  repay_os = 0;
  balance_percent = 0;
  balance_perc_effect = 0;
  for
  select id_list from region_list(:rep_region_id,1,3,1)
  into reg_id
  do
  begin
    -- портфель на начало
    oks_balance = null;
    select sum(oks_balance) from oksummary
    where :end_last_date between oks_date_begin and oks_date_end
    and oks_region_id = :reg_id
    and oks_close_state = 0
    into oks_balance;
    balance_begin = balance_begin + coalesce(oks_balance, 0);

    -- портфель на конец
    oks_balance = null;
    oks_balance_perc = null;
    oks_balance_perc_eff = null;
    select sum(oks_balance),
           sum(oks_balance/ 100 * k_percent),
           sum(oks_balance/ 100 * k_percent_effect)
    from oksummary, kontract
    where :rep_date_end between oks_date_begin and oks_date_end
    and oks_region_id = :reg_id
    and oks_close_state = 0
    and oks_from_k_id = k_id
    into oks_balance, oks_balance_perc, oks_balance_perc_eff;
    balance_end = balance_end + coalesce(oks_balance, 0);
    balance_percent = balance_percent + coalesce(oks_balance_perc, 0);
    balance_perc_effect = balance_perc_effect + coalesce(oks_balance_perc_eff, 0);

    -- выданные за текущий период
    k_amount_approved = null;
    select sum(k_amount_approved)
    from kontstate, kontract
    where ks_region_id = :reg_id
    and ks_date_begin between :rep_date_begin and :rep_date_end
    and ks_state = 1
    and ks_from_k_id = k_id
    into k_amount_approved;
    disb_sum = disb_sum + coalesce(k_amount_approved, 0);

    -- Погашение кредитного портфеля в отчетном периоде
    ok_os = null;
    select -sum(ok_os) from operkred
    where ok_ol_is_active = 1
    and ok_ol_date between :rep_date_begin and :rep_date_end
    and ok_ol_reg_id = :reg_id
    and ok_type = 4
    into ok_os;
    repay_os = repay_os + coalesce(ok_os, 0);

  end

  if (balance_end <> 0) then
  begin
    perc_nominal = balance_percent / balance_end * 100;
    perc_effect = balance_perc_effect / balance_end * 100;
  end

  balance_begin = balance_begin / 1000;
  balance_end = balance_end / 1000;
  disb_sum = disb_sum / 1000;
  repay_os = repay_os / 1000;

  suspend;
end
