CREATE PROCEDURE HP_ANALITICA_GET_LIST(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_EXCLUDE_ACTIVE DATE,
    REP_LO_ID DECIMAL(18,0))
RETURNS (
    RES_K_ID DECIMAL(18,0),
    RES_BEGIN_IS_ACTIVE INTEGER,
    RES_BEGIN_REGION INTEGER,
    RES_BEGIN_KC DECIMAL(18,0),
    RES_BEGIN_BALANCE DECIMAL(18,4),
    RES_BEGIN_DELINQ_DATE DATE,
    RES_END_IS_ACTIVE INTEGER,
    RES_END_REGION INTEGER,
    RES_END_KC DECIMAL(18,0),
    RES_END_BALANCE DECIMAL(18,4),
    RES_END_DELINQ_DATE DATE,
    RES_DIFF_BALANCE DECIMAL(18,4),
    KS_DATE_BEGIN DATE)
AS
declare variable rep_date_beginning date;
--declare variable ks_date_begin date;
declare variable ks_date_end date;
declare variable is_office integer;
begin
-- Процедура возвращает все активные займы на начало и на конец
-- Если займ на начало периода был в другом офисе, все равно получаем их биланс
  rep_date_beginning = rep_date_begin - 1;
  if (rep_lo_id is null) then
    rep_lo_id = 0;
  if (rep_exclude_active is null) then
    rep_exclude_active = '01.01.1990';

  if (rep_region_id = 1000000) then -- Отчет на уровне региона
  begin
    for select ks_from_k_id, ks_date_begin, ks_date_end
        from kontstate
        where ks_date_begin between :rep_exclude_active and :rep_date_end and
              ks_date_end >= :rep_date_beginning and
              ks_state = 1 and
              ks_ol_is_delete = 0
        into :res_k_id, ks_date_begin, ks_date_end
    do
    begin
      res_begin_is_active = 0;
      res_begin_region = 0;
      res_begin_kc = 0;
      res_begin_balance = 0;
      res_begin_delinq_date = null;
      res_end_is_active = 0;
      res_end_region = 0;
      res_end_kc = 0;
      res_end_balance = 0;
      res_end_delinq_date = null;
      res_diff_balance = 0;

      if (ks_date_begin <= :rep_date_beginning and
          ks_date_end >= :rep_date_beginning) then
      begin
        select oks_region_id, oks_kc_id, oks_balance, oks_delinq_date
        from oksummary
        where oks_from_k_id = :res_k_id and
              oks_close_state = 0 and
              oks_date_begin <= :rep_date_beginning and
              oks_date_end >= :rep_date_beginning
        into :res_begin_region, :res_begin_kc, :res_begin_balance, :res_begin_delinq_date;
        if (res_begin_balance > 0) then
          res_begin_is_active = 1;
        if (res_begin_delinq_date > :rep_date_beginning) then
          res_begin_delinq_date = null;
        if (res_begin_region = 0) then
          res_begin_kc = 0;
      end

      if (ks_date_begin <= :rep_date_end and
          ks_date_end >= :rep_date_end) then
      begin
        select oks_region_id, oks_kc_id, oks_balance, oks_delinq_date
        from oksummary
        where oks_from_k_id = :res_k_id and
              oks_close_state = 0 and
              oks_date_begin <= :rep_date_end and
              oks_date_end >= :rep_date_end
        into :res_end_region, :res_end_kc, :res_end_balance, :res_end_delinq_date;
        if (res_end_balance > 0) then
          res_end_is_active = 1;
        if (res_end_delinq_date > :rep_date_end) then
          res_end_delinq_date = null;
        if (res_end_region = 0) then
          res_end_kc = 0;
      end

      if (ks_date_end >= :rep_date_beginning and
          ks_date_end < :rep_date_end) then
      begin
        res_end_is_active = 0;
        select oks_region_id, oks_kc_id
        from oksummary
        where oks_from_k_id = :res_k_id and
              oks_close_state = 0 and
              oks_date_begin <= :rep_date_end
        order by oks_from_k_id desc, oks_date_begin desc
        rows 1
        into :res_end_region, :res_end_kc;
        if (res_end_region = 0) then
          res_end_kc = 0;
      end

      res_diff_balance = res_end_balance - res_begin_balance;
      suspend;
    end
    exit;
  end -- Конец приготовления на уровне компании

  if (not(exists(select *
                 from temp_analitica_regions))) then
  begin
    insert into temp_analitica_regions (reg_id, result_reg_id)
    with tab
    as (select id_list, 0 as result_reg_id
        from region_list(1000000, 1, 3, 1)
        union
        select id_list, id_list as result_reg_id
        from region_list(:rep_region_id, 1, 3, 1))
    select id_list, max(result_reg_id)
    from tab
    group by id_list;
  end

  if (rep_lo_id <> 0) then
  begin -- Отчет на уровне одного офиса
    for select distinct oks_from_k_id, ks_date_begin, ks_date_end
        from kontstate, oksummary
        where ks_state = 1 and
              ks_ol_is_delete = 0 and
              ks_date_begin >= :rep_exclude_active and
              oks_date_begin <= :rep_date_end and
              oks_date_end >= :rep_date_beginning and
              oks_region_id = :rep_region_id and
              oks_kc_id = :rep_lo_id and
              oks_from_k_id = ks_from_k_id
        into :res_k_id, ks_date_begin, ks_date_end
    do
    begin
      res_begin_is_active = 0;
      res_begin_region = 0;
      res_begin_kc = 0;
      res_begin_balance = 0;
      res_begin_delinq_date = null;
      res_end_is_active = 0;
      res_end_region = 0;
      res_end_kc = 0;
      res_end_balance = 0;
      res_end_delinq_date = null;
      res_diff_balance = 0;

      if (ks_date_begin <= :rep_date_beginning and
          ks_date_end >= :rep_date_beginning) then
      begin
        select result_reg_id, oks_kc_id, oks_balance, oks_delinq_date
        from oksummary, temp_analitica_regions
        where oks_from_k_id = :res_k_id and
              oks_close_state = 0 and
              oks_date_begin <= :rep_date_beginning and
              oks_date_end >= :rep_date_beginning and
              oks_region_id = reg_id
        into :res_begin_region, :res_begin_kc, :res_begin_balance, :res_begin_delinq_date;
        if (res_begin_balance > 0) then
          res_begin_is_active = 1;
        if (res_begin_delinq_date > :rep_date_beginning) then
          res_begin_delinq_date = null;
        if (res_begin_region = 0) then
          res_begin_kc = 0;
      end

      if (ks_date_begin <= :rep_date_end and
          ks_date_end >= :rep_date_end) then
      begin
        select result_reg_id, oks_kc_id, oks_balance, oks_delinq_date
        from oksummary, temp_analitica_regions
        where oks_from_k_id = :res_k_id and
              oks_close_state = 0 and
              oks_date_begin <= :rep_date_end and
              oks_date_end >= :rep_date_end and
              oks_region_id = reg_id
        into :res_end_region, :res_end_kc, :res_end_balance, :res_end_delinq_date;
        if (res_end_balance > 0) then
          res_end_is_active = 1;
        if (res_end_delinq_date > :rep_date_end) then
          res_end_delinq_date = null;
        if (res_end_region = 0) then
          res_end_kc = 0;
      end

      if (ks_date_end >= :rep_date_beginning and
          ks_date_end < :rep_date_end) then
      begin
        res_end_is_active = 0;
        select result_reg_id, oks_kc_id
        from oksummary, temp_analitica_regions
        where oks_from_k_id = :res_k_id and
              oks_close_state = 0 and
              oks_date_begin <= :rep_date_end and
              oks_region_id = reg_id
        order by oks_from_k_id desc, oks_date_begin desc
        rows 1
        into :res_end_region, :res_end_kc;
        if (res_end_region = 0) then
          res_end_kc = 0;
      end

      res_diff_balance = res_end_balance - res_begin_balance;
      if (res_end_region <> 0 or res_begin_region <> 0) then
      begin
        suspend;
      end
    end
    exit;
  end

  if (exists(select *
             from region
             where reg_unicode = :rep_region_id and
                   reg_status = 3)) then
    is_office = 1;
  else
    is_office = 0;

  if (is_office = 0) then -- Отчет на уровне региона
  begin
    for select ks_from_k_id, ks_date_begin, ks_date_end
        from kontstate
        where ks_date_begin between :rep_exclude_active and :rep_date_end and
              ks_date_end >= :rep_date_beginning and
              ks_state = 1 and
              ks_ol_is_delete = 0
        into :res_k_id, ks_date_begin, ks_date_end
    do
    begin
      res_begin_is_active = 0;
      res_begin_region = 0;
      res_begin_kc = 0;
      res_begin_balance = 0;
      res_begin_delinq_date = null;
      res_end_is_active = 0;
      res_end_region = 0;
      res_end_kc = 0;
      res_end_balance = 0;
      res_end_delinq_date = null;
      res_diff_balance = 0;

      if (ks_date_begin <= :rep_date_beginning and
          ks_date_end >= :rep_date_beginning) then
      begin
        select result_reg_id, oks_kc_id, oks_balance, oks_delinq_date
        from oksummary, temp_analitica_regions
        where oks_from_k_id = :res_k_id and
              oks_close_state = 0 and
              oks_date_begin <= :rep_date_beginning and
              oks_date_end >= :rep_date_beginning and
              oks_region_id = reg_id
        into :res_begin_region, :res_begin_kc, :res_begin_balance, :res_begin_delinq_date;
        if (res_begin_balance > 0) then
          res_begin_is_active = 1;
        if (res_begin_delinq_date > :rep_date_beginning) then
          res_begin_delinq_date = null;
        if (res_begin_region = 0) then
          res_begin_kc = 0;
      end

      if (ks_date_begin <= :rep_date_end and
          ks_date_end >= :rep_date_end) then
      begin
        select result_reg_id, oks_kc_id, oks_balance, oks_delinq_date
        from oksummary, temp_analitica_regions
        where oks_from_k_id = :res_k_id and
              oks_close_state = 0 and
              oks_date_begin <= :rep_date_end and
              oks_date_end >= :rep_date_end and
              oks_region_id = reg_id
        into :res_end_region, :res_end_kc, :res_end_balance, :res_end_delinq_date;
        if (res_end_balance > 0) then
          res_end_is_active = 1;
        if (res_end_delinq_date > :rep_date_end) then
          res_end_delinq_date = null;
        if (res_end_region = 0) then
          res_end_kc = 0;
      end

      if (ks_date_end >= :rep_date_beginning and
          ks_date_end < :rep_date_end) then
      begin
        res_end_is_active = 0;
        select result_reg_id, oks_kc_id
        from oksummary, temp_analitica_regions
        where oks_from_k_id = :res_k_id and
              oks_close_state = 0 and
              oks_date_begin <= :rep_date_end and
              oks_region_id = reg_id
        order by oks_from_k_id desc, oks_date_begin desc
        rows 1
        into :res_end_region, :res_end_kc;
        if (res_end_region = 0) then
          res_end_kc = 0;
      end

      res_diff_balance = res_end_balance - res_begin_balance;
      if (res_end_region <> 0 or res_begin_region <> 0) then
      begin
        suspend;
      end
    end
  end
  else
  begin -- Отчет на уровне одного офиса
    for select distinct oks_from_k_id, ks_date_begin, ks_date_end
        from kontstate, oksummary
        where ks_state = 1 and
              ks_ol_is_delete = 0 and
              ks_date_begin >= :rep_exclude_active and
              oks_date_begin <= :rep_date_end and
              oks_date_end >= :rep_date_beginning and
              oks_region_id = :rep_region_id and
              oks_from_k_id = ks_from_k_id
        into :res_k_id, ks_date_begin, ks_date_end
    do
    begin
      res_begin_is_active = 0;
      res_begin_region = 0;
      res_begin_kc = 0;
      res_begin_balance = 0;
      res_begin_delinq_date = null;
      res_end_is_active = 0;
      res_end_region = 0;
      res_end_kc = 0;
      res_end_balance = 0;
      res_end_delinq_date = null;
      res_diff_balance = 0;

      if (ks_date_begin <= :rep_date_beginning and
          ks_date_end >= :rep_date_beginning) then
      begin
        select result_reg_id, oks_kc_id, oks_balance, oks_delinq_date
        from oksummary, temp_analitica_regions
        where oks_from_k_id = :res_k_id and
              oks_close_state = 0 and
              oks_date_begin <= :rep_date_beginning and
              oks_date_end >= :rep_date_beginning and
              oks_region_id = reg_id
        into :res_begin_region, :res_begin_kc, :res_begin_balance, :res_begin_delinq_date;
        if (res_begin_balance > 0) then
          res_begin_is_active = 1;
        if (res_begin_delinq_date > :rep_date_beginning) then
          res_begin_delinq_date = null;
        if (res_begin_region = 0) then
          res_begin_kc = 0;
      end

      if (ks_date_begin <= :rep_date_end and
          ks_date_end >= :rep_date_end) then
      begin
        select result_reg_id, oks_kc_id, oks_balance, oks_delinq_date
        from oksummary, temp_analitica_regions
        where oks_from_k_id = :res_k_id and
              oks_close_state = 0 and
              oks_date_begin <= :rep_date_end and
              oks_date_end >= :rep_date_end and
              oks_region_id = reg_id
        into :res_end_region, :res_end_kc, :res_end_balance, :res_end_delinq_date;
        if (res_end_balance > 0) then
          res_end_is_active = 1;
        if (res_end_delinq_date > :rep_date_end) then
          res_end_delinq_date = null;
        if (res_end_region = 0) then
          res_end_kc = 0;
      end

      if (ks_date_end >= :rep_date_beginning and
          ks_date_end < :rep_date_end) then
      begin
        res_end_is_active = 0;
        select result_reg_id, oks_kc_id
        from oksummary, temp_analitica_regions
        where oks_from_k_id = :res_k_id and
              oks_close_state = 0 and
              oks_date_begin <= :rep_date_end and
              oks_region_id = reg_id
        order by oks_from_k_id desc, oks_date_begin desc
        rows 1
        into :res_end_region, :res_end_kc;
        if (res_end_region = 0) then
          res_end_kc = 0;

        suspend;
      end
    end
  end
