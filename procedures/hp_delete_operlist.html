CREATE PROCEDURE HP_DELETE_OPERLIST(
    UN_ID D_ID,
    OLR_ID D_ID,
    IS_DEL D_SMALLINT,
    RD_ID D_INTEGER)
RETURNS (
    ERR_TEXT VARCHAR(255))
AS
declare variable ol_id decimal(18,0);
declare variable k_id d_id;
declare variable oper_kind d_str_code;
begin
  err_text = '';
  if (is_del = 1) then
  begin

     select opr.olr_from_ol_id
     from operlistappr opr
     where opr.olr_id = :olr_id
     into :ol_id;

     select opc.opc_from_k_id,
            f_enum_code('cashoperkind', opc.opc_oper_kind)
     from opercash opc
     where opc_from_ol_id = :ol_id and
           opc_is_delete = 0
     rows 1
     into k_id,
          oper_kind;

    update operlistappr
    set olr_appr_rd_id = :rd_id,
        olr_appr_status = :is_del,
        olr_appr_user = :un_id,
        olr_appr_datetime = current_timestamp
    where (olr_id = :olr_id);

    rdb$set_context('USER_TRANSACTION', 'CASH_DELETING', '1');

    update opercash
    set opc_is_delete = :is_del
    where opc_from_ol_id = :ol_id and
          opc_is_delete = 0;

    update operlist
    set is_delete = :is_del
    where (ol_id = :ol_id);

    if (oper_kind = 'minus.loan.disb') then
    begin
      select res_text
      from hp_annual_insurance_loan(:k_id, :un_id)
      into err_text;
    end
  end
  else
  if (is_del = 2) then
  begin
    update operlistappr
    set olr_appr_rd_id = :rd_id,
        olr_appr_status = :is_del,
        olr_appr_user = :un_id,
        olr_appr_datetime = current_timestamp
    where (olr_id = :olr_id);
  end
  suspend;
end
