CREATE PROCEDURE AAREP_CALC_AMORT_FULL_TEMP(
    REG_ID INTEGER,
    DD DATE)
RETURNS (
    PCA_BILL_ID DECIMAL(18,0),
    PCA_BILL_LINE_NO INTEGER,
    PCA_START_YEAR INTEGER,
    PCA_START_MONTH INTEGER,
    PCA_YEAR_NO INTEGER,
    PCA_MONTH_NO INTEGER,
    PCA_AGE INTEGER,
    PCA_TOTAL_OS DECIMAL(18,4),
    PCA_TOTAL_AMORT DECIMAL(18,4),
    PCA_LIFETIME INTEGER,
    PCA_WP_ID DECIMAL(18,0),
    PCA_MO_ID DECIMAL(18,0),
    PCA_ML_ID DECIMAL(18,0),
    PCA_MC_ID DECIMAL(18,0),
    PCA_INVENT_STR VARCHAR(20),
    PCA_CURR_OS DECIMAL(18,4),
    PCA_CURR_AMORT DECIMAL(18,4),
    PCA_CURR_AMORT_OTHER DECIMAL(18,4),
    PCA_LIFETIME_SALDO INTEGER,
    PCA_COST DECIMAL(18,4),
    PCA_AMORT_ACC VARCHAR(20),
    PCA_REGION_ID INTEGER)
AS
declare variable year_no integer;
declare variable month_no integer;
begin
  /* Procedure Text */
  year_no = extract(year from dd);
  month_no = extract(month from dd);

  pca_start_year = 0;
  pca_start_month = 0;
  pca_year_no = 0;
  pca_month_no = 0;
  pca_age = 0;
  pca_total_os = 0;
  pca_total_amort = 0;
  pca_lifetime = 0;
  pca_curr_os = 0;
  pca_curr_amort = 0;
  pca_curr_amort_other = 0;
  pca_lifetime_saldo = 0;

  for select mbal_summa, mbal_bill_id, mbal_bill_line_no, mbal_wp_id, mbal_mo_id, mbal_ml_id, mbal_mc_id,
             mbal_invent_str, mbal_region_id, mc_acc_amort
      from matlist_balance_list(:reg_id, 0, :dd, 1, 1), matcateg
      where mc_id = mbal_mc_id
      into :pca_cost, :pca_bill_id, :pca_bill_line_no, :pca_wp_id, :pca_mo_id, :pca_ml_id, :pca_mc_id, :pca_invent_str,
           :pca_region_id, :pca_amort_acc
  do
  begin
    suspend;
  end

  pca_cost = 0;
  pca_wp_id = 0;
  pca_mo_id = 0;
  pca_ml_id = 0;
  pca_mc_id = 0;
  pca_invent_str = '';
  pca_region_id = 0;
  pca_amort_acc = '';

  for select pca_bill_id, pca_bill_line_no, pca_start_year, pca_start_month, pca_year_no, pca_month_no, pca_age,
             sum(pca_total_os), sum(pca_total_amort), max(pca_lifetime), sum(pca_curr_os), sum(pca_curr_amort),
             sum(pca_curr_amort_other), sum(pca_lifetime_saldo)
      from aarep_calc_amort_temp(:reg_id, :year_no, :month_no)
      group by pca_bill_id, pca_bill_line_no, pca_start_year, pca_start_month, pca_year_no, pca_month_no, pca_age
      into :pca_bill_id, :pca_bill_line_no, :pca_start_year, :pca_start_month, :pca_year_no, :pca_month_no, :pca_age,
           :pca_total_os, :pca_total_amort, :pca_lifetime, :pca_curr_os, :pca_curr_amort, pca_curr_amort_other,
           :pca_lifetime_saldo
  do
  begin
    suspend;
  end
end
