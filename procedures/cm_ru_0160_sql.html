CREATE PROCEDURE CM_RU_0160_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable acl_date date;
declare variable rd_id d_integer;
begin
  res_error_is_critical = 1;

  for select kont.k_id
      from kontract kont
      inner join product pr on pr.pr_id = kont.k_from_pr_id
      where kont.k_id = :k_id and
            ((pr.pr_code like '%.zalog_car.%') or
            (pr.pr_code like '%.education.%'))
      into :k_id
  do
  begin
    select rd_id
    from hp_get_check_loan_data(:k_id)
    into :rd_id;

    select cast(acl_last_datetime as date)
    from actionlist
    where acl_object_kind = 9 and
          acl_object_id = :k_id and
          acl_from_rd_id = :rd_id and
          acl_status = 2
    into :acl_date;
    if (acl_date is not null) then
    begin
      if (acl_date > cast('18.03.2020' as date)) then
      begin
        res_error_text = 'Для продукта Образование и Автозайм для выдачи допустимо подтверждение КК не позднее 18.03.2020';
        suspend;
      end
    end
  end
end
