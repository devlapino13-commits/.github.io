CREATE PROCEDURE CM_RU_0030_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable k_amount_approved curr_type;
declare variable k_kolich_repay integer;
declare variable pr_code varchar(20);
declare variable in_rows1 integer;
declare variable rows1_cli_id id_type;
declare variable kc_from_c_id id_type;
declare variable k_kommis_perc perc_type;
declare variable k curr_type;
declare variable kol_cli_group integer;
declare variable C_CURR_BONUS_STATUS integer;
declare variable PRL_PERCENT_MAX perc_type;
declare variable pr_id integer;
declare variable k_percent perc_type;
declare variable c_citizen integer;
declare variable rd_code varchar(255);
begin
  for
  select k_amount_approved,
         k_kolich_repay,
         pr_code,
         k_disb_date,
         k_kommis_perc,
         k_kolich_client_approved,
         pr_id,
         k_percent
  from kontract, product
  where k_id = :k_id
  and k_from_pr_id = pr_id
  and pr_code similar to '%(.FAMILY.|.FAMILY PLUS.|.BARAKAT.|.BARAKAT PLUS.)%'
  into k_amount_approved,
       k_kolich_repay,
       pr_code,
       kf_disb_date,
       k_kommis_perc,
       kol_cli_group,
       pr_id,
       k_percent
  do
  begin
    C_CURR_BONUS_STATUS = null;
    select C_CURR_BONUS_STATUS
    from HP_KONTSTATE_LOAN_STATUS(:K_ID, 0, 0, :kf_disb_date, null)
    into C_CURR_BONUS_STATUS;
    if (C_CURR_BONUS_STATUS is null) then C_CURR_BONUS_STATUS = 0;
    PRL_PERCENT_MAX = null;
    select max(PRL_PERCENT_MAX)
    from prodlogic
    where prl_from_pr_id = :pr_id
    and :kf_disb_date between prl_date_begin and prl_date_end
    and :kol_cli_group between prl_kolich_min and prl_kolich_max
    and :k_kolich_repay between prl_repay_min and prl_repay_max
    into PRL_PERCENT_MAX;
    if (PRL_PERCENT_MAX is null) then PRL_PERCENT_MAX = 100;
--    if (K_Percent <> (PRL_PERCENT_MAX - C_CURR_BONUS_STATUS * 0.2)) then
--    begin
--       res_error_text = 'Процентная ставка не правильно указано!';
--       res_error_is_critical = 2;
--       suspend;
--    end 
--  проверка по продукту Семья
    if (pr_code containing '.FAMILY.') then
    begin
      if (kol_cli_group <> 2) then
      begin
        res_error_text = 'Кол-во клиентов данного продукта не правильно указано!';
        res_error_is_critical = 2;
        suspend;
      end 
      if (k_kommis_perc <> 1.25) then
      begin
        res_error_text = 'Для данного продукта Доп. процент не правильно указано!';
        res_error_is_critical = 2;
        suspend;
      end
    end
--  проверка по продукту Семья Плюс
    if (pr_code containing '.FAMILY PLUS.') then
    begin
      if (kol_cli_group <> 3) then
      begin
        res_error_text = 'Кол-во клиентов данного продукта не правильно указан!';
        res_error_is_critical = 2;
        suspend;
      end 
      if (k_kommis_perc <> 1.00) then
      begin
        res_error_text = 'Для данного продукта Доп. процент не правильно указан!';
        res_error_is_critical = 2;
        suspend;
      end
    end 
--  проверка членов семья должны быть родственниками
--  Семья и Семья+
    if (pr_code containing '.FAMILY') then
    begin
      in_rows1 = 1;
      rows1_cli_id = 0;
      for
      select kc_from_c_id, kc_amount_approved
      from knclient
      where kc_from_k_id = :k_id
      and kc_amount_approved > 0
      into kc_from_c_id, kc_amount
      do
      begin
        select c_name_fio
        from clients
        where c_id = :kc_from_c_id
        into res_info_text;
        if (in_rows1 = 1) then
        begin
          in_rows1 = 0;
          rows1_cli_id = kc_from_c_id;
        end else
        begin
          if ((:kc_from_c_id not in (select CREL_OTHER_C_ID
                                     from HP_CLIENT_RELATION_LIST(:rows1_cli_id,0))
              )) then
          begin
            res_error_text = 'Клиент '||res_info_text||' не является родственником!';
            res_error_is_critical = 0;
            suspend;
          end 
        end
        if (pr_code containing '.FAMILY.') then k = 35000;
        else k = 30000;
        if (kc_amount not in (15000, 25000, :k)) then
        begin
          res_error_text = 'У клиента указана не правильная сумма!';
          res_error_is_critical = 2;
          suspend;
        end
      end
    end
--  проверка по продукту Семья
    if (pr_code containing '.BARAKAT.') then
    begin
      if (kol_cli_group <> 1) then
      begin
        res_error_text = 'Кол-во клиентов данного продукта не правильно указан!';
        res_error_is_critical = 2;
        suspend;
      end 
      if (k_kommis_perc <> 1.50) then
      begin
        res_error_text = 'Для данного продукта Доп. процент не правильно указан!';
        res_error_is_critical = 2;
        suspend;
      end
      for
      select kc_from_c_id, kc_amount_approved
      from knclient
      where kc_from_k_id = :k_id
      and kc_amount_approved > 0
      into kc_from_c_id, kc_amount
      do
      begin
        select c_name_fio, c_citizen
        from clients
        where c_id = :kc_from_c_id
        into res_info_text, c_citizen;
        select rd_code from refdata
        where rd_id = :c_citizen
        into rd_code;

--    Обязательно наличие у клиента Патента или Разрешения на работу
          if (not exists(
                select *
                from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                where CDS_FROM_C_ID = :kc_from_c_id
                and cds_from_pk_id = pk_id
                and pk_from_pdt_id = pdt_id
                and cds_is_delete = 0
                and :kf_disb_date between CDS_DOC_DATE_BEGIN and iif(CDS_DOC_DATE_END is null and pdt_code = 'job_agreement','31.12.9999', CDS_DOC_DATE_END)
                and (pdt_code in ('work_permit', -- Разрешение на работу
                                 'work_patent')  -- Патент
                     or (:rd_code in ('Citizen.KG', 'Citizen.RU')
                         and pdt_code = 'job_agreement') -- Трудовой договор или справа с места работы
                    ))) then
          begin
            res_error_text = 'У клиента должно быть патент или разрешения на работу!';
            res_error_is_critical = 1;
            suspend;
          end
          if (kc_amount not in (10000
--                              , 15000 пока без этого, но потом нужно дополнительную проверку по пукнтку 3,2 нужно сделать
--                              , 20000
                              )) then
          begin
            res_error_text = 'У клиента указана не правильная сумма!';
            res_error_is_critical = 2;
            suspend;
          end
      end
    end 
--  проверка по продукту Семья Плюс
    if (pr_code containing '.BARAKAT PLUS.') then
    begin
      if (kol_cli_group <> 1) then
      begin
        res_error_text = 'Кол-во клиентов данного продукта не правильно указан!';
        res_error_is_critical = 2;
        suspend;
      end 
      if (k_kommis_perc <> 1.50) then
      begin
        res_error_text = 'Для данного продукта Доп. процент не правильно указан!';
        res_error_is_critical = 2;
        suspend;
      end
      for
      select kc_from_c_id, kc_amount_approved
      from knclient
      where kc_from_k_id = :k_id
      and kc_amount_approved > 0
      into kc_from_c_id, kc_amount
      do
      begin
        select c_name_fio, c_citizen
        from clients
        where c_id = :kc_from_c_id
        into res_info_text, c_citizen;
        select rd_code from refdata
        where rd_id = :c_citizen
        into rd_code;
--    Обязательно наличие у клиента Разрешения на временное проживание в РФ или Вид на жительство
          if (not exists(
                select *
                from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                where CDS_FROM_C_ID = :kc_from_c_id
                and cds_from_pk_id = pk_id
                and pk_from_pdt_id = pdt_id
                and cds_is_delete = 0
                and :kf_disb_date between CDS_DOC_DATE_BEGIN and iif(CDS_DOC_DATE_END is null and pdt_code = 'job_agreement','31.12.9999', CDS_DOC_DATE_END)
                and (pdt_code in ('registration_rvp', -- Разрешение на временное проживание (РВП)
                                 'registration_vnj') -- Вид на жительство (ВНЖ)
                    or (:rd_code in ('Citizen.KG', 'Citizen.RU')
                         and pdt_code = 'job_agreement')) -- Трудовой договор или справа с места работы
                    )) then
          begin
              res_error_text = 'У клиента должно быть Разрешения на временное проживание или Вид на жительство';
              res_error_is_critical = 1;
              suspend;
          end
          if (kc_amount not in (10000, 15000
--                              , 20000, 30000 пока без этого, потом по пукнту 3,2 нужно доработать
                              )) then
          begin
            res_error_text = 'У клиента указано не правильная сумма!';
            res_error_is_critical = 2;
            suspend;
          end
      end
    end
  end
end
