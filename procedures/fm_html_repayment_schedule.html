CREATE PROCEDURE FM_HTML_REPAYMENT_SCHEDULE(
    OBJ_ID D_ID,
    OBJ_KIND D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_HTML D_BLOB_TEXT)
AS
declare variable RES_FIELD_NAME D_STR_2048;
declare variable RES_LINK D_STR_LONG;
declare variable LANG D_STR_LANG;
declare variable ID D_INTEGER;
declare variable C_NAME_FIO D_STR_255;
declare variable FORMATTED_DATE D_STR_20;
declare variable TOTAL_AMOUNT decimal(18,2);
declare variable K_STATUS D_ID;
begin
    lang = f_get_lang_code(un_id, null);

    if (obj_kind = f_objkind('miniapp', 'id')) then

        select cl.c_name_fio
        from miniappclient mac
        inner join clients cl on cl.c_id = mac.mac_from_c_id
        inner join miniapp maa on maa.ma_id = mac.mac_from_ma_id
        where mac.mac_from_ma_id = :obj_id
        rows 1
        into :c_name_fio;
    else
        select cl.c_name_fio, kn.k_status
        from clients cl
        join knclient kc on kc.kc_from_c_id = cl.c_id
        join kontract kn on kn.k_id = kc.kc_from_k_id
        where kn.k_id = :obj_id
        rows 1
        into :c_name_fio, :k_status;

    res_html = blob_append(res_html, f_format('<h2 style="text-align: center;">%1</h2>', c_name_fio));

    -- Заголовок таблицы с учётом статуса
    res_html = blob_append(res_html,
        '<table width="100%" border="1" cellpadding="5" cellspacing="0">' || 
        '<tr bgcolor="#008080" style="font-weight: bold; color: white;">' ||
        '<th style="width:10%; background-color: #008080; color: white;">№</th>' ||
        '<th style="width:28%;">Дата</th>' ||
        '<th style="width:30%;">Всего с НСП</th>' ||
        '</tr>');

    for select
        sh_id, 
        f_datetostr(sh_date) as formatted_date,
        round(sh_os + sh_perc + sh_perc_tax, 2) as total_amount
    from fm_schedule_list(:obj_kind, :obj_id)
    where sh_id > 0
    order by sh_id
    into :id, :formatted_date, :total_amount
    do
    begin
        if (:k_status = 1) then
        begin
            -- Только отображение, без ссылок
            res_field_name = 
                '<tr><td align="center" style="width:10%; background-color: #008080; color: white;">' || :id || '</td>' ||
                '<td align="center" style="width:28%;">' || :formatted_date || '</td>' ||
                '<td align="center" style="width:30%;">' || :total_amount || '</td></tr>';
        end
        else
        begin
            res_field_name = 
                '<tr><td align="center" style="width:10%; background-color: #008080; color: white;">' || :id || '</td>' ||
                '<td align="center" style="width:28%;">' || :formatted_date || '</td>' ||
                '<td align="center" style="width:30%;">' || :total_amount || '</td></tr>';
    
            res_link = f_uri_sql(
            f_lang(lang, 'Изменить график'),
            'set_due_date',
            trim(iif(obj_kind = 9, 'kontract', 'miniapp')),
            :obj_id,
            'number',
            f_lang(lang, 'Укажите новую дату погашения'),
            '',
            2,
            2,
            '',
            f_lang(lang, ''),
            'dialog', null);
        end
        res_html = blob_append(res_html, res_field_name);
    end
    res_html = blob_append(res_html, '</table>'); 
    res_html = blob_append(res_html, '<div align="center" style="margin-top: 20px;">');
    res_html = blob_append(res_html, '<button style="padding: 10px 20px; font-size: 16px; cursor: pointer;">' || res_link || '</button>');
    res_html = blob_append(res_html, '</div>');

    suspend;
end
