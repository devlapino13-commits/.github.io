CREATE PROCEDURE HP_KIB_SCORE_DEGREE(
    KIB_SCORE D_INTEGER)
RETURNS (
    RES_TEXT D_STR_255,
    RES_IS_ERROR D_BOOL)
AS
begin
  -- возвращает пояснение степени риска КИБ (RFCKG-3057)

  res_is_error = 0;
  res_text = case
               when kib_score between 743 and 900 then 'Очень низкий'
               when kib_score between 731 and 742 then 'Очень низкий'
               when kib_score between 713 and 730 then 'Очень низкий'
               when kib_score between 701 and 712 then 'Низкий'
               when kib_score between 689 and 700 then 'Низкий'
               when kib_score between 677 and 688 then 'Низкий'
               when kib_score between 665 and 676 then 'Средний'
               when kib_score between 653 and 664 then 'Средний'
               when kib_score between 641 and 652 then 'Средний'
               when kib_score between 620 and 640 then 'Высокий'
               when kib_score between 604 and 619 then 'Высокий'
               when kib_score between 574 and 603 then 'Высокий'
               when kib_score between 530 and 573 then 'Очень высокий'
               when kib_score between 251 and 529 then 'Очень высокий'
               when kib_score between 1 and 250 then 'Очень высокий'
               when kib_score = 0 then 'Не может быть рассчитан'
             end;

  if (coalesce(res_text, '') = '') then
  begin
    res_is_error = 1;
    res_text = 'Ошибочное значение kib_score';
  end

  suspend;
end
