CREATE PROCEDURE CM_RU_0034_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare kf_client_pass_no varchar(30);
declare variable kc_from_c_id id_type;
declare variable c_citizen integer;
declare variable rd_code varchar(255);
declare variable C_PASS_DATE date;
declare variable C_PASS_DATE_EXPIRE date;
declare variable k_percent_for_days integer;
declare variable pr_percent_for_days integer;
declare variable db_id integer;
declare variable k_kolich_repay integer;
declare variable kc_circle integer;
declare variable k_kommis curr_type;
declare variable KC_KOMMIS_LGOTA curr_type;
declare variable k_kredline_end date;
declare variable kf_client_pass_no_tmp varchar(30);
declare variable c_pass_date_tmp date;
declare variable c_pass_date_expire_tmp date;
begin
  select com.cm_database_id
  from company com
  into :db_id;
  if (not exists (select *
                    from kontract, product
                    where k_id = :k_id
                    and k_from_pr_id = pr_id
                    and pr_code containing '.EMP.')) then
  
  for select kc_from_c_id,
             kc_amount_approved,
             coalesce(k_percent_for_days,-777),
             coalesce(pr_percent_for_days,-777),
             k_disb_date,
             k_kolich_repay,
             kc_circle,
             KC_AMOUNT_APPROVED*K_KOMMIS_PERC/100 as k_kommis,
             KC_KOMMIS_LGOTA,
             kont.k_kredline_end
  from knclient
  inner join kontract kont on k_id = kc_from_k_id
  inner join product on pr_id  = k_from_pr_id
  where kc_from_k_id = :k_id
  and kc_amount_approved > 0
  into kc_from_c_id,
       kc_amount,
       k_percent_for_days,
       pr_percent_for_days,
       kf_disb_date,
       k_kolich_repay,
       kc_circle,
       k_kommis,
       KC_KOMMIS_LGOTA,
       k_kredline_end
  do
  begin
    If (k_kommis < KC_KOMMIS_LGOTA) then
    begin
        res_error_text = 'Дополнительный процент указан не правильно!';
        res_error_is_critical = 2;
        suspend;
    end
    if (k_percent_for_days <> pr_percent_for_days) then
    begin
        res_error_text = 'Неправильное значение ежедневной ставки. Просим заново перестроить график';
        res_error_is_critical = 1;
        suspend;
    end
    c_citizen = null;
    rd_code   = null;
    select c_name_fio,
           c_citizen,
           c_pass_no,
           C_PASS_DATE,
           C_PASS_DATE_EXPIRE
    from clients
    where c_id = :kc_from_c_id
    into res_info_text,
         c_citizen,
         KF_CLIENT_PASS_NO,
         C_PASS_DATE,
         C_PASS_DATE_EXPIRE;

    select rd_code from refdata
    where rd_id = :c_citizen
    into rd_code;
    if (rd_code not in ('Citizen.RU','Citizen.KG','Citizen.TJ','Citizen.UZ')) then
    begin
      res_error_text = 'Клиент не является гражданином след. республик: РФ, КГ, ТЖ и УЗ';
      res_error_is_critical = 1;
      suspend;
      Exit;
    end
    if (kf_disb_date > dateadd(month, -6, :c_pass_date_expire)
        and k_kredline_end > c_pass_date_expire  -- если срок паспорта заканчивается позже срока кредита то не проверять
       ) then
    begin
      kf_client_pass_no_tmp = null;
      c_pass_date_tmp = null;
      c_pass_date_expire_tmp = null;
      select cds_doc_no,
             cds_doc_date_begin,
             cds_doc_date_end
      from clientdocs, passkind, passdoctype
      where cds_from_c_id = :kc_from_c_id
      and cds_from_pk_id = pk_id
      and pk_from_pdt_id = pdt_id
      and cds_is_delete = 0
      and pdt_is_main_doc = 1 -- Паспорт, загран и другие.
      and cds_is_main_doc = 0 -- Это строка исключает прикрепленный основной паспорт.
      order by cds_doc_date_end desc
      rows 1
      into kf_client_pass_no_tmp,
           c_pass_date_tmp,
           c_pass_date_expire_tmp;
      if (kf_client_pass_no_tmp is not null) then
      begin
        if (c_pass_date_expire < c_pass_date_expire_tmp) then
        begin
          kf_client_pass_no = kf_client_pass_no_tmp;
          c_pass_date = c_pass_date_tmp;
          c_pass_date_expire = c_pass_date_expire_tmp;
        end 
      end 
      if (kf_disb_date > dateadd(month, -6, :c_pass_date_expire)
          and rd_code <> 'Citizen.RU'   -- добавлено от 25.04.2017 Улан
           ) then
      begin
        res_error_text = 'Срок паспорта заканчивается в течении 6 месяцев';
        res_error_is_critical = 1;
        suspend;
      end
    end

    if (C_PASS_DATE < cast('01.01.1990' as date) or C_PASS_DATE > current_date) then
    begin
      res_error_text = 'Дата выдачи паспорта, указана не верно!';
      res_error_is_critical = 1;
      suspend;
    end
    if (not(current_date between C_PASS_DATE and C_PASS_DATE_EXPIRE)) then
    begin
      res_error_text = 'Срок паспорта не действителен!';
      res_error_is_critical = 1;
      suspend;
    end 
     -- << Добавлял Музаффар
    if (rd_code <> 'Citizen.RU') then
    begin
        if ((k_kolich_repay between 3 and 5) and kc_circle > 1) then
        begin
            if (datediff(day, k_kredline_end, C_PASS_DATE_EXPIRE)<15) then
            begin
                res_error_text = 'Разница между датой окончания паспорта и датой погашения займа не должна быть менее 15 дней';
                  if(:rd_code in ('Citizen.TJ', 'Citizen.KG') and exists (select distinct 1
                                                        from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                                                        where CDS_FROM_C_ID = :kc_from_c_id
                                                        and cds_from_pk_id = pk_id
                                                        and pk_from_pdt_id = pdt_id
                                                        and cds_is_delete = 0
                                                        and :k_kredline_end+15  < CDS_DOC_DATE_END
                                                        and :kf_disb_date between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END
                                                        and pdt_code containing 'passport')) then
      res_error_is_critical = 0; else res_error_is_critical = 1;
                suspend;    
            end
        end
      /*  else  -- Данную проверку поднял на верх, Улан от 23.06.2016
        begin
          if (current_date > dateadd(month, -6, :C_PASS_DATE_EXPIRE)) then
          begin
            res_error_text = 'Срок паспорта заканчивается в течении 6 месяцев';
            res_error_is_critical = 1;
            suspend;
          end
        end */
    end
    -- >> конец добавлении
    if (rd_code <> 'Citizen.RU') then
    begin
      if (not exists(select *
                     from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                     where CDS_FROM_C_ID = :kc_from_c_id
                     and cds_from_pk_id = pk_id
                     and pk_from_pdt_id = pdt_id
                     and cds_is_delete = 0
                     and :kf_disb_date between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END
                     and pdt_code in ('magration_card',
                                      'registration_rvp',
                                      'registration_vnj')
           )) then
      begin
        res_error_text = 'У иностранного гражданина должен быть указан хотя бы один из документов: миграц. карта, РВП, ВНЖ';
        res_error_is_critical = 1;
        suspend;
      end
    end
  end
end
