CREATE PROCEDURE AAREP_CALC_PERC_LIST(
    REPORT_REGION_ID D_INTEGER,
    REPORT_DATE_BEGIN D_DATE,
    REPORT_DATE_END D_DATE,
    REPORT_CURRENCY D_STR_LANG)
RETURNS (
    AAREP_K_ID D_ID,
    AAREP_BALANCE_BEGIN D_CURRENCY,
    AAREP_PERC_BEGIN D_CURRENCY,
    AAREP_PERC_CALCED D_CURRENCY,
    AAREP_PERC_PAYED D_CURRENCY,
    AAREP_PERC_WRITEOFF D_CURRENCY,
    AAREP_PERC_END D_CURRENCY,
    AAREP_PERC_TRANSFER D_CURRENCY,
    AAREP_BALANCE_END D_CURRENCY,
    AAREP_NSP_BALANCE_BEGIN DECIMAL(18,2),
    AAREP_NSP_CALCED DECIMAL(18,2),
    AAREP_NSP_PAYED DECIMAL(18,2),
    AAREP_NSP_WRITEOFF DECIMAL(18,2),
    AAREP_NSP_BALANCE_END DECIMAL(18,2),
    AAREP_KONTRACT_NO D_STR_20,
    AAREP_CLIENT_NAME D_STR_100,
    AAREP_IS_INDIV D_STR_10,
    AAREP_PRODUCT_NAME D_STR_CODE,
    AAREP_PRODUCT_KIND D_STR_20,
    AAREP_DISB_DATE D_DATE,
    AAREP_DISB_AMOUNT D_CURRENCY,
    REP_DATE_END D_DATE,
    REP_MEMO D_STR_255,
    REP_CURRENCY D_STR_LANG,
    K_CLIENTS_CASHIER_ID D_STR_20,
    C_ID D_ID,
    K_STOP_PERC D_DATE,
    AAREP_FROM_REG_ID D_INTEGER,
    REG_NAME VARCHAR(60),
    AAREP_OUT_PERC_BEGIN D_CURRENCY,
    AAREP_OUT_PERC_CALCED D_CURRENCY,
    AAREP_OUT_PERC_END D_CURRENCY,
    AAREP_K_PERIOD DECIMAL(15,2),
    AAREP_DELAYED_PERC_BEGIN D_CURRENCY,
    AAREP_DELAYED_PERC_END D_CURRENCY,
    AAREP_DELAYED_PERC_TRANSFER D_CURRENCY,
    AAREP_MARKUP_MBH D_CURRENCY,
    AAREP_WRITEOFF D_CURRENCY,
    IS_MBH BOOLEAN,
    AAREP_LGOTA D_CURRENCY)
AS
declare variable K_PREV_ID D_ID;
declare variable AAREP_LOAN_T D_INTEGER;
declare variable DB_ID D_INTEGER;
declare variable PR_ID D_INTEGER;
declare variable PR_ID_PRIOR D_INTEGER;
declare variable PR_CODE D_STR_100;
declare variable REP_PERIOD_BEGIN D_INTEGER;
declare variable REP_PERIOD_END D_INTEGER;
declare variable AAREP_FROM_REG_ID_PRIOR D_INTEGER;
declare variable AAREP_PERC D_CURRENCY;
declare variable PMW_AMOUNT D_CURRENCY;
declare variable PR_PERCENT_KIND D_SMALLINT;
begin
  db_id = f_get_db_id();
  k_prev_id = 0;

  -- Ведомость начисленных процентов
  rep_period_begin = 0;
  rep_period_end = 0;
  if (extract(day from :report_date_begin) = 1) then
    rep_period_begin = extract(year from :report_date_begin) * 100 + extract(month from :report_date_begin);
  if (extract(day from (:report_date_end + 1)) = 1) then
    rep_period_end = extract(year from :report_date_end) * 100 + extract(month from :report_date_end);
  if ((rep_period_begin = 0) or (rep_period_end = 0)) then
    exit;

  rep_date_end = report_date_end;
  rep_memo = 'Проценты по кредиту';
  rep_currency = '';
  pr_id_prior = 0;
  aarep_from_reg_id_prior = 0;
  for
    select
      k_id,
      k_amount_approved,
      k_kontract_no,
      k_stop_perc,
      k_clients_cashier_id,
      k_currency,
      k_from_pr_id,
      k_disb_date,
      aarep_perc_begin,
      aarep_perc_calced,
      aarep_perc_payed,
      aarep_perc_writeoff,
      aarep_perc_end,
      aarep_perc_transfer,
      aarep_delayed_perc_transfer,
      aarep_from_reg_id,
      aarep_out_perc_begin,
      aarep_out_perc_calced,
      aarep_out_perc_end,
      k_period,
      aarep_delayed_perc_begin,
      aarep_delayed_perc_end,
      pr_percent_kind
    from aarep_calc_perc_list_tmp(:report_region_id, :report_date_begin, :report_date_end)
      inner join kontract on (k_id = aarep_k_id)
      inner join product on (k_from_pr_id = pr_id)
    order by k_id
  into
    :aarep_k_id,
    :aarep_disb_amount,
    :aarep_kontract_no,
    :k_stop_perc,
    :c_id,
    :rep_currency,
    :pr_id,
    :aarep_disb_date,
    :aarep_perc_begin,
    :aarep_perc_calced,
    :aarep_perc_payed,
    :aarep_perc_writeoff,
    :aarep_perc_end,
    :aarep_perc_transfer,
    :aarep_delayed_perc_transfer,
    :aarep_from_reg_id,
    :aarep_out_perc_begin,
    :aarep_out_perc_calced,
    :aarep_out_perc_end,
    :aarep_k_period,
    :aarep_delayed_perc_begin,
    :aarep_delayed_perc_end,
    :pr_percent_kind
  do
  begin
    --if (db_id in (39801, 41701) and aarep_perc_end < 0) then        RFCKG-3046 Вернуть условиие после 01.10.2024
      --aarep_perc_end = 0;
    aarep_nsp_balance_begin = 0;
    aarep_nsp_calced = 0;
    aarep_nsp_payed = 0;
    aarep_nsp_writeoff = 0;
    aarep_nsp_balance_end = 0;

    -- RFCKZ-204: Убираем НСП для Казахстана
    if (db_id = 41701 and k_prev_id <> aarep_k_id) then
    begin
      k_prev_id = aarep_k_id;
      if (aarep_disb_date >= '26.10.2013') then
        aarep_nsp_calced = aarep_perc_calced * 2 / 100;

      select
        coalesce(sum(ob_summa), 0),
        coalesce(sum(iif(ob_short_acc_dt = '21131/21', ob_summa, 0)), 0)
      from operbook
        inner join (
            select distinct ok_from_ol_id
            from operkred
            where ok_from_k_id = :aarep_k_id
              and ok_ol_is_active = 1
              and ok_ol_date between :report_date_begin and :report_date_end) as a
          on a.ok_from_ol_id = ob_from_ol_id
      where ob_ol_is_active = 1
        and ob_short_acc_kt in ('11531/4', '11531/6')
      into
        :aarep_nsp_payed,
        :aarep_nsp_writeoff;

      aarep_nsp_payed = aarep_nsp_payed - aarep_nsp_writeoff;
      aarep_nsp_balance_end = aarep_nsp_balance_begin + aarep_nsp_calced - aarep_nsp_payed - aarep_nsp_writeoff;
    end

    aarep_balance_begin = 0;
    select
      oks_balance
    from oksummary
    where oks_from_k_id = :aarep_k_id
      and oks_date_begin <= (:report_date_begin - 1)
      and oks_date_end >= (:report_date_begin - 1)
      and oks_region_id = :aarep_from_reg_id
      and oks_close_state = 0
    into
      :aarep_balance_begin;

    aarep_balance_end = 0;
    select
      oks_balance
    from oksummary
    where oks_from_k_id = :aarep_k_id
      and :report_date_end between oks_date_begin and oks_date_end
      and oks_region_id = :aarep_from_reg_id
      and oks_close_state = 0
    into
      :aarep_balance_end;

    if (pr_id <> pr_id_prior) then
    begin
      select
        pr_loan_type,
        pr_name,
        pr_code,
        iif(pr_code containing '.IPF.', 'ИПФ', 'Классика')
      from product
      where pr_id = :pr_id
      into
        :aarep_loan_t,
        :aarep_product_name,
        :pr_code,
        :aarep_product_kind;
      pr_id_prior = pr_id;

      if (aarep_loan_t = 1) then
        aarep_is_indiv = 'Индив.';
      else
        aarep_is_indiv = 'Группа';
    end

    aarep_client_name = '';
    if (c_id is not null) then
    begin
      select
        c_name_fio
      from clients
      where c_id = :c_id
      into
        :aarep_client_name;
    end

    k_clients_cashier_id = 'N' || cast(c_id as varchar(15));
    if (aarep_from_reg_id_prior <> aarep_from_reg_id) then
    begin
      select
        reg_code || ' ' || reg_name
      from region
      where reg_unicode = :aarep_from_reg_id
      into
        :reg_name;
      aarep_from_reg_id_prior = aarep_from_reg_id;
    end

    is_mbh = (db_id in (39801, 41701)) and (rep_period_end >= 202308) and (pr_code containing '.murabaha.');
    aarep_writeoff = null;
    aarep_markup_mbh = null;
    aarep_lgota = 0;

    if ((pr_percent_kind = 2 and db_id = 41701 and f_ipf_newrepmode(report_date_end, aarep_k_id, 0) = 1) or f_mbh_annu(aarep_k_id)) then
    begin
      select -sum(ok_sh_perc)
      from operkred
      where ok_from_k_id = :aarep_k_id and
            ok_ol_date between :report_date_begin and :report_date_end and
            ok_ol_is_active = 1 and
            ok_type = 3
      into aarep_lgota;

      if (aarep_lgota is null) then
        aarep_lgota = 0;
    end

    if (is_mbh) then
    begin
      k_stop_perc = (
        select
          rep_delinq_begin + 90
        from aarep_kont_delinq_status(:aarep_k_id, :report_date_end)
        where rep_delinq_days >= 90);

      if (aarep_perc_writeoff > 0) then
      begin
        -- списание
        aarep_delayed_perc_end = 0;
        aarep_perc = aarep_perc_begin - aarep_perc_payed - aarep_lgota;
        if (aarep_perc > aarep_delayed_perc_begin) then
        begin
          aarep_writeoff = aarep_delayed_perc_begin - aarep_perc_calced - aarep_lgota;

          select sum(pmw.pmw_amount)
          from partial_markup_writeoff pmw
          where pmw.pmw_from_k_id = :aarep_k_id
          and pmw.pmw_date between :report_date_end and :report_date_end
          and pmw.pmw_is_delete = 0
          into :pmw_amount;

          pmw_amount = coalesce(pmw_amount, 0);

          aarep_perc_writeoff = aarep_perc - aarep_writeoff - pmw_amount;
        end else
        begin
          aarep_writeoff = aarep_perc;
          aarep_perc_writeoff = 0;
        end
      end

      if (aarep_disb_date between report_date_begin and report_date_end) then
      begin
        aarep_markup_mbh = aarep_delayed_perc_begin;
        aarep_perc_begin = 0;
        aarep_delayed_perc_begin = 0;
      end
    end
    if (db_id <> 41701 or
       (coalesce(aarep_balance_begin, 0) <> 0 or
        coalesce(aarep_balance_end, 0) <> 0 or
        coalesce(aarep_perc_begin, 0) <> 0 or
        coalesce(aarep_perc_calced, 0) <> 0 or
        coalesce(aarep_perc_payed, 0) <> 0 or
        coalesce(aarep_perc_writeoff, 0) <> 0 or
        coalesce(aarep_perc_end, 0) <> 0 or
        coalesce(aarep_perc_transfer, 0) <> 0 or
        coalesce(aarep_nsp_balance_begin, 0) <> 0 or
        coalesce(aarep_nsp_calced, 0) <> 0 or
        coalesce(aarep_nsp_payed, 0) <> 0 or
        coalesce(aarep_nsp_writeoff, 0) <> 0 or
        coalesce(aarep_nsp_balance_end, 0) <> 0)) then
    begin
        suspend;
    end
  end
end
