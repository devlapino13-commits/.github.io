CREATE PROCEDURE HP_BKI_MANUAL_CREATE_PACKET(
    BMM_ID D_ID,
    THIS_DATES_ONLY D_BOOL,
    CHANGE_REASON D_INTEGER,
    UN_ID D_ID)
AS
declare variable bkk_id d_id;
declare variable action_code d_str_code;
declare variable c_id d_id;
declare variable k_id d_id;
declare variable ma_id d_id;
declare variable bkp_id d_id;
declare variable object_id d_id;
declare variable object_kind d_integer;
declare variable counter d_integer;
declare variable event_date d_date;
declare variable packet_date d_date;
declare variable packet_name d_str_code;
declare variable bkk_code d_str_code;
declare variable bkt_id d_id;
declare variable bke_id d_id;
declare variable bkc_id d_id;
declare variable bkl_id d_id;
declare variable bmm_from_bkk_id d_id;
declare variable bkc_add d_str_long;
declare variable packet_format d_smallint;
begin
  select f_enum('bki_packet_action', bkt_action, 'code'), bkt_from_bkk_id, bkk_code, bkt_id, bmm_from_bkk_id, bkt_format
  from bki_manual_main, bki_packet_type, bki_kind
  where bkt_id = bmm_from_bkt_id
    and bkk_id = bkt_from_bkk_id
    and bmm_id = :bmm_id
    and bmm_status = 0
  into action_code, bkk_id, bkk_code, bkt_id, bmm_from_bkk_id, packet_format;

  if (bkk_id is null) then
    exception e_default 'not found packet with bmm_id = ' || coalesce(bmm_id, '<NULL>');

  if (bmm_from_bkk_id is distinct from bkk_id) then
    exception e_message 'Тип пакета не соответствует БКИ';

  packet_name = null;
  packet_date = current_date;
  counter = 0;

  select packet_name, new_counter
  from hp_bki_get_packet_name(:packet_date, :bkk_code, :counter, :packet_format)
  into packet_name, counter;

  insert into bki_packet (bkp_code, bkp_from_bkk_id, bkp_date, bkp_number, bkp_from_bkt_id)
  values (:packet_name, :bkk_id, :packet_date, :counter, :bkt_id)
  returning bkp_id
  into :bkp_id;

  for select distinct bml_object_kind, bml_object_id
  from bki_manual_list
  where bml_from_bmm_id = :bmm_id
  order by 1,2
  into object_kind, object_id
  do
  begin
    if (object_kind = 11) then
    begin
      ma_id = object_id;
      k_id = null;

      select k_id from kontract
      where k_from_ma_app_id = :ma_id
        and k_status >= 0
      into k_id;

      select mac_from_c_id
      from miniappclient
      where mac_from_ma_id = :ma_id
      rows 1
      into c_id;
    end
    else
    begin
      k_id = object_id;
      ma_id = null;

      select k_from_ma_app_id
      from kontract
      where k_id = :k_id
      into ma_id;

      select kc_from_c_id
      from knclient kc
      where kc_from_k_id = :k_id
        and kc_amount_approved > 0
      rows 1
      into c_id;
    end

    insert into bki_loans (bkl_from_bkp_id, bkl_object_kind, bkl_object_id)
    values (:bkp_id, :object_kind, :object_id)
    returning bkl_id
    into bkl_id;

    -- найдем самую младшую дату для объекта
    select min(bml_date)
    from bki_manual_list
    where bml_from_bmm_id = :bmm_id
      and bml_object_kind = :object_kind
      and bml_object_id = :object_id
    into event_date;

    if (event_date is null) then
      exception e_default 'Not found min event_date for object_id = ' || coalesce(object_id, '<NULL>');

    bkc_add = iif(action_code in ('reset', 'recreate'), f_format('{"start_date":"%1", "change_reason":"%2"}', event_date, coalesce(change_reason, 1)), iif(action_code = 'delete', f_format('{"change_reason":"%1"}', coalesce(change_reason, 1)), null));

    insert into bki_clients (bkc_from_bkl_id, bkc_from_c_id, bkc_add)
    values (:bkl_id, :c_id, :bkc_add)
    returning bkc_id
    into bkc_id;
  
    for select bke_id
    from bki_errors, bki_events, bki_clients, bki_loans
    where bke_from_bev_id = bev_id
      and bkc_id = bev_from_bkc_id
      and bkl_id = bkc_from_bkl_id
      and bkl_object_kind = :object_kind
      and bkl_object_id = :object_id
      and bke_fixed = 0
    into bke_id
    do
      update bki_errors
      set bke_fixed = 1,
          bke_fixed_un_id = :un_id
      where bke_id = :bke_id;

    for select distinct bml_date
    from bki_manual_list
    where bml_from_bmm_id = :bmm_id
      and bml_object_kind = :object_kind
      and bml_object_id = :object_id
    order by 1
    into event_date
    do
    begin
      if (action_code = 'recreate') then
        select k_disb_date
        from kontract
        where k_id = :k_id
        into event_date;

      insert into bki_events (bev_from_bkc_id, bev_event_date, bev_tag)
      values (:bkc_id, :event_date, iif(:this_dates_only = 1, 55, 5));

      if (action_code in ('reset', 'recreate')) then
        break;
    end
  end

  update bki_manual_main
  set bmm_status = 1,
      bmm_from_bkp_id = :bkp_id
  where bmm_id = :bmm_id;
end
