CREATE PROCEDURE AAREP_PLACES_LANG_LINK(
    PL_ID D_ID,
    LANG_KIND D_SMALLINT)
RETURNS (
    PLACE_FULL D_STR_255,
    PLACE_FULL_REGION D_STR_255)
AS
declare variable mas_pl_id decimal(18,0);
declare variable mas_pl_name varchar(100);
declare variable plt_short_name varchar(100);
declare variable plt_short_name_side integer;
begin
  if (lang_kind is null) then
    lang_kind = 0;

  if (lang_kind <> 1) then
    lang_kind = 0;

  place_full = '';
  place_full_region = '';

  for with recursive pl_list
      as (select p1.pl_id, p1.parent, p1.pl_status,
          iif(:lang_kind = 0 or coalesce(p1.pl_name_komitet, '') = '', p1.pl_name, p1.pl_name_komitet) as pl_name
          from places p1
          where p1.pl_id = :pl_id and
                p1.is_delete = 0
          union all
          select p2.pl_id, p2.parent, p2.pl_status,
          iif(:lang_kind = 0 or coalesce(p2.pl_name_komitet, '') = '', p2.pl_name, p2.pl_name_komitet) as pl_name
          from places p2
          join pl_list on (pl_list.parent = p2.pl_id)
          where p2.is_delete = 0)
      select pl_id, pl_name,
             iif(:lang_kind = 0, plt_short_name, plt_alt_short_name),
             iif(:lang_kind = 0, plt_short_name_side, plt_alt_short_name_side)
      from pl_list, placetype
      where pl_status = plt_id and
            plt_show_in_address = 1
      into :mas_pl_id, :mas_pl_name, plt_short_name, plt_short_name_side
  do
  begin
    if ((plt_short_name is not null) and
        (plt_short_name <> '')) then
    begin
      if (plt_short_name_side = -1) then
      begin
        if (right(plt_short_name, 1) <> '.') then
          plt_short_name = plt_short_name || ' ';
        mas_pl_name = plt_short_name || mas_pl_name;
      end
      else
      if (plt_short_name_side = 1) then
        mas_pl_name = mas_pl_name || ' ' || plt_short_name;
    end
    if (place_full = '') then
      place_full = :mas_pl_name;
    else
      place_full = :mas_pl_name || ', ' || place_full;
    if (mas_pl_id <> pl_id) then
    begin
      if (place_full_region = '') then
        place_full_region = :mas_pl_name;
      else
        place_full_region = :mas_pl_name || ', ' || place_full_region;
    end
  end
  if (place_full = '') then
    place_full = 'Не указан';
  suspend;
end
