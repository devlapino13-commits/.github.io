CREATE PROCEDURE AAREP_KONTRACT_INFO_AZ(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    PAL_MAIN_ACC VARCHAR(20),
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    K_KOLICH_REPAY INTEGER,
    KAL_NEW_KF_NAME VARCHAR(50),
    K_KREDLINE_END DATE,
    KD_PERCENT CURR_TYPE,
    KD_PERCENT_EFFECT CURR_TYPE,
    PR_LOAN_TYPE VARCHAR(20),
    PR_NAME VARCHAR(50),
    LK_PERCENT CURR_TYPE,
    CS_NAME VARCHAR(50),
    IS_RESTRUCT INTEGER,
    RESTRUCT_DATE DATE,
    CLOSE_DATE_AFTER_REST DATE,
    K_STATUS VARCHAR(20),
    DELINQ_DAY INTEGER,
    DELINQ_DATE DATE,
    IS_ZALOG SMALLINT,
    VID_ZALOG VARCHAR(50),
    PORUCH_FIO VARCHAR(100),
    KOL_PORUCH INTEGER,
    C_NAME_FIO VARCHAR(100),
    KC_AMOUNT CURR_TYPE,
    ZAD_OS_CLI CURR_TYPE,
    SH_CALC_PERC_CLI CURR_TYPE,
    DELINQ_OS_CLI CURR_TYPE,
    DELINQ_PERC_CLI CURR_TYPE,
    SH_CALC_FINE_CLI CURR_TYPE,
    KC_LOAN_TARGET_STR VARCHAR(50),
    BS_NAME VARCHAR(100),
    BT_NAME VARCHAR(100),
    LK_BALANCE_CLI CURR_TYPE,
    LK_CALCPERC_CLI CURR_TYPE,
    LK_TOTAL_CLI CURR_TYPE,
    KD_PERCENT_REST CURR_TYPE,
    REST_BALANCE_CLI CURR_TYPE,
    REG_NAME VARCHAR(100))
AS
declare variable K_FROM_KF_ID ID_TYPE;
declare variable K_AMOUNT_APPROVED CURR_TYPE;
declare variable K_FROM_PR_ID integer;
declare variable LK_STATUS integer;
declare variable LK_BALANCE CURR_TYPE;
declare variable LK_CALCPERC CURR_TYPE;
declare variable KS_STATE integer;
declare variable ZAD_SUM CURR_TYPE;
declare variable ZAD_OS CURR_TYPE;
declare variable SH_CALC_PERC CURR_TYPE;
declare variable SH_CALC_FINE CURR_TYPE;
declare variable DELINQ_OS CURR_TYPE;
declare variable DELINQ_PERC CURR_TYPE;
declare variable Z_TYPE integer;
declare variable Z_NAME_FIO varchar(100);
declare variable Z_SUBTYPE integer;
declare variable REG_ID integer;
declare variable PR_FROM_PAT_ID integer;
declare variable C_ID ID_TYPE;
declare variable DOLYA double precision;
declare variable KC_LOAN_TARGET integer;
declare variable KC_LOAN_TARGET_PRIOR integer;
declare variable BUSUNESS_SUBTYPE integer;
declare variable REST_BALANCE CURR_TYPE;
declare variable REP_K_ID decimal(18,0);
declare variable REG_ID_PRIOR integer;
declare variable KS_DATE_BEGIN date;
declare variable KS_DATE_END date;
begin
  reg_id_prior = 0;
  for select k_id, k_kontract_no, k_from_kf_id, k_disb_date, k_amount_approved, k_kolich_repay, k_from_pr_id,
             k_kredline_end, kal_new_kf_name, kal_new_reg_id, ks_date_begin, ks_date_end
      from kontstate, kontract, knaktlist, region_list(:rep_region_id, 1, 3, 1)
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin <= :rep_date_end and
            ks_from_k_id = k_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            kal_new_reg_id = id_list and
            :rep_date_end between kal_date and kal_date_end
--  and k_kontract_no = '0201-005481-03'
      into rep_k_id, k_kontract_no, k_from_kf_id, k_disb_date, k_amount_approved, k_kolich_repay, k_from_pr_id,
           close_date_after_rest, kal_new_kf_name, reg_id, ks_date_begin, ks_date_end
           /*
      for
      select k_kontract_no,
             k_from_kf_id,
             k_disb_date,
             k_amount_approved,
             k_kolich_repay,
             k_from_pr_id,
             k_kredline_end
      from kontract
      where k_id = :rep_k_id
        into k_kontract_no,
             k_from_kf_id,
             k_disb_date,
             k_amount_approved,
             k_kolich_repay,
             k_from_pr_id,
             close_date_after_rest
      */
  do
  begin
    /*
        select kal_new_kf_name,
                kal_new_reg_id
        from knaktlist
        where kal_from_kf_id = :k_from_kf_id
        and :rep_date_end between kal_date and kal_date_end
        and knaktlist.is_delete = 0
        into kal_new_kf_name,
             reg_id;
        */
    if (rep_date_end between ks_date_begin and ks_date_end) then
      ks_state = 1;
    else
      ks_state = 2;

    if (reg_id_prior <> reg_id) then
    begin
      reg_id_prior = reg_id;
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :reg_id
      into reg_name;
    end

    select max(sh_date)
    from schedule
    where sh_from_k_id = :rep_k_id
    into k_kredline_end;

    select kd_percent_per_day * 365, kd_percent_effect
    from kontdata
    where kd_from_k_id = :rep_k_id and
          :k_disb_date between kd_ol_date and kd_date_end and
          kd_ol_is_delete = 0
    into kd_percent, kd_percent_effect;

    zad_os = 0;
    sh_calc_perc = 0;
    delinq_os = 0;
    delinq_perc = 0;
    sh_calc_fine = 0;

    lk_percent = null;
    lk_status = null;
    lk_balance = null;
    lk_calcperc = null;
    cs_name = null;
    delinq_date = null;
    if (ks_state = 1) then
    begin
      select sum(zad_os), sum(sh_calc_perc), sum(iif(sh_date <= :rep_date_end, zad_os, 0)),
             sum(iif(sh_date <= :rep_date_end, sh_calc_perc, 0)), sum(sh_calc_fine)
      from aarep_make_zad_list(:rep_k_id, :rep_date_end, 1)
      where sh_date is not null
      into zad_os, sh_calc_perc, delinq_os, delinq_perc, sh_calc_fine;

      select lk_percent, lk_status, lk_balance, lk_calcperc
      from llc_kont
      where lk_from_k_id = :rep_k_id and
            :rep_date_end between lk_date and lk_date_end
      into lk_percent, lk_status, lk_balance, lk_calcperc;

      lk_balance = lk_balance * lk_percent / 100;
      lk_calcperc = lk_calcperc * lk_percent / 100;

      select cs_name
      from classific
      where cs_id = :lk_status
      into cs_name;

      select oks_delinq_date
      from oksummary
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_from_k_id = :rep_k_id
            and oks_close_state = 0 
      into delinq_date;

      if (delinq_date > rep_date_end) then
        delinq_date = null;

    end

    select pr_name, iif(pr_loan_type = 1, 'индив.', 'групп.'), pr_from_pat_id
    from product
    where pr_id = :k_from_pr_id
    into pr_name, pr_loan_type, pr_from_pat_id;

    select pal_main_acc
    from prodacclist
    where pal_from_pat_id = :pr_from_pat_id and
          pal_from_at_id = 1
    into pal_main_acc;

    select is_restruct, restruct_date
    from aarep_get_restructuring(:rep_k_id, :rep_date_end, 94401)
    into is_restruct, restruct_date;

    if (is_restruct = 0) then
      close_date_after_rest = null;

    kd_percent_rest = null;
    if (is_restruct = 1) then
    begin
      select oks_balance
      from oksummary
      where :restruct_date between oks_date_begin and oks_date_end and
            oks_from_k_id = :rep_k_id
            and oks_close_state = 0 
      into rest_balance;

      select kd_percent_per_day * 365
      from kontdata
      where kd_from_k_id = :rep_k_id and
            :rep_date_end between kd_ol_date and kd_date_end and
            kd_ol_is_delete = 0
      into kd_percent_rest;

    end

--  статус кредита
    /*
        select ks_state from kontstate
        where ks_ol_is_delete = 0
        and :rep_date_end between ks_date_begin and ks_date_end
        and ks_from_k_id = :rep_k_id
        into ks_state;
        */
    k_status = case ks_state
                 when 0 then 'в процессе'
                 when 1 then 'активный'
                 when 2 then 'закрытый'
               end;

    if (ks_state = 2) then
    begin
      zad_sum = null;
      select sum(oko_os + oko_sh_os + oko_perc + oko_sh_perc)
      from operkroff, operlist
      where oko_from_k_id = :rep_k_id and
            oko_type < 7 and
            oko_from_ol_id = ol_id and
            operlist.is_delete = 0
      into zad_sum;
      zad_sum = coalesce(zad_sum, 0);

      if (zad_sum > 0) then
        k_status = 'вне балансовый';
    end

    if (delinq_date is not null) then
      delinq_day = :rep_date_end - delinq_date;
    else
      delinq_day = 0;

    poruch_fio = null;
    kol_poruch = 0;
    is_zalog = 0;
    vid_zalog = null;
    for select z_type, z_name_fio, z_subtype
        from zalog
        where z_from_k_id = :rep_k_id
        into z_type, z_name_fio, z_subtype
    do
    begin
      if (z_type = 0) then
      begin
        kol_poruch = kol_poruch + 1;
        if (poruch_fio is null) then
          poruch_fio = z_name_fio;
      end
      if (z_type >= 1) then
      begin
        is_zalog = 1;
        if (vid_zalog is null) then
        begin
          select zs_name
          from zalogsubtype
          where zs_from_zt_id = :z_type and
                zs_id = :z_subtype
          into vid_zalog;
        end
      end
    end

    kc_loan_target_prior = 0;
    for select kc_from_c_id, kc_amount_approved, kc_loan_target, kc_busuness_subtype
        from knclient
        where kc_from_k_id = :rep_k_id and
              kc_amount_approved > 0
        into c_id, kc_amount, kc_loan_target, busuness_subtype
    do
    begin
      dolya = kc_amount / k_amount_approved;
      select c_name_fio
      from clients
      where c_id = :c_id
      into c_name_fio;

      zad_os_cli = dolya * zad_os;
      sh_calc_perc_cli = dolya * sh_calc_perc;
      delinq_os_cli = dolya * delinq_os;
      delinq_perc_cli = dolya * delinq_perc;
      sh_calc_fine_cli = dolya * sh_calc_fine;

--  Цель кредита
      if (kc_loan_target_prior <> kc_loan_target) then
      begin
        kc_loan_target_prior = kc_loan_target;
        select rd_name
        from refdata
        where rd_id = :kc_loan_target
        into :kc_loan_target_str;
      end

      select bs_name, bt_name
      from bussubtype, bustype
      where bs_id = :busuness_subtype and
            bs_from_bt_id = bt_id
      into bs_name, bt_name;

      lk_balance_cli = coalesce(dolya * lk_balance, 0);
      lk_calcperc_cli = coalesce(dolya * lk_calcperc, 0);

      lk_total_cli = lk_balance_cli + lk_calcperc_cli;

      rest_balance_cli = dolya * rest_balance;
      suspend;
    end
  end

end
