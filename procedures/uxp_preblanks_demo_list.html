CREATE PROCEDURE UXP$PREBLANKS_DEMO_LIST(
    PREBLANK_KIND D_STR_CODE,
    PREBLANK_CODE D_STR_CODE)
RETURNS (
    RES_BLN_ID D_INTEGER,
    RES_BLN_NAME D_STR_255)
AS
declare variable in_obj_kind d_str_code;
declare variable in_obj_id d_str_code = null;
begin
  if (coalesce(preblank_code, '') = '') then
    exit;

  preblank_kind = f_enum('uxaction_preblank_kind', preblank_kind, 'code');

  if (preblank_kind <> 'code') then
    in_obj_id = preblank_code;

  if (in_obj_id <> '') then
  begin
    if (preblank_kind = 'hr') then
      in_obj_kind = 'sa_worktype';
    else
    if (preblank_kind = 'edoc') then
      in_obj_kind = 'edoc_cfg';
    else
    if (preblank_kind = 'eblank') then
      in_obj_kind = 'edoc_list';
    else
    if (preblank_kind = 'loan') then
      in_obj_kind = 'product';
    else
    if (preblank_kind = 'blank') then
      in_obj_kind = 'blankname';
  end

  for select res_bln_id, bln_name
  from blankname, uxp$preblank_actions(:preblank_kind, :preblank_code, :in_obj_kind, :in_obj_id)
  where bln_id = res_bln_id
  order by res_bln_id
  into :res_bln_id, res_bln_name
  do
    suspend;
end
