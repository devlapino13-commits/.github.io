CREATE PROCEDURE HP_CARDAKTCLIENT_SAVE(
    C_ID D_ID,
    RD_ID D_INTEGER,
    CARD_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    CAC_ID D_INTEGER,
    ERR_TEXT D_STR_255)
AS
declare variable k_id d_id;
declare variable cst_id d_integer;
declare variable cst_name d_str_code;
declare variable cpl_id d_integer;
declare variable reg_id d_integer;
declare variable cashier_id d_id;
declare variable cashier_name d_str_100;
declare variable dt_curr d_date;
declare variable rd_code d_str_255;
declare variable found_approve d_bool;
declare variable found_active d_bool;
declare variable card_act_id d_integer;
declare variable reason_id d_integer;
declare variable cac_id1 d_integer;
declare variable card_id1 d_integer;
declare variable cac_status d_integer;
declare variable is_delayed boolean;
declare variable is_pincover d_bool;
declare variable obj_kind d_integer;
declare variable cca_id d_integer;
declare variable cs_id d_integer;
declare variable edl_id d_id;
begin
  err_text = '';
  dt_curr = current_date;
  obj_kind = f_objkind('cardaktclient', 'id');
  is_pincover = 0;
  rd_code = f_get_rd_code(rd_id);
  is_delayed = rd_code in ('ReceivCard_LossTheft', 'ReceivCard_Spoilage');

  select
    cac_id,
    rd_id
  from cardaktclient
    inner join refdata
      on rd_id = cac_from_rd_id
  where cac_card_id = :card_id
    and cac_status >= 0
  into
    :cac_id,
    :reason_id;

  if (cac_id > 0) then
    err_text = 'Карта №' || card_id || ' уже фигурирует в акте выдачи №' || cac_id;

  if (rd_code = 'ReceivCard_Primary') then
  begin
    if (f_edoc_status('mbulak_card.agreement', :c_id, :dt_curr) <> 'signed') then
      err_text = 'Договор на выпуск и обслуж. карт М Булак не подписан!, клиенту необходимо сначала подписать Договор';
  end

  if (err_text = '') then
  begin
    select
      cal_status,
      f_enum('cardlist_status', cal_status, 'name'),
      cal_last_cpl_id,
      cem_is_pincover
    from cardlist
      inner join cardembomain
        on cem_id = cal_from_cem_id
    where cal_card_id = :card_id
    into
      :cst_id,
      :cst_name,
      :cpl_id,
      :is_pincover;
    if (row_count = 0) then
      err_text = 'Карта №' || card_id || ' не найдена среди готовых к использованию';
    if (err_text = '' and cst_id <> 1) then
      err_text = 'Карта №' || card_id || ' имеет текущий статус "' || cst_name || '", что исключает возможность выдачи';
  end

  if (err_text = '') then
  begin
    select
      cpl_from_reg_id,
      c_id,
      c_name_fio
    from cardplace
      inner join clients
        on c_id = cpl_from_c_id
    where cpl_id = :cpl_id
      and cpl_is_transfer = 0
    into
      :reg_id,
      :cashier_id,
      :cashier_name;
    if (row_count = 0) then
      err_text = 'Местоположение карты №' || card_id || ' не определено, либо карта находится "в пути"';
    if (err_text = '' and not exists(select * from username where un_id = :un_id and un_from_c_id = :cashier_id)) then
      err_text = 'Несоответствие МОЛ. Карта №' || card_id || ' числится за: ' || cashier_name;
  end

  if (err_text = '') then
  begin
    card_act_id = (select card_id from hp_client_bank_card_mb(:c_id, current_date));
    if (card_act_id > 0) then
      err_text = f_format('У клиента уже есть активная карта(%1)', card_act_id);
  end

  if (err_text = '') then
  begin
    select first 1
      cac_id,
      cac_card_id
    from cardaktclient
      inner join cardlist
        on cal_card_id = cac_card_id
          and cal_status = f_enum_id('cardlist_status', 'ready_to_use')
    where cac_from_c_id = :c_id
      and cac_status >= 0
    into
      :cac_id1,
      :card_id1;
    if (card_id1 > 0) then
      err_text = f_format('Найдена карта %1 (акт %2), уже находящаяся в процессе выдачи этому клиенту.', card_id1, cac_id1);
  end

  if (err_text = '') then
  begin
    found_approve = 0;
    if (rd_code in ('ReceivCard_Primary')) then
    begin
      k_id = (
        select first 1 k_id
        from knclient
          inner join kontract
            on k_id = kc_from_k_id
              and k_disb_kind = 6
              and k_status = 1
          inner join kontstate
            on ks_from_k_id = k_id
              and :dt_curr between ks_date_begin and ks_date_end
              and ks_state = 0
              and ks_ol_is_delete = 0
        where kc_from_c_id = :c_id
          and kc_amount_approved > 0
        order by ks_state desc, ks_date_begin desc);
      if (k_id > 0) then
        found_approve = 1;
    end

    found_active = 0;
    if ((found_approve = 0) and (rd_code in ('ReceivCard_LossTheft', 'ReceivCard_Spoilage', 'ReceivCard_Expiration'))) then
    begin
      k_id = (
        select first 1 k_id
        from knclient
          inner join kontract
            on k_id = kc_from_k_id
              and k_disb_kind = 6
              and k_status = 1
          inner join kontstate
            on ks_from_k_id = k_id
              and :dt_curr between ks_date_begin and ks_date_end
              and ks_state in (0, 1)
              and ks_ol_is_delete = 0
        where kc_from_c_id = :c_id
          and kc_amount_approved > 0
        order by ks_state desc, ks_date_begin desc);
      if (k_id > 0) then
        found_active = 1;
    end

    if (found_approve = 0 and found_active = 0) then
      err_text = f_format('У клиента должен быть %1%2%3 карточный займ',
        trim(iif(rd_code in ('ReceivCard_Primary', 'ReceivCard_Expiration'), 'утверждённый, но еще не выданный', '')),
        trim(iif(rd_code = 'ReceivCard_Expiration', ' или ', '')),
        trim(iif(rd_code in ('ReceivCard_LossTheft', 'ReceivCard_Spoilage', 'ReceivCard_Expiration'), 'активный либо утверждённый, но еще не выданный', '')));
  end

  if (err_text = '') then
  begin
    -- новый акт выдачи карты
    insert into cardaktclient(
      cac_from_rd_id,
      cac_from_k_id,
      cac_from_c_id,
      cac_card_id,
      cac_from_reg_id,
      cac_from_un_id
    )
    values(
      :rd_id,
      :k_id,
      :c_id,
      :card_id,
      :reg_id,
      :un_id
    )
    returning cac_id
    into :cac_id;

    if (is_delayed) then
    begin
      -- отложенная выдача, запрос баланса в МПЦ
      cca_id = (
        select
          max(cca_id)
        from cardclientid
          inner join cardclientacc
            on cca_from_cci_id = cci_id
              and cca_status = 0
        where cci_from_c_id = :c_id
          and cci_is_stopped = 0
          and cci_status = 1);

      insert into cardsync(
        cs_from_csc_id,
        cs_card_id,
        cs_from_cca_id,
        cs_object_kind,
        cs_object_id,
        cs_request_json,
        cs_office_id,
        cs_from_k_id,
        cs_from_c_id,
        cs_from_un_id
      )
      values (
        f_get_cardsynccfg_id('check_balance'),
        :card_id,
        :cca_id,
        :obj_kind,
        :cac_id,
        f_format('{"card_id": %1}', :card_id),
        :reg_id,
        :k_id,
        :c_id,
        :un_id
      )
      returning cs_id
      into :cs_id;

      update cardaktclient set
        cac_from_cs_id = :cs_id,
        cac_status = f_enum_id('cardaktclient', 'issued_delayed')
      where cac_id = :cac_id;
    end else
    begin
      -- формирование эл.документа на подпись
      select
        edl_id,
        res_text
      from hp_edoc_insert(
        iif(:is_pincover = 1, 'bankcard_pin.mbulak', 'bankcard.mbulak'),
        :obj_kind,
        :cac_id,
        :un_id
      )
      into
        :edl_id,
        :err_text;

      if (err_text = '') then
      begin
        update cardaktclient set
          cac_from_edl_id = :edl_id
        where cac_id = :cac_id;

        update verify_webcam_photo set
          vwp_card_id = :card_id
        where vwp_k_id = :k_id
          and vwp_c_id = :c_id
          and vwp_card_id is null;
      end else
        exception e_message :err_text;

    end
  end
  suspend;
end
