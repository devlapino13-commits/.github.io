    CPP_ID INTEGER,

    RES_TEXT VARCHAR(255))
AS
declare variable rd_id integer;
declare variable stage integer;
declare variable res_date_begin date;
declare variable res_date_end date;
declare variable rep_period integer;
declare variable res_period_id smallint;
declare variable mo_c_id d_id;
declare variable reg_id integer;
declare variable res_is_working integer;
declare variable res_jobname varchar(100);
declare variable res_jobkind varchar(100);
declare variable res_jobmemo varchar(100);
declare variable res_lo_fio varchar(100);
declare variable res_region_name varchar(100);
declare variable res_team_id integer;
declare variable res_team_name varchar(100);
declare variable res_team_leader integer;
declare variable res_team_date date;
declare variable cpr_id integer;
declare variable cpd_points_1 decimal(18,2) = 0;
declare variable cpd_points_2 decimal(18,2) = 0;
declare variable cpd_points_3 decimal(18,2) = 0;
declare variable cpd_points_4 decimal(18,2) = 0;
declare variable cpd_points_5 decimal(18,2) = 0;
declare variable cpd_points_6 decimal(18,2) = 0;
declare variable cpd_points_7 decimal(18,2) = 0;
declare variable cpd_points_8 decimal(18,2) = 0;
declare variable cpd_points_9 decimal(18,2) = 0;
declare variable cpd_points_all decimal(18,2) = 0;
declare variable cpd_comment blob sub_type 1 segment size 160;
declare variable cpd_staff_kolich integer;
declare variable growth_without_delinq decimal(15,2);
declare variable reg_par30_end decimal(15,2);
declare variable reg_par30_diff decimal(15,2);
declare variable reg_par30_begin decimal(15,2);
declare variable reg_qty smallint;
declare variable growth_without_delinq_rank decimal(15,2);
declare variable super_pokazatel decimal(15,2);
declare variable sp_rank smallint;
declare variable avg_res_qual_koef decimal(15,2);
declare variable avg_res_qual_koef_rank smallint;
declare variable second_decade smallint;
declare variable new_disb_rep_period decimal(15,2);
declare variable new_disb_rank smallint;
declare variable new_disb_rank_temp smallint;
declare variable disb_1_15 decimal(15,2);
declare variable disb_1_15_rank smallint;
declare variable disb_1_15_rank_temp smallint;
declare variable cpr_additional_points decimal(15,2);
declare variable old_points smallint;
declare variable avg_disb_loan d_currency;
declare variable penalty_points smallint;
declare variable disb_loan_avg decimal(15,2);
declare variable diff_client_avg decimal(15,2);
declare variable new_cpr_final_orderno smallint;
declare variable par7_end decimal(15,2);
declare variable par7_end_rank smallint;
declare variable lo_qty_par30_2 decimal(15,2);
declare variable super_pokazatel_txt d_str_255;
declare variable mo_c_id_tmp d_id;
declare variable kol_kc d_integer;
declare variable res_lo_id d_id;
declare variable reg_lo30_begin decimal(15,2);
declare variable lo_bal_par30 decimal(15,2);
declare variable lo_qty_par30_1 decimal(15,2);
declare variable lo_par30_begin decimal(15,2);
declare variable chl_c_id d_id;
declare variable cpd_add_data_6 decimal(18,2) = 0;
declare variable cpd_add_data_7 decimal(18,2) = 0;
declare variable cpd_add_data_6_temp decimal(18,2) = 0;
declare variable cpd_add_data_7_temp decimal(18,2) = 0;
declare variable cpd_add_data_8 d_integer;
declare variable cpr1_id d_id;
declare variable k_balance_begin decimal(15,2);
declare variable k_balance_end decimal(15,2);
declare variable lo_par30_diff decimal(15,5);
declare variable reg_par30_end_tmp decimal(15,5);
declare variable reg_par30_begin_tmp decimal(15,5);
declare variable reg_balance_begin decimal(15,2);
declare variable reg_balance_end decimal(15,2);
declare variable cpd_id integer;
declare variable cpd_object_id integer;
declare variable sp_avg decimal(15,2);
declare variable kolvo integer;
declare variable cpd_add_data_1 decimal(15,2);
declare variable cpd_add_data_2 decimal(15,2);
declare variable point decimal(15,2);
begin
  /* Олимпийские игры МО */
  -->> Удаляем существующие расчеты
  delete from competresult
  where (cpr_from_cpp_id = :cpp_id);

  res_text = '';
  select cpp_from_rd_id, cpp_stage, cpp_date_begin, cpp_date_end
  from competperiod
  where cpp_id = :cpp_id
  into :rd_id, :stage, :res_date_begin, :res_date_end;

  /* Stage:      0 - период (неделя)
                 1 - Промежуточные итоги
                 2 - Итоги
                 3 - Финальные результаты  */

  rep_period = 100 * extract(year from res_date_end) + extract(month from res_date_end);

  if (rd_id is null) then
  begin
    res_text = 'Период для расчета не указан';
    suspend;
    exit;
  end

  if (res_date_end > current_date) then
  begin
    res_text = 'Период еще не наступил. Расчет отменяется.';
    suspend;
    exit;
  end

  if (extract(day from res_date_end) = 15) then
    second_decade = 1;
  else
    second_decade = 0;
  mo_c_id_tmp = 0;

  -- Формируем в памяти данные по компании
  -- Тип отчета передаем "olimpic", чтобы система учитывала особенности расчета для олимпиады
  execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0,
      'standart')
      returning_values :res_period_id;

  if (res_period_id is null) then
  begin
    res_text = 'Не удалось расчитать данные за период';
    suspend;
    exit;
  end

  for --Список сотрудников, окно добавление в команд
      select distinct res_lo_id as mo_c_id, --
                      res_region_id as reg_id, --
                      res_is_working, --
                      res_jobname, --
                      res_jobkind, --
                      res_jobmemo, --
                      res_lo_fio, --
                      res_region_name, --
                      res_team_id, --
                      res_team_name, --
                      res_team_leader, --
                      res_team_date --
      from --
      aarep_staffteam_list(1000000, :res_date_end)
      where res_jobkind = 'Кредитный менеджер' and
            res_jobmemo <> 'Декретный отпуск' /*and
            :res_date_end - res_team_date >= 15   */
      into mo_c_id, --
           reg_id, --
           res_is_working, --
           res_jobname, --
           res_jobkind, --
           res_jobmemo, --
           res_lo_fio, --
           res_region_name, --
           res_team_id, --
           res_team_name, --
           res_team_leader, --
           res_team_date --
  do
  begin

    cpd_points_1 = 0.0;
    cpd_points_2 = 0.0;
    cpd_points_3 = 0.0;
    cpd_points_4 = 0.0;
    cpd_points_5 = 0.0;
    cpd_points_6 = 0.0;
    cpd_points_7 = 0.0;
    cpd_points_8 = 0.0;
    cpd_points_9 = 0.0;
    cpd_comment = 'Дата вступления: <b>' || f_datetostr(res_team_date) || '</b><br>';

    if (mo_c_id <> mo_c_id_tmp) then
    begin
      mo_c_id_tmp = mo_c_id;
      insert into competresult (cpr_id, cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id,
                                cpr_calced_points, cpr_additional_points, cpr_order_in_group, cpr_final_points,
                                cpr_final_orderno, cpr_result_kind, cpr_comment, cpr_from_un_id, cpr_datetime)
      values (null, :cpp_id, :res_jobname, 0, :mo_c_id, 0, 0, 0, 0, 0, -1, '', :un_id, current_timestamp)
      returning cpr_id
      into :cpr_id;
    end

    cpd_staff_kolich = 0;
    cpd_points_all = -->>
    cpd_points_1 + cpd_points_2 + cpd_points_3 + cpd_points_4 + cpd_points_5 + cpd_points_6 + cpd_points_7 + cpd_points_8 + cpd_points_9;

    insert into competdetails (cpd_from_cpr_id, cpd_id, cpd_object_kind, cpd_object_id, --
                               cpd_points_all, cpd_points_1, cpd_points_2, cpd_points_3, cpd_points_4, --
                               cpd_points_5, cpd_points_6, cpd_points_7, cpd_points_8, cpd_points_9, --
                               cpd_comment, --
                               cpd_add_data_1, cpd_add_data_2, cpd_add_data_3, cpd_add_data_4, cpd_add_data_5,
                               cpd_add_data_6, cpd_add_data_7, cpd_add_data_8, cpd_add_data_9, cpd_staff_kolich)
    values (:cpr_id, null, 7, :reg_id, --
            :cpd_points_all, :cpd_points_1, :cpd_points_2, :cpd_points_3, :cpd_points_4, --
            :cpd_points_5, :cpd_points_6, :cpd_points_7, :cpd_points_8, :cpd_points_9, --
            :cpd_comment, --
            0, 0, 0, 0, 0, --
            0, 0, 0, 0, :cpd_staff_kolich); -- cpd_staff_kolich 1 если не отпуск, иначе 0

  end

  for select chl.chl_c_id, chl.chl_reg_id
      from compet_hr_list chl
      where :res_date_end between chl.chl_date_begin and chl.chl_date_end
      into chl_c_id, reg_id
  do
  begin
    if (not exists(select *
                   from competresult cpr
                   inner join competdetails cpd on cpd.cpd_from_cpr_id = cpr.cpr_id and
                         cpd.cpd_object_kind = f_objkind('region', 'id') and
                         cpd.cpd_object_id = :reg_id
                   where cpr.cpr_from_cpp_id = :cpp_id)) then
    begin
      insert into competresult (cpr_id, cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id,
                                cpr_calced_points, cpr_additional_points, cpr_order_in_group, cpr_final_points,
                                cpr_final_orderno, cpr_result_kind, cpr_comment, cpr_from_un_id, cpr_datetime)
      values (null, :cpp_id, 'Менеджер офиса', 0, :chl_c_id, 0, 0, 0, 0, 0, -1, '', :un_id,
              current_timestamp)
      returning cpr_id
      into :cpr_id;

      insert into competdetails (cpd_from_cpr_id, cpd_id, cpd_object_kind, cpd_object_id, --
                                 cpd_points_all, cpd_points_1, cpd_points_2, cpd_points_3, cpd_points_4, --
                                 cpd_points_5, cpd_points_6, cpd_points_7, cpd_points_8, cpd_points_9, --
                                 cpd_comment, --
                                 cpd_add_data_1, cpd_add_data_2, cpd_add_data_3, cpd_add_data_4, cpd_add_data_5,
                                 cpd_add_data_6, cpd_add_data_7, cpd_add_data_8, cpd_add_data_9, cpd_staff_kolich)
      values (:cpr_id, null, 7, :reg_id, --
              :cpd_points_all, :cpd_points_1, :cpd_points_2, :cpd_points_3, :cpd_points_4, --
              :cpd_points_5, :cpd_points_6, :cpd_points_7, :cpd_points_8, :cpd_points_9, --
              :cpd_comment, --
              0, 0, 0, 0, 0, --
              0, 0, 0, 0, :cpd_staff_kolich);
    end
  end

  reg_qty = 0;
  select count(distinct cpr_id)
  from competresult cr
  inner join competdetails cd on cd.cpd_from_cpr_id = cr.cpr_id and
        cd.cpd_object_id > 0
  where cpr_from_cpp_id = :cpp_id
  into reg_qty;
  cpr_additional_points = 0;
  if (stage = 0) then
  begin
    for with t_reg_list
        as (select cd.cpd_object_id as reg_id
            from competresult cr
            inner join competdetails cd on cr.cpr_id = cd.cpd_from_cpr_id and
                  cd.cpd_staff_kolich = 0
            where cr.cpr_from_cpp_id = :cpp_id),
        t_decade_info
        as (select reg_id, --
                   --Сумма выдач новым клиентам за период >>>>
                   coalesce((select sum(kc_amount_approved) new_disb_rep_period
                             from kontstate ks
                             inner join knclient kc on ks_from_k_id = kc_from_k_id and
                                   kc_amount_approved > 0 and
                                   kc_circle = 1
                             where :second_decade = 0 and
                                   ks.ks_region_id = reg_id and
                                   ks_ol_is_delete = 0 and
                                   ks_date_begin between :res_date_begin and :res_date_end and
                                   ks_state = 1), 0) new_disb_rep_period, --
                   --Выдачи с 1 по 15 число >>>>
                   coalesce((select sum(kc_amount_approved) disb_1_15
                             from kontstate ks
                             inner join knclient kc on ks_from_k_id = kc_from_k_id and
                                   kc_amount_approved > 0
                             where :second_decade = 1 and
                                   ks.ks_region_id = reg_id and
                                   ks_ol_is_delete = 0 and
                                   ks_date_begin between :res_date_begin and :res_date_end and
                                   ks_state = 1), 0) disb_1_15, --
                   coalesce((select count(*)
                             from aarep_personal_list(reg_id, :res_date_end, 1)
                             where (((cast(:res_date_end as date) - ps_date_begin) >= 5) or
                                   (ho_text like '%Перевести%')) and
                                   jl_name in ('Специалист по финансированию', 'Старший специалист по финансированию') and
                                   ps_curr_status not in ('Отпуск без содерж.по уходу за ребенком', 'Отпуск по беременности и родам', 'Трудовой отпуск декретницы')), 0) kol_kc
            from t_reg_list)
        select reg_id, --
               new_disb_rep_period, --
               -- rank() over(order by new_disb_rep_period / kol_kc desc) new_disb_rank, --
               disb_1_15, kol_kc
               --  rank() over(order by disb_1_15 / kol_kc desc) disb_1_15_rank
        from t_decade_info
        into :reg_id, new_disb_rep_period, disb_1_15, kol_kc -- new_disb_rank, disb_1_15, disb_1_15_rank
    do
    begin

      /*   kol_kc = coalesce(kol_kc, 1);
      cpd_comment = '';
      new_disb_rep_period = new_disb_rep_period / kol_kc; -- средняя сумма выдачи по новым клиентам за период
      disb_1_15 = disb_1_15 / kol_kc; -- средняя сумма выдачи с 1 по 15 число  */
      --   if (:second_decade = 0) then
      -- begin
      for select cpr_id, cd.cpd_add_data_6, cd.cpd_add_data_7, cpd_add_data_8
          from competdetails cd
          inner join competresult cr on cd.cpd_from_cpr_id = cr.cpr_id and
                cr.cpr_from_cpp_id = :cpp_id
          where cd.cpd_object_kind = 7 and
                cd.cpd_object_id = :reg_id
          into cpr_id, cpd_add_data_6, cpd_add_data_7, cpd_add_data_8
      do
      begin

        update competdetails
        set --cpd_comment = cpd_comment || :cpd_comment,
            cpd_add_data_6 = iif(coalesce(:cpd_add_data_6, 0) <> 0, :cpd_add_data_6 + :disb_1_15, :disb_1_15),
            cpd_add_data_7 = iif(coalesce(:cpd_add_data_7, 0) <> 0, :cpd_add_data_7 + :new_disb_rep_period, :new_disb_rep_period),
            cpd_add_data_8 = iif(coalesce(:cpd_add_data_8, 0) <> 0, :cpd_add_data_8 + :kol_kc, :kol_kc)
        where cpd_from_cpr_id = :cpr_id;/* and
              cpd_object_id = :reg_id;  */

        /*  update competresult
        set cpr_additional_points = :cpr_additional_points
        where cpr_id = :cpr_id; */
      end
    end
    for select cpr_id, cd.cpd_add_data_6, cd.cpd_add_data_7, cpd_add_data_8
        from competdetails cd
        inner join competresult cr on cd.cpd_from_cpr_id = cr.cpr_id and
              cr.cpr_from_cpp_id = :cpp_id
        where cd.cpd_object_kind = 7
        into cpr_id, cpd_add_data_6, cpd_add_data_7, cpd_add_data_8
    do
    begin
      if (:kol_kc = 0) then
        kol_kc = 1;
      if (cpd_add_data_6 <> 0) then
        cpd_add_data_6_temp = :cpd_add_data_6 / :cpd_add_data_8;
      if (cpd_add_data_7 <> 0) then
        cpd_add_data_7_temp = :cpd_add_data_7 / :cpd_add_data_8;

      update competdetails
      set cpd_add_data_6 = :cpd_add_data_6_temp,
          cpd_add_data_7 = :cpd_add_data_7_temp
      where cpd_from_cpr_id = :cpr_id;
    end

    cpr1_id = 0;
    new_disb_rank_temp = 0;
    disb_1_15_rank_temp = 0;
    if (:second_decade = 0) then
    begin
      for select distinct cpr_id, cd.cpd_add_data_7, rank() over(order by cpd_add_data_7 desc) new_disb_rank
          from competdetails cd
          inner join competresult cr on cd.cpd_from_cpr_id = cr.cpr_id and
                cr.cpr_from_cpp_id = :cpp_id
          order by cpd_add_data_7 desc
          into cpr_id, cpd_add_data_7, new_disb_rank
      do
      begin
        if (cpr_id <> cpr1_id) then
        begin
          cpr1_id = :cpr_id;
          new_disb_rank_temp = iif(:new_disb_rank - :new_disb_rank_temp = 1, :new_disb_rank, :new_disb_rank_temp + 1);

          cpd_comment = 'Сумма выдач новым клиентам: <b>' || cpd_add_data_7 || '</b><br>';
          cpr_additional_points = 0;

          if (cpd_add_data_7 > 0 and
              new_disb_rank_temp between 1 and 10) then
          begin
            cpr_additional_points = 11 - new_disb_rank_temp;
            cpd_comment = cpd_comment || '    Доп.балл за выдачи новым клиентам: <b>' || cpr_additional_points || ';</b><br>    Ранг: ' || new_disb_rank_temp || ';<br>';
          end
          update competdetails
          set cpd_comment = cpd_comment || :cpd_comment
          where cpd_from_cpr_id = :cpr_id;

          update competresult
          set cpr_additional_points = :cpr_additional_points
          where cpr_id = :cpr_id;
        end
      end
    end

    if (:second_decade = 1) then
    begin
      for select distinct cpr_id, cd.cpd_add_data_6, rank() over(order by cpd_add_data_6 desc) disb_1_15_rank,
                          cd.cpd_comment, cr.cpr_additional_points
          from competdetails cd
          inner join competresult cr on cd.cpd_from_cpr_id = cr.cpr_id and
                cr.cpr_from_cpp_id = :cpp_id
          order by cpd_add_data_6 desc
          into cpr_id, cpd_add_data_6, disb_1_15_rank, cpd_comment, cpr_additional_points
      do
      begin
        if (cpr_id <> cpr1_id) then
        begin
          cpr1_id = :cpr_id;
          disb_1_15_rank_temp = iif(:disb_1_15_rank - :disb_1_15_rank_temp = 1, :disb_1_15_rank, :disb_1_15_rank_temp + 1);

          cpd_comment = cpd_comment || 'Сумма выдач с 1-15 число: <b>' || cpd_add_data_6 || '</b><br>';

          if (cpd_add_data_6 > 0 and
              disb_1_15_rank_temp between 1 and 15) then
          begin
            cpr_additional_points = cpr_additional_points + 5;
            cpd_comment = cpd_comment || '    Доп.балл за выдачи новым клиентам: <b>5;</b><br>    Ранг: ' || disb_1_15_rank_temp || ';<br>';
          end
          else
          if (disb_1_15_rank_temp between reg_qty - 14 and reg_qty) then
          begin
            cpr_additional_points = -5;
            cpd_comment = cpd_comment || '    Доп.балл за выдачи новым клиентам: <b>5;</b><br>    Ранг: ' || disb_1_15_rank_temp || ';<br>';
          end
        end

        update competdetails
        set cpd_comment = cpd_comment || :cpd_comment
        where cpd_from_cpr_id = :cpr_id;

        update competresult
        set cpr_additional_points = cpr_additional_points + :cpr_additional_points
        where cpr_id = :cpr_id;
      end
    end
  end
  if (stage = 1) then
  begin
    for select distinct cd.cpd_object_id as reg_id
        from competresult cr
        inner join competdetails cd on cr.cpr_id = cd.cpd_from_cpr_id and
              cd.cpd_staff_kolich = 0
        where cr.cpr_from_cpp_id = :cpp_id
        into :reg_id -- ID офиса
    do
    begin
      cpd_comment = '';
      for select cpr_id, cd.cpd_add_data_8
          from competdetails cd
          inner join competresult cr on cd.cpd_from_cpr_id = cr.cpr_id and
                cr.cpr_from_cpp_id = :cpp_id
          where cd.cpd_object_kind = 7 and
                cd.cpd_object_id = :reg_id
          into cpr_id, cpd_add_data_8
      do
      begin
        for select count(*)
            from aarep_personal_list(:reg_id, :res_date_end, 1)
            where (((cast(:res_date_end as date) - ps_date_begin) >= 15) or
                  (ho_text like '%Перевести%')) and
                  jl_name in ('Специалист по финансированию', 'Старший специалист по финансированию') and
                  ps_curr_status not in ('Отпуск без содерж.по уходу за ребенком', 'Отпуск по беременности и родам', 'Трудовой отпуск декретницы')
            into kol_kc
        do
        begin
          if (:reg_id = 1070300 and
              :rep_period = 202102) then
            kol_kc = 14; -- одного сотрудника перевели с другого офиса, он не считается (по коду 15 КС)
          update competdetails

            union
            select cpr_id, diff_client_avg, 3, --
                   rank() over(order by diff_client_avg desc) diff_client_avg_rank
            from t_cpr)

        select cpr_id, par30_end_avg, numb
        from t_rank
        where avg_rank = 1
        into cpr_id, disb_loan_avg, sp_rank
    do
    begin
      if (sp_rank = 1) then
        cpd_comment = 'PAR30 на конец: ' || round(:disb_loan_avg) || '. Доп.балл 5; ';
      else
      if (sp_rank = 2) then
        cpd_comment = 'Кол.выдач в ср.: ' || round(:disb_loan_avg) || '. Доп.балл 5; ';
      else
      if (sp_rank = 3) then
        cpd_comment = 'Прирост акт.кл в ср.: ' || round(:disb_loan_avg) || '. Доп.балл 5; ';

      update competresult cr
      set cr.cpr_additional_points = cpr_additional_points + 5,
          cpr_comment = cpr_comment || :cpd_comment
      where cpr_from_cpp_id = :cpp_id and
            cpr_id = :cpr_id;
    end
  end

  old_points = 0;
  for select cpd.cpd_object_id, cur_cr.cpr_id, sum(old_cr.cpr_final_points) old_points
      from competresult cur_cr
      inner join competresult old_cr on cur_cr.cpr_object_id = old_cr.cpr_object_id
      inner join competdetails cpd on cpd.cpd_from_cpr_id = cur_cr.cpr_id
      inner join competperiod cp on old_cr.cpr_from_cpp_id = cp.cpp_id and
            cp.cpp_date_begin between :res_date_begin and :res_date_end and
            cp.cpp_stage = :stage - 1
      where cur_cr.cpr_from_cpp_id = :cpp_id
      group by 1, 2
      into reg_id, cpr_id, old_points
  do
  begin

    if (stage = 1) then
    begin
      cpd_points_all = 0;
      select sum(distinct(cpd_points_1 + cpd_points_2 + cpd_points_3 + cpd_points_4))
      from competdetails cd
      where cpd_from_cpr_id = :cpr_id
      into cpd_points_all;
      cpd_points_all = coalesce(cpd_points_all, 0);

      cpd_comment = '<br>Баллы с декад: ' || round(:old_points, 2) || '.<br>';
      update competdetails
      set cpd_points_all = :cpd_points_all,
          cpd_comment = cpd_comment || :cpd_comment
      where cpd_from_cpr_id = :cpr_id;

      update competresult cr
      set cpr_additional_points = cpr_additional_points + :old_points,
          cr.cpr_calced_points = :cpd_points_all
      where cpr_id = :cpr_id;
    end
    else
      update competresult cr
      set cr.cpr_calced_points = cpr_calced_points + :old_points
      where cpr_id = :cpr_id;
  end
  if (:stage = 1) then
  begin

    for select cpr.cpr_id
        from competdetails cpd
        inner join competresult cpr on cpr.cpr_id = cpd.cpd_from_cpr_id and
              cpr.cpr_from_cpp_id = :cpp_id
        where cpd.cpd_id > 1
        into cpr_id
    do
    begin

      update competdetails
      set cpd_points_all = (cpd_points_all / 2)
      where cpd_from_cpr_id = :cpr_id;

      update competresult cr
      set cpr_additional_points = (cpr_additional_points / 2),
          cr.cpr_calced_points = (cpr_calced_points / 2)
      where cpr_id = :cpr_id;
    end
  end
  execute procedure hp_competition_recalc_order(:cpp_id);

  /*    5.3.    При равенстве всех итоговых баллов победитель определяется с учетом показателей в абсолютных значениях:
  первый критерий по КОЛИЧЕСТВУ ВЫДАЧ НА ОДНОГО ВИПС ОФИСА, в котором работают/работает МНМ, за отчетный период;
  в случае равенства по первому критерию, то расчет осуществляется
  ПО РОСТУ КОЛИЧЕСТВА АКТИВНЫХ КЛИЕНТОВ НА ОДНОГО ВИПС ОФИСА.  */
  disb_loan_avg = 0;
  diff_client_avg = 0;
  for with t_cpr
      as (select cr.cpr_id, --
                 cr.cpr_final_orderno, --
                 avg(ad_disb_loan) disb_loan_avg, --Ср.выдачи на одного СВЗ
                 avg(ad_diff_client) diff_client_avg --Ср.росту количества активных клиентов
          from competresult cr
          inner join competdetails cd on cr.cpr_id = cd.cpd_from_cpr_id and
                cd.cpd_object_id > 0
          inner join hp_analitica_report('reg_id, lo_id', 'tree_group', 31, 999999, 0, 999999, 8, 999999, 31, 999999, 181, 999999, 1065, 0, 'standart') on res_field_1 = cd.cpd_object_id and
                res_field_2 > 1000000000010 and
                ad_active_balance > 0.0
          where cr.cpr_from_cpp_id = :cpp_id and
                cpr_object_id > 0
          group by 1, 2)

      select cpr_id, disb_loan_avg, diff_client_avg, --
             rank() over(order by cpr_final_orderno, disb_loan_avg desc, diff_client_avg desc) new_cpr_final_orderno
      from t_cpr
      into cpr_id, disb_loan_avg, diff_client_avg, new_cpr_final_orderno
  do
  begin
    cpd_comment = 'Ср.выдачи: ' || round(:disb_loan_avg) || '; Ср.прирост актив.кл.: ' || round(:diff_client_avg) || '; ';

    update competresult
    set cpr_final_orderno = :new_cpr_final_orderno,
        cpr_comment = cpr_comment || :cpd_comment
    where cpr_from_cpp_id = :cpp_id and
          cpr_id = :cpr_id;
  end

  if (stage = 1) then
    update competresult
    set cpr_additional_points = cpr_additional_points + 1,
        cpr_comment = cpr_comment || 'Доп. бал за 1 место: 1'
    where cpr_from_cpp_id = :cpp_id and
          cpr_final_orderno = 1;
  else
  if (stage = 2) then
  begin
    for select cpr.cpr_id
        from competdetails cpd
        inner join competresult cpr on cpr.cpr_id = cpd.cpd_from_cpr_id and
              cpr.cpr_from_cpp_id = :cpp_id
        where cpd.cpd_id > 1
        into cpr_id
    do
    begin

      update competdetails
      set cpd_points_all = (cpd_points_all / 2)
      where cpd_from_cpr_id = :cpr_id;

      update competresult cr
      set cpr_additional_points = (cpr_additional_points / 2),
          cr.cpr_calced_points = (cpr_calced_points / 2)
      where cpr_id = :cpr_id;
    end

    execute procedure hp_competition_recalc_order(:cpp_id);

    update competresult
    set cpr_additional_points = cpr_additional_points + 10,
        cpr_comment = cpr_comment || 'Доп. бал за 1-место: 10'
    where cpr_from_cpp_id = :cpp_id and
          cpr_final_orderno = 1;

    update competresult
    set cpr_additional_points = cpr_additional_points + 6,
        cpr_comment = cpr_comment || 'Доп. бал за 2-место: 6'
    where cpr_from_cpp_id = :cpp_id and
          cpr_final_orderno = 2;

    update competresult
    set cpr_additional_points = cpr_additional_points + 3,
        cpr_comment = cpr_comment || 'Доп. бал за 3-место: 3'
    where cpr_from_cpp_id = :cpp_id and
          cpr_final_orderno = 3;
  end

  suspend;
end
