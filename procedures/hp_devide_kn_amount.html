CREATE PROCEDURE HP_DEVIDE_KN_AMOUNT(
    OK_FROM_K_ID ID_TYPE,
    OK_FROM_KC_ID ID_TYPE,
    AMOUNT CURR_TYPE,
    ROUND_SCALE INTEGER = 0)
RETURNS (
    KC_ID ID_TYPE,
    RES_AMOUNT CURR_TYPE)
AS
declare variable k_amount curr_type;
declare variable kc_amount curr_type;
declare variable kc_count d_integer;
declare variable curr_line d_integer;
declare variable saldo_amount curr_type;
declare variable koef double precision;
declare variable diff_amount curr_type;
begin
  if (ok_from_kc_id is null) then
    ok_from_kc_id = 0;

  if (ok_from_k_id is null) then
  begin
    select kc_from_k_id
    from knclient
    where kc_id = :ok_from_kc_id
    into ok_from_k_id;
  end

  select k_amount_approved, k_kolich_client_approved
  from kontract
  where k_id = :ok_from_k_id
  into k_amount, kc_count;

-- если участников группы только 1, возвращаем как есть
  if (kc_count = 1) then
  begin
    select kc_id
    from knclient
    where kc_from_k_id = :ok_from_k_id and
          kc_amount_approved > 0
    into :kc_id;

    res_amount = amount;
    suspend;
    exit;
  end

  if (amount < kc_count) then
     round_scale = 2;

  saldo_amount = amount;
  diff_amount = 0;
  curr_line = 0;
  for select kc_id, kc_amount_approved
      from knclient
      where kc_from_k_id = :ok_from_k_id and
            kc_amount_approved > 0
      into kc_id, kc_amount
  do
  begin
    curr_line = curr_line + 1;
    koef = cast(kc_amount as double precision) / k_amount;

    if (curr_line < kc_count) then
    begin
      res_amount = round(koef * amount + diff_amount, round_scale);

      if (kc_count = 2) then
        res_amount = (select res
                      from fhp_round_bank(:koef * :amount, :round_scale));
      else
        diff_amount = round(koef * amount + diff_amount, 4) - res_amount;

      saldo_amount = saldo_amount - res_amount;
    end
    else
      res_amount = saldo_amount;

    if (kc_id = ok_from_kc_id) then
    begin
      suspend;
      exit;
    end else
      suspend;
  end
end
