CREATE PROCEDURE HP_VERIFY_CONSENT(
    MA_ID D_INTEGER,
    C_ID ID_TYPE)
RETURNS (
    IS_VERIFIED D_BOOL)
AS
declare variable spouse_id id_type;
begin
  -- если c_id является супругом(ой) какому-либо клиенту заявки,
  -- то is_verified = 1, иначе is_verified = 0

  is_verified = 0;
  spouse_id = 0;

  select crel_other_c_id
  from hp_client_relation_list(:c_id, 0)
  where crel_from_rel_id in (1, 2) -- муж/жена
  into spouse_id;

  if (spouse_id is not null and
      exists(select mac_from_c_id
             from miniappclient mac
             where mac_from_c_id = :spouse_id and
                   mac_from_ma_id = :ma_id and
                   mac_amount > 0)) then
    is_verified = 1;

  suspend;
end
