CREATE PROCEDURE AAREP_CLIENTS_APP_COUNT(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    COUNT_ALL_MINIAPP D_INTEGER,
    COUNT_UNBLOCKED_VIPS D_INTEGER,
    COUNT_BLOCKED_VIPS_ST0 D_INTEGER,
    COUNT_BLOCK_VIPS D_INTEGER,
    COUNT_BLOCKED_VIPS_ST1 D_INTEGER,
    COUNT_ISSUED D_INTEGER,
    COUNT_BLOCKED_VIPS_REFUSAL D_INTEGER,
    COUNT_MINIAPP_CREDIT D_INTEGER,
    COUNT_MINIAPP_OTK D_INTEGER,
    COUNT_MINIAPP_PRO D_INTEGER,
    COUNT_MINIAPP D_INTEGER,
    COUNT_PROSESS D_INTEGER,
    COUNT_PROSECC_KR D_INTEGER,
    KONSEVR D_CURRENCY,
    SHARE_UNBLOCKED_VIPS D_CURRENCY,
    SHARE_VIPS_REFUSALS_ZAV D_CURRENCY,
    RD_NAME D_STR_255,
    COUNT_REFUSAL D_INTEGER)
AS
declare variable KOLVO_RD D_INTEGER;
declare variable EMA_RESULT_RD_ID D_ID;
begin

  ---кол-во заявок
  select count(*)
  from extminiapp
  inner join ticket t on t.tkt_id = ema_from_tkt_id
  where ema_date between :rep_date_begin and :rep_date_end and
        ema_from_c_id is not null and
        t.tkt_source = 2
  into count_all_miniapp;


  --не заблокировано випсом
  count_unblocked_vips = null;
  select count(*)
  from extminiapp ex
  inner join ticket tk on tk.tkt_id = ex.ema_from_tkt_id
  where ex.ema_from_c_id is not null and
        ex.ema_date between :rep_date_begin and :rep_date_end and
        ex.ema_block_datetime is null and
        tk.tkt_source = 2
  into count_unblocked_vips;


  -----Заблокиров Випсом со статусом процессе
  count_blocked_vips_st0 = null;
  select count(*)
  from extminiapp ex
  inner join ticket tk on tk.tkt_id = ex.ema_from_tkt_id
  where ex.ema_from_c_id is not null and
        ex.ema_date between :rep_date_begin and :rep_date_end and
        ex.ema_block_datetime is not null and
        ex.ema_status in (0) and
        tk.tkt_source = 2
  into count_blocked_vips_st0;

   -----Заблокиров Випсом со статусом процессе
  count_blocked_vips_st1 = null;
  select count(*)
  from extminiapp ex
  inner join ticket tk on tk.tkt_id = ex.ema_from_tkt_id
  where ex.ema_from_c_id is not null and
        ex.ema_date between :rep_date_begin and :rep_date_end and
        ex.ema_block_datetime is not null and
        ex.ema_status in (1) and
        tk.tkt_source = 2
  into count_blocked_vips_st1;


  ----Количество отказов
  count_blocked_vips_refusal = null;
  select count (*)
  from extminiapp ex
  inner join ticket tk on tk.tkt_id = ex.ema_from_tkt_id
  where ex.ema_from_c_id is not null and
        ex.ema_date between :rep_date_begin and :rep_date_end and
        ex.ema_block_datetime is not null and
        ex.ema_status = -1 and
        tk.tkt_source = 2
  into count_blocked_vips_refusal;


  count_block_VIPS = 0;
  count_block_VIPS = count_blocked_vips_st0 + count_blocked_vips_st1 + count_blocked_vips_refusal;


  count_miniapp_credit = null;
  select count (*)
  from miniapp ma
  inner join ticket tk on tk.tkt_id = ma.ma_from_tkt_id
  inner join extminiapp ex on ex.ema_from_tkt_id = tk.tkt_id
  where ma.ma_date_app between :rep_date_begin and :rep_date_end
  and ma.ma_date_end = '31.12.9999'
  and tk.tkt_source = 2
  into count_miniapp_credit;




  count_miniapp_otk = null;
  select count (*)
  from miniapp ma
  inner join ticket tk on tk.tkt_id = ma.ma_from_tkt_id
  inner join extminiapp ex on ex.ema_from_tkt_id = tk.tkt_id
  where ma.ma_date_app between :rep_date_begin and :rep_date_end
  and ma.ma_date_end = '31.12.9999'
  and ma.ma_is_active = -1
  and tk.tkt_source = 2
  into count_miniapp_otk;



  count_miniapp = null;
  select count (*)
  from miniapp ma
  inner join ticket tk on tk.tkt_id = ma.ma_from_tkt_id
  inner join extminiapp ex on ex.ema_from_tkt_id = tk.tkt_id
  where ma.ma_date_app between :rep_date_begin and :rep_date_end
      and ma.ma_date_end = '31.12.9999'
      and ma.ma_is_active = 1
      and tk.tkt_source = 2
  into count_miniapp;

  --- кол-во выданных
  select count(*)
  from kontstate ks
  inner join kontract kt on kt.k_id = ks.ks_from_k_id
  inner join ticket tk on tk.tkt_id = kt.k_from_tkt_id
  inner join extminiapp ex on ex.ema_from_tkt_id = tk.tkt_id
  where ks.ks_ol_is_delete = 0 and
        ks.ks_date_begin between :rep_date_begin and :rep_date_end and
        ks.ks_state = 1 and
        tk.tkt_source = 2
  into count_issued;


  count_miniapp_pro = 0;
  count_prosecc_kr  = 0;
  count_miniapp_pro =count_issued- count_miniapp;
  count_prosecc_kr = count_blocked_vips_st1 - count_miniapp_credit;

 ------------------------------------------------------------------------------------------

   if (count_all_miniapp = 0) then
      konsevr = 0;
  else
  konsevr = (cast(count_issued  as d_currency )/ cast (count_all_miniapp as d_currency)) * 100;


  -------------------------------------------------------------------------------------------
   
  if (count_all_miniapp = 0) then
      share_unblocked_vips = 0;
  else
  share_unblocked_vips = (cast(count_unblocked_vips as d_currency) / cast (count_all_miniapp as d_currency)) * 100;


 -----------------------------------------------------------------------------------------------
  if (count_all_miniapp = 0) then
      share_vips_refusals_zav = 0;
  else
  share_vips_refusals_zav = (cast(count_blocked_vips_refusal as d_currency )/ cast (count_all_miniapp  as d_currency))* 100;


  for select count(*), iif (rd.rd_name is null, 'Не заполнено' ,rd.rd_name)
      from extminiapp ex
      inner join ticket tk on tk.tkt_id = ex.ema_from_tkt_id
      left join refdata rd on rd.rd_id = ex.ema_result_rd_id
      where ex.ema_from_c_id is not null and
            ex.ema_date between :rep_date_begin and :rep_date_end and
            ex.ema_status = -1 and
            tk.tkt_source = 2 and
            ex.ema_lo_id is not null
      group by ema_result_rd_id, rd_name
      into count_refusal, rd_name
  do
  begin
    suspend;
  end
end
