CREATE PROCEDURE HP_DO_KN_OPERBOOK_FIX_SALDO_TAX(
    REP_DATE D_DATE,
    BALANCE_ACCOUNT D_STR_CODE)
AS
declare variable ol_id d_id;
declare variable diff_amount d_currency;
declare variable acc_dt d_id;
declare variable acc_kt d_id;
declare variable line_id d_integer;
declare variable acc_id d_id;
declare variable kob_summa_bal d_currency;
declare variable kob_summa_tax d_currency;
declare variable tax_kind d_smallint;
declare variable add_ab_id d_id;
begin
  -- Процедура показывает и если надо исправляет сальдо НУ по лицевым счетам через 0-й счет.

  -- Если передать NEED_CREATE_OPERLIST = 1, будет создан мем.ордер для исправления остатков.
  -- Иначе будет отображен только как список.

  select ab_id, ab_tax_kind
  from ref_accbalance
  where ab_code = :balance_account
  into add_ab_id, tax_kind;

  if (row_count = 0) then
    exit;

  line_id = 0;
  for select acc_id
  from ref_account
  where acc_from_add_ab_id = :add_ab_id
  into acc_id
  do
  begin
    for select kobs_bal_sum, kobs_bal_tax
    from kn_obsummary
    where kobs_acc_id = :acc_id and
          kobs_date <= :rep_date
    order by kobs_acc_id desc, kobs_date desc
    rows 1
    into kob_summa_bal, kob_summa_tax
    do
    begin
      if (tax_kind = 0 and kob_summa_bal = kob_summa_tax) then
        Continue;

      if (tax_kind = 1 and kob_summa_tax = 0) then
        Continue;

      line_id = line_id + 1;
  
      --0 - Всегда БУ = НУ
      --1 - Всегда НУ = 0
  
      if (ol_id is null) then
      begin
        insert into operlist (ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind, ol_doc_no, ol_doc_no_str,
                              ol_memo, ol_parent_id, is_delete, ol_user_id)
        values (1010100, 2, 11, :rep_date, 9, 0, '', 'Корректировка суммы НУ по счету ' || :balance_account, null, 0, 1000000000001)
        returning ol_id
        into :ol_id;
      end
  
      if (tax_kind = 0) then
        diff_amount = kob_summa_bal - kob_summa_tax;
      else
        diff_amount = -kob_summa_tax;
  
      if (diff_amount = 0) then
        Continue;
  
      if (diff_amount > 0) then
      begin
        acc_dt = acc_id;
        acc_kt = 0;
      end else
      begin
        acc_dt = 0;
        acc_kt = acc_id;
        diff_amount = -diff_amount;
      end
  
      insert into kn_operbook (kob_currency, kob_curr_rate, kob_acc_dt, kob_acc_kt, kob_summa, kob_summa_val,
                               kob_summa_tax_dt, kob_summa_tax_kt, kob_from_ol_id, kob_ol_date, kob_ol_reg_id, kob_ol_is_active)
      values ('810', 1, :acc_dt, :acc_kt, 0, 0,
              :diff_amount, :diff_amount, :ol_id, :rep_date, 1010100, 1);
    end
  end
end
