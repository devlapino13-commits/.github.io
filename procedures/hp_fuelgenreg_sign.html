CREATE PROCEDURE HP_FUELGENREG_SIGN(
    FG_ID D_INTEGER,
    SIGN_BOOKER BOOLEAN = false)
AS
declare variable dt_sign d_datetime;
declare variable ol_id d_id;
declare variable err_text d_str_255;
begin
  dt_sign = current_timestamp;

  if (sign_booker) then
  begin
    select
      ol_id,
      err_text
    from hp_fuelgenreg_accounting_entry (:fg_id)
    into
      :ol_id,
      :err_text;

    update fuelgenreg set
      fg_booker_dtsign = :dt_sign,
      fg_ol_id = :ol_id,
      fg_status = 1
    where fg_id = abs(:fg_id)
      and fg_booker_dtsign is null
      and fg_driver_dtsign is not null
      and fg_manager_dtsign is not null
      and fg_status = 0;
  end else
  begin
    if (fg_id < 0) then
      update fuelgenreg set
        fg_driver_dtsign = :dt_sign
      where fg_id = abs(:fg_id)
        and fg_driver_dtsign is null
        and fg_status = 0;
    else
      update fuelgenreg set
        fg_manager_dtsign = :dt_sign
      where fg_id = abs(:fg_id)
        and fg_manager_dtsign is null
        and fg_status = 0;
  end
end
