CREATE PROCEDURE AAREP_SB_STATISTIC(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    SBL_ID TYPE OF COLUMN SB_LIST.SBL_ID,
    SBL_NAME TYPE OF COLUMN SB_LIST.SBL_NAME,
    SBL_TERRITORY TYPE OF COLUMN SB_LIST.SBL_TERRITORY,
    SBL_TEAM_NAME TYPE OF COLUMN SB_LIST.SBL_TEAM_NAME,
    SBL_CATEGORY TYPE OF COLUMN SB_LIST.SBL_CATEGORY,
    SBL_COORDINATOR TYPE OF COLUMN SB_LIST.SBL_COORDINATOR,
    SBL_MEMO TYPE OF COLUMN SB_LIST.SBL_MEMO,
    SBL_DATE_BEGIN TYPE OF COLUMN SB_LIST.SBL_DATE_BEGIN,
    SBL_DATE_END TYPE OF COLUMN SB_LIST.SBL_DATE_END,
    SBL_PHONE TYPE OF COLUMN SB_LIST.SBL_PHONE,
    SBL_EMAIL TYPE OF COLUMN SB_LIST.SBL_EMAIL,
    SBL_TYPE_STR VARCHAR(20),
    SBL_REGION_NAME VARCHAR(100),
    REP_LOAN_COUNT_BEGIN INTEGER,
    REP_CLOSED_COUNT_BEGIN INTEGER,
    REP_LOAN_COUNT_END INTEGER,
    REP_CLOSED_COUNT_END INTEGER,
    REP_LOAN_FROM_LO INTEGER,
    REP_LOAN_FROM_SB INTEGER,
    REP_LOAN_FROM_COLLECT INTEGER,
    REP_LOAN_TO_LO INTEGER,
    REP_LOAN_TO_SB INTEGER,
    REP_LOAN_TO_COLLECT INTEGER,
    REP_BALANCE_BEGIN DECIMAL(18,4),
    REP_BALANCE_END DECIMAL(18,4),
    REPAY_OS DECIMAL(18,4),
    REPAY_PERC DECIMAL(18,4),
    REPAY_FINE DECIMAL(18,4),
    REPAY_LOAN_COUNT INTEGER,
    REP_BALANCE_FROM_LO DECIMAL(18,4),
    REP_BALANCE_FROM_SB DECIMAL(18,4),
    REP_BALANCE_FROM_COLLECT DECIMAL(18,4),
    REP_BALANCE_TO_LO DECIMAL(18,4),
    REP_BALANCE_TO_SB DECIMAL(18,4),
    REP_BALANCE_TO_COLLECT DECIMAL(18,4),
    REP_DELINQ_COUNT_BEGIN INTEGER,
    REP_DELINQ_BALANCE_BEGIN DECIMAL(18,4),
    REP_DELINQ_COUNT_END INTEGER,
    REP_DELINQ_BALANCE_END DECIMAL(18,4),
    REP_HAS_NO_DELINQ_BEGIN INTEGER,
    REP_HAS_NO_DELINQ_END INTEGER)
AS
declare variable sbl_from_reg_id integer;
declare variable sbl_type integer;
declare variable sbl_okrug_id integer;
declare variable sbl_from_c_id decimal(18,0);
declare variable k_id decimal(18,0);
declare variable k_id_priod decimal(18,0);
declare variable sba_date_begin date;
declare variable sba_date_end date;
declare variable sb_type_tmp integer;
declare variable sb_id_tmp integer;
declare variable k_id_prior decimal(18,0);
declare variable oks_balance_tmp decimal(18,4) = 0;
declare variable dd1 date;
declare variable dd2 date;
declare variable repay_os_tmp decimal(18,4) = 0;
declare variable repay_perc_tmp decimal(18,4) = 0;
declare variable repay_fine_tmp decimal(18,4) = 0;
declare variable oks_delinq_date date;
begin
  for select sbl_id, sbl_name, sbl_from_c_id, sbl_okrug_id, sbl_territory, sbl_team_name, sbl_category, sbl_coordinator,
             sbl_memo, sbl_date_begin, sbl_date_end, sbl_type, sbl_phone, sbl_email, sbl_from_reg_id
      from sb_list
      where sbl_id > 0
      order by sbl_id
      into :sbl_id, :sbl_name, :sbl_from_c_id, :sbl_okrug_id, :sbl_territory, :sbl_team_name, :sbl_category,
           :sbl_coordinator, :sbl_memo, :sbl_date_begin, :sbl_date_end, :sbl_type, :sbl_phone, :sbl_email,
           :sbl_from_reg_id
  do
  begin
    if (sbl_type = 0) then
      sbl_type_str = 'СБ';
    else
      sbl_type_str = 'Коллекторы';

    sbl_region_name = '';
    if (sbl_from_reg_id is not null) then
    begin
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :sbl_from_reg_id
      into :sbl_region_name;
    end

    k_id_prior = 0;
    rep_loan_count_begin = 0;
    rep_loan_count_end = 0;

    rep_loan_from_lo = 0;
    rep_loan_from_sb = 0;
    rep_loan_from_collect = 0;

    rep_loan_to_lo = 0;
    rep_loan_to_sb = 0;
    rep_loan_to_collect = 0;

    rep_balance_begin = 0;
    rep_balance_end = 0;
    rep_closed_count_begin = 0;
    rep_closed_count_end = 0;

    repay_os = 0;
    repay_perc = 0;
    repay_fine = 0;
    repay_loan_count = 0;

    rep_balance_from_lo = 0;
    rep_balance_from_sb = 0;
    rep_balance_from_collect = 0;

    rep_balance_to_lo = 0;
    rep_balance_to_sb = 0;
    rep_balance_to_collect = 0;

    rep_delinq_count_begin = 0;
    rep_delinq_balance_begin = 0;
    rep_delinq_count_end = 0;
    rep_delinq_balance_end = 0;

-- Просмотр каждого кредита у данного СБ
    for select sba_from_k_id, sba_date_begin, sba_date_end
        from sb_akt
        where sba_from_sbl_id = :sbl_id and
              sba_date_begin <= :rep_date_end and
              sba_date_end >= :rep_date_begin and
              sba_date_begin <= sba_date_end and
              sba_status = 2 and
              sba_is_delete = 0
        order by sba_from_k_id, sba_date_begin
        into :k_id, :sba_date_begin, :sba_date_end
    do
    begin
      oks_balance_tmp = null;
-- вычисление баланса на начало периода
      select oks_balance, oks_delinq_date
      from oksummary
      where oks_from_k_id = :k_id and
            oks_date_begin <= :rep_date_begin and
            oks_date_end >= :rep_date_begin and
            oks_close_state = 0
      into :oks_balance_tmp, oks_delinq_date;
      if (oks_balance_tmp is not null) then
      begin
        if (sba_date_begin >= :rep_date_begin) then
          dd1 = sba_date_begin;
        else
          dd1 = :rep_date_begin;

        if (sba_date_end <= :rep_date_end) then
          dd2 = sba_date_end;
        else
          dd2 = :rep_date_end;

        repay_os_tmp = 0;
        repay_perc_tmp = 0;
        repay_fine_tmp = 0;

        select sum(-ok_os), sum(-ok_perc), sum(-ok_fine)
        from operkred
        where ok_from_k_id = :k_id and
              ok_ol_date between :dd1 and :dd2 and
              ok_ol_is_active = 1 and
              ok_type = 4 and
              ok_from_c_id = 0
        into :repay_os_tmp, :repay_perc_tmp, :repay_fine_tmp;

        if (repay_os_tmp is null) then
          repay_os_tmp = 0;
        if (repay_perc_tmp is null) then
          repay_perc_tmp = 0;
        if (repay_fine_tmp is null) then
          repay_fine_tmp = 0;

        repay_os = repay_os + repay_os_tmp;
        repay_perc = repay_perc + repay_perc_tmp;
        repay_fine = repay_fine + repay_fine_tmp;
        if (repay_os_tmp + repay_perc_tmp + repay_fine_tmp <> 0) then
          repay_loan_count = repay_loan_count + 1;

        if ((sba_date_begin <= :rep_date_begin) and
            (sba_date_end >= :rep_date_begin)) then
        begin
-- кол-во на учете на начало периода
          rep_loan_count_begin = rep_loan_count_begin + 1;
          rep_balance_begin = rep_balance_begin + :oks_balance_tmp;

          if (oks_delinq_date < :rep_date_begin) then
          begin
            rep_delinq_count_begin = rep_delinq_count_begin + 1;
            rep_delinq_balance_begin = rep_delinq_balance_begin + oks_balance_tmp;
          end
        end
        if ((sba_date_begin <= :rep_date_end) and
            (sba_date_end >= :rep_date_end)) then
        begin
-- кол-во на учете на конец периода
          rep_loan_count_end = rep_loan_count_end + 1;

-- вычисление баланса на конец периода
          oks_balance_tmp = null;
          select oks_balance, oks_delinq_date
          from oksummary
          where oks_from_k_id = :k_id and
                oks_date_begin <= :rep_date_end and
                oks_date_end >= :rep_date_end and
                oks_close_state = 0
          into :oks_balance_tmp, oks_delinq_date;
          if (oks_balance_tmp is not null) then
          begin
            rep_balance_end = rep_balance_end + :oks_balance_tmp;
            if (oks_delinq_date < :rep_date_end) then
            begin
              rep_delinq_count_end = rep_delinq_count_end + 1;
              rep_delinq_balance_end = rep_delinq_balance_end + oks_balance_tmp;
            end
          end
          begin
-- кол-во закрытых, но на учете данного СБ на конец периода
            if ((sba_date_begin <= :rep_date_end) and
                (sba_date_end >= :rep_date_end)) then
            begin
              rep_closed_count_end = rep_closed_count_end + 1;
            end
          end
        end
        if (sba_date_begin between :rep_date_begin and :rep_date_end) then
        begin
-- принятые за период

          oks_balance_tmp = null;
          select oks_balance
          from oksummary
          where oks_from_k_id = :k_id and
                oks_date_begin <= :sba_date_begin and
                oks_date_end >= :sba_date_begin and
                oks_close_state = 0
          into :oks_balance_tmp;
          if (oks_balance_tmp is null) then
            oks_balance_tmp = 0;

          sb_type_tmp = null;
          sb_id_tmp = null;
          select sbl_type, sbl_id
          from sb_list, sb_akt
          where sba_from_k_id = :k_id and
                sba_date_end = (:sba_date_begin - 1) and
                sba_is_delete = 0 and
                sba_status = 2 and
                sba_from_sbl_id <> :sbl_id and
                sba_date_begin <= sba_date_end and
                sba_from_sbl_id = sbl_id rows 1
          into :sb_type_tmp, sb_id_tmp;
          if (sb_type_tmp is null) then
          begin
            rep_loan_from_lo = rep_loan_from_lo + 1; -- от СВЗ
            rep_balance_from_lo = rep_balance_from_lo + oks_balance_tmp;
          end
          if (sb_type_tmp = 0) then
          begin
            if (sb_id_tmp <> 0) then
            begin
              rep_loan_from_sb = rep_loan_from_sb + 1; -- от СБ
              rep_balance_from_sb = rep_balance_from_sb + oks_balance_tmp;
            end
            else
            begin
              rep_loan_from_lo = rep_loan_from_lo + 1; -- от СВЗ
              rep_balance_from_lo = rep_balance_from_lo + oks_balance_tmp;
            end
          end
          if (sb_type_tmp = 1) then
          begin
            rep_loan_from_collect = rep_loan_from_collect + 1; -- от коллекторов
            rep_balance_from_collect = rep_balance_from_collect + oks_balance_tmp;
          end
        end

        if (sba_date_end between :rep_date_begin and :rep_date_end) then
        begin
-- переданные от него за период

          oks_balance_tmp = null;
          select oks_balance
          from oksummary
          where oks_from_k_id = :k_id and
                oks_date_begin <= :sba_date_end and
                oks_date_end >= :sba_date_end and
                oks_close_state = 0
          into :oks_balance_tmp;
          if (oks_balance_tmp is null) then
            oks_balance_tmp = 0;

          sb_type_tmp = null;
          sb_id_tmp = null;
          select sbl_type, sbl_id
          from sb_list, sb_akt
          where sba_from_k_id = :k_id and
                sba_date_begin = (:sba_date_end + 1) and
                sba_is_delete = 0 and
                sba_status = 2 and
                sba_from_sbl_id <> :sbl_id and
                sba_date_begin <= sba_date_end and
                sba_from_sbl_id = sbl_id rows 1
          into :sb_type_tmp, sb_id_tmp;

          if (sb_type_tmp is null) then
          begin
            rep_loan_to_lo = rep_loan_to_lo + 1; -- возврат СВЗ
            rep_balance_to_lo = rep_balance_to_lo + oks_balance_tmp;
          end
          if (sb_type_tmp = 0) then
          begin
            if (sb_id_tmp <> 0) then
            begin
              rep_loan_to_sb = rep_loan_to_sb + 1; -- передан другому СБ
              rep_balance_to_sb = rep_balance_to_sb + oks_balance_tmp;
            end
            else
            begin
              rep_loan_to_lo = rep_loan_to_lo + 1; -- возврат СВЗ
              rep_balance_to_lo = rep_balance_to_lo + oks_balance_tmp;
            end
          end
          if (sb_type_tmp = 1) then
          begin
            rep_loan_to_collect = rep_loan_to_collect + 1; -- передан коллекторам
            rep_balance_to_collect = rep_balance_to_collect + oks_balance_tmp;
          end

        end
      end
      else
      begin
        if ((sba_date_begin <= :rep_date_begin) and
            (sba_date_end >= :rep_date_begin)) then
        begin
-- кол-во закрытых, но на учете данного СБ на начало периода
          rep_closed_count_begin = rep_closed_count_begin + 1;
          rep_loan_count_begin = rep_loan_count_begin + 1;
        end
      end
    end
    if (rep_balance_end <> rep_delinq_balance_end) then
      rep_has_no_delinq_end = 1;
    else
      rep_has_no_delinq_end = 0;
    if (rep_balance_begin <> rep_delinq_balance_begin) then
      rep_has_no_delinq_begin = 1;
    else
      rep_has_no_delinq_begin = 0;
    suspend;
  end
end
