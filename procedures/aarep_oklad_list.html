CREATE PROCEDURE AAREP_OKLAD_LIST(
    REPORT_DATE_END DATE)
RETURNS (
    SAO_FROM_C_ID DECIMAL(18,0),
    SAO_OKLAD_SUMMA DECIMAL(18,4),
    SAO_CHILD_SUMMA DECIMAL(18,4),
    SAO_LGOTA_SUMMA DECIMAL(18,4),
    SAO_OKLAD_ID INTEGER,
    SAO_CHILD_ID INTEGER,
    SAO_LGOTA_ID INTEGER,
    SAO_MOUNTAIN_PERC DECIMAL(18,7),
    SAO_MOUNT_STAGE_ID INTEGER,
    SAO_MOUNT_KOEF_ID INTEGER,
    SAO_OKLAD_HOUR_ID INTEGER,
    SAO_HOUR_SUMMA DECIMAL(18,4),
    SAO_SF_LGOTA_ID INTEGER,
    SAO_SF_LGOTA DECIMAL(18,7),
    SAO_PDNAL_LGOTA_ID INTEGER,
    SAO_PDNAL_LGOTA DECIMAL(18,4),
    SAO_DONATION_ID INTEGER,
    SAO_DONATION DECIMAL(18,4),
    SAO_FOR_TRANSPORT_ID INTEGER,
    SAO_FOR_TRANSPORT DECIMAL(18,4))
AS
declare variable sum_minus decimal(18,4);
declare variable tt decimal(18,7);
declare variable sao_child_kolich integer;
declare variable oklad decimal(18,4);
declare variable pdn decimal(18,7);
declare variable ijd decimal(18,4);
declare variable krp decimal(18,4);
declare variable rp decimal(18,4);
declare variable sf decimal(18,7);
declare variable sao_is_lgota integer;
declare variable sao_is_net integer;
declare variable sao_is_hour integer;
declare variable db_id integer;
begin

  :db_id = f_get_db_id();

  /* Ставка соц.фонда */
  select scv_value / 100
  from sa_consvalue
  where scv_from_scn_id = 3 and
        scv_date_begin <= :report_date_end
  order by scv_date_begin desc
  rows 1
  into :sf;
  if (sf is null) then
    sf = 0;

  /* Расчетный показатель */
  select scv_value
  from sa_consvalue
  where scv_from_scn_id = 1 and
        scv_date_begin <= :report_date_end
  order by scv_date_begin desc
  rows 1
  into :rp;
  if (rp is null) then
    rp = 0;

  /* Коэф.расчетный показателя для вычета из Подох.налог */
  select scv_value
  from sa_consvalue
  where scv_from_scn_id = 2 and
        scv_date_begin <= :report_date_end
  order by scv_date_begin desc
  rows 1
  into :krp;
  if (krp is null) then
    krp = 0;

  /* Коэф.расчетный показателя для вычета из Подох.налог */
  select scv_value
  from sa_consvalue
  where scv_from_scn_id = 6 and
        scv_date_begin <= :report_date_end
  order by scv_date_begin desc
  rows 1
  into :ijd;
  if (ijd is null) then
    ijd = 0;

  select sac_priority_order
  from sa_config
  where sac_is_credit = 3
  into :sao_oklad_id;
  if (sao_oklad_id is null) then
    sao_oklad_id = 0;

  select sac_priority_order
  from sa_config
  where sac_is_credit = 4
  into :sao_lgota_id;
  if (sao_lgota_id is null) then
    sao_lgota_id = 0;

  select sac_priority_order
  from sa_config
  where sac_is_credit = 5
  into :sao_child_id;
  if (sao_child_id is null) then
    sao_child_id = 0;

  select sac_priority_order
  from sa_config
  where sac_is_credit = 6
  into :sao_mount_stage_id;
  if (sao_mount_stage_id is null) then
    sao_mount_stage_id = 0;

  select sac_priority_order
  from sa_config
  where sac_is_credit = 7
  into :sao_mount_koef_id;
  if (sao_mount_koef_id is null) then
    sao_mount_koef_id = 0;

  select sac_priority_order
  from sa_config
  where sac_is_credit = 8
  into :sao_oklad_hour_id;
  if (sao_oklad_hour_id is null) then
    sao_oklad_hour_id = 0;

  select sac_priority_order
  from sa_config
  where sac_is_credit = 12
  into :sao_sf_lgota_id;
  if (sao_sf_lgota_id is null) then
    sao_sf_lgota_id = 0;

  select sac_priority_order
  from sa_config
  where sac_is_credit = 13
  into :sao_pdnal_lgota_id;
  if (sao_pdnal_lgota_id is null) then
    sao_pdnal_lgota_id = 0;
  sao_pdnal_lgota = 0;

  select sac_priority_order
  from sa_config
  where sac_is_credit = 14
  into :sao_donation_id;
  if (sao_donation_id is null) then
    sao_donation_id = 0;
  sao_donation = 0;

  select sac_priority_order
  from sa_config
  where sac_is_credit = 15
  into :sao_for_transport_id;
  if (sao_for_transport_id is null) then
    sao_for_transport_id = 0;
  sao_for_transport = 0;

  select sap_percent / 100
  from sa_pdnalog
  rows 1
  into :pdn;
  if (pdn is null) then
    pdn = 0;

  sao_oklad_summa = 0;
  sao_is_net = 0;
  sao_child_kolich = 0;
  sao_is_lgota = 0;
  for select sao_from_c_id, sum(sao_summa), sum(sao_is_net), sum(sao_child_kolich), sum(sao_is_lgota),
             sum(sao_mountain_perc), sum(sao_is_hour), sum(sao_sf_lgota), sum(sao_pdnal_lgota), sum(sao_donation)
      from sa_oklad
      where sao_date <= :report_date_end
      group by sao_from_c_id
      having sum(sao_summa) <> 0
      into :sao_from_c_id, :oklad, :sao_is_net, :sao_child_kolich, :sao_is_lgota, :sao_mountain_perc, :sao_is_hour,
           :sao_sf_lgota, :sao_pdnal_lgota, sao_donation
  do
  begin
    sao_oklad_summa = 0;
    sao_hour_summa = 0;
    if (oklad is null) then
      oklad = 0;
    if (sao_is_net is null) then
      sao_is_net = 0;
    if (sao_child_kolich is null) then
      sao_child_kolich = 0;
    if (sao_is_lgota is null) then
      sao_is_lgota = 0;
    if (sao_is_hour is null) then
      sao_is_hour = 0;
    if (sao_mountain_perc is null) then
      sao_mountain_perc = 0;
    if (sao_sf_lgota is null) then
      sao_sf_lgota = 0;
    if (sao_pdnal_lgota is null) then
      sao_pdnal_lgota = 0;
    if (sao_donation is null) then
      sao_donation = 0;
    if (sao_is_hour = 0) then
    begin
      if (sao_is_net = 1) then
      begin
        if (sao_is_lgota = 0) then
        begin
          rp = 0;
          krp = 0;
        end
        sum_minus = (:rp * :krp + :sao_child_kolich * :ijd) * :pdn;
        sao_oklad_summa = :oklad - :sum_minus;
        tt = 1 - sf - pdn + sf * pdn;
        sao_oklad_summa = sao_oklad_summa / tt;
      end
      else
        sao_oklad_summa = :oklad;
    end
    else
      sao_hour_summa = :oklad;
    sao_oklad_summa = cast(sao_oklad_summa as decimal(15,2));
    sao_hour_summa = cast(sao_hour_summa as decimal(15,2));
    sao_child_summa = :sao_child_kolich * :ijd;
    sao_lgota_summa = (:rp * :krp) * sao_is_lgota;

    /* сумма компенсации за транспорт если это КС в КР */
    sao_for_transport = 0;
    if (db_id in (39801, 41701)) then
    begin
      if (exists(select *
                 from hr_pstatus, joblist, jobkind
                 where ps_from_c_id = :sao_from_c_id and
                       ps_date_begin <= :report_date_end and
                       ps_date_end >= :report_date_end and
                       ps_status = 1 and
                       ps_from_jl_id = jl_id and
                       jl_kind = 5 and
                       ps_is_working = 1)) then
        sao_for_transport = 1234.58;
    end
    suspend;
  end
end
