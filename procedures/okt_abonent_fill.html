  from
  (
      select pantk.pant_id,

              pi_date_end >= current_date

    where kont.k_id = :k_id
    into timezone_ru;

    timezone_ru = coalesce(timezone_ru, 3);
    timezone = (case when phone starting('996') then 6
                     when phone starting('998') then 5
                     when phone starting('992') then 5
                     else timezone_ru end);

    phone = left(f_cont_info('clients', :object_id, 'cell_main', 'ok'), 20);
    if (phone <> '') then
    begin

                        from okt_active_abonent_list t

            values (:phone, :object_id, :object_kind, :timezone, :pant_id, :project_code, :task_id);
        end
    end
  end
