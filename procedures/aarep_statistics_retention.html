CREATE PROCEDURE AAREP_STATISTICS_RETENTION(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    KF_OFFICER_NAME VARCHAR(100),
    KNC_NAME_FIO VARCHAR(100),
    K_KONTRACT_NO VARCHAR(20),
    KC_CIRCLE_DISB INTEGER,
    IS_DISB SMALLINT,
    IS_CLOSED SMALLINT,
    MAX_DELINQ_DAY INTEGER,
    OFFICE_NAME VARCHAR(50),
    BRANCH_NAME VARCHAR(50),
    OKRUG_NAME VARCHAR(50),
    PR_NAME VARCHAR(60),
    JS_NAME VARCHAR(100),
    K_KONTRACT_CATEGORY VARCHAR(20),
    SBL_NAME VARCHAR(255))
AS
declare variable kc_from_c_id id_type;
declare variable ks_kc_id id_type;
declare variable ks_region_id integer;
declare variable k_from_kf_id id_type;
declare variable st_staff integer;
declare variable st_client integer;
declare variable ks_kc_id_prior id_type;
declare variable ks_region_id_prior integer;
declare variable k_from_pr_id integer;
declare variable k_id id_type;
declare variable k_id_prior id_type;
declare variable ks_date_begin date;
begin
-- oksummary_fix

  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  ks_kc_id_prior = 0;
  ks_region_id_prior = 0;
  js_name = 'не определен';
  k_id_prior = 0;
  sbl_name = '';
  for with tabl1
      as (
-- Выданные
      select kc_from_c_id, kc_circle, 1 as kc_disb, 0 as kc_closed, ks_kc_id, ks_region_id, k_from_kf_id,
             0 max_delinq_day, 0 k_id, cast('01.01.1900' as date) ks_date_begin
      from kontstate, knclient, kontract, region_list(:rep_region_id, 1, -1, 1)
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
            ks_from_k_id = kc_from_k_id and
            kc_amount_approved > 0 and
            ks_from_k_id = k_id and
            ks_region_id = id_list
      union
-- Закрытые
      select kc_from_c_id, 0 kc_circle, 0 as kc_disb, 1 as kc_closed, ks_kc_id, ks_region_id, k_from_kf_id,
             (select max(iif(oks_date_end <= :rep_date_end, iif(oks_date_end - oks_delinq_date + 1 > 0, oks_date_end - oks_delinq_date + 1, 0), 0))
              from oksummary
              where oks_from_k_id = ks_from_k_id and
                    oks_close_state = 0) as max_delinq_day, k_id, ks_date_begin
      from kontstate, knclient, kontract, region_list(:rep_region_id, 1, -1, 1)
      where ks_ol_is_delete = 0 and
            ks_state = 2 and
            cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
            ks_from_k_id = kc_from_k_id and
            kc_amount_approved > 0 and
            ks_from_k_id = k_id and
            ks_region_id = id_list)

      select kc_from_c_id, ks_kc_id, ks_region_id, k_from_kf_id, max(kc_circle) kc_circle, max(kc_disb) is_disb,
             max(kc_closed) is_closed, max(max_delinq_day) max_delinq_day, max(k_id) k_id,
             max(ks_date_begin) ks_date_begin
      from tabl1
      group by kc_from_c_id, ks_kc_id, ks_region_id, k_from_kf_id
      order by ks_region_id, ks_kc_id
      into kc_from_c_id, ks_kc_id, ks_region_id, k_from_kf_id, kc_circle_disb, is_disb, is_closed, max_delinq_day, k_id,
           ks_date_begin
  do
  begin
    if (k_id = 0) then
    begin
      k_kontract_category = '';
      sbl_name = '';
    end
    else
    if (k_id <> k_id_prior) then
    begin
      k_id_prior = k_id;
      select kont_category
      from aarep_get_category_id(:k_id, :ks_date_begin - 1, null, null, null, null, null, null)
      into k_kontract_category;
--    Вычисляет СБ
      sbl_name = null;
      select sbl_name
      from sb_akt, sb_list
      where sba_from_sbl_id = sbl_id and
            :ks_date_begin between sba_date_begin and sba_date_end and
            sba_is_delete = 0 and
            sba_from_sbl_id <> 0 and
            sba_from_k_id = :k_id and
            sbl_type = 1
      into sbl_name;
      if (sbl_name is null) then
        sbl_name = '';
    end
--  Номер контракта
    select k_kontract_no, k_from_pr_id
    from knaktlist, kontract
    where knaktlist.is_delete = 0 and
          :rep_date_end between kal_date and kal_date_end and
          kal_from_kf_id = :k_from_kf_id and
          kal_from_kf_id = k_from_kf_id and
          k_disb_date <= :rep_date_end
    order by k_disb_date desc
    rows 1
    into k_kontract_no, k_from_pr_id;
    pr_name = (select pr_id || ' ' || pr_name
               from product
               where pr_id = :k_from_pr_id);
--  ФИО СВЗ
    if (ks_kc_id <> ks_kc_id_prior) then
    begin
      ks_kc_id_prior = ks_kc_id;
      select c_name_fio
      from clients
      where c_id = :ks_kc_id
      into kf_officer_name;
-- Системная должность
      js_name = null;
      select rd_name
      from hr_pstatus, joblist, jobkind, refdata
      where ps_from_c_id = :ks_kc_id and
            ps_is_working = 1 and
            ps_status = 1 and
            :rep_date_end between ps_date_begin and ps_date_end and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_id = rd_id
      into js_name;
      if (js_name is null) then
        js_name = 'не определен';

--  Регион
      if (ks_region_id <> ks_region_id_prior) then
      begin
        ks_region_id_prior = ks_region_id;
        select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, r3.reg_code || ' ' || r3.reg_name
        from region r1, region r2, region r3
        where r1.reg_unicode = :ks_region_id and
              r1.parent = r2.reg_unicode and
              r2.parent = r3.reg_unicode
        into office_name, branch_name, okrug_name;
      end
    end
--  ФИО клиента
    select c_name_fio
    from clients
    where c_id = :kc_from_c_id
    into knc_name_fio;

    suspend;
  end
end
