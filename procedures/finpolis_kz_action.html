CREATE PROCEDURE FINPOLIS_KZ_ACTION(
    IS_DELETED D_STR_10,
    UID D_STR_CODE,
    INN D_STR_CODE,
    FIRST_NAME D_STR_100,
    SURNAME D_STR_100,
    PATRONIMIC D_STR_100,
    BIRTH_DATE D_DATE,
    LIST_CATEGORY D_STR_CODE,
    PHONE_LIST D_STR_8100)
AS
declare variable FP_ID D_ID;
declare variable FIO D_STR_255;
begin
    phone_list = ''; -- Обнуляем список телефонных номеров т.к. по ним мы не осуществляем поиск

    for select fpu.fpu_from_fp_id
    from finpolis_uid fpu
    where fpu.fpu_uid = :uid
    into :fp_id
    do
    begin
        delete from finpolis fp
        where fp.fp_id = :fp_id;
    end

    if (is_deleted = 'false') then
    begin
        fio = trim(coalesce(surname, '') || ' ' || coalesce(first_name, '') || ' ' || coalesce(patronimic, ''));
        
        insert into finpolis (nameu, descript, nd, gr, terr_categ, phones)
        values (:fio, :uid, :inn, :birth_date, :list_category, :phone_list);
    end
end
