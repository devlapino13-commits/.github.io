CREATE PROCEDURE AAREP_SCHEDULE_CLIENT(
    KONT_ID DECIMAL(18,0))
RETURNS (
    SH_ID INTEGER,
    SH_DATE DATE,
    SH_OS DECIMAL(18,4),
    SH_PERC DECIMAL(18,4),
    SH_KOMMIS DECIMAL(18,4),
    SH_DISB_AMOUNT DECIMAL(18,4),
    SH_NAME_FIO VARCHAR(100),
    SH_PERC_TAX DECIMAL(18,4),
    REG_NAME VARCHAR(50),
    REG_ADDRESS VARCHAR(255),
    KF_NAME VARCHAR(100),
    KF_OFFICER VARCHAR(100),
    OFFISER_PHONE VARCHAR(20),
    SIGNMEN VARCHAR(100),
    SIGNMEN_PHON VARCHAR(20),
    REG_PHONE VARCHAR(255),
    C_LSCHET_KG D_STR_255)
AS
declare variable sh_os1 decimal(18,4);
declare variable sh_perc1 decimal(18,4);
declare variable sh_kommis1 decimal(18,4);
declare variable disb_all decimal(18,4);
declare variable sh_perc_tax1 decimal(18,4);
declare variable st_staff integer;
declare variable st_client integer;
declare variable k_disb_date date;
declare variable reg_address_fact decimal(18,0);
declare variable k_nsp_percent decimal(18,7);
declare variable db_id integer;
declare variable c_uniq_acc d_str_100;
declare variable c_birth_date d_date;
declare variable c_id d_id;
begin
  :db_id = f_get_db_id();
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  k_nsp_percent = 0;
  c_uniq_acc = 'XXX';
  -- Вытаскивает параметры офиса и КС
  select reg_name, reg_address, kal_new_kf_name, c_name_fio,
         (select cont_value
          from hp_contdata_get_cell_main(0, c_id)), k_disb_date, reg_phone, reg_address_fact
  from kontract, knaktlist, region, clients
  where k_from_kf_id = kal_from_kf_id and
        knaktlist.is_delete = 0 and
        k_disb_date between kal_date and kal_date_end and
        k_id = :kont_id and
        reg_unicode = kal_new_reg_id and
        c_id = kal_new_kf_officer
  into reg_name, reg_address, kf_name, kf_officer, offiser_phone, k_disb_date, reg_phone, reg_address_fact;

  if (exists(select *
             from nspconfig_data
             where nsd_from_nsm_id = 0 and
                   nsd_date_begin <= :k_disb_date and
                   nsd_date_end >= :k_disb_date)) then
  begin
    select nsd_percent
    from nspconfig_data
    where nsd_from_nsm_id = -1 -- нсп от процента
          and
          nsd_date_begin <= :k_disb_date and
          nsd_date_end >= :k_disb_date
    into :k_nsp_percent;
    if (k_nsp_percent is null) then
      k_nsp_percent = 0;
  end

  -- вытаскивает уполномоченное лицо кто утверждал в последний раз
  if (reg_address_fact > 0) then
  begin
    select adl_address_str
    from addrlist
    where adl_id = :reg_address_fact
    into :reg_address;
  end

  select un_fio,
         (select cont_value
          from hp_contdata_get_cell_main(0, c_id))
  from kontlog, username, kontsignmen, clients
  where kl_state = 1 and
        kl_user_id = un_id and
        ksm_is_active = 1 and
        ksm_from_c_id = un_from_c_id and
        un_from_c_id = c_id and
        kl_from_k_id = :kont_id and
        cast(kl_datetime as date) <= :k_disb_date
  order by kl_datetime desc rows 1
  into signmen, signmen_phon;
  if (reg_phone is null) then
    reg_phone = signmen_phon;

  for select k_amount_approved, c_name_fio, c_birth_date, kc_amount_approved, kc_from_c_id
      from knclient, clients, kontract
      where kc_from_c_id = c_id and
            kc_from_k_id = :kont_id and
            kc_amount_approved <> 0 and
            k_id = :kont_id
      order by kc_number
      into :disb_all, :sh_name_fio, :c_birth_date, :sh_disb_amount, :c_id
  do
  begin
    c_uniq_acc = 'XXX';
    select uniq_acc_str
    from hp_show_clientuniq_acc_no(:c_id)
    into :c_uniq_acc;

    if (c_uniq_acc = 'XXX') then
      c_uniq_acc = 'Не закреплен';
    else
      c_lschet_kg = extract(year from c_birth_date) || right(0 || extract(month from c_birth_date), 2) || right(0 || extract(day from c_birth_date), 2) || '-' || c_uniq_acc;

    for select sh_id, sh_date, sh_os, sh_perc, sh_kommis, sh_perc_tax
        from schedule
        where sh_from_k_id = :kont_id and
              (:db_id in (39801, 41701) and
              sh_id > 0 or :db_id not in (39801, 41701) and
              sh_id >= 0)
        order by sh_date
        into :sh_id, :sh_date, :sh_os1, :sh_perc1, :sh_kommis1, :sh_perc_tax1
    do
    begin
      --sh_perc_tax1=round((sh_perc1+sh_kommis1)*k_nsp_percent/100);
      sh_perc_tax1 = (select res
                      from fhp_round_bank((:sh_perc1 + :sh_kommis1) * :k_nsp_percent / 100, 0));

      --   sh_perc_tax1 = (sh_perc1 + sh_kommis1) * k_nsp_percent / 100;

      sh_os = cast(sh_os1 * (sh_disb_amount / disb_all) as decimal(18,2));
      sh_perc = cast(sh_perc1 * (sh_disb_amount / disb_all) as decimal(18,2));
      sh_kommis = cast(sh_kommis1 * (sh_disb_amount / disb_all) as decimal(18,2));

      sh_perc_tax = 0;

      sh_perc_tax = cast(sh_perc_tax1 * (sh_disb_amount / disb_all) as decimal(18,2));
      suspend;
    end
  end
end
