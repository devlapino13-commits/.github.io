    OFFICER_ID ID_TYPE,
    LIST_TYPE SMALLINT,
    REP_REGION_ID ID_TYPE)

    K_ID ID_TYPE,
    K_DISB_DATE DATE,
    UN_ID ID_TYPE,

      if (officer_id = 1010100033350) then
        rep_region_id = 1000000; -- Гусев И. - доступ РФ
      if (officer_id = 1010100003532) then
        rep_region_id = 1000000; -- Абыдынов Акылбек Кошбаевич - доступ РФ

      for select kont.k_id, kncl.kc_from_c_id, kont.k_disb_date, kont.k_from_kf_id, kont.k_from_ma_app_id
          from kontstate ks, kontract kont, knclient kncl, region_list(:rep_region_id, 1, -1, 1)
          where ks.ks_kc_id = 1000000000004 -- КЦ Сирена
                and
                ks.ks_region_id = id_list and
                ks_ol_is_delete = 0 and
                ks_state = 0 and
                :rep_date between ks_date_begin and ks_date_end and
                ks_from_k_id = k_id and
                k_status = 0 and
                kncl.kc_from_k_id = kont.k_id and
                kncl.kc_amount_approved > 0
          order by kont.k_disb_date, kont.k_from_kf_id, kont.k_id, kncl.kc_from_c_id
          into k_id, c_id, k_disb_date, :kf_id, :ma_app_id
      do
      begin
        un_id = null;
        check_result = null;

        select cont.contr_from_un_id, rd_res.rd_code, cont.contr_checked_date
        from controller cont, refdata rd_type, refdata rd_res
        where cont.contr_from_k_id = :k_id and
              cont.contr_from_c_id = :c_id and
              rd_type.rd_id = cont.contr_rd_type and
              rd_res.rd_id = cont.contr_check_result and
              rd_type.rd_code in ('ControllerType.Specialist_Visitor', 'ControllerType.LO') and
              cont.contr_is_delete = 0
        order by cont.contr_id desc
        rows 1
        into :un_id, :check_result, :contr_checked_date;

        if (contr_checked_date is not null) then
          if (contr_checked_date < rep_date - 7) then
          begin
            check_result = 'ControllerCheckStatus.NotChecked';
            un_id = null;
            contr_checked_date = null;
          end
        check_result = coalesce(check_result, 'ControllerCheckStatus.NotChecked');

-- Для ВО отображаю участников группы, если они не заблокированны на проверку либо заблокированны, но еще не проверены
        if (check_result = 'ControllerCheckStatus.NotChecked') then
        begin
-- Если это клиент еще не проверялся в течении 7-и дней

      end

        inner join kontract kont on kont.k_id = ks.ks_from_k_id
        inner join knclient kncl on kncl.kc_from_k_id = ks.ks_from_k_id
        inner join region reg on reg.reg_unicode = ks.ks_region_id
        where ks.ks_date_begin between :rep_date - 7 and :rep_date and

              (:rep_region_id = 1000000)) and

              ks_from_k_id = k_id and
              k_status = 0 and
              kncl.kc_from_k_id = kont.k_id and
              kncl.kc_amount_approved > 0
        order by kont.k_disb_date, kont.k_from_kf_id, kont.k_id, kncl.kc_from_c_id
        into k_id, c_id, k_disb_date, :kf_id, :ma_app_id

    begin
      un_id = null;

      check_result = null;

  begin

    begin
      for select k_id, c_id, checked_date, disb_date, kf_id, ma_app_id, data_type

                 kont.k_disb_date as disb_date, kont.k_from_kf_id as kf_id, kont.k_from_ma_app_id as ma_app_id,

          inner join region_list(:rep_region_id, 1, -1, 1) on cast(kont.k_last_reg_id as id_type) = id_list -- Москва 1010000

                cont.contr_checked_date >= :rep_date - 7 -- чтобы отсеять лишние записи
                and

                cont.contr_is_delete = 0 and
                kont.k_disb_date >= :rep_date - 7 and
                kncl.kc_amount_approved > 0 and
                reg.reg_logic_label containing ('.mega_office.')
----------------------
          union
--------- СВЗ --------
          select distinct kont.k_id, kncl.kc_from_c_id as c_id, ks.ks_date_begin as checked_date,
                          kont.k_disb_date as disb_date, kont.k_from_kf_id as kf_id, kont.k_from_ma_app_id as ma_app_id,
                          2 as data_type
          from kontract kont
          inner join knclient kncl on kncl.kc_from_k_id = kont.k_id
          inner join kontstate ks on kont.k_id = ks.ks_from_k_id
          inner join region_list(:rep_region_id, 1, -1, 1) on id_list = kont.k_last_reg_id
          inner join region reg on reg.reg_unicode = id_list
          inner join hr_pstatus hps on hps.ps_from_c_id = ks.ks_kc_id

      begin

          end

          begin
            if (kf_id <> kf_id_tmp) then
            begin
              kf_counter = kf_counter + 1;
              kf_id_tmp = kf_id;
            end
            suspend;
          end
        end
      end
    end
    else
    begin
      for select k_id, k_disb_date, un_id, check_result, kf_id, kf_counter, c_id, ma_app_id
          from controller_for_block_list_old(:rep_date, 3, :rep_region_id)
          into :k_id, :k_disb_date, :un_id, :check_result, :kf_id, :kf_counter, :c_id, :ma_app_id

        suspend;
      end
    end
  end
