CREATE PROCEDURE HP_TMP_LOAN_LIST_ACTIVE(
    REP_DATE D_DATE NOT NULL,
    ONLY_OWN_LOANS D_BOOL NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_K_ID D_ID,
    RES_DATE_BEGIN D_DATE,
    RES_IS_OWN D_BOOL)
AS
declare variable reg_id d_integer;
declare variable c_id d_id;
declare variable lml_id d_integer;
declare variable jk_system_code d_str_code;
declare variable lmk_from_lml_id d_integer;
declare variable rep_region_id d_integer;
declare variable need_region_suspend d_bool;
declare variable orig_c_id d_id;
begin
  select un_from_c_id
  from username
  where un_id = :un_id
  into orig_c_id;

  for select res_c_id
  from hp_staff_changer_list(:un_id, iif(:only_own_loans = 1, -1, 1))
  into :c_id
  do
  begin
    res_is_own = 0;
    if (c_id = orig_c_id) then
      res_is_own = 1;

    need_region_suspend = 1;

    select ps_from_reg_id, jk_system_code
    from hr_pstatus, joblist, jobkind
    where ps_from_c_id = :c_id and
          ps_is_working = 1 and
          ps_date_begin <= :rep_date and
          ps_date_end >= :rep_date and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id
    into rep_region_id, jk_system_code;
  
    if (row_count = 0) then
      Continue;

    if (jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Partner', 'JobSystem.Specialist_Expert')) then
    begin
      -- Кредитный специалист
      for select oks_from_k_id, oks_date_begin
      from oksummary
      where oks_kc_id = :c_id and
            oks_date_begin + 0 <= :rep_date and -- Искуственно отключаем использование индекса
            oks_date_end >= :rep_date and
            oks_close_state = 0
      into res_k_id, res_date_begin
      do
        suspend;
  
      need_region_suspend = 0;
    end else
    begin
      lml_id = null;

      select lml_id
      from loanmgrlist
      where lml_from_c_id = :c_id and
            lml_date_begin <= :rep_date and
            lml_date_end >= :rep_date
      into lml_id;
  
      if (lml_id is not null) then
      begin
        -- Менеджер мега-офиса, получаем список подчиненных специалистов
        for select ke_from_p_id, ke_from_reg_id
        from krexpert
        where ke_from_lml_id = :lml_id and
              ke_date <= :rep_date and
              ke_date_end >= :rep_date
        into c_id, reg_id
        do
        begin
          for select oks_from_k_id, oks_date_begin
          from oksummary
          where oks_kc_id = :c_id and
                oks_region_id + 0 = :reg_id and     -- Искуственно отключаем использование индекса
                oks_date_begin + 0 <= :rep_date and -- Искуственно отключаем использование индекса
                oks_date_end >= :rep_date and
                oks_close_state = 0
          into res_k_id, res_date_begin
          do
            suspend;
        end
  
        for select distinct lmk_from_k_id
        from loanmgrkont
        where lmk_from_lml_id = :lml_id and
              lmk_date_begin <= :rep_date
        into res_k_id
        do
        begin
          res_date_begin = null;
          select oks_date_begin
          from oksummary
          where oks_from_k_id = :res_k_id and
                oks_date_begin <= :rep_date and
                oks_date_end >= :rep_date and
                oks_kc_id = 1000000000004 and -- КЦ Сирена
                oks_close_state = 0
          into res_date_begin;
  
          if (res_date_begin is not null) then
          begin
            select lmk_from_lml_id
            from loanmgrkont
            where lmk_from_k_id = :res_k_id and
                  lmk_date_begin <= :rep_date
            order by lmk_date_begin desc
            rows 1
            into lmk_from_lml_id;
  
            if (lmk_from_lml_id = lml_id) then
              suspend;
          end
        end
  
        need_region_suspend = 0;
      end else
      begin
        -- Если не сотрудник фронт-офиса, завершаем работу
        if (jk_system_code not in ('JobSystem.Region_Director', 'JobSystem.Region_KM', 'JobSystem.Specialist_SB')) then
          need_region_suspend = 0;
      end
    end

    if (need_region_suspend = 1) then
    begin
      -- Список займов в офисе.
      for select id_list
      from region_list(:rep_region_id, 1, 3, 1)
      into reg_id
      do
      begin
        for select oks_from_k_id, oks_date_begin
        from oksummary
        where oks_region_id = :reg_id and
              oks_date_begin <= :rep_date and
              oks_date_end >= :rep_date and
              oks_close_state = 0
        into res_k_id, res_date_begin
        do
          suspend;
      end
    end
  end
end
