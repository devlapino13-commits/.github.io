CREATE PROCEDURE MILLICARD_CREATE_OPERBOOK(
    ESM_ID INTEGER)
AS
declare variable k_disb_date date;
declare variable c_name_fio str_100;
declare variable k_kontract_no str_100;
declare variable kf_name str_100;
declare variable id_no id_type;
declare variable ol_id id_type;
declare variable reg_id integer;
declare variable amount curr_type;
declare variable ob_memo varchar(1024);
declare variable acc_dt varchar(20);
declare variable acc_kt varchar(20);
declare variable c_id id_type;
declare variable esm_project str_100;
declare variable acc_dt_str str_100;
declare variable acc_kt_str str_100;
begin
  select k_disb_date, c_name_fio, k_kontract_no, kal_new_kf_name, kal.kal_new_reg_id, em.esm_amount, cl.c_id, em.esm_project
  from extsysmain em
  inner join kontract kont on kont.k_id = em.esm_from_k_id
  inner join clients cl on cl.c_id = em.esm_object_id
  inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
  where em.esm_id = :esm_id and
        kont.k_disb_date between kal.kal_date and kal.kal_date_end and
        kal.is_delete = 0
  into :k_disb_date, :c_name_fio, :k_kontract_no, :kf_name, :reg_id, :amount, :c_id, :esm_project;

  select kv
  from get_kv(1010100, 5)
  into :id_no;

  ol_id = :id_no;

  ob_memo = f_datetostr(:k_disb_date) || ' tarixli ' || k_kontract_no || ' nomreli muqavileye esasen kredit buraxilishi. Kredit papkasi № ' || kf_name || ' / ' || c_name_fio;

  insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_status, is_delete,
                        ol_user_id, ol_document_client)
  values (:ol_id, 1010100, 2, :k_disb_date, 9, null, 0, 0, 1000000000000, :c_id);

  select al_id
  from find_acc_list('25170') t
  where al_from_reg_id = 1010100 and
        left(al_name, 4) = (select reg_code
                            from region reg
                            where reg.reg_unicode = :reg_id)
  into :acc_dt;

  acc_kt_str = '15010/12';
  if (esm_project = 'millicard_biz_disb') then
    acc_kt_str = '15010/13';

  select al_id
  from find_acc_list(:acc_kt_str) t
  where al_from_reg_id = 1010100
  into :acc_kt;

  insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                        ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                        ob_from_sac_id)
  values (:ol_id, 1, '944', 1, null, :acc_dt, :amount, :amount, null, null, null, :acc_kt, null, :ob_memo, null);

  insert into extsysanswer (esa_from_esm_id, esa_message)
  values (:esm_id, 'ol_id = ' || :ol_id);
end
