CREATE PROCEDURE CC_WMA_ANKETA_GET(
    WMA_ID INTEGER,
    GIVE_ANKETA_IN_HTML SMALLINT = 0)
RETURNS (
    PHONE_NO STR_100,
    NAME STR_255,
    GENDER SMALLINT,
    BIRTHDAY DATE,
    CITY STR_100,
    CITY_ID ID_TYPE,
    CITIZENSHIP STR_100,
    LANGUAGE STR_100,
    MEETING_POINT STR_255,
    MEETING_ADDRESS STR_255,
    CLIENTINCOMPANYSTATUS STR_100,
    PHONEADD STR_100,
    PASSNO STR_100,
    DOCLIST STR_255,
    WORKSTATUS STR_100,
    CURRENTLOANMBF SMALLINT,
    ACTIVELOANOTHECOMPANY SMALLINT,
    LOANPRODUCT STR_100,
    LOANPERIOD SMALLINT,
    LOANAMOUNT INTEGER,
    MEETINGADDRESS STR_255,
    MEETINGDATE DATE,
    OFFICE INTEGER,
    ANKETA_HTML D_BLOB_TEXT,
    INFO_SOURCE STR_100,
    INFO_SOURCE_TEXT STR_255,
    LOAN_PURPOSE STR_100)
AS
declare variable citizenship_name str_100;
declare variable language_name str_100;
declare variable meeting_point_name str_100;
declare variable client_in_company_status_name str_100;
declare variable work_status_name str_100;
declare variable loan_product_name str_100;
declare variable office_name str_100;
declare variable gender_name str_100;
declare variable city_name str_100;
declare variable active_loan_othe_company_name str_100;
declare variable current_loan_mbf_name str_100;
declare variable wma_status_name str_100;
declare variable wma_status smallint;
declare variable close_name str_100;
declare variable refusal_reason_name str_100;
declare variable info_source_name str_100;
declare variable loan_purpose_name str_100;
begin
  select wma_phone_no,
         wmat_name,
         wmat_gender,

         wmat_birthday,
         wmat_city,
         wmat_city_id,

         rd_c.rd_code  as citizenship,
         rd_l.rd_code  as language,
         rd_mp.rd_code as meeting_point,

         wmat_meeting_address,
         rd_ccs.rd_code as client_in_company_status,
         wmat_phone_add,

         wmat_pass_no,
         wmat_doclist,
         rd_ws.rd_code as wmat_work_status,

         wmat_current_loan_mbf,
         wmat_active_loan_othe_company,
         rd_pr.rd_code as wmat_loan_product,

         wmat_loan_period,
         wmat_loan_amount,
         wmat_meeting_address,

         wmat_meeting_date,
         wmat_office,

         coalesce(rd_c.rd_name, '-')   as citizenship_name,
         coalesce(rd_l.rd_name, '-')   as language_name,
         coalesce(rd_mp.rd_name, '-')  as meeting_point_name,
         coalesce(rd_ccs.rd_name, '-') as client_in_company_status_name,
         coalesce(rd_ws.rd_name, '-')  as work_status_name,
         coalesce(rd_pr.rd_name, '-')  as loan_product_name,
         coalesce(reg.reg_name, '-')   as office_name,
         iif(wmt.wmat_gender = 0, 'Женский', 'Мужской'),
         coalesce(pl.pl_name, '-')     as city_name,
         iif(wmat_current_loan_mbf = 0, 'Нету', 'Есть') as current_loan_mbf_name,
         iif(wmat_active_loan_othe_company = 0, 'Нету', 'Есть') as active_loan_othe_company_name,
         (case wma_status
                    when -1 then 'Закрыта. (Негатив. статус)'
                    when 0  then 'Новая'
                    when 1  then 'Перезвонить'
                    when 2  then 'Передана в офис'
         end) as wma_status_name,
         wma_status,
         coalesce(rd_close.rd_name, '-'),
         coalesce(rd_refusal_reason.rd_name, ''),

         rd_info_source.rd_code as info_source,
         coalesce(rd_info_source.rd_name, '-') as info_source_name,
         coalesce(wmt.wmat_info_source_text, ''),
         rd_loan_purpose.rd_code as loan_purpose,
         coalesce(rd_loan_purpose.rd_name, '-') as loan_purpose_name
  from cc_website_miniapp wma
  inner join cc_website_miniapp_tmp wmt on wmt.wmat_id = wma_last_wmat_id
  left join cc_website_miniapp_close wmc on wmc.wmc_from_wma_id = wma.wma_id
  left join refdata rd_c on rd_c.rd_id = wmat_citizenship
  left join refdata rd_l on rd_l.rd_id = wmat_language
  left join refdata rd_mp on rd_mp.rd_id = wmat_meeting_point
  left join refdata rd_ws on rd_ws.rd_id = wmt.wmat_work_status
  left join refdata rd_pr on rd_pr.rd_id = wmt.wmat_loan_product
  left join refdata rd_ccs on rd_ccs.rd_id = wmt.wmat_client_in_company_status
  left join region reg on reg.reg_unicode = wmat_office
  left join places pl on pl.pl_id = wmt.wmat_city_id
  left join refdata rd_close on rd_close.rd_id = wmc.wmc_closed_rd_id
  left join refdata rd_refusal_reason on rd_refusal_reason.rd_id = wmc.wmc_refuse_id
  left join refdata rd_info_source on rd_info_source.rd_id = wmat_info_source
  left join refdata rd_loan_purpose on rd_loan_purpose.rd_id = wmat_loan_purpose
  -- left join refdata rd_svz_lang on rd_svz_lang.rd_id = wmt.wmat_svz_lang
  where wma.wma_id = :wma_id
  into phone_no,
       name,
       gender,

       birthday,
       city,
       city_id,

       citizenship,
       language,
       meeting_point,

       meeting_address,
       clientincompanystatus,
       phoneadd,

       passno,
       doclist,
       workstatus,

       currentloanmbf,
       activeloanothecompany,
       loanproduct,

       loanperiod,
       loanamount,
       meetingaddress,

       meetingdate,
       office,

       citizenship_name,
       language_name,
       meeting_point_name,
       client_in_company_status_name,
       work_status_name,
       loan_product_name,
       office_name,
       gender_name,
       city_name,
       current_loan_mbf_name,
       active_loan_othe_company_name,
       wma_status_name,
       wma_status,
       close_name,
       refusal_reason_name,

       info_source,
       info_source_name,
       info_source_text,
       loan_purpose,
       loan_purpose_name;

   anketa_html = '';

   loanperiod = coalesce(loanperiod, 0);
   loanamount = coalesce(loanamount, 0);
   gender = coalesce(gender, 0);

   if (give_anketa_in_html = 1) then
   begin
        anketa_html = BLOB_APPEND(anketa_html,'<h3>Статус заявки: '||wma_status_name||'</h3>');

        anketa_html = BLOB_APPEND(anketa_html,'Клиента компании: <b>'||client_in_company_status_name||'</b><br>');
        anketa_html = BLOB_APPEND(anketa_html,'Язык общения: <b>'||language_name||'</b><br>');
        anketa_html = BLOB_APPEND(anketa_html,'Пол: <b>'||gender_name||'</b><br>');
        anketa_html = BLOB_APPEND(anketa_html,'ФИО: <b>'||name||'</b><br>');
        anketa_html = BLOB_APPEND(anketa_html,'Номер телефона: <b>'||iif(phone_no = '', '-', phone_no)||'</b><br>');
        anketa_html = BLOB_APPEND(anketa_html,'Доп. номер телефона: <b>'||iif(phoneadd = '',  '-',phoneadd)||'</b><br>');
        anketa_html = BLOB_APPEND(anketa_html,'Гражданство: <b>'||citizenship_name||'</b><br>');
        anketa_html = BLOB_APPEND(anketa_html,'Город (из списка): <b>'||city_name||'</b><br>');
        anketa_html = BLOB_APPEND(anketa_html,'Город (произвольно): <b>'||city||'</b><br>');
        anketa_html = BLOB_APPEND(anketa_html,'Номер паспорта: <b>'||passno||'</b><br>');
        anketa_html = BLOB_APPEND(anketa_html,'Дата рождения: <b>'||iif(birthday is null, '-', f_datetostr(birthday))||'</b><br>');
        anketa_html = BLOB_APPEND(anketa_html,'Документы: <b>'||doclist||'</b><br>');
        anketa_html = BLOB_APPEND(anketa_html,'Работа: <b>'||work_status_name||'</b><br>');
        anketa_html = BLOB_APPEND(anketa_html,'Текущий заем в '||iif(f_get_db_id() = 81001, 'М Булак: ', 'М Булак: ')||'<b>'||current_loan_mbf_name||'</b><br>');
        anketa_html = BLOB_APPEND(anketa_html,'Текущий заем в другой компании: <b>'||active_loan_othe_company_name||'</b><br>');
        anketa_html = BLOB_APPEND(anketa_html,'Вид займа: <b>'||loan_product_name||'</b><br>');
        anketa_html = BLOB_APPEND(anketa_html,'Срок займа: <b>'||loanperiod||'</b><br>');
        anketa_html = BLOB_APPEND(anketa_html,'Сумма займа: <b>'||loanamount||'</b><br>');
        anketa_html = BLOB_APPEND(anketa_html,'Место встречи: <b>'||meeting_point_name||'</b><br>');
        anketa_html = BLOB_APPEND(anketa_html,'Адрес встречи с ВИПС: <b>'||meeting_address||'</b><br>');
        anketa_html = BLOB_APPEND(anketa_html,'Дата встречи с ВИПС: <b>'||iif(meetingdate is null, '-', f_datetostr(meetingdate))||'</b><br>');
        anketa_html = BLOB_APPEND(anketa_html,'Офис: <b>'||office_name||'</b><br>');
        anketa_html = BLOB_APPEND(anketa_html,'Цель займа: <b>'||loan_purpose_name||'</b><br>');
        anketa_html = BLOB_APPEND(anketa_html,'Откуда узнали номер горячей линии: <b>'||info_source_name||''||iif(info_source = 'hot_line_info_source.Other', ' ['||info_source_text||']', '')||'</b><br>');

        if (wma_status = -1) then
        begin
            anketa_html = BLOB_APPEND(anketa_html,'<br>');
            anketa_html = BLOB_APPEND(anketa_html,'Причина закрытия заявки: <b>'||close_name||'</b><br>');
            if (refusal_reason_name <> '') then
                anketa_html = BLOB_APPEND(anketa_html,'Причина отказа: <b>'||refusal_reason_name||'</b><br>');
        end
   end
   suspend;
end
