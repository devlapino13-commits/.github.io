CREATE PROCEDURE HP_NEW_NEGATIV_KIB_QUERY(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable INN D_STR_20;
declare variable JSON_STR D_STR_255;
declare variable EIL_ID D_ID;
declare variable CFG D_STR_CODE;
declare variable BLS_ID D_ID;
begin
  -- RFCKG-4215 ПКБ Негативная кредитная история клиента: Отправка Запроса
  cfg = 'pkb_cib_report_negative';
  res_text = '';

  select c.c_inn
  from clients c
  where c.c_id = :c_id
  into inn;

  select first 1 eil.eil_from_bls_id
  from extinfolist eil
  inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
  where eic.eic_code = 'pkb_available_reports' and
        eil.eil_object_kind = f_objkind('clients', 'id') and
        eil.eil_object_id = :c_id and
        eil.eil_status = 2 and
        eil.eil_from_bls_id = not null
  order by eil.eil_status_datetime desc
  into :bls_id;

  if (f_pkb_available_reports(:bls_id) = 1) then
  begin
    if (coalesce(inn, '') = '') then
      res_text = 'У клиента отсутствует ИИН';

    if (res_text = '') then
    begin
      json_str = f_format('{"report_type": 13, "doc1_type": 14, "doc1_number": "%1", "consent_confirmed": true}', inn);
    
      execute procedure hp_extinfo_insert(:cfg, 'clients', :c_id, 0, :un_id);
    
      select eil.eil_id
      from extinfolist eil
      inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
      where eil.eil_from_c_id = :c_id
      and eic.eic_code = :cfg
      order by eil.eil_id desc
      rows 1
      into :eil_id;
    
      insert into negativ_kib_history (nkh_from_c_id,  nkh_from_un_id, nkh_from_eil_id, nkh_json)
      values (:c_id, :un_id, :eil_id, :json_str);
    
      post_event :cfg;
      res_text = 'Запрос отправлен';
    end

  end
  else
   res_text = 'У клиента отсутствует негативный отчет';

  suspend;
end
