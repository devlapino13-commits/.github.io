CREATE PROCEDURE FM_HTML_EXTINFO_EQUIFAX_FS(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_HTML D_BLOB_TEXT)
AS
declare variable lang d_str_lang;
declare variable eil_date d_date;
declare variable calc_status d_str_255;
declare variable bls_id d_id;
declare variable c_name_fio d_str_255;
declare variable k_kontract_no d_str_code;
declare variable html_loan d_str_8100;
declare variable row_fmt d_str_255;
declare variable row_fmt_fg d_str_255;
declare variable cred_id d_str_code;
declare variable cred_no d_integer;
declare variable cred_partner_type d_integer;
declare variable cred_partner_type_str d_str_code;
declare variable cred_type_str d_str_255;
declare variable cred_sum d_currency;
declare variable cred_date d_date;
declare variable cred_end_date d_date;
declare variable cred_sum_debt d_currency;
declare variable cred_sum_overdue d_currency;
declare variable cred_day_overdue d_integer;
declare variable cred_max_overdue d_currency;
declare variable cred_update d_date;
declare variable cred_active_str d_str_255;
declare variable cred_active_order d_integer;
declare variable cred_active_order_prv d_integer;
declare variable cred_active_order_str d_str_code;
declare variable cred_uid d_str_code;
declare variable active_cnt d_integer;
declare variable sell_cnt d_integer;
declare variable collect_cnt d_integer;
declare variable closed_cnt d_integer;
declare variable overdue_cnt d_integer;
declare variable html_list d_blob_text;
declare variable kc_id d_id;
declare variable error_txt d_str_255;
begin
  lang = f_get_lang_code(un_id, null);

  -- Строки формата для разных строк. С раскраской и без.
  row_fmt = '<tr><td>%1</td><td><b>%2</b></td>';
  row_fmt_fg = '<tr><td>%1</td><td><font color="%3"><b>%2</b></font></td>';
  res_html = (select res_style from fm_html_styles);

  /************************************************  Скоринг Бюро **************************************/
  res_html = BLOB_APPEND(res_html, '<body><h1>Источник №1</h1>');
  -- Счетчики для хранения количества кредитных историй по категориям
  active_cnt = 0;
  sell_cnt = 0;
  collect_cnt = 0;
  closed_cnt = 0;
  overdue_cnt = 0;

  select eil_date, eil_from_bls_id
  from extinfolist, extinfocfg
  where eil_from_c_id = :c_id
    and eil_from_eic_id = eic_id
    and eic_code = 'equifax_40'
    and eil_is_json = 1
    and eil_status = 2
  order by eil_date desc, eil_id
  rows 1
  into eil_date, bls_id;

  if (bls_id is null) then
    calc_status = f_lang(lang, 'Кредитная история не получена');
  else
    calc_status = f_format('<br>%1: %2', 'Кредитная история получена', f_datetostr(eil_date));

  select c_name_fio
  from clients
  where c_id = :c_id
  into c_name_fio;

  res_html = BLOB_APPEND(res_html, f_format('<body><a name="top"></a><h2>%1</h2>%2<br>', c_name_fio, calc_status));

  if (bls_id is null) then
  begin
    suspend;
    exit;
  end 

  html_list = '';
  cred_active_order_prv = -1;

  for select cred_id, cred_no, cred_partner_type, cred_partner_type_str, cred_type_str, cred_sum,
    cred_date, coalesce(cred_end_date_fact, cred_end_date), cred_sum_debt, cred_sum_overdue,
    cred_day_overdue, cred_max_overdue,
    cred_update, cred_active_str, cred_active_order, cred_active_order_str, cred_uid, error_txt
  from aarep_equifax_cred(:bls_id, 'equifax_40')
  order by cred_active_order, cred_end_date_fact desc, cred_date
  into cred_id, cred_no, cred_partner_type, cred_partner_type_str, cred_type_str, cred_sum,
    cred_date, cred_end_date, cred_sum_debt, cred_sum_overdue,
    cred_day_overdue, cred_max_overdue,
    cred_update, cred_active_str, cred_active_order, cred_active_order_str, cred_uid, error_txt
  do
  begin
    if (coalesce(error_txt, '') <> '') then
    begin
      res_html = BLOB_APPEND(res_html, f_format('<h1 style="color: red;">%1</h1></body>', error_txt));
      suspend;
      exit;
    end

    html_loan = '';

    if (cred_active_order = 1) then
    begin
      active_cnt = active_cnt + 1;
      if (cred_sum_overdue > 0) then
        overdue_cnt = overdue_cnt + 1;

      if (active_cnt > 100) then
        continue;
    end

    else if (cred_active_order = 2) then
    begin
      sell_cnt = sell_cnt + 1;

      if (active_cnt + sell_cnt > 100) then
        continue;
    end

    else if (cred_active_order = 3) then
    begin
      collect_cnt = collect_cnt + 1;

      if (active_cnt + sell_cnt + collect_cnt > 100) then
        continue;
    end

    else if (cred_active_order = 4) then
    begin
      closed_cnt = closed_cnt + 1;

      if (active_cnt + sell_cnt + collect_cnt > 0) then
      begin
        if (closed_cnt > 1) then
          continue;
      end
      else
        if (closed_cnt > 3) then
          continue;
    end

    -- Если кредит выдан МФО, проверяем, может этот кредит выдали мы
    kc_id = null;
    if (cred_partner_type = 3) then
    begin
      for select kc_id, k_kontract_no
      from knclient, kontstate, kontract
      where kc_from_c_id = :c_id and
            kc_amount_approved = :cred_sum and
            kc_from_k_id = ks_from_k_id and
            kc_from_k_id = k_id and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin = :cred_date and
            f_uuid_cbr(f_uuid('knclient', kc_id)) = :cred_uid
      rows 1
      into kc_id, k_kontract_no
      do
      begin
        cred_partner_type = -1;
        cred_partner_type_str = 'М Булак';
      end
    end

    -- Заголовок группы: Активные, Закрытые, Проданные ...
    if (cred_active_order <> cred_active_order_prv) then
    begin
      html_loan = html_loan ||
        f_format('<a name ="%1"><h2>%1<h2></a>', f_lang(lang, cred_active_order_str));
      cred_active_order_prv = cred_active_order;
    end

    -- Заголовок таблицы
    html_loan = html_loan || f_format(
      '<table width="100%"><col width="60%"><col width="40%"><th>%1</th><th>%2</th>',
      f_format('№ %1, ID дог: %2', cred_no, coalesce(cred_uid, cred_id)), cred_active_str);

    html_loan = html_loan || f_format(row_fmt,
      f_lang(lang, 'Источник информации'), cred_partner_type_str);

    if (kc_id is null) then
      html_loan = html_loan || f_format(row_fmt, f_lang(lang, 'Тип кредита'), cred_type_str);
    else
    begin
      html_loan = html_loan || f_format(row_fmt, f_lang(lang, '№ договора'), k_kontract_no);
    end

    html_loan = html_loan || f_format(row_fmt,
      f_lang(lang, 'Сумма выдачи'), f_numtostr(cred_sum, 2));

    html_loan = html_loan || f_format(row_fmt,
      f_lang(lang, 'Дата выдачи'),
      f_datetostr(cred_date));

    html_loan = html_loan || f_format(row_fmt,
      f_lang(lang, 'Дата погашения'),
      f_datetostr(cred_end_date));

    if (cred_active_order <> 2) then
    begin
      -- По закрытым кредитам эти поля не нужны
      html_loan = html_loan || f_format(row_fmt,
        f_lang(lang, 'Текущая задолженность'),
        f_numtostr(cred_sum_debt, 2));
  
      html_loan = html_loan || f_format(row_fmt_fg,
        f_lang(lang, 'Текущая просроченная задолженность'),
        f_numtostr(cred_sum_overdue, 2),
        iif(cred_sum_overdue = 0, '#008300', '#FF0000'));
  
      html_loan = html_loan || f_format(row_fmt_fg,
        f_lang(lang, 'Текущая просроченная задолженность дней'),
        cred_day_overdue,
        iif(cred_day_overdue = 0, '#008300', '#FF0000'));
    end

    html_loan = html_loan || f_format(row_fmt_fg,
      f_lang(lang, 'Максимальная сумма просроченной задолженности'),
      f_numtostr(cred_max_overdue, 2),
      iif(cred_max_overdue = 0, '#008300', '#FF0000'));

    html_loan = html_loan || f_format(row_fmt,
      f_lang(lang, 'Дата актуальности информации'),
      f_datetostr(cred_update));

    html_loan = html_loan ||
      f_format('</table><a href="#top">%1</a><br>', f_lang(lang, 'Вверх'));

    html_list = BLOB_APPEND(html_list, '<br>' || html_loan);
  end

  if ((cred_active_order_prv= -1) and (coalesce(bls_id, 0) > 0)) then
    res_html = BLOB_APPEND(res_html, f_lang(lang, 'У клиента отсутствует кредитная история.'));

  html_loan = '';
  if (active_cnt <> 0) then
    html_loan = html_loan || f_format('%1: %2 <font color="%3">(%4: %5)</font><br>',
                                        f_lang(lang, 'Активные договоры'),
                                        active_cnt,
                                        iif(overdue_cnt = 0, '#008300', '#FF0000'),
                                        f_lang(lang, 'просроченные'),
                                        overdue_cnt);

  if (sell_cnt <> 0) then
    html_loan = html_loan || f_format('%1: %2<br>', f_lang(lang, 'Проданные договоры'), sell_cnt);
  if (collect_cnt <> 0) then
    html_loan = html_loan || f_format('%1: %2<br>', f_lang(lang, 'Переданные коллекторам договоры'), collect_cnt);
  if (closed_cnt <> 0) then
    html_loan = html_loan || f_format('%1: %2<br>', f_lang(lang, 'Закрытые договоры'), closed_cnt);
  if(f_get_db_id() = 81001) then
    html_loan = html_loan || f_format('%1: %2<br>', 'Самозапрет', iif(f_is_selfblock(:c_id, current_date) = 1, 'да', 'нет'));

  res_html = BLOB_APPEND(res_html, html_loan, html_list);



  /********************** НБКИ *********************************/
  -- Счетчики для хранения количества кредитных историй по категориям
  active_cnt = 0;
  sell_cnt = 0;
  collect_cnt = 0;
  closed_cnt = 0;
  overdue_cnt = 0;
  bls_id = null;

  select eil_from_bls_id, eil_date
  from extinfolist, extinfocfg
  where eil_from_c_id = :c_id
    and eil_from_eic_id = eic_id
    and eic_code = 'nbki_40'
    and eil_is_json = 1
    and eil_status = 2
    and eil_date + eic_info_actual_days >= current_date
  order by eil_date desc, eil_id
  rows 1
  into bls_id, eil_date;

  if (bls_id is null) then
  begin
    suspend;
    exit;
  end 

  calc_status = f_format('<br>%1: %2', 'Кредитная история получена', f_datetostr(eil_date));
  res_html = BLOB_APPEND(res_html, f_format('<h1>Источник №2</h1><a name="top"></a>%1<br><br>', calc_status));

  html_list = '';
  cred_active_order_prv = -1;
  cred_max_overdue = 0;

  for select cred_no, cred_partner_type, cred_partner_type_str, cred_type_str, cred_sum,
    cred_date, coalesce(cred_end_date_fact, cred_end_date), cred_sum_debt, cred_sum_overdue,
    cred_day_overdue,
    cred_update, cred_active_str, cred_active_order, cred_active_order_str, cred_uid
  from aarep_ko_nbki_cred(:bls_id)
  order by cred_active_order, cred_end_date_fact desc, cred_date
  into cred_no, cred_partner_type, cred_partner_type_str, cred_type_str, cred_sum,
    cred_date, cred_end_date, cred_sum_debt, cred_sum_overdue,
    cred_day_overdue,
    cred_update, cred_active_str, cred_active_order, cred_active_order_str, cred_uid
  do
  begin
    html_loan = '';

    if (cred_max_overdue < cred_sum_overdue) then
      cred_max_overdue = cred_sum_overdue;

    if (cred_active_order = 1) then
    begin
      active_cnt = active_cnt + 1;
      if (cred_sum_overdue > 0) then
        overdue_cnt = overdue_cnt + 1;

      if (active_cnt > 100) then
        continue;
    end

    else if (cred_active_order = 2) then
    begin
      sell_cnt = sell_cnt + 1;

      if (active_cnt + sell_cnt > 100) then
        continue;
    end

    else if (cred_active_order = 3) then
    begin
      collect_cnt = collect_cnt + 1;

      if (active_cnt + sell_cnt + collect_cnt > 100) then
        continue;
    end

    else if (cred_active_order = 4) then
    begin
      closed_cnt = closed_cnt + 1;

      if (active_cnt + sell_cnt + collect_cnt > 0) then
      begin
        if (closed_cnt > 1) then
          continue;
      end
      else
        if (closed_cnt > 3) then
          continue;
    end

    -- Если кредит выдан МФО, проверяем, может этот кредит выдали мы
    kc_id = null;
    if (cred_partner_type = 3) then
    begin
      select kc_id, k_kontract_no
      from knclient, kontstate, kontract
      where kc_from_c_id = :c_id and
            kc_amount_approved = :cred_sum and
            kc_from_k_id = ks_from_k_id and
            kc_from_k_id = k_id and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin = :cred_date
      rows 1
      into kc_id, k_kontract_no;
      if (kc_id is not null) then
      begin
        if (f_uuid_cbr(f_uuid('knclient', kc_id)) = cred_uid) then
        begin
          cred_partner_type = -1;
          cred_partner_type_str = 'М Булак';
        end
        else kc_id = null;
      end
    end

    -- Заголовок группы: Активные, Закрытые, Проданные ...
    if (cred_active_order <> cred_active_order_prv) then
    begin
      html_loan = html_loan ||
        f_format('<a name ="%1"><h2>%1<h2></a>', f_lang(lang, cred_active_order_str));
      cred_active_order_prv = cred_active_order;
    end

    -- Заголовок таблицы
    html_loan = html_loan || f_format(
      '<table width="100%"><col width="60%"><col width="40%"><th>%1</th><th>%2</th>',
      f_format('№ %1, ID дог: %2', cred_no, coalesce(cred_uid, cred_id)), cred_active_str);

    html_loan = html_loan || f_format(row_fmt,
      f_lang(lang, 'Источник информации'), cred_partner_type_str);

    if (kc_id is null) then
      html_loan = html_loan || f_format(row_fmt, f_lang(lang, 'Тип кредита'), cred_type_str);
    else
    begin
      html_loan = html_loan || f_format(row_fmt, f_lang(lang, '№ договора'), k_kontract_no);
    end

    html_loan = html_loan || f_format(row_fmt,
      f_lang(lang, 'Сумма выдачи'), f_numtostr(cred_sum, 2));

    html_loan = html_loan || f_format(row_fmt,
      f_lang(lang, 'Дата выдачи'),
      f_datetostr(cred_date));

    html_loan = html_loan || f_format(row_fmt,
      f_lang(lang, 'Дата погашения'),
      f_datetostr(cred_end_date));

    if (cred_active_order <> 2) then
    begin
      -- По закрытым кредитам эти поля не нужны
      html_loan = html_loan || f_format(row_fmt,
        f_lang(lang, 'Текущая задолженность'),
        f_numtostr(cred_sum_debt, 2));
  
      html_loan = html_loan || f_format(row_fmt_fg,
        f_lang(lang, 'Текущая просроченная задолженность'),
        f_numtostr(cred_sum_overdue, 2),
        iif(cred_sum_overdue = 0, '#008300', '#FF0000'));
  
      html_loan = html_loan || f_format(row_fmt_fg,
        f_lang(lang, 'Текущая просроченная задолженность дней'),
        cred_day_overdue,
        iif(cred_day_overdue = 0, '#008300', '#FF0000'));
    end

    html_loan = html_loan || f_format(row_fmt_fg,
      f_lang(lang, 'Максимальная сумма просроченной задолженности'),
      f_numtostr(cred_max_overdue, 2),
      iif(cred_max_overdue = 0, '#008300', '#FF0000'));

    html_loan = html_loan || f_format(row_fmt,
      f_lang(lang, 'Дата актуальности информации'),
      f_datetostr(cred_update));

    html_loan = html_loan ||
      f_format('</table><a href="#top">%1</a><br>', f_lang(lang, 'Вверх'));

    html_list = BLOB_APPEND(html_list, '<br>' || html_loan);
  end

  if ((cred_active_order_prv= -1) and (coalesce(bls_id, 0) > 0)) then
    res_html = BLOB_APPEND(res_html, f_lang(lang, 'У клиента отсутствует кредитная история.'));

  html_loan = '';
  if (active_cnt <> 0) then
    html_loan = html_loan || f_format('%1: %2 <font color="%3">(%4: %5)</font><br>',
                                        f_lang(lang, 'Активные договоры'),
                                        active_cnt,
                                        iif(overdue_cnt = 0, '#008300', '#FF0000'),
                                        f_lang(lang, 'просроченные'),
                                        overdue_cnt);

  if (sell_cnt <> 0) then
    html_loan = html_loan || f_format('%1: %2<br>', f_lang(lang, 'Проданные договоры'), sell_cnt);
  if (collect_cnt <> 0) then
    html_loan = html_loan || f_format('%1: %2<br>', f_lang(lang, 'Переданные коллекторам договоры'), collect_cnt);
  if (closed_cnt <> 0) then
    html_loan = html_loan || f_format('%1: %2<br>', f_lang(lang, 'Закрытые договоры'), closed_cnt);
  if(f_get_db_id() = 81001) then
    html_loan = html_loan || f_format('%1: %2<br>', 'Самозапрет', iif(f_is_selfblock(:c_id, current_date) = 1, 'да', 'нет'));

  res_html = BLOB_APPEND(res_html, html_loan, html_list, '</body>');
  suspend;
end
