CREATE PROCEDURE HP_CASHBOOKMAIN_SIGN(
    BOOK_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable edl_id d_id;
declare variable edl_status d_smallint;
declare variable eds_id d_id;
begin
  res_is_error = 0;
  res_text = 'Кассовый журнал подписан';

  select cbm_from_edl_id, edl_status
  from cashbookmain
       left join edoc_list on cbm_from_edl_id = edl_id
  where cbm_id = :book_id
  into edl_id, edl_status;

  if (edl_id is null) then
  begin
    res_is_error = 1;
    res_text = 'Электронный документ еще не сформирован';
  end
  else
  if (edl_status = 1) then
    res_text = 'Кассовый журнал уже подписан';
  else
  if (edl_status = -1) then
  begin
    res_is_error = 1;
    res_text = 'Кассовый журнал ранее аннулирован';
  end
  else
  begin
    for select eds_id
    from edoc_sign
    where eds_from_edl_id = :edl_id and
          eds_status = 0
    into eds_id
    do
    begin
      update edoc_msg
      set edm_status = 1
      where edm_from_eds_id = :eds_id and
            edm_status = 0;

      update edoc_sign
      set eds_status = 1,
          eds_enum_edoc_sign_kind = f_enum('edoc_sign_kind', 'finstart', 'id')
      where eds_id = :eds_id;
    end
  end

  suspend;
end
