CREATE PROCEDURE CB_KG_LO_2013_05(
    REP_REGION_ID INTEGER,
    REP_BONUS_ID INTEGER,
    REP_USER_ID ID_TYPE,
    REP_DATE_END DATE)
RETURNS (
    KC_BALANCE CURR_TYPE,
    DELINQ_BALANCE CURR_TYPE,
    PAR DECIMAL(5,5),
    AA_KOLICH_CL_DISB INTEGER,
    AA_AMOUNT_DISB CURR_TYPE,
    KOL_CLI_DISB INTEGER,
    KOL_CLI_CLOSED INTEGER,
    KOL_CLI_CLOS_WOFF INTEGER,
    KOL_CLI_CLOS_SB INTEGER,
    DISB_AVG CURR_TYPE,
    BONUS_ORIG CURR_TYPE,
    BONUS_TOTAL CURR_TYPE,
    REG_NAME VARCHAR(60),
    BRANCH_NAME VARCHAR(60),
    OKRUG_NAME VARCHAR(60),
    JL_NAME VARCHAR(100),
    PS_IS_ON_STAFF_STR VARCHAR(50),
    PS_IS_WORKING_STR VARCHAR(20),
    AA_CURR_KC_FIO VARCHAR(100),
    PERC_UDERJANIE DECIMAL(15,5),
    DISB_CL_UDERJ INTEGER,
    CLOSE_CL_UDERJ INTEGER,
    BONUS_KOEF CURR_TYPE,
    BONUS_SUM CURR_TYPE,
    BONUS_MEMO VARCHAR(255),
    BONUS_END CURR_TYPE,
    KM_ID ID_TYPE,
    KM_FIO VARCHAR(100),
    KM_MEMO VARCHAR(100),
    CURR_REG_ID INTEGER,
    PS_IS_WORKING SMALLINT)
AS
declare variable rep_date_begin date;
declare variable k_id id_type;
declare variable kal_date date;
declare variable ks_kc_id id_type;
declare variable cr_rate curr_type;
declare variable k_index_curr_code varchar(3);
declare variable k_index_curr_rate curr_type;
declare variable ok_os curr_type;
declare variable is_own smallint;
declare variable ps_is_on_staff smallint;
declare variable curr_kc_id decimal(18,0);
declare variable fact1 decimal(18,4);
declare variable fact2 decimal(18,4);
declare variable fact3 decimal(18,4);
declare variable tt decimal(18,4);
begin
  rep_date_begin = '01.' || extract(month from :rep_date_end) || '.' || extract(year from :rep_date_end); -- Начало периода
  cr_rate = null;
  select cr_rate
  from opercurr
  where cr_from_cu_id = '840' and
        :rep_date_end between cr_date and cr_date_end
  into cr_rate;
  if (cr_rate is null) then
    cr_rate = 1;
-- Вытаскивает всех КСников которые за отчетный период прозводили какое либо операцию или на конец есть активный портфель
  for with tabl1
      as (select distinct ks_region_id as curr_reg_id, ks_kc_id as curr_kc_id
          from kontstate, region_list(:rep_region_id, 1, 3, 1)
          where ks_ol_is_delete = 0 and
                ks_state in (1, 2) and
                cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
                ks_region_id = id_list
          union
          select distinct kal_new_reg_id, kal_new_kf_officer
          from kontstate, kontract, knaktlist, region_list(:rep_region_id, 1, 3, 1)
          where ks_ol_is_delete = 0 and
                ks_state = 1 and
                :rep_date_end between ks_date_begin and ks_date_end and
                ks_from_k_id = k_id and
                k_from_kf_id = kal_from_kf_id and
                knaktlist.is_delete = 0 and
                :rep_date_end between kal_date and kal_date_end and
                kal_new_reg_id = id_list)
      select distinct curr_reg_id, curr_kc_id
      from tabl1
      into curr_reg_id, curr_kc_id
  do
  begin
    kc_balance = 0;
    delinq_balance = 0;
--  Расчет ПАР кредитного служащего
    for select k_id, kal_date, -- дата передачи или дата создание папки
               ks_kc_id, -- выдавший КС
               k_index_curr_code, k_index_curr_rate
        from knaktlist, kontract, kontstate
        where knaktlist.is_delete = 0 and
              :rep_date_end between kal_date and kal_date_end and
              kal_new_reg_id = :curr_reg_id and
              kal_new_kf_officer = :curr_kc_id and
              kal_from_kf_id = k_from_kf_id and
              ks_from_k_id = k_id and
              ks_ol_is_delete = 0 and
              ks_state = 1 and
              :rep_date_end between ks_date_begin and ks_date_end
        into k_id, kal_date, ks_kc_id, k_index_curr_code, k_index_curr_rate
    do
    begin
--    Вычисляет Баланс и Пар
      ok_os = null;
      select sum(ok_os)
      from operkred
      where ok_ol_is_active = 1 and
            ok_ol_date <= :rep_date_end and
            ok_from_k_id = :k_id and
            ok_type in (4, 5, 8) and
            ok_from_c_id = 0
      into ok_os; -- Портфель задаваемого контракта k_id
      if (ok_os is null) then
        ok_os = 0;
      if (ok_os <> 0 and
          k_index_curr_code = '840') then
      begin
        if (k_index_curr_rate > cr_rate) then
          ok_os = ok_os * k_index_curr_rate;
        else
          ok_os = ok_os * cr_rate;
      end
      kc_balance = kc_balance + ok_os;
--    Если на момент передачи небыло ПАР>7 то этих кредитов тоже включает в пар
      is_own = 1;
      if (ks_kc_id <> curr_kc_id) then
        if (coalesce((select rep_delinq_days
                      from aarep_kont_delinq_status(:k_id, :kal_date)), 0) > 7) then
          is_own = 0;

      if (is_own = 1) then
        if (coalesce((select rep_delinq_days
                      from aarep_kont_delinq_status(:k_id, :rep_date_end)), 0) between 8 and 180) then
        begin
          delinq_balance = delinq_balance + ok_os;
        end
    end
    par = 0;
    if (kc_balance <> 0) then
      par = round(delinq_balance / kc_balance * 100, 5);
    aa_kolich_cl_disb = null;
    aa_amount_disb = null;
    kol_cli_disb = null;
    kol_cli_closed = null;
    kol_cli_clos_woff = null;
    kol_cli_clos_sb = null;

    disb_cl_uderj = null;
    close_cl_uderj = null;
    select count(
           case
             when ks_state = 1 then kc_from_c_id
           end),
-- кол-во выданных кредитов без учета параллельных и семья
           sum(
           case
             when ks_state = 1 then kc_amount_approved
           end),
-- сумма выдачи
           count(
           case -- кол-во выдачи кроме ниже 10тыс
             when (ks_state = 1 and kc_amount_approved >= 10000) then kc_from_c_id
           end),
           count(
           case -- кол-во закрытых кредитов
             when (ks_state = 2 and kc_amount_approved >= 10000) then kc_from_c_id
           end),
           sum(
           case -- кол-во закрытых со списаниями
             when (ks_state = 2 and kc_amount_approved >= 10000) then case
                                                                        when exists(select 1
                                                                                    from operkred
                                                                                    where ok_from_ol_id = ks_from_ol_id and
                                                                                          ok_from_k_id = ks_from_k_id and
                                                                                          ok_type = 5 and
                                                                                          ok_os + ok_perc + ok_fine <> 0 and
                                                                                          ok_from_c_id = 0 and
                                                                                          not exists(select 1
                                                                                                     from sb_akt, sb_list
                                                                                                     where sba_from_k_id = ks_from_k_id and
                                                                                                           sba_is_delete = 0 and
                                                                                                           sba_from_sbl_id <> 0 and
                                                                                                           sba_status = 2 and
                                                                                                           ks_date_begin between sba_date_begin and sba_date_end and
                                                                                                           sba_from_sbl_id = sbl_id and
                                                                                                           sbl_type = 1)) then 1
                                                                        else 0
                                                                      end
           end),
           sum(
           case -- кол-во закрытых переданные в СБ
             when (ks_state = 2 and kc_amount_approved >= 10000) then case
                                                                        when exists(select 1
                                                                                    from sb_akt, sb_list
                                                                                    where sba_from_k_id = ks_from_k_id and
                                                                                          sba_is_delete = 0 and
                                                                                          sba_from_sbl_id <> 0 and
                                                                                          sba_status = 2 and
                                                                                          ks_date_begin between sba_date_begin and sba_date_end and
                                                                                          sba_from_sbl_id = sbl_id and
                                                                                          sbl_type = 1) then 1
                                                                        else 0
                                                                      end
           end),
           count(
           case
             when (ks_state = 1 and kc_circle > 1) then kc_from_c_id
           end),
-- кол-во кл., получивших повторный основной кредит
           count(
           case
             when (ks_state = 2) then kc_from_c_id
           end) -- кол-во кл., полностью погасивших основной кредит
    from kontstate, kontract, knclient
    where ks_ol_is_delete = 0 and
          ks_state in (1, 2) and
          cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
          ks_from_k_id = k_id and
          ks_kc_id = :curr_kc_id and
          ks_region_id = :curr_reg_id and
          kc_from_k_id = ks_from_k_id and
          kc_amount_approved > 0 and
          not exists(select pr_id, pr_name
                     from product -- Исключает продукт семья и параллельные кредиты
                     where pr_id = k_from_pr_id and
                           (pr_name like '%сем_я%' or
                           pr_product_kind = 2))
    into aa_kolich_cl_disb, -- кол-во выданных клиентов, для вычисления средней суммы выдачи
         aa_amount_disb, kol_cli_disb, -- кол-во выданных клиентов у которых сумма >=10000
         kol_cli_closed, kol_cli_clos_woff, kol_cli_clos_sb, disb_cl_uderj, close_cl_uderj;

    if (aa_kolich_cl_disb is null) then
      aa_kolich_cl_disb = 0;
    if (aa_amount_disb is null) then
      aa_amount_disb = 0;

    if (kol_cli_disb is null) then
      kol_cli_disb = 0;
    if (kol_cli_closed is null) then
      kol_cli_closed = 0;
    if (kol_cli_clos_woff is null) then
      kol_cli_clos_woff = 0;
    if (kol_cli_clos_sb is null) then
      kol_cli_clos_sb = 0;

    if (disb_cl_uderj is null) then
      disb_cl_uderj = 0;
    if (close_cl_uderj is null) then
      close_cl_uderj = 0;
--  Вычисление средней суммы выдачи
    disb_avg = 0;
    if (aa_kolich_cl_disb <> 0) then
      disb_avg = round(aa_amount_disb / aa_kolich_cl_disb, 4);
    fact1 = 0;
    if (disb_avg < 36000) then
      fact1 = 0.2;
    fact2 = case
              when (par <= 1) then 0
              when (par > 1 and par <= 1.5) then 0.25
              when (par > 1.5 and par <= 2.0) then 0.50
              when (par > 2.0) then 1.0
            end;
    perc_uderjanie = 100;
    if (close_cl_uderj <> 0) then
      perc_uderjanie = cast(disb_cl_uderj as double precision) / close_cl_uderj * 100;

    fact3 = case
              when (perc_uderjanie < 75) then 0.2
              else 0
            end;

    bonus_orig = (kol_cli_closed - kol_cli_clos_woff - kol_cli_clos_sb) * 300 + (kol_cli_disb * 200);

    tt = (1 - fact1) * (1 - fact2) * (1 - fact3);
    bonus_total = bonus_orig * tt;

    select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, r3.reg_code || ' ' || r3.reg_name
    from region r1, region r2, region r3
    where r1.reg_unicode = :curr_reg_id and
          r1.parent = r2.reg_unicode and
          r2.parent = r3.reg_unicode
    into reg_name, branch_name, okrug_name;
    jl_name = null;
    ps_is_on_staff = null;
    ps_is_working = null;
    select jl_name, ps_is_on_staff, ps_is_working
    from hr_pstatus, joblist
    where ps_from_c_id = :curr_kc_id and
          :rep_date_end between ps_date_begin and ps_date_end and
          ps_from_jl_id = jl_id
    order by ps_is_working desc
    rows 1
    into jl_name, ps_is_on_staff, ps_is_working;
    if (ps_is_working is null) then
      ps_is_working = 0;
    if (ps_is_working = 0) then
      bonus_total = 0; -- если сотрудник уволен то не начислять бонус
--  вытаскивает данные по выговорам и бонус коэффициент
    select bonus_koef, bonus_sum, bonus_memo
    from hp_get_vigovor_koef(:curr_kc_id, :rep_date_end)
    into bonus_koef, bonus_sum, bonus_memo;

    bonus_end = bonus_total * bonus_koef / 100 - bonus_sum;
    if (bonus_end < 0) then
      bonus_end = 0;

--  Параметры кредитного менеджера
    km_id = null;
    km_fio = '';
    km_memo = '';

    if (ps_is_on_staff is not null) then
    begin
      if (ps_is_on_staff = 1) then
        ps_is_on_staff_str = 'Штатный';
      else
        ps_is_on_staff_str = 'Внештатный';
      if (ps_is_working = 0) then
      begin
        ps_is_on_staff_str = ps_is_on_staff_str || ', ' || 'уволен';
        ps_is_working_str = 'Уволен';
      end
      else
      if (ps_is_working = 1) then
        ps_is_working_str = 'Работает';
    end
    else
      ps_is_on_staff_str = 'Приказ не оформлен';
    aa_curr_kc_fio = null;
    select c_name_fio
    from clients
    where c_id = :curr_kc_id
    into aa_curr_kc_fio;

    suspend;
  end
end
