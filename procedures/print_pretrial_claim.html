CREATE PROCEDURE PRINT_PRETRIAL_CLAIM(
    OBJ_ID D_ID)
RETURNS (
    ISK_HEADER D_STR_LONG,
    ISK_DATE D_STR_20,
    ISK_AMOUNT D_STR_20,
    ISK_OS D_STR_20,
    ISK_PERC D_STR_20,
    K_NO D_STR_CODE,
    K_PERIOD D_SMALLINT,
    K_DATE_END D_DATE,
    KC_AMOUNT_APPROVED D_STR_20,
    PURPOSE D_STR_100,
    SLD_ID D_ID,
    SLD_OUTGOING_NUMBER D_STR_CODE)
AS
declare variable k_id d_id;
declare variable k_fact_disb_date d_date;
declare variable k_kontract_no d_str_20;
declare variable zad_os d_currency;
declare variable zad_perc d_currency;
declare variable c_name_fio d_str_100;
declare variable address_yur d_str_255;
declare variable c_birth_date d_date;
declare variable citizen d_str_255;
declare variable c_pass_no d_str_code;
declare variable c_pass_who d_str_255;
declare variable c_pass_date d_date;
declare variable lb d_str_10;
begin
  -- данные досудебной претензии (RFCRU-3114)
  -- входящий параметр obj_id - id запроса (заявки) судебного модуля

  lb = ascii_char(10); -- перенос строки (line break)

  for select trim(c_name_fio),
             c_birth_date,
             c_pass_no,
             c_pass_who,
             c_pass_date,
             rd_name,
             kc_from_k_id,
             round(kc_amount_approved, 2),
             slc_address
      from sud_loanclient
      inner join knclient on kc_id = slc_from_kc_id
      inner join clients on c_id = kc_from_c_id
      inner join refdata on rd_id = c_citizen
      where slc_from_smr_id = :obj_id
      into c_name_fio,
           c_birth_date,
           c_pass_no,
           c_pass_who,
           c_pass_date,
           citizen,
           k_id,
           kc_amount_approved,
           address_yur
  do
  begin
    isk_header = null;
    isk_date = null;
    isk_amount = null;
    isk_os = null;
    isk_perc = null;
    k_no = null;
    k_kontract_no = null;
    k_period = null;
    k_date_end = null;
    purpose = null;
    sld_id = 0;
    sld_outgoing_number = null;

    isk_header = c_name_fio || ',' || lb || f_datetostr(c_birth_date) || ' г.р., уроженец ' || citizen || ',' || lb;
    isk_header = isk_header || 'паспортные данные ' || c_pass_no || ' ' || c_pass_who || ' от ' || f_datetostr(c_pass_date) || ' г.,' || lb;
    isk_header = isk_header || 'Адрес: ' || address_yur;

    isk_date = f_datetostr(current_date) || ' г.';

    select sum(zad_os),
           sum(zad_perc)
    from aarep_make_zad_list(:k_id, current_date, 1)
    into zad_os,
         zad_perc;

    isk_os = cast(round(zad_os, 2) as d_str_20);
    isk_perc = cast(round(zad_perc, 2) as d_str_20);
    isk_amount = cast(round(zad_os + zad_perc, 2) as d_str_20);

    select k_kontract_no,
           k_fact_disb_date,
           k_period
    from kontract
    where k_id = :k_id
    into k_kontract_no,
         k_fact_disb_date,
         k_period;

    k_no = k_kontract_no || ' от ' || f_datetostr(k_fact_disb_date) || ' г.';
    purpose = k_kontract_no || ', ' || c_name_fio || '.';

    k_date_end = dateadd(month, k_period, k_fact_disb_date) - 1;

    select sld_id,
           sld_outgoing_number
    from sud_loandocs
    where sld_from_smd_id = :obj_id and
          sld_doc_kind = 10
    into sld_id,
         sld_outgoing_number; -- исходящий номер (RFCRU-4792)

    suspend;
  end
end
