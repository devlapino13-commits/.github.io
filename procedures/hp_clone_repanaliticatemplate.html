CREATE PROCEDURE HP_CLONE_REPANALITICATEMPLATE(
    RAT_ID D_INTEGER,
    NEW_USERNAME D_STR_CODE)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable un_id d_id;
declare variable new_rat_id d_integer;
declare variable orig_un_id d_id;
declare variable orig_un_name d_str_code;
begin
  res_text = '';

  if (trim(coalesce(new_username, '')) = '') then
    res_text = 'Пользователь не указан';

  if (res_text = '') then
  begin
    select un_id
    from username
    where un_name = :new_username
    into un_id;
  
    if (un_id is null) then
      res_text = 'Пользователь не найден: ' || new_username;
  end

  if (res_text = '') then
  begin
    select un_id, un_name
    from repanaliticatemplate, username
    where rat_from_un_id = un_id and
          rat_id = :rat_id
    into orig_un_id, orig_un_name;

    if (orig_un_id = un_id) then
      res_text = 'Укажите логин пользователя, с которым хотите поделиться шаблоном';
  end

  if (res_text = '') then
  begin
    insert into repanaliticatemplate (rat_from_un_id, rat_name, rat_memo, rat_rep_kind)
    select :un_id, rat_name || ' от ' || :orig_un_name, rat_memo, rat_rep_kind
    from repanaliticatemplate
    where rat_id = :rat_id
    returning rat_id
    into new_rat_id;

    insert into repanaliticaconfig (rac_from_rat_id, rac_id, rac_section, rac_object_name, rac_object_value)
    select :new_rat_id, rac_id, rac_section, rac_object_name, rac_object_value
    from repanaliticaconfig
    where rac_from_rat_id = :rat_id;
  end

  suspend;
end
