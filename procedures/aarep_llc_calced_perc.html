CREATE PROCEDURE AAREP_LLC_CALCED_PERC(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    AAREP_PERC_BEGIN D_CURRENCY,
    AAREP_PERC_CALCED D_CURRENCY,
    AAREP_PERC_PAYED D_CURRENCY,
    AAREP_PERC_WRITEOFF D_CURRENCY,
    AAREP_PERC_END D_CURRENCY,
    AAREP_BALANCE_END D_CURRENCY,
    AAREP_KONTRACT_NO D_STR_20,
    AAREP_IS_INDIV D_STR_10,
    AAREP_PRODUCT_NAME D_STR_CODE,
    REP_CURRENCY D_STR_LANG,
    OFFICE_NAME VARCHAR(60),
    BRANCH_NAME VARCHAR(60),
    REP_DELIQ_DAYS_KOLICH D_INTEGER,
    LLC_SUM_PERC D_CURRENCY,
    K_STOP_PERC D_DATE,
    AAREP_DELAYED_PERC_BEGIN D_CURRENCY,
    AAREP_DELAYED_PERC_END D_CURRENCY,
    IS_MBH BOOLEAN)
AS
declare variable def_curr d_str_lang;
declare variable aarep_k_id d_id;
declare variable k_index_curr_rate d_currency;
declare variable cr_rate d_currency;
begin
  select
    cm_default_currency
  from company
  into
    :def_curr;

  cr_rate = null;
  select
    cr_rate
  from opercurr
  where :rep_date_end between cr_date and cr_date_end
    and cr_from_cu_id = 840
  into
    :cr_rate;
  if (cr_rate is null) then
    cr_rate = 1;

  for
    select
      aarep_k_id,
      aarep_perc_begin,
      aarep_perc_calced,
      aarep_perc_payed,
      aarep_perc_writeoff,
      aarep_perc_end,
      aarep_balance_end,
      aarep_kontract_no,
      aarep_is_indiv,
      aarep_product_name,
      rep_currency,
      k_stop_perc,
      coalesce(aarep_delayed_perc_begin, 0),
      coalesce(aarep_delayed_perc_end, 0),
      is_mbh
    from aarep_calc_perc_list(:rep_region_id, :rep_date_begin, :rep_date_end, '---')
  into
    :aarep_k_id,
    :aarep_perc_begin,
    :aarep_perc_calced,
    :aarep_perc_payed,
    :aarep_perc_writeoff,
    :aarep_perc_end,
    :aarep_balance_end,
    :aarep_kontract_no,
    :aarep_is_indiv,
    :aarep_product_name,
    :rep_currency,
    :k_stop_perc,
    :aarep_delayed_perc_begin,
    :aarep_delayed_perc_end,
    :is_mbh
  do
  begin
    k_index_curr_rate = null;
    office_name = null;
    branch_name = null;

    select
      k_index_curr_rate,
      r1.reg_code || ' ' || r1.reg_name,
      r2.reg_code || ' ' || r2.reg_name
    from kontract
      inner join knaktlist
        on kal_from_kf_id = k_from_kf_id
          and :rep_date_end between kal_date and kal_date_end
          and is_delete = 0
      inner join region r1
        on r1.reg_unicode = kal_new_reg_id
      inner join region r2
        on r2.reg_unicode = r1.parent
    where k_id = :aarep_k_id
    into
      :k_index_curr_rate,
      :office_name,
      :branch_name;

    k_index_curr_rate = coalesce(k_index_curr_rate, 1);
    rep_deliq_days_kolich = coalesce((
      select rep_delinq_days
      from aarep_kont_delinq_status(:aarep_k_id, :rep_date_end)), 0);
    llc_sum_perc = 0;

    if (rep_currency <> def_curr) then
      if (k_index_curr_rate > cr_rate) then
      begin
        aarep_perc_end = aarep_perc_end * k_index_curr_rate;
        aarep_balance_end = aarep_balance_end * k_index_curr_rate;
      end
      else
      begin
        aarep_perc_end = aarep_perc_end * cr_rate;
        aarep_balance_end = aarep_balance_end * cr_rate;
      end

    if (rep_deliq_days_kolich >= 90 or coalesce(k_stop_perc, '31.12.9999') <= :rep_date_end) then
    begin
      llc_sum_perc = aarep_perc_end;
    end

    if (aarep_product_name containing 'МУРАБАХА' or
      aarep_product_name containing 'Иджара Мунтахийя Биттамлик') then
    begin
      if (aarep_perc_end > aarep_delayed_perc_end) then
        llc_sum_perc = (aarep_perc_end - aarep_delayed_perc_end) *
          iif(rep_deliq_days_kolich < 90,
            coalesce((
              select
                lkc_percent
              from llc_kont_cli
                inner join llc_main
                  on lm_id = lkc_from_lm_id
                    and lm_date_status = :rep_date_end
                    and lm_is_signed = 1
              where lkc_from_k_id = :aarep_k_id), 0), 100) / 100;
      else
        llc_sum_perc = 0;
    end

    suspend;
  end
end
