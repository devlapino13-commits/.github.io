CREATE PROCEDURE HP_BKI_SET_ERROR(
    PACKET_NAME D_STR_CODE NOT NULL,
    REC_NO D_INTEGER NOT NULL,
    ERROR_CODE D_STR_10 NOT NULL,
    ERROR_TEXT D_STR_LONG NOT NULL,
    REC_NO_3_2 D_INTEGER = null)
AS
declare variable packet_id d_id;
declare variable bev_id d_id;
declare variable bkc_id d_id;
declare variable action_code d_str_code;
declare variable jso d_integer;
declare variable bkc_add d_str_long;
declare variable rec_no_tmp d_integer;
declare variable f boolean;

declare procedure int_hp_save_err (bev_id d_id, rec_no d_integer, packet_name d_str_code, error_code d_str_10, error_text d_str_long)
as
declare variable mml_id d_id;
begin
  if (bev_id is null) then
    exception e_default 'record №' || rec_no || ' not found in packet "' || packet_name || '".';

  mml_id = f_memo_id(error_text);

  insert into bki_errors (bke_from_bev_id, bke_date, bke_error_code, bke_error_id)
  values (:bev_id, current_date, :error_code, :mml_id);
end
begin
  select bkp_id, f_enum('bki_packet_action', bkt_action, 'code')
  from bki_packet, bki_packet_type
  where bkp_code = :packet_name
    and bkt_id = bkp_from_bkt_id
  into packet_id, action_code;

  if (packet_id is null) then
    exception e_default 'packet "' || :packet_name || '" not found';

  if (action_code in ('reset', 'recreate') and rec_no_3_2 is not null) then
    rec_no = rec_no_3_2;

  if (action_code not in ('reset', 'recreate') or rec_no_3_2 is not null) then
  begin
    select bev_id
    from bki_loans, bki_clients, bki_events
    where bkl_from_bkp_id = :packet_id
      and bkc_from_bkl_id = bkl_id
      and bev_from_bkc_id = bkc_id
      and bev_rec_no      = :rec_no
    into bev_id;

    execute procedure int_hp_save_err(bev_id, rec_no, packet_name, error_code, error_text);
  end
  else
  begin
    f = false;

    for select distinct bkc_id, bkc_add
    from bki_loans, bki_clients, bki_events
    where bkl_from_bkp_id = :packet_id
      and bkc_from_bkl_id = bkl_id
      and bev_from_bkc_id = bkc_id
      and bev_event_id is not null
    into bkc_id, bkc_add
    do
    begin
      jso = iif(coalesce(bkc_add, '') = '', json.obj_create(), json.parse(bkc_add));
      rec_no_tmp = json.obj_get_val(jso, 'orderNum');

      if (rec_no_tmp is null) then
        exception e_default 'Not found orderNum in bkc_id = ' || coalesce(bkc_id, '<NULL>');

      if (rec_no_tmp <> rec_no) then
        continue;

      f = true;

      for select bev_id
      from bki_events
      where bev_from_bkc_id = :bkc_id
      into bev_id
      do
        execute procedure int_hp_save_err(bev_id, rec_no, packet_name, error_code, error_text);
    end

    if (not f) then
      exception e_default 'Not found orderNum = ' || coalesce(rec_no, '<NULL>');
  end 
end
