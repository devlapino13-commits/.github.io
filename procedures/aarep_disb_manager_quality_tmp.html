CREATE PROCEDURE AAREP_DISB_MANAGER_QUALITY_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_MANAGER_C_ID DECIMAL(18,0),
    FOR_IS_BONUS SMALLINT = 0)
RETURNS (
    C_ID DECIMAL(18,0),
    RES_DISB_PERIOD INTEGER,
    RES_DELINQ1 INTEGER,
    RES_DELINQ2 INTEGER,
    RES_DELINQ3 INTEGER,
    RES_TOTAL1 INTEGER,
    RES_TOTAL2 INTEGER,
    RES_TOTAL3 INTEGER,
    K_ID DECIMAL(18,0),
    LO_EXPERIENCE DECIMAL(4,2),
    SH_DATE DATE,
    IS_NEW_CL D_INTEGER,
    KAL_NEW_REG_ID D_INTEGER)
AS
declare variable SH_ID integer;
declare variable DD_TMP date;
declare variable kc_from_c_id id_type;
begin
  if (rep_date_end is null) then
  begin
    rep_date_begin = dateadd(month, -3, current_date);
    rep_date_begin = rep_date_begin - extract(day from rep_date_begin) + 1;
    rep_date_end = current_date;
  end
-- Вытаскаиваем список выдач за периода
  if (rep_manager_c_id = 0) then
  begin
    for select sh_from_k_id, sh_id, sh_date, k_approved_manager, ks_lo_experience, kc_from_c_id, kal_new_reg_id
        from kontstate, kontract, schedule, knaktlist, region_list(:rep_region_id, 1, 3, 1), knclient
        where ks_state = 1 and
              ks_ol_is_delete = 0 and
              ks_date_begin <= :rep_date_end and
              ks_date_end >= :rep_date_begin and
              ks_from_k_id = k_id and
              sh_from_k_id = k_id and
              kc_from_k_id = k_id and
              sh_id between 1 and 3 and
              sh_date between :rep_date_begin and :rep_date_end and
              k_from_kf_id = kal_from_kf_id and
              kc_amount_approved > 0 and
              knaktlist.is_delete = 0 and
              :rep_date_end between kal_date and kal_date_end and
              kal_new_reg_id = id_list
        into :k_id, :sh_id, :sh_date, :c_id, lo_experience, kc_from_c_id, kal_new_reg_id
    do
    begin
      res_disb_period = extract(year from sh_date) * 100 + extract(month from sh_date);
      res_total1 = 0;
      res_total2 = 0;
      res_total3 = 0;
      res_delinq1 = 0;
      res_delinq2 = 0;
      res_delinq3 = 0;

      if (sh_id = 1) then
        res_total1 = 1;
      if (sh_id = 2) then
        res_total2 = 1;
      if (sh_id = 3) then
        res_total3 = 1;

      if ((select count(*)
           from knclient
           where kc_from_c_id = :kc_from_c_id and
                 kc_date_disb is not null) > 1) then
        is_new_cl = 0;
      else
        is_new_cl = 1;

      dd_tmp = minvalue(sh_date + 7, minvalue(rep_date_end + 7, current_date));
--      dd_tmp=minvalue(sh_date+7, rep_date_end);
      if (exists(select *
                 from oksummary
                 where oks_date_begin <= :dd_tmp and
                       oks_date_end >= :dd_tmp and
                       oks_delinq_date < :dd_tmp and
                       oks_from_k_id = :k_id and
                       oks_close_state = 0 and
                       oks_balance > 0)) then
      begin
        if (dd_tmp = sh_date + 7) then
        begin
          if (sh_id = 1) then
            res_delinq1 = 1;
          if (sh_id = 2) then
            res_delinq2 = 1;
          if (sh_id = 3) then
            res_delinq3 = 1;
        end
        else
        begin
          if (sh_id = 1) then
            res_total1 = 0;
          if (sh_id = 2) then
            res_total2 = 0;
          if (sh_id = 3) then
            res_total3 = 0;
        end
      end
      suspend;
    end
  end
  else
  begin
    for select sh_from_k_id, sh_id, sh_date, k_approved_manager, ks_lo_experience
        from kontstate, kontract, schedule, region_list(:rep_region_id, 1, 3, 1)
        where ks_state = 1 and
              ks_ol_is_delete = 0 and
              ks_date_begin <= :rep_date_end and
              ks_date_end >= :rep_date_begin and
              ks_from_k_id = k_id and
              ks_region_id = id_list and
              (k_approved_manager = :rep_manager_c_id or
              (:for_is_bonus = 1 and
              k_disb_date >= '01.12.2018' and
              exists(select jl_name, jk_system_code
                     from hr_pstatus, joblist, jobkind
                     where ps_from_c_id = k_approved_manager and
                           ps_from_reg_id = :rep_region_id and
                           :rep_date_end between ps_date_begin and ps_date_end and
                           ps_is_working = 1 and
                           ps_status = 1 and
                           ps_from_jl_id = jl_id and
                           jl_kind = jk_id and
                           jk_system_code = 'JobSystem.LO_Standart'))) and
              sh_from_k_id = k_id and
              sh_id between 1 and 3 and
              sh_date between :rep_date_begin and :rep_date_end
        into :k_id, :sh_id, :sh_date, :c_id, lo_experience
    do
    begin
      if (for_is_bonus = 1 and
          c_id <> rep_manager_c_id) then
        c_id = rep_manager_c_id;
      res_disb_period = extract(year from sh_date) * 100 + extract(month from sh_date);
      res_total1 = 0;
      res_total2 = 0;
      res_total3 = 0;
      res_delinq1 = 0;
      res_delinq2 = 0;
      res_delinq3 = 0;

      if (sh_id = 1) then
        res_total1 = 1;
      if (sh_id = 2) then
        res_total2 = 1;
      if (sh_id = 3) then
        res_total3 = 1;

      dd_tmp = minvalue(sh_date + 7, minvalue(rep_date_end + 7, current_date));
--      dd_tmp=minvalue(sh_date+7, rep_date_end);
      if (exists(select *
                 from oksummary
                 where oks_date_begin <= :dd_tmp and
                       oks_date_end >= :dd_tmp and
                       oks_delinq_date < :dd_tmp and
                       oks_from_k_id = :k_id and
                       oks_close_state = 0 and
                       oks_balance > 0)) then
      begin
        if (dd_tmp = sh_date + 7) then
        begin
          if (sh_id = 1) then
            res_delinq1 = 1;
          if (sh_id = 2) then
            res_delinq2 = 1;
          if (sh_id = 3) then
            res_delinq3 = 1;
        end
        else
        begin
          if (sh_id = 1) then
            res_total1 = 0;
          if (sh_id = 2) then
            res_total2 = 0;
          if (sh_id = 3) then
            res_total3 = 0;
        end
      end
      suspend;
    end
  end
end
