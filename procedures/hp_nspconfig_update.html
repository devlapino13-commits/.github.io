CREATE PROCEDURE HP_NSPCONFIG_UPDATE(
    NSM_ID INTEGER)
AS
declare variable dd_last date;
declare variable nsd_id integer;
declare variable nsd_date_begin date;
declare variable nsd_date_end date;
begin
  if (nsm_id <> 0) then
  begin
    dd_last = '31.12.9999';
    for select nsd_id, nsd_date_begin, nsd_date_end
        from nspconfig_data
        where nsd_from_nsm_id = :nsm_id
        order by nsd_date_begin desc, nsd_id desc
        into :nsd_id, :nsd_date_begin, :nsd_date_end
--      as cursor cur
    do
    begin
      if ((nsd_date_end <> dd_last) or (nsd_date_end is null)) then
      begin
        update nspconfig_data
        set nsd_date_end = :dd_last
        where nsd_id = :nsd_id;
--          where current of cur;
      end
      dd_last = nsd_date_begin - 1;
    end
  end
end
