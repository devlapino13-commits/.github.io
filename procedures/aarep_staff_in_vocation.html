CREATE PROCEDURE AAREP_STAFF_IN_VOCATION(
    REP_REGION_ID INTEGER)
RETURNS (
    RES_C_ID DECIMAL(18,0),
    RES_C_NAME_FIO VARCHAR(100),
    RES_REGION_NAME VARCHAR(100),
    RES_JL_NAME VARCHAR(100),
    HRV_FROM_HO_ID DECIMAL(18,0),
    HRV_NUMBER INTEGER,
    HRV_DATE_BEGIN DATE,
    HRV_DATE_END_FACT DATE,
    HRV_DATE_END DATE,
    HRV_CHANGER_ID DECIMAL(18,0),
    HRV_CHANGER_FIO VARCHAR(100),
    SAWT_NAME VARCHAR(100),
    HO_ORDER_NO VARCHAR(20))
AS
begin
  for select ps_from_c_id, c_name_fio, reg_code || ' ' || reg_name, jl_name
      from hr_pstatus, clients, region, joblist, region_list(:rep_region_id, 1, 3, 1)
      where ps_from_c_id = c_id and
            ps_from_reg_id = reg_unicode and
            ps_from_jl_id = jl_id and
            ps_date_end >= '31.12.2999' and
            ps_is_working = 1 and
            ps_status = 1 and
            ps_from_reg_id = id_list
      into :res_c_id, :res_c_name_fio, :res_region_name, res_jl_name
  do
  begin
    for select hrv_from_ho_id, hrv_number, hrv_date_begin, hrv_date_end, hrv_date_end_fact, hrv_changer_id, sawt_name,
               ho_order_no
        from hr_voctravel, hr_orders, sa_worktype
        where hrv_from_c_id = :res_c_id and
              hrv_status = 1 and
              hrv_date_begin <= current_date and
              hrv_date_end >= current_date and
              ho_order_type <> 20 -- Не включать отзывы
              and
              hrv_from_ho_id = ho_id and
              ho_order_type = sawt_id
        into :hrv_from_ho_id, :hrv_number, :hrv_date_begin, :hrv_date_end, :hrv_date_end_fact, :hrv_changer_id,
             sawt_name, ho_order_no
    do
    begin
      hrv_changer_fio = '';
      if (hrv_changer_id = 0) then
        hrv_changer_fio = 'Заменяющий не требуется';
      else
      begin
        select c_name_fio
        from clients
        where c_id = :hrv_changer_id
        into :hrv_changer_fio;
      end
      suspend;
    end
  end
end
