declare variable k_id bigint;

declare variable oks_balance d_currency;
declare variable oks_bal_perc d_currency;
declare variable oks_bal_zad_os d_currency;

    --  where line_no>=:obeg and line_no<=:oend

  do
  begin
  oks_balance_01 = 0;

  oks_balance_03 = 0;

  oks_bal_perc_02 = 0;

  oks_pmt_perc_02 = 0;
  oks_pmt_perc_03 = 0;
  oks_rvpz_01 = 0;
  oks_rvpz_02 = 0;
  oks_rvpz_03 = 0;
  oks_ecl_vbs_03 = 0;

  period_no = 0;
  while (period_no < max_period) do
  begin
    period_no = period_no + 1;
    rep_date_end = dateadd(month, period_no, rep_date_begin) - 1;

    select oks_balance, oks_bal_perc, oks_bal_zad_os, oks_bal_zad_perc, oks_delinq_days,
           oks_pmt_os, oks_pmt_perc, oks_llr_os_nadzor, oks_llr_perc_nadzor, oks_ecl_vbs
    from aarep_centerbank_rep_data_2025(:rep_date_end, :rep_query_date, :k_id)
    into :oks_balance, :oks_bal_perc, :oks_bal_zad_os, :oks_bal_zad_perc, :oks_delinq_days,
         :oks_pmt_os, :oks_pmt_perc, :oks_llr_os_nadzor, :oks_llr_perc_nadzor, :oks_ecl_vbs;

  if (period_no = 1) then
      oks_balance_01 = oks_balance;
    if (period_no = 2) then
      oks_balance_02 = oks_balance;
    if (period_no = 3) then
      oks_balance_03 = oks_balance;
   if (period_no = 1) then
      oks_bal_perc_01 = oks_bal_perc;
    if (period_no = 2) then
      oks_bal_perc_02 = oks_bal_perc;
    if (period_no = 3) then
      oks_bal_perc_03 = oks_bal_perc;
    if (period_no = 1) then
      oks_bal_zad_os_01 = oks_bal_zad_os;
    if (period_no = 2) then

    if (period_no = 2) then
      oks_bal_zad_perc_02 = oks_bal_zad_perc;
    if (period_no = 3) then
      oks_bal_zad_perc_03 = oks_bal_zad_perc;
    if (period_no = 1) then
      oks_delinq_days_01 = oks_delinq_days;
    if (period_no = 2) then
      oks_delinq_days_02 = oks_delinq_days;
    if (period_no = 3) then
      oks_delinq_days_03 = oks_delinq_days;
    if (period_no = 1) then
      oks_pmt_os_01 = oks_pmt_os;
    if (period_no = 2) then
      oks_pmt_os_02 = oks_pmt_os;
    if (period_no = 3) then
      oks_pmt_os_03 = oks_pmt_os;
    if (period_no = 1) then
      oks_pmt_perc_01 = oks_pmt_perc;
    if (period_no = 2) then
      oks_pmt_perc_02 = oks_pmt_perc;
    if (period_no = 3) then
      oks_pmt_perc_03 = oks_pmt_perc;
    -------------Сформированный РВПЗ (суммарно по ОД, процентам и иным платежам) на дату, руб.
    if (period_no = 1) then
      oks_rvpz_01 = oks_llr_os_nadzor + oks_llr_perc_nadzor;

      oks_rvpz_02 = oks_llr_os_nadzor + oks_llr_perc_nadzor;
