CREATE PROCEDURE HP_EDOC_BLANK_SEND_CHECK(
    EDB_ID D_ID,
    REP_REGION_ID D_INTEGER,
    UN_ID D_ID)
AS
declare variable reg_id d_integer;
declare variable c_id d_id;
declare variable can_send d_smallint;
declare variable eds_status d_smallint;
begin
  can_send = 0;
  if (exists(select *
             from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.SendRequestWithoutDoc')
             where has_access = 1)) then
    can_send = 1;

  if (coalesce(edb_id, 0) <> 0) then
  begin
    if (f_get_db_id() = 81001) then
    begin
      eds_status = 0;
      c_id = 0;

      select eds.eds_from_c_id, eds.eds_status
      from edoc_blank edb, edoc_list edl, edoc_sign eds
      where edl.edl_id = edb.edb_from_edl_id and
            edl.edl_id = eds.eds_from_edl_id and
            eds.eds_from_c_id > 0 and
            edb.edb_id = :edb_id
      into :c_id, eds_status;

      if (eds_status = 0) then
        exception e_message 'Документ на изменение номера не подписан клиентом';

      if (eds_status = 1) then
      begin
        if (can_send = 0) then
        begin

          if (not exists(select *
                         from clientdocs cd
                         where cd.cds_from_c_id = :c_id and
                               cd.cds_from_pk_id = 355 and
                               current_date between cd.cds_doc_date_begin and cd.cds_doc_date_end and
                               cd.cds_is_delete = 0 and
                               cd.cds_verify_status in (-3, 1))) then
            exception e_message 'Закрепите фото клиента для смены номера';
        end
      end
    end
    update edoc_blank_scan
    set edbs_status = 1,
        edbs_status_un_id = :un_id
    where edbs_from_edb_id = :edb_id and
          edbs_from_bls_id is not null and
          edbs_status in (-1, 0);

  end
  else
  begin
    for select id_list
        from region_list(:rep_region_id, 1, 3, 1)
        into reg_id
    do
    begin
      for select edbp_from_edb_id
          from edoc_blank_pending, edoc_blank_scan
          where edbp_from_reg_id = :reg_id and
                edbp_from_edb_id = edbs_from_edb_id and
                edbs_from_bls_id is not null and
                edbs_status in (-1, 0)
          into edb_id
      do
      begin
        if (f_get_db_id() = 81001) then
        begin
          eds_status = 0;
          c_id = 0;
          select eds.eds_from_c_id, eds.eds_status
          from edoc_blank edb, edoc_list edl, edoc_sign eds
          where edl.edl_id = edb.edb_from_edl_id and
                edl.edl_id = eds.eds_from_edl_id and
                eds.eds_from_c_id > 0 and
                edb.edb_id = :edb_id
          into :c_id, eds_status;

          if (eds_status = 0) then
            exception e_message 'Документ на изменение номера не подписан клиентом';

          if (can_send = 0) then
          begin

            if (not exists(select *
                           from clientdocs cd
                           where cd.cds_from_c_id = :c_id and
                                 cd.cds_from_pk_id = 355 and
                                 cd.cds_is_delete = 0 and
                                 cd.cds_verify_status in (-3, 1) and
                                 current_date between cd.cds_doc_date_begin and cd.cds_doc_date_end)) then
              exception e_message 'Закрепите фото клиента для смены номера';
          end
        end
        update edoc_blank_scan
        set edbs_status = 1,
            edbs_status_un_id = :un_id
        where edbs_from_edb_id = :edb_id;
      end
    end
  end
end
