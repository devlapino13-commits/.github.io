CREATE PROCEDURE HP_MAKE_LOGINFO_TRIGGER(
    TABLE_NAME D_STR_CODE,
    LOG_FIELDS D_STR_LONG,
    STATUS_FIELD_NAME D_STR_CODE,
    IS_DELETE_FIELD_NAME D_STR_CODE,
    IS_ACTIVE_FIELD_NAME D_STR_CODE,
    UN_ID_FIELD_NAME D_STR_CODE,
    DONT_LOG_INSERT D_BOOL = 0)
RETURNS (
    TRIGGER_SCRIPT D_STR_8100)
AS
declare variable key_field d_str_code;
declare variable block_str d_str_8100;
declare variable exec_proc d_str_8100;
declare variable if_str d_str_8100;
declare variable field_name d_str_code;
declare variable proc_params d_str_8100;
declare variable json_type d_str_code;
declare variable res_field_type d_str_code;
begin
  -- Процедура готовит скрипт для создания триггера логирования
  -- После получения коды выполните скрипт через IBExpert

  -- Значение LOG_FIELDS:
  --   Пусто если не нужно логировать поля.
  --   Full - если полное логирование,
  --   иначе через запятую передать список полей

  log_fields = trim(coalesce(log_fields, ''));
  status_field_name = trim(coalesce(status_field_name, ''));
  is_delete_field_name = trim(coalesce(is_delete_field_name, ''));
  is_active_field_name = trim(coalesce(is_active_field_name, ''));
  un_id_field_name = trim(coalesce(un_id_field_name, ''));
  if (dont_log_insert is null) then
    dont_log_insert = 0;

  if (is_delete_field_name <> '' and is_active_field_name <> '') then
  begin
    trigger_script = f_format('-- В таблице не дожлно быть одновременно поля %1 и %2',
                              is_delete_field_name, is_active_field_name);
    suspend;
    exit;
  end

  if (log_fields = '' and
      status_field_name = '' and
      is_delete_field_name = '' and
      is_active_field_name = '' and
      un_id_field_name = '') then
  begin
    trigger_script = '-- Не указаны поля, нет условий для логирования';
    suspend;
    exit;
  end

  proc_params = 'select trigger_script from hp_make_loginfo_trigger(''' || table_name || ''', ''%1'', ''%2'', ''%3'', ''%4'', ''%5'', ' || dont_log_insert || ')';
  proc_params = f_format(proc_params, log_fields, status_field_name, is_delete_field_name, is_active_field_name, un_id_field_name);

  key_field = f_objkind(table_name, 'field');

  exec_proc = '';
  exec_proc = 'select lgi_id
    from hp_save_loginfo(''%1'', :id, :prv_id, :action_str, %2, null, %3)
    into lgi_id;';
  if (is_delete_field_name <> '') then
    exec_proc = 'select lgi_id
    from hp_save_loginfo(''%1'', :id, :prv_id, :action_str, %2, new.' || is_delete_field_name || ', %3)
    into lgi_id;';
  if (is_active_field_name <> '') then
    exec_proc = 'select lgi_id
    from hp_save_loginfo(''%1'', :id, :prv_id, :action_str, %2, 1 - new.' || is_active_field_name || ', %3)
    into lgi_id;';

  exec_proc = f_format(exec_proc, table_name,
                       iif(status_field_name <> '', 'new.' || status_field_name, 'null'),
                       iif(un_id_field_name <> '', 'new.' || un_id_field_name, 'null'));

  if (dont_log_insert = 0) then
  begin
    trigger_script = 'SET TERM ^ ;

create or alter trigger trg_%1_loginfo for %1
active after insert or update or delete position 999
AS
declare variable need_log d_bool;
declare variable action_str d_str_code;
declare variable prv_id d_id;
declare variable lgi_id d_id;
declare variable id d_id;
begin
  -- Код этого триггера сгеренирован через вызов процедуры:
  -- %2
  -- Если необходимо добавить или исключить поля логирования, получите новый скрипт через процедуру.

  need_log = 0;
  prv_id = null;

  if (inserting) then
  begin
    action_str = ''insert'';
    id = new.%3;

    %4

    exit;
  end

  if (deleting) then
  begin
    action_str = ''delete'';
    id = old.%3;
  end
';

    trigger_script = f_format(trigger_script, table_name, proc_params, key_field, exec_proc);
  end else
  begin
    trigger_script = 'SET TERM ^ ;

create or alter trigger trg_%1_loginfo for %1
active after update or delete position 999
AS
declare variable need_log d_bool;
declare variable action_str d_str_code;
declare variable prv_id d_id;
declare variable lgi_id d_id;
declare variable id d_id;
begin
  -- Код этого триггера сгеренирован через вызов процедуры:
  -- %2
  -- Если необходимо добавить или исключить поля логирования, получите новый скрипт через процедуру.

  need_log = 0;
  prv_id = null;

  if (deleting) then
  begin
    action_str = ''delete'';
    id = old.%3;
  end
';

    trigger_script = f_format(trigger_script, table_name, proc_params, key_field);
  end

  block_str = f_format(block_str, exec_proc);
  trigger_script = trigger_script || block_str;

  exec_proc = replace(exec_proc, 'new.', 'old.');

  block_str = '
  if (updating) then
  begin
    action_str = ''update'';
    id = new.%1;

    if (old.%1 is distinct from new.%1) then
    begin
      need_log = 1;
      prv_id = old.%1;
    end
';

  block_str = f_format(block_str, key_field);
  trigger_script = trigger_script || block_str;

  if_str = '
    if (need_log = 0 and old.%1 is distinct from new.%1) then
      need_log = 1;
';

  if (status_field_name <> '') then
  begin
    block_str = f_format(if_str, status_field_name);
    trigger_script = trigger_script || block_str;
  end

  if (is_delete_field_name <> '') then
  begin
    block_str = f_format(if_str, is_delete_field_name);
    trigger_script = trigger_script || block_str;
  end

  if (is_active_field_name <> '') then
  begin
    block_str = f_format(if_str, is_active_field_name);
    trigger_script = trigger_script || block_str;
  end

  if (log_fields <> '') then
  begin
    if (log_fields <> 'full') then
    begin
      for select lower(trim(res_str))
      from hp_split_str(:log_fields, ',')
      into :field_name
      do
      begin
        if (field_name = key_field or
            field_name = status_field_name or
            field_name = is_delete_field_name or
            field_name = is_active_field_name or
            field_name = un_id_field_name) then
          Continue;
           
        if (exists(select * from hp_uxfield_one(:table_name, :field_name)
                   where res_field_type in ('memo', 'blob'))) then
          Continue;

        block_str = f_format(if_str, field_name);
        trigger_script = trigger_script || block_str;
      end
    end else
    begin
      for select lower(trim(r.rdb$field_name))
      from rdb$fields f, rdb$relation_fields r
      where f.rdb$field_name = r.rdb$field_source and
            r.rdb$system_flag = 0 and
            r.rdb$relation_name = upper(:table_name)
      order by r.rdb$field_position
      into field_name
      do
      begin
        if (field_name = key_field or
            field_name = status_field_name or
            field_name = is_delete_field_name or
            field_name = is_active_field_name or
            field_name = un_id_field_name) then
          Continue;

        if (exists(select * from hp_uxfield_one(:table_name, :field_name)
                   where res_field_type in ('memo', 'blob'))) then
          Continue;

        block_str = f_format(if_str, field_name);
        trigger_script = trigger_script || block_str;
      end
    end
  end

  block_str = '  end

  if (deleting or need_log = 1) then
  begin
    %1
';

  block_str = f_format(block_str, exec_proc);
  trigger_script = trigger_script || block_str;


  if (log_fields <> '') then
  begin
    if (log_fields <> 'full') then
    begin
      for select lower(trim(res_str))
      from hp_split_str(:log_fields, ',')
      into :field_name
      do
      begin
        if (field_name = key_field or
            field_name = status_field_name or
            field_name = is_delete_field_name or
            field_name = is_active_field_name or
            field_name = un_id_field_name) then
          Continue;

        select res_json_pair_type, res_field_type
        from hp_uxfield_one(:table_name, :field_name)
        into json_type, res_field_type;

        if (res_field_type in ('memo', 'blob')) then
          Continue;

        if (json_type = 'str') then
        begin
          if_str = '
    if ((deleting and coalesce(old.%1, '''') <> '''') or
        (updating and old.%1 is distinct from new.%1)) then
      execute procedure hp_save_logdata(''%2'', ''%1'', :lgi_id, old.%1);
';
        end else
        begin
          if_str = '
    if ((deleting and old.%1 is not null) or
        (updating and old.%1 is distinct from new.%1)) then
      execute procedure hp_save_logdata(''%2'', ''%1'', :lgi_id, old.%1);
';
        end

        block_str = f_format(if_str, field_name, :table_name);
        trigger_script = trigger_script || block_str;
      end
    end else
    begin
      for select lower(trim(r.rdb$field_name))
      from rdb$fields f, rdb$relation_fields r
      where f.rdb$field_name = r.rdb$field_source and
            r.rdb$system_flag = 0 and
            r.rdb$relation_name = upper(:table_name)
      order by r.rdb$field_position
      into field_name
      do
      begin
        if (field_name = key_field or
            field_name = status_field_name or
            field_name = is_delete_field_name or
            field_name = is_active_field_name or
            field_name = un_id_field_name) then
          Continue;

        select res_json_pair_type, res_field_type
        from hp_uxfield_one(:table_name, :field_name)
        into json_type, res_field_type;

        if (res_field_type in ('memo', 'blob')) then
          Continue;

        if (json_type = 'str') then
        begin
          if_str = '
    if ((deleting and coalesce(old.%1, '''') <> '''') or
        (updating and old.%1 is distinct from new.%1)) then
      execute procedure hp_save_logdata(''%2'', ''%1'', :lgi_id, old.%1);
';
        end else
        begin
          if_str = '
    if ((deleting and old.%1 is not null) or
        (updating and old.%1 is distinct from new.%1)) then
      execute procedure hp_save_logdata(''%2'', ''%1'', :lgi_id, old.%1);
';
        end

        block_str = f_format(if_str, field_name, table_name);
        trigger_script = trigger_script || block_str;
      end
    end
  end

  block_str = '  end
end
^

SET TERM ; ^

reconnect;

execute procedure hp_uxfield_sync(''%1'');

';

  block_str = f_format(block_str, table_name);
  trigger_script = trigger_script || block_str;

  suspend;
end
