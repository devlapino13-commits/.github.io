CREATE PROCEDURE HP_CASH_CHECK_MEMBERS(
    REG_ID INTEGER,
    LIST_KIND INTEGER)
RETURNS (
    PERS_ID ID_TYPE,
    PERS_FIO STR_100)
AS
begin
  pers_id = -1;
  pers_fio = 'Выбрать через поиск...';
  suspend;

  if (f_get_db_id() in (39801, 41701)) then begin
    for
      select
        c_id,
        c_name_fio
      from hr_pstatus
        inner join clients
          on ps_from_c_id = c_id
      where
        ps_from_reg_id = :reg_id and
        -- Менеджер офиса | Кассир / оператор базы данных
        ps_from_jl_id = iif(:list_kind=0, 225, 36) and
        ps_date_end >= '31.12.2999' and
        ps_is_working = 1
      order by c_name_fio
    into
      :pers_id,
      :pers_fio
    do suspend;
  end

end
