CREATE PROCEDURE PRINT_EXA_SCHEDULE(
    EXA_ID D_INTEGER)
RETURNS (
    SH_ID VARCHAR(10),
    SH_DATE DATE,
    SH_OS DECIMAL(18,4),
    SH_PERC DECIMAL(18,4),
    SH_TOTAL DECIMAL(18,4),
    SH_KOMMIS DECIMAL(18,4),
    SH_PERC_TAX DECIMAL(18,4),
    SH_DISB_AMOUNT D_CURRENCY,
    SH_BALANCE D_CURRENCY,
    SH_TOTAL_WITH_NSP D_CURRENCY)
AS
declare variable LOAN_DISB_DATE D_DATE;
declare variable LOAN_PRODUCT_ID D_INTEGER;
declare variable LOAN_AMOUNT D_CURRENCY;
declare variable LOAN_PERCENT D_PERCENT;
declare variable LOAN_REPAY_KOLICH D_INTEGER;
declare variable LOAN_REPAY_DAY D_INTEGER;
declare variable K_NSP_PERCENT PERC_TYPE;
begin
  -- предварительный график для отправки клиенту
  select exa_disb_date, exa_from_pr_id, exa_amount, exa_period, exa_percent, exa_repay_day
  from expressapp
  where exa_id = :exa_id
  into :loan_disb_date, :loan_product_id, :loan_amount, :loan_repay_kolich, :loan_percent, :loan_repay_day;

  if (row_count = 0) then
    Exit;

  for select sh_id, sh_date, sh_os, sh_perc, sh_total, sh_kommis, sh_perc_tax, sh_disb_amount, sh_balance
  from hp_make_schedule_by_param(:loan_disb_date, :loan_product_id, :loan_amount, :loan_percent, 0, :loan_repay_kolich, :loan_repay_day)
  into :sh_id, :sh_date, :sh_os, :sh_perc, :sh_total, :sh_kommis, :sh_perc_tax, sh_disb_amount, sh_balance
  do
  begin
    if (exists(select *
             from nspconfig_data
             where nsd_from_nsm_id = 0 and
                   nsd_date_begin <= :loan_disb_date and
                   nsd_date_end >= :loan_disb_date)) then
    begin
      select nsd_percent
      from nspconfig_data
      where nsd_from_nsm_id = -1 -- нсп от процента
              and
      nsd_date_begin <= :loan_disb_date and
      nsd_date_end >= :loan_disb_date
      into :k_nsp_percent;
      if (k_nsp_percent is null) then
          k_nsp_percent = 0;
    end

    sh_perc_tax = (select res
                   from fhp_round_bank(((:sh_perc + :sh_kommis) * :k_nsp_percent / 100.0), 0));
    sh_total_with_nsp = sh_total + sh_perc_tax;

    suspend;
  end
end
