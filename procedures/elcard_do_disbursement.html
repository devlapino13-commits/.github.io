CREATE PROCEDURE ELCARD_DO_DISBURSEMENT(
    K_ID ID_TYPE)
AS
declare variable CARD_ID STR_100;
declare variable C_ID D_ID;
declare variable DISB_AMOUNT CURR_TYPE;
begin
  -- если имеется актианя задача "на проверку карты" или "переод на карту" или эта задача уже успешно обработкана,
  -- тогда перевыдача займа ни к чему не приведет.
  -- Если же после выдачи займа, Элкарт вернул ошибку и деньги не ушли, перевыдача займа заново сформирует задачу на переод денег и деньги попытаются уйти

  if (exists(select *
             from extsysmain em
             where em.esm_project = 'elcard_disb' and
                   em.esm_action = 'CardDisb' and
                   em.esm_from_k_id = :k_id and
                   em.esm_status in (0, 1))) then
    exit;

  if (exists(select *
             from extsysmain em
             where em.esm_project = 'elcard_disb' and
                   em.esm_action = 'CardCheck' and
                   em.esm_from_k_id = :k_id and
                   em.esm_status = 0)) then
    exit;

  -- если вдруг ранее уже была успешная проверка карты, обнуляю результат этой проверки,
  -- чтобы она не повлияла на формирвоание задач по текущей выдаче ------------>
  update extsysmain em
  set em.esm_status = -2
  where em.esm_project = 'elcard_disb' and
        em.esm_action = 'CardCheck' and
        em.esm_from_k_id = :k_id and
        em.esm_status = 1;
  ---------------------------- < -----------------------------------------------

  for select kncl.kc_from_c_id, kncl.kc_amount_approved
  from knclient kncl
  where kncl.kc_from_k_id = :k_id
  and kncl.kc_amount_approved > 0
  into :c_id, disb_amount
  do
  begin
      -------------------------------------------------<
      -- получаю последнюю активную карту клиента ----->
      card_id = (select token
               from hp_client_bank_card(:c_id));
    
      if (card_id is null) then
        exception e_default 'Card ID not exists';
      -----------------------------------------------<
    
      -- создаю задачу на проверку карты
      insert into extsysmain (esm_project, esm_action, esm_date, esm_object_kind, esm_object_id, esm_status,
                              esm_full_source, esm_amount, esm_from_k_id, esm_account)
      values ('elcard_disb', 'CardCheck', current_date, 0, :c_id, 0, '', :disb_amount, :k_id, :card_id);
   end
end
