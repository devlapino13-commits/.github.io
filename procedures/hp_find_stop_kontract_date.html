CREATE PROCEDURE HP_FIND_STOP_KONTRACT_DATE(
    K_ID DECIMAL(18,0))
RETURNS (
    DATE_STOP_KONTRACT D_DATE)
AS
declare variable ok_sh_date_max date;
declare variable k_stop_perc d_date;
declare variable delinq_day d_integer;
begin
  /*что бы подтягивалась дата расторжения, если же договор закончился раньше даты расторжения,
  то дата окончания договора должна подтягивать.*/

  select k.k_stop_perc
  from kontract k
  where k.k_id = :k_id
  into :k_stop_perc;

  if (k_stop_perc is not null) then
    date_stop_kontract = k_stop_perc;
  else
  begin
    select max(ok.ok_sh_date)
    from operkred ok
    where ok.ok_from_k_id = :k_id and
          ok.ok_type = 1 and
          ok.ok_ol_is_active = 1
    into :ok_sh_date_max;
    date_stop_kontract = ok_sh_date_max;

    if (ok_sh_date_max < current_date) then
    begin
      select coalesce(iif((cast(:ok_sh_date_max as date) - oks.oks_delinq_date < 0), 0, cast(:ok_sh_date_max as date) - oks.oks_delinq_date), 0) -- число дней активной просрочки
      from oksummary oks
      where :ok_sh_date_max between oks.oks_date_begin and oks.oks_date_end and
            oks.oks_from_k_id = :k_id and
            oks.oks_close_state = 0
      into :delinq_day;
      if (delinq_day > 91) then
        date_stop_kontract = null;
    end

    else
    begin
      date_stop_kontract = null;
    end
  end
  suspend;
end
