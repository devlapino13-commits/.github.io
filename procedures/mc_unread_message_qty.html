CREATE PROCEDURE MC_UNREAD_MESSAGE_QTY(
    MCL_ID D_ID)
RETURNS (
    MESSAGE_QTY D_INTEGER)
AS
declare variable mcl_message_qty d_integer;
declare variable c_message_qty d_integer;
begin
  -- Процедура отвечает за предостлаения числа новых (непрочтенных) сообщений пользователю МП
  if (coalesce(mcl_id, 0) <= 0) then
    exception e_default 'parametr "mcl_id" has invalid value';

  mcl_message_qty = null;
  c_message_qty = null;

  select mco.mco_message_qty
  from mc_message_counter mco
  where mco.mco_object_kind = f_objkind('mc_login', 'id')
  and mco.mco_object_id = :mcl_id
  into :mcl_message_qty;

  mcl_message_qty = coalesce(mcl_message_qty, 0);
  c_message_qty = coalesce(c_message_qty, 0);

  message_qty = 0;
  message_qty = mcl_message_qty + c_message_qty;

  suspend;
end
