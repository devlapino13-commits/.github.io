CREATE PROCEDURE HP_PANTERA_IGNORE_EXTRACT_DAYS_LIMIT(
    REP_DATE D_DATE)
AS
declare variable PI_ID D_INTEGER;
declare variable PROJECT_CODE D_STR_255;
declare variable DAYS_IN_IGNORE_LIST D_INTEGER;
declare variable REASON_CODE D_STR_255;
declare variable DO_EXTRACT D_BOOL;
begin
  for select pi.pi_id,
             pt.rd_code,
             cast(:rep_date as date) - pi.pi_date_begin,
             rd.rd_code
  from pantera_ignore pi
  inner join pantera_kontracts pak on pak.pant_id = pi.pi_from_pant_id
  inner join refdata rd on rd.rd_id = pi.pi_reason_rd_id
  inner join refdata pt on pt.rd_id = pak.pant_rd_project_type
  where pi.pi_date_begin <= :rep_date and
        pi.pi_date_end = '31.12.9999' and
        pi.pi_is_active = 1
  into :pi_id,
       :project_code,
       :days_in_ignore_list,
       :reason_code
  do
  begin
    do_extract = 0;

    /* -- закомментировал согласно задаче RFCRU-5482

    if ((project_code in ('oktell.projects.auto_informer', 'oktell.projects.ks_finance_2')) and (days_in_ignore_list >= 5)) then -- RFCRU-5394
        dp_extract = 1;

    if ((project_code not in ('oktell.projects.auto_informer', 'oktell.projects.ks_finance_2')) and  -- старая логика, относится ко всему что не является "Автоинформатором" и "КС Финанс 2"
       (days_in_ignore_list >= 3) and
       (reason_code in ('OKTELL.SirenaResults.CellNoNotServed',
                        'OKTELL.SirenaResults.CantCalled',
                        'OKTELL.SirenaResults.CellNoTempUnavailable',
                        'OKTELL.SirenaResults.CalledToOneCantact')))  then
        dp_extract = 1;
    */
    if (days_in_ignore_list >= 5) then do_extract = 1; -- RFCRU-5482

    if (do_extract = 1) then
    begin
        update pantera_ignore pi
        set pi.pi_date_end = :rep_date
        where pi.pi_id = :pi_id;        
    end
  end
end
