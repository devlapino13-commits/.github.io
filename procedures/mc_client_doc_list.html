CREATE PROCEDURE MC_CLIENT_DOC_LIST(
    MCL_ID D_ID)
RETURNS (
    CDS_ID D_ID,
    PASSKIND_ID D_INTEGER,
    PASSKIND_NAME D_STR_100,
    IS_MAIN_DOC D_SMALLINT,
    API_CODE D_STR_255,
    BLS_1 D_ID,
    BLS_2 D_ID,
    STATUS_CODE D_STR_CODE,
    TITLE D_STR_CODE,
    DESCRIPTION D_STR_255)
AS
declare variable db_id d_id;
declare variable c_id d_id;
declare variable passkind_id_old d_integer;
declare variable create_date d_date;
begin
  if (coalesce(mcl_id, 0) <= 0) then
    exception e_default 'parametr "mcl_id" has invalid value';

  select mcl.mcl_from_c_id
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into :c_id;

  db_id = f_get_db_id();
  passkind_id_old = 0;

  if (c_id is null) then
  begin
    for select
      pdt.pdt_is_main_doc,
      drm.drm_api_code,
      drm.drm_name,
      drl.drl_from_pk_id,
      drl.drl_from_bls_1,
      drl.drl_from_bls_2,
      cast(drl.drl_create_datetime as date)
    from docrecoglist drl
    inner join actionlist al on al.acl_id = drl.drl_from_acl_id
      and al.acl_status >= 0 -- учитываю только документы по которым статус проверки <> Отказ
    inner join passkind pask on pask.pk_id = drl.drl_from_pk_id
    inner join passdoctype pdt on pdt.pdt_id = pask.pk_from_pdt_id
    inner join docrecogmain drm on drm.drm_id = pask.pk_from_drm_id
      and drm.drm_show_in_client_mobile = 1
    where drl.drl_from_mcl_id = :mcl_id
      and ((coalesce(drl.drl_from_bls_1, 0) > 0) or (coalesce(drl.drl_from_bls_2, 0) > 0))
    order by drl.drl_id desc, drm.drm_name
    into
      :is_main_doc,
      :api_code,
      :passkind_name,
      :passkind_id,
      :bls_1,
      :bls_2,
      :create_date
    do
    begin
      title = '';
      status_code = '';
      description = '';

      if (passkind_id = passkind_id_old) then
        continue;

      -- Фото с места работы
      -- Фото с места работы из галереи
      -- Фото с места жительства
      if (db_id in (81001, 94401)
        and passkind_id in (79, 88, 299)
        and datediff(day, :create_date, current_date) > 90
      ) then
        continue; -- если с момента создания записи прошло 3 месяца и более

      select res_value
      from mc_doc_name_change(:passkind_id, :passkind_name)
      into :passkind_name;

      if (is_main_doc = 1) then
      begin
        title = 'Находится в обработке';
        status_code = 'in_process';
        description = 'Паспорт находится в обработке. Пожалуйста подождите.';
      end

      suspend;
      passkind_id_old = passkind_id;

    end
  end
  else
  begin
    for select
      null as cds_id,
      drl.drl_from_pk_id,
      pask.pk_name,
      pdt.pdt_is_main_doc,
      drm.drm_api_code,
      drl.drl_from_bls_1,
      drl.drl_from_bls_2,
      drl.drl_create_datetime
    from docrecoglist drl
    inner join actionlist al on al.acl_id = drl.drl_from_acl_id
      and al.acl_status >= 0 -- учитываю только документы по которым статус проверки <> Отказ
    inner join passkind pask on pask.pk_id = drl.drl_from_pk_id
    inner join passdoctype pdt on pdt.pdt_id = pask.pk_from_pdt_id
    inner join docrecogmain drm on drm.drm_id = pask.pk_from_drm_id
      and drm.drm_show_in_client_mobile = 1
    where drl.drl_from_c_id = :c_id
      and coalesce(drl.drl_from_mcl_id, 0) > 0
      and ((coalesce(drl.drl_from_bls_1, 0) > 0) or (coalesce(drl.drl_from_bls_2, 0) > 0))
    order by
      pdt.pdt_is_main_doc desc,
      pask.pk_name,
      pask.pk_id,
      drl.drl_create_datetime desc
    into
      :cds_id,
      :passkind_id,
      :passkind_name,
      :is_main_doc,
      :api_code,
      :bls_1,
      :bls_2,
      :create_date
    do
    begin
      title = '';
      status_code = '';
      description = '';

      if (passkind_id = passkind_id_old) then
        continue;

      -- Фото с места работы
      -- Фото с места работы из галереи
      -- Фото с места жительства
      if (db_id in (81001, 94401)
        and passkind_id in (79, 88, 299)
        and datediff(day, :create_date, current_date) > 90
      ) then
        continue; -- если с момента создания записи прошло 3 месяца и более

      select res_value
      from mc_doc_name_change(:passkind_id, :passkind_name)
      into :passkind_name;

      if (is_main_doc = 1) then
        select status_code, title, description
        from mc_verify_status(:mcl_id)
        into :status_code, :title, :description;

      suspend;
      passkind_id_old = passkind_id;

    end
  end
end
