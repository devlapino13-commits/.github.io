CREATE PROCEDURE CM_RU_0015_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare variable dopusk integer;
declare variable c_citizen integer;
declare variable kc_circle integer;
declare variable c_id id_type;
declare variable last_k_id id_type;
declare variable ks_date_end date;
declare variable max_delinq_day integer;
declare variable kol_delinq integer;
declare variable delinq_day integer;
begin
-- Разработка по приказу от 6го марта 2015года
-- каждого участника проверяет, что гражданин КР и не имеет в предыдущем цикле просрочку
   dopusk = 1;
   for
   select c_id, c_citizen, kc_circle
   from knclient, clients
   where kc_from_k_id = :k_id
   and kc_amount_approved > 0
   and kc_from_c_id = c_id
   into c_id, c_citizen, kc_circle
   do
   begin
     if (dopusk = 1) then
     begin
--   если не гражданин КР или новый клиент или не имеет действующую регср. карту и временную регистрацию то условие по приказ от 6 марта не будет срабатывать
       if (c_citizen <> 8 or kc_circle < 2
           or not exists(select * from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                        where CDS_FROM_C_ID = :c_id
                        and cds_from_pk_id = pk_id
                        and pk_from_pdt_id = pdt_id
                        and cds_is_delete = 0
                        and current_date between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END
                        and pdt_code in ('registration_fiz', -- Временная регистрация, физ.лицо
                                         'magration_card'))  -- Миграционная карта
       ) then
       begin
         dopusk = 0;
       end 
     end
--   если не имеются выше ограничение то проверяется следующее
     if (dopusk = 1) then
     begin
--     Проверка просрочки предыдущего цикла
--     вычисление предыдущего контракта
       last_k_id = null;
       select ks_from_k_id, ks_date_end
       from kontstate, knclient
       where ks_ol_is_delete = 0
       and ks_state = 1
       and ks_from_k_id = kc_from_k_id
       and kc_amount_approved > 0
       and kc_from_c_id = :c_id
       and ks_from_k_id <> :k_id
       and ks_date_begin < current_date
       order by ks_date_begin desc
       rows 1
       into last_k_id, ks_date_end;
       if (last_k_id is null) then
       begin
         res_error_is_critical = 1;
         res_error_text = 'У клиента цикл не правильно указан, прошу исправить!';
         suspend;
       end 
--     проверка предыдущего контракта
       if (last_k_id is not null) then
       begin
--       вычисление максимальное кол-во просроченных дней предыдущего цикла
         max_delinq_day = null;
         kol_delinq = null;
         select max(ok_ol_date - ok_sh_date), count(*)
         from operkred
         where ok_ol_is_active = 1
         and ok_from_c_id = 0
         and ok_ol_date > ok_sh_date
         and ok_from_k_id = :last_k_id
         and ok_type in (4,5,9) -- просроченные выплаты или списании
         and ok_os + ok_perc < 0
         into max_delinq_day, kol_delinq;
         if (max_delinq_day is null) then max_delinq_day = 0;
         if (kol_delinq is null) then kol_delinq = 0;
         if (kol_delinq > 1 or max_delinq_day > 14) then
         begin
           dopusk = 0;
         end 
--       если предыдущий контракт активный то проверяет на просрочку
         if (ks_date_end >= current_date and dopusk = 1) then
         begin
           delinq_day = 0;
           select current_date - oks_delinq_date
           from oksummary
           where oks_from_k_id = :last_k_id
           and oks_close_state = 0
           and current_date between oks_date_begin and oks_date_end
           and current_date > oks_delinq_date
           into delinq_day;
           if (delinq_day is null) then delinq_day = 0;
--         Если имеется просрочка то по приказу не будет работать
           if (delinq_day > 14) then dopusk = 0;
         end
       end
     end
   end
   if (dopusk = 0) then -- добавлено от 06.02.2015 Улан
   begin
     res_error_is_critical = 0;
     res_error_text = '1. Отсутствуют необходимые документы. Приказ от 07.02.2015 №04-02-15о/д, пункт 2.2';
     suspend;
   end
end
