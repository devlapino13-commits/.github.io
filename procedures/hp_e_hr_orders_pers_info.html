CREATE PROCEDURE HP_E_HR_ORDERS_PERS_INFO(
    IN_C_ID D_ID,
    IN_DATE D_DATE)
RETURNS (
    RES_ACTUAL_OFFICE D_STR_100,
    RES_ACTUAL_JOBLIST D_STR_255,
    RES_ACTUAL_REG_ID D_INTEGER,
    RES_AGREEMENT_NO D_STR_10,
    RES_FULL_INFO D_STR_LONG,
    PS_FROM_REG_ID D_INTEGER,
    PS_FROM_DEP_ID D_INTEGER,
    PS_FROM_JL_ID D_INTEGER,
    PS_IS_MOUNT D_SMALLINT,
    PS_TABEL_NO D_STR_10,
    PS_TRANSFER_OFFICE D_INTEGER,
    RES_WORK_KIND D_INTEGER)
AS
declare variable ps_id d_integer;
declare variable ho_id d_id;
declare variable dep_name d_str_100;
begin
  if (in_date is null) then
    in_date = current_date;

  res_actual_office = '';
  res_actual_joblist = '';
  res_full_info = '';

  if (in_c_id is null) then
  begin
    suspend;
    exit;
  end

  res_actual_joblist = 'Нет приказа на ' || f_datetostr(in_date);
  res_agreement_no = '';

  select ps_id, ps_from_ho_id, ps_from_reg_id, reg_code || ' ' || reg_name,
         trim(iif(ps_is_working = 0, 'Уволен: ', '')) || jl_name, dep_name,

         ps_from_reg_id, ps_from_dep_id, ps_from_jl_id, ps_is_mount, ps_tabel_no, ps_from_reg_id as ps_transfer_office,
         iif(ps_is_part = 1, 2, ps_is_on_staff) as res_work_kind
  from hr_pstatus, region, joblist, depart
  where ps_from_c_id = :in_c_id and
        ps_date_begin <= :in_date and
        ps_from_reg_id = reg_unicode and
        ps_from_jl_id = jl_id and
        ps_from_dep_id = dep_id and
        ps_status = 1
  order by ps_from_c_id desc, ps_date_begin desc, ps_date_end desc
  rows 1
  into ps_id, ho_id, res_actual_reg_id, res_actual_office, res_actual_joblist, dep_name,
       ps_from_reg_id, ps_from_dep_id, ps_from_jl_id, ps_is_mount, ps_tabel_no, ps_transfer_office, res_work_kind;

  res_full_info = res_actual_office || ', ' || res_actual_joblist || ', ' || dep_name;

  if (ps_id is not null) then
  begin
    select psa_agreement_no
    from hr_pstatus_agreement
    where psa_from_ho_id = :ho_id and
          psa_from_ps_id = :ps_id
    into res_agreement_no;
  end

  suspend;
end
