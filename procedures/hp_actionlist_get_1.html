                 'Документ:      <b>' || pk_name || '</b><br>' || 'Номер:         <b>' || coalesce(cds_doc_no, '') || '</b><br>' || 'Дата выдачи:   <b>' || iif(cds_doc_date_begin is not null, right('0' || extract(day from cds_doc_date_begin), 2) || '.' || right('0' || extract(month from cds_doc_date_begin), 2) || '.' || extract(year from cds_doc_date_begin), 'Не указан') || '</b><br>' || 'Срок действия: <b>' || iif(cds_doc_date_end is not null, right('0' || extract(day from cds_doc_date_end), 2) || '.' || right('0' || extract(month from cds_doc_date_end), 2) || '.' || extract(year from cds_doc_date_end), 'Не указан') || '</b><br>' || 'Место выдачи:  <b>' || coalesce(cds_doc_issued_by, 'Не указан') || '</b><br>' || 'Для компании:  <b>' || coalesce(cds_doc_for_company, 'Не указан') || '</b><br>' || 'Терриятория действия: <b>' || coalesce(cds_doc_territory, 'Не указан') || '</b><br>' || iif(coalesce(cds_memo_1, '') <> '', '<b>' || coalesce(cds_memo_1, '') || '</b><br>', '') || iif(coalesce(cds_memo_2, '') <> '', '<b>' || coalesce(cds_memo_2, '') || '</b><br>', '') || iif(coalesce(cds_memo_3, '') <> '', '<b>' || coalesce(cds_memo_3, '') || '</b><br>', '') || iif(coalesce(cds_memo_4, '') <> '', '<b>' || coalesce(cds_memo_4, '') || '</b><br>', '') || iif(coalesce(cds_memo_5, '') <> '', '<b>' || coalesce(cds_memo_5, '') || '</b><br>', ''),
                 cds_doc_blob_1, cds_doc_blob_2
          from clientdocs, passkind, passdoctype
          where cds_from_c_id = :rep_object_id and
                cds_from_pk_id <> :c_doc_type and
                (cds_doc_date_end is null or
                cds_doc_date_end >= current_date) and
                cds_is_delete = 0 and
                cds_from_pk_id = pk_id and
                pk_from_pdt_id = pdt_id and
                (pdt_code <> 'mkr_agreement' and
                pdt_code <> 'credit_history')
          into :pk_name, :res_object_detail_tmp, bls_id_1, bls_id_2
      do
      begin
        doc_count = doc_count + 1;
        res_object_detail = BLOB_APPEND(res_object_detail, '<br><br>', res_object_detail_tmp);
        if (coalesce(bls_id_1, 0) > 0) then
        begin
          res_blob_ids = res_blob_ids || :bls_id_1 || ';';
          res_object_blob_names = BLOB_APPEND(res_object_blob_names, '[' || :bls_id_1 || '-' || :pk_name || ']');
        end
        if (coalesce(bls_id_2, 0) > 0) then
        begin
          res_blob_ids = res_blob_ids || :bls_id_2 || ';';
          res_object_blob_names = BLOB_APPEND(res_object_blob_names, '[' || :bls_id_2 || '-' || :pk_name || ', стр.2]');
        end
      end
      if (doc_count = 0) then
        res_object_detail = BLOB_APPEND(res_object_detail, ' отсутствуют.<br>');
      if (res_blob_ids = '') then
        res_object_detail = BLOB_APPEND(res_object_detail, '<font color="red">Копии документов не закреплены</font><br>');
      suspend;
    end
    exit;
  end
