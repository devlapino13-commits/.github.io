CREATE PROCEDURE UXP$UXCODE_CLONE(
    UX_UUID D_UUID,
    NEW_SCODE D_STR_CODE,
    UX_UUID_CODE D_STR_CODE = null)
RETURNS (
    NEW_UX_UUID D_UUID)
AS
declare variable ux_scode d_str_100;
declare variable ux_name d_str_100;
declare variable us_uuid d_uuid;
begin
  if (coalesce(ux_uuid, '') = '') then
    ux_uuid = uxf$uniq(ux_uuid_code);

  insert into ux$uniq (uu_kind)
  select uu_kind
  from ux$uniq
  where uu_id = :ux_uuid
  returning uu_id
  into new_ux_uuid;

  for select ux_us_uuid, ux_scode, ux_name
  from ux$code
  where ux_uuid = :ux_uuid
  into us_uuid, ux_scode, ux_name
  do
  begin
    if (coalesce(new_scode, '') = '') then
      new_scode = ux_scode || '_copy';

    update or insert into ux$code (ux_us_uuid, ux_uuid, ux_scode, ux_name)
    values (:us_uuid, :new_ux_uuid, :new_scode, :ux_name)
    matching (ux_uuid);
  end

  suspend;
end
