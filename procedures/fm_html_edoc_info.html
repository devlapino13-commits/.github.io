CREATE PROCEDURE FM_HTML_EDOC_INFO(
    EDC_CODE D_STR_CODE,
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    CANCEL_INACTIVE D_SMALLINT,
    UN_ID D_ID)
RETURNS (
    RES_HTML D_STR_8100)
AS
declare variable eds_id d_id;
declare variable edl_id d_id;
declare variable eds_status d_smallint;
declare variable msg_c_id d_id;
declare variable msg_text d_str_8100;
declare variable link d_str_long;
declare variable lang d_str_lang;
declare variable edoc_state d_str_code;
declare variable c_name_fio d_str_100;
declare variable tmp_str d_str_100;
declare variable color_green d_str_code;
declare variable edc_name d_str_100;
declare variable db_id d_integer;
declare variable edl_blank_status d_smallint;
declare variable uri_text d_str_100;
declare variable uri_kind d_str_code;
declare variable uri_description d_str_100;
declare variable uri_add_params d_str_code;
declare variable edc_id d_integer;
declare variable msg_kind d_smallint;
declare variable temp_link d_str_long;
declare variable temp_html d_str_long;
declare variable spouse_name d_str_100;
declare variable spouse_c_id d_id;
declare variable spouse_edoc_state d_str_code;
declare variable c_id d_id;
declare variable self_prohibition_status d_integer;
declare variable spm_query_qty d_integer;
begin
  -- Блок для отображения информации об электронном документе
  -- Позволяет увидеть статус, отправить на подписание, в т.ч. повторно
  -- Текущая реализация обрабатывает заявку на кредит и договор:
  -- loan.application, loan.agreement
  -- Допустимо расширять для других документов

  res_html = '';
  link = '';

  if (f_edoc_state2(edc_code, current_date, un_id) = 'inactive') then
  begin
    suspend;
    exit;
  end

  color_green = '#02B316';
  db_id = f_get_db_id();
  lang = f_get_lang_code(un_id, null);

  select edc_id, edc_name
  from edoc_cfg
  where edc_code = :edc_code
  into edc_id, edc_name;

  if (db_id = 41701 and
      edc_code = 'loan.agreement' and
      f_get_const('self_prohibition') = 1) then   -- RFCKG-4711
  begin
    for select kc.kc_from_c_id
    from knclient kc
    where kc.kc_from_k_id = :obj_id and
          kc.kc_amount_approved > 0
    into :c_id do
    begin
      select res,
             spm_query_qty
      from hp_self_prohibition_get_status(:c_id, :obj_kind, :obj_id)
      into self_prohibition_status,
           spm_query_qty;

      self_prohibition_status = coalesce(self_prohibition_status, -777);
    end
  end

  edoc_state = f_edoc_status(edc_code, obj_id, current_date);

  if (edc_code = 'consulting.agreement') then
  begin
    select f_edoc_status(:edc_code, kc.kc_id, current_date)  -- RFCKG-4363
    from knclient kc
    where kc.kc_id = :obj_id
    into edoc_state;
  end

  select msg_kind
  from hp_edoc_get_msg_kind(:obj_kind, :obj_id, :edc_id)
  into msg_kind;

  if (edoc_state in ('empty', 'cancelled', 'signed')) then
  begin
    res_html = f_format('<h2>%1</h2><table width="100%"><col width="40%"><col width="60%">', edc_name);

    if (edoc_state in ('empty', 'cancelled')) then
    begin
      -- Ссылка на создание электронного документа
      uri_description = 'Подтвердите создание эл. документа и отправку СМС';
      uri_kind = null;
      uri_add_params = null;

      if (edc_code = 'loan.application') then
      begin
        uri_text = 'Отправить клиенту/ам';
        uri_kind = 'edoc_ins_loan_app';
      end
      else
      if (edc_code = 'consulting.agreement') then
      begin
        uri_text = 'Отправить клиенту/ам';
        uri_kind = 'edoc_ins_ca';
      end
      else
      if (edc_code = 'loan.specification') then
      begin
        uri_text = 'Отправить подрядчику';
        uri_kind = 'edoc_ins_spec';
      end
      else
      if (edc_code = 'mbulak_card.agreement') then
      begin
        uri_text = 'Отправить клиенту/ам';
        uri_kind = 'edoc_ins_mbc';
      end
      else
      if (edc_code = 'purchase_act.agreement') then
      begin
        uri_text = 'Отправить продавцу';
        uri_kind = 'edoc_ins_p_act';
      end
      else
      -- RFCKG-1918  АКТ приема передачи в договоре Истиснаа
      if (edc_code = 'act_from_builder.agreement') then
      begin
        uri_text = 'Отправить подрядчику';
        uri_kind = 'edoc_ins_act_builder';
      end
      else
      if (edc_code = 'act_from_company.agreement') then
      begin
        uri_text = 'Отправить клиенту';
        uri_kind = 'edoc_ins_act_company';
      end
      -- RFCKG-1918 конец
      else
      if (edc_code = 'ad.agreement') then
      begin
        uri_text = 'Отправить клиенту';
        uri_kind = 'edoc_ins_ad_agreement';
      end
      else
      if (edc_code = 'bki.agreement' and f_get_const('doc_optimize_pack2') <> 1) then
      begin
        uri_text = 'Отправить клиенту';
        uri_kind = 'edoc_ins_bki_agreement';
      end
      else
      if (edc_code = 'pdata.agreement' and f_get_const('doc_optimize_pack2') <> 1) then
      begin
        uri_text = 'Отправить клиенту';
        uri_kind = 'edoc_ins_pdata_agreement';
      end
      else
      if (edc_code = 'loan.agreement') then
      begin
        uri_text = 'Отправить клиенту/ам';
        uri_kind = 'edoc_ins_loan_agreement';
      end
      else
      if (edc_code in ('insurance.application', 'insurance.kyrgyz', 'insurance.kyrgyz.ipf')) then
      begin
        uri_text = 'Отправить клиенту/ам';
        uri_kind = 'edoc_ins_insurance_policy';
      end
      else
      if (edc_code = 'bki.agreement.zalog') then
      begin
        uri_text = 'Отправить залогодателю';
        uri_kind = 'edoc_ins_bki_agreement_zalog';
      end
      else
      if (edc_code = 'pdata.agreement.zalog') then
      begin
        uri_text = 'Отправить залогодателю';
        uri_kind = 'edoc_ins_pdata_agreement_zalog';
      end
      else
      if (edc_code = 'loan.collateral') then
      begin
        uri_text = 'Отправить залогодателю';
        uri_kind = 'edoc_ins_loan_collateral';
      end
      if (edc_code = 'loan.pdn') then
      begin
        uri_text = 'Отправить клиенту';
        uri_kind = 'edoc_ins_loan_pdn';
      end
      if (edc_code = 'idjara_goods.contragent_act') then
      begin
        uri_text = 'Отправить продавцу';
        uri_kind = 'edoc_idjara_goods_contragent';
      end
      if (edc_code = 'pers_data_consent') then    -- RFCKG-4210
      begin
        uri_text = 'Сформировать документ';
        uri_kind = 'edoc_ins_pers_data_consent';
      end
      if (edc_code = 'personaldata.consent') then  -- RFCKG-4573
      begin
        uri_text = 'Отправить клиенту/ам';
        uri_kind = 'edoc_ins_personaldata_consent';
      end
       if (edc_code = 'persdata.consent') then  -- RFCKG-4573
      begin
        uri_text = 'Отправить клиенту/ам';
        uri_kind = 'edoc_ins_persdata_consent';
      end
      --Для подписания при помощи селфи
      if (msg_kind in (4, 5)) then
      begin
        uri_text = 'Сформировать документ';
        uri_description = 'Подтвердите создание эл. документа';
      end
    end
    else -- edoc_state = 'signed'
    begin
      -- Аннулирование для подписанных документов

      -- В РФ через МП нельзя аннулировать документы.
      if (db_id = 81001) then
        cancel_inactive = 1;

      if (cancel_inactive = 0) then
      begin
        uri_text = 'Аннулировать';
        uri_kind = 'edoc_cancel';
        uri_description = 'Подтвердите аннулирование эл. документа';
        uri_add_params = edc_code;
      end
    end

    if (db_id = 41701 and
        edc_code = 'loan.agreement' and
        f_get_const('self_prohibition') = 1 and
        edoc_state in ('empty', 'cancelled')) then -- RFCKG-4711
    begin
      if(self_prohibition_status = -777 or self_prohibition_status = -3) then
      begin
        uri_kind = 'send_self_prohibition_query';
        uri_add_params = :c_id;
        uri_description = 'Вы подтверждаете отправку запроса?';
      end

      if (self_prohibition_status = -1) then
      begin
        if (spm_query_qty <= 1) then
        begin
          uri_text = 'Сформировать документ(1/2)';
          uri_kind = 'send_self_prohibition_query';
          uri_add_params = :c_id;
          uri_description = 'Вы подтверждаете отправку запроса?';
        end
        else
        begin
          uri_kind = null;
          link = '<span style="color: red;">Формирование документа невозможно</span>';
        end
      end

      if(self_prohibition_status = -2) then
      begin
        uri_kind = null;
        link = '<span style="color: red;">Формирование документа невозможно</span>';
      end

      if(self_prohibition_status = 0) then
      begin
        uri_kind = null;
        link = 'Документ формируется. Подождите 1 минуту';
      end

      if(self_prohibition_status = 1) then
      begin
        uri_kind = null;
        link = '<span style="color: red;">Формирование документа невозможно. У клиента действует самозапрет</span>';
      end
    end

    if (uri_kind is not null) then
      link = f_uri_sql(f_lang(lang, uri_text), uri_kind, obj_kind, obj_id, 'none', f_lang(lang, uri_description), '', null, null, null, null, 'dialog', uri_add_params);

    if (edoc_state = 'signed') then
      link = f_lang(lang, 'Подписан') || ' ' || link;

    res_html = res_html || f_html_table_row(f_lang(lang, 'Электронный документ'), link, '', '', -1, -1, 0, 0, 2, '');
  end

  if (edoc_state = 'unsigned') then
  begin
    -- Отображение списка участников, со ссылкой на повторную отправку
    res_html = f_format('<h2>%1</h2><table width="100%"><col width="40%"><col width="30%"><col width="30%">', edc_name);

    if (edc_code = 'consulting.agreement') then  -- RFCKG-4363
    begin
      select kc.kc_id, 'knclient'
      from knclient kc
      where kc.kc_id = :obj_id
      into obj_id, obj_kind;
    end

    select edl_id, edl_blank_status
    from edoc_list
    where edl_object_kind = f_objkind(:obj_kind, 'id') and
          edl_object_id = :obj_id and
          edl_status = 0 and
          edl_from_edc_id = :edc_id
    order by edl_id desc
    rows 1
    into edl_id, edl_blank_status;

    -- edl_id будет найден, т.к. статус "не подписан"
    if (edl_blank_status < 0) then
    begin
      msg_text = f_lang(lang, 'Ошибка при формировании эл. документа. Ожидайте 15 минут, затем обратитесь в тех. поддержку');
      res_html = res_html || '<tr><td colspan="3">' || msg_text || '</td></tr>';
    end
    else
    if (edl_blank_status = 0) then
    begin
      msg_text = f_lang(lang, 'Документ формируется. Подождите 1 минуту');
      res_html = res_html || '<tr><td colspan="3">' || msg_text || '</td></tr>';
    end
    else
    begin
      for select eds_id, eds_from_c_id, c_name_fio, eds_status
          from edoc_sign, clients
          where eds_from_edl_id = :edl_id and
                eds_from_c_id = c_id
          into eds_id, msg_c_id, c_name_fio, eds_status
      do
      begin
        link = '';

        if (db_id in (39801, 41701) and
            edc_code = 'loan.application') then
        begin
          if (exists(select *
                     from miniappclient mac
                     where mac.mac_from_ma_id = :obj_id and
                           mac.mac_from_c_id = :msg_c_id and
                           mac.mac_is_poruch = -1)) then
            continue;

        end

        if (eds_status = 0) then
        begin
          link = f_uri_sql(f_lang(lang, 'Отправить повторно'), 'edoc_msg_ins', obj_kind, edl_id, 'none', c_name_fio, f_lang(lang, 'Подтвердите повторную отправку СМС'), null, null, null, null, 'dialog', eds_id);

          --Показываем кнопку подписания через видеоидентифкацию только тем у кого есть доступ и если не было отправлено смс по текущему эл. док-ту
          --Если же смс было отправлено, то переменная link останется неизменённой с текстом "Отправить повторно"
          if (msg_kind = 4) then
            link = f_uri_open('Подписать клиенту\ам', 'edoc', 'client', msg_c_id, edc_name, 'eds_id=' || eds_id || '&save_photo=0', un_id);

          if (msg_kind = 5) then
            link = f_uri_open('Подписать клиенту\ам', 'edoc', 'client', msg_c_id, edc_name, 'eds_id=' || eds_id || '&save_photo=1', un_id);

          select msg_text
          from hp_edoc_draft_message(:edc_code, :obj_id, :msg_c_id)
          into :msg_text;
        end

        if (eds_status = 1) then
          tmp_str = f_format('<font color="%1"<b>%2', color_green, f_lang(lang, 'Подписан'));
        else
          tmp_str = f_lang(lang, iif(msg_kind in (4, 5), trim('Ожидание подписи через биометрию'), trim('СМС отправлен')));

      res_html = res_html || f_html_table_row(c_name_fio, tmp_str, link, '', -1, 0, 0, 0, 3, '');

      if (msg_text <> '') then
        res_html = res_html || '<tr><td colspan="3">' || msg_text || '</td></tr>';
    end
    end
  end
  res_html = res_html || '</table>';

  if (db_id in (39801, 41701) and
      edc_code = 'loan.application') then
  begin
    for select cl.c_name_fio, cl.c_id
        from miniappclient mac
        inner join clients cl on cl.c_id = mac.mac_from_c_id
        where mac.mac_from_ma_id = :obj_id and
              mac.mac_is_poruch = -1
        into spouse_name, spouse_c_id
    do
    begin

      if (coalesce(spouse_c_id, 0) <> 0) then
      begin
        edc_id = f_edoc_kind('loan.spouses_consent', 'id');
        spouse_edoc_state = f_edoc_status('loan.spouses_consent', :obj_id, current_date);

        select msg_kind
        from hp_edoc_get_msg_kind('clients', :spouse_c_id, :edc_id)
        into msg_kind;

        if (spouse_edoc_state in ('empty', 'cancelled')) then
        begin
          uri_text = 'Сформировать документ';
          uri_description = 'Подтвердите создание эл. документа';
          uri_kind = 'edoc_spouse_agreement_create';
          uri_add_params = null;

          link = f_uri_sql(uri_text, uri_kind, obj_kind, obj_id, 'none', uri_description, '', null, null, null, null, 'dialog', uri_add_params);

          temp_html = f_html_table_row(:spouse_name, link, '', '', -1, -1, 0, 0, 2, '');

          res_html = res_html || iif(coalesce(temp_html, '') <> '', trim(f_format('<h2>%1</h2><table width="100%"><col width="40%"><col width="60%">', 'Согласие супруга') || temp_html || '</table>'), trim(''));
        end
        else
        if (spouse_edoc_state = 'signed') then
        begin
          uri_text = 'Аннулировать';
          uri_kind = 'edoc_cancel';
          uri_description = 'Подтвердите аннулирование эл. документа';
          uri_add_params = 'loan.spouses_consent';

          link = f_uri_sql(uri_text, uri_kind, obj_kind, obj_id, 'none', uri_description, '', null, null, null, null, 'dialog', uri_add_params);

          temp_html = f_html_table_row(:spouse_name, link, f_format('<font color="%1"<b>%2', color_green, 'Подписан'), '', -1, 0, -1, 0, 3, '');

          res_html = res_html || iif(coalesce(temp_html, '') <> '', f_format('<h2>%1</h2><table width="100%"><col width="40%"><col width="60%">', 'Согласие супруга') || temp_html || '</table>', '');
        end
        else
        if (spouse_edoc_state = 'unsigned') then
        begin
          res_html = res_html || f_format('<h2>%1</h2><table width="100%"><col width="40%"><col width="60%">', 'Согласие супруга');

          select edl_id, edl_blank_status
          from edoc_list
          where edl_object_kind = f_objkind(:obj_kind, 'id') and
                edl_object_id = :obj_id and
                edl_status = 0 and
                edl_from_edc_id = :edc_id
          order by edl_id desc
          rows 1
          into edl_id, edl_blank_status;

          -- edl_id будет найден, т.к. статус "не подписан"
          if (edl_blank_status < 0) then
          begin
            msg_text = f_lang(lang, 'Ошибка при формировании эл. документа. Ожидайте 15 минут, затем обратитесь в тех. поддержку');
            res_html = res_html || '<tr><td colspan="3">' || msg_text || '</td></tr>';
          end
          else
          if (edl_blank_status = 0) then
          begin
            msg_text = f_lang(lang, 'Документ формируется. Подождите 1 минуту');
            res_html = res_html || '<tr><td colspan="3">' || msg_text || '</td></tr>';
          end
          else
          begin
            for select eds_id, eds_from_c_id, c_name_fio, eds_status
                from edoc_sign, clients
                where eds_from_edl_id = :edl_id and
                      eds_from_c_id = c_id
                into eds_id, msg_c_id, c_name_fio, eds_status
            do
            begin
              link = '';

              if (exists(select *
                         from edoc_msg edm
                         where edm.edm_from_eds_id = :eds_id) and
                  eds_status = 0) then
              begin
                link = f_uri_sql('Отправить повторно', 'edoc_msg_ins', obj_kind, edl_id, 'none', c_name_fio, 'Подтвердите повторную отправку СМС', null, null, null, null, 'dialog', eds_id);

                res_html = res_html || f_html_table_row(:spouse_name, 'СМС отправлено', link, '', -1, 0, 0, 0, 3, '');
                suspend;
                exit;
              end

              if (eds_status = 0) then
              begin
                uri_text = 'Отправить СМС';
                uri_kind = 'edoc_spouse_agreement_sms';
                uri_description = 'Подтвердите отправку СМС лицу дающему согласие';
                uri_add_params = null;

                --Формируем ещё одну ссылку
                temp_link = f_uri_sql(uri_text, uri_kind, 'edoc_list', edl_id, 'none', uri_description, '', null, null, null, null, 'dialog', uri_add_params);

                if (msg_kind = 4) then
                  link = f_uri_open('Подписать через биометрию', 'edoc', 'client', msg_c_id, edc_name, 'eds_id=' || eds_id || '&save_photo=0', un_id);
                else
                  link = 'Подпись через биометрию недоступна';
              end

              res_html = res_html || f_html_table_row(:spouse_name, link, temp_link, '', -1, 0, 0, 0, 3, '');
            end
          end
        end
      end
    end
  end

  suspend;
end
