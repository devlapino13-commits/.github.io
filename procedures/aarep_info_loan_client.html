CREATE PROCEDURE AAREP_INFO_LOAN_CLIENT(
    REP_DATE_END D_DATE)
RETURNS (
    CLIENT D_STR_100,
    KONTRACTS D_STR_255,
    PHONE D_STR_20,
    STATUS D_STR_100,
    SEND_TIME D_STR_20)
AS
declare variable ilc_id d_id;
declare variable ilc_phone d_str_100;
declare variable ilc_juridical_c_id id_type;
declare variable c_id id_type;
begin
  -- отчет рассылки СМС "Уведомление о кредитной задолженности" (RFCKG-1970)
  for select ilc_id,
             ilc_from_c_id,
             ilc_juridical_c_id,
             c_name_fio
      from info_loan_client
      inner join clients on c_id = ilc_from_c_id
      where ilc_date = :rep_date_end
      order by c_name_fio
      into ilc_id,
           c_id,
           ilc_juridical_c_id, -- id юр. лица
           client -- ФИО, кому отправлен СМС
  do
  begin
    kontracts = '';
    phone = '';
    status = '';
    send_time = '';

    select list(k_kontract_no, ', ')
    from info_loan_debt
    inner join kontract on k_id = ild_from_k_id
    where ild_from_ilc_id = :ilc_id
    into kontracts;

    if (ilc_juridical_c_id > 0) then
      select c_name_fio || '(' || :client || ')'
      from clients
      where c_id = :ilc_juridical_c_id
      into client;

    for select msg_mail_to,
               sds_name,
               left(msg_datetime, 16)
        from fsrobot_msgbox
        inner join sms_delivery_status on sds_id = msg_send_status
        where msg_object_kind = 0 and
              msg_object_id = :c_id and
              msg_type = 1 and
              msg_body starting with 'Uvedomlenie o kreditnom dolge' and
              extract(year from msg_datetime) = extract(year from :rep_date_end) + 1 -- отправка за отчетный год производится в след. году
        into phone,
             status,
             send_time
    do
      suspend;
  end
end
