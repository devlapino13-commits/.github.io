CREATE PROCEDURE UXP$UXTREE_LIST
RETURNS (
    UTR_UUID_CODE D_STR_CODE,
    UTR_PARENT_UUID_CODE D_STR_CODE,
    UTR_UUID TYPE OF D_UUID,
    UTR_PARENT_UUID TYPE OF D_UUID,
    UTR_CODE TYPE OF D_STR_CODE,
    UTR_NAME TYPE OF D_STR_100,
    UTR_MEMO TYPE OF D_STR_LONG,
    UTR_POSITION D_INTEGER,
    RES_CAPTION D_STR_255)
AS
begin
  for select utr_uuid,
             utr_parent_uuid,
             utr_code,
             utr_name,
             utr_position,
             umo_memo
      from ux$tree left join ux$memo on (utr_uuid = umo_uuid)
      into :utr_uuid,
           :utr_parent_uuid,
           :utr_code,
           :utr_name,
           :utr_position,
           :utr_memo
  do
  begin
    res_caption = utr_name;
    if (res_caption not containing utr_code) then
      res_caption = res_caption || '    / ' || utr_code;

    utr_uuid_code = uuid_to_char(utr_uuid);
    utr_parent_uuid_code = uuid_to_char(utr_parent_uuid);

    suspend;
  end
end
