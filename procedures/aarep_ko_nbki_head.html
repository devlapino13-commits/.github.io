declare variable jso d_integer;
declare variable okato d_str_code;

declare function getcatval(kind d_str_code, code d_str_code)
  returns d_str_255
  as
  begin
    return coalesce((select bct_name from bki_catalog where bct_kind = :kind and bct_code = :code), 'Неопределен');
  end

declare function f_my_obj_get_arr(handle d_integer, path d_str_long)
returns d_integer
as
  declare variable jsa d_integer;
  declare variable val_type d_str_code;
begin
  if (json.contains(handle, path) = 0) then
    return null;

  val_type = json.val_get_type_name(handle, path);

  if (val_type = 'Array') then
    return json.obj_get_arr(handle, path);
  else if (val_type = 'Object') then
  begin
    handle = json.obj_get_obj(handle, path);
    jsa = json.arr_create();
    json.arr_set_obj(jsa, -1, handle);

    return jsa;
  end
  else
    exception e_default path || ' is not Array/Object';
end

declare function my_get_last_el(jso_in d_integer)
returns d_integer
as
  declare variable jso d_integer;
  declare variable res d_integer;
  declare variable res_tmp d_integer;
  declare variable max_date d_date;
  declare variable cur_date d_date;
begin
  res = null;
  if (jso_in is not null) then
  begin
    max_date = '01.01.1900';
  
    for select jso
    from hp_bki_json_get_list(:jso_in, 1)
    into jso
    do
    begin
      res_tmp = jso;
  
      cur_date = json.obj_get_val(jso, 'last_updated_dt', current_date);
  
      if (max_date < cur_date) then
      begin
        max_date = cur_date;
        res = res_tmp;
      end
    end
  end

  if (res is null) then
    res = json.obj_create();

  return res;
end

  select list(json.obj_get_val(jso, 'id_type_text', '') || ' ' ||

                  json.obj_get_val(jso, 'id_num', '') || ' выдан ' ||

  from hp_bki_json_get_list(json.obj_get_arr(:jso, 'prequest.req.id_req'), 1)
  into request_pass;

  select json.obj_get_val(jso, 'name1', '') || ' ' ||

        json.obj_get_val(jso, 'paternal', ''),

        json.obj_get_val(jso, 'place_of_birth', '')
  from hp_bki_json_get_list(json.obj_get_arr(:jso, 'preply.report.person_reply'), 1)
  order by f_bki_json_get_date(jso, 'last_updated_dt', '01.01.1900') desc
  rows 1
  into fio, birth_data;

  -- ДУЛ
  select json.obj_get_val(jso, 'id_type_text', '') || ' ' ||
        json.obj_get_val(jso, 'series_number', '') || ' ' ||
        json.obj_get_val(jso, 'id_num', '') || ' выдан ' ||

        json.obj_get_val(jso, 'issue_authority', '')
  from hp_bki_json_get_list(json.obj_get_arr(:jso, 'preply.report.id_reply'), 1)
  order by f_bki_json_get_date(jso, 'last_updated_dt', '01.01.1900') desc

  rows 1

  if (okato is not null) then
  begin
    with recursive t1 (lvl, parent_id, name, code, okato) as(
      select 0, parent, pl_name || ' ' || plt_short_name, pl_code, pl_add_code
      from places, placetype
      where plt_id = pl_status
        and pl_add_code = :okato
    union all
      select lvl + 1, parent, pl_name || ' ' || plt_short_name || ', ' || name, pl_code, pl_add_code
      from t1, places, placetype
      where pl_id = parent_id
        and parent is not null
        and plt_id = pl_status
    )
    select name || ', ' || :address_reg
    from t1
    order by lvl desc
    rows 1
