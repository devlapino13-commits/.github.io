CREATE PROCEDURE AAREP_REPORT_OKUD(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    KC_ID ID_TYPE,
    K_KONTRACT_NO VARCHAR(20),
    CLIENT_NAME_FIO VARCHAR(100),
    C_BIRTH_DATE DATE,
    C_NAME_FIO VARCHAR(100),
    K_DISB_DATE DATE,
    CLI_BALANCE DECIMAL(18,2),
    CLI_PERC_SALDO DECIMAL(18,2),
    TERM_IN_DAYS INTEGER,
    PERCENT_EFFECT PERC_TYPE,
    KC_AMOUNT DECIMAL(18,2),
    PR_NAME D_STR_CODE,
    OKS_KONT_LAST_DATE D_DATE,
    OKS_LAST_SH_DATE D_DATE,
    Z_COST_LIKVID DECIMAL(18,2),
    K_STOP_PERC D_DATE,
    FINAL_PDN CURR_TYPE,
    OKS_DELINQ_DAYS D_INTEGER,
    CRED_FULL_COST DECIMAL(15,3),
    K_PERCENT_FLAT_ANNUAL DECIMAL(15,3),
    KC_ANKETA_INCOME DECIMAL(18,2),
    KC_STAT_AVG_INCOME DECIMAL(18,2),
    KC_FINAL_INCOME DECIMAL(18,2),
    STATUS_ZAP D_STR_100,
    PERCENT_YEAR_R DECIMAL(15,2),
    DATE_OFF DATE,
    STR_T1 STR_10,
    STR_T2 STR_100,
    STR_T3 D_STR_255,
    OKS_IS_RESCHEDULE_LLR SMALLINT,
    OKS_IS_ZALOG_LLR SMALLINT,
    STR_T4 D_STR_255,
    STR_T5 D_STR_20,
    STR_T6 D_STR_20,
    STR_T7 D_STR_64,
    STR_T8 D_STR_255)
AS
declare variable C_ID ID_TYPE;
declare variable PR_CODE STR_100;
declare variable MAX_SH_DATE date;
declare variable PR_ID_OLD integer;
declare variable SAB_ID D_ID;
declare variable SAB_ID_ZALOG_COST D_ID;
declare variable K_ID decimal(18,0);
declare variable PR_ID D_INTEGER;
declare variable K_ID_PRIOR ID_TYPE;
declare variable PR_ID_PRIOR ID_TYPE;
begin
  pr_id_old = 0;

  select sab_id
  from sys_accbalance
  where sab_full_code = 'Микрозаймы.Внебаланс.Поручительство'
  into sab_id;

  select sab_id
  from sys_accbalance
  where sab_full_code = 'Микрозаймы.Внебаланс.Залог'
  into :sab_id_zalog_cost;

  for
  --------списанные
      with tabl1
      as (select oko_from_k_id
          from operkroff
          where oko_ol_is_active = 1 and
                oko_ol_date between :rep_date_begin and :rep_date_end and
                oko_type = 1 and
                (oko_sh_os > 0 or
                oko_sh_perc > 0)
          group by 1)
      select 'списан' status_zap, k_id, k_kontract_no, k_disb_date, k_stop_perc,
             cast(k_percent_effect_fifd as decimal(15,3)) as cred_full_cost, k_from_pr_id

      from tabl1, kontract
      where oko_from_k_id = k_id and
            k_disb_date > '30.09.2019'
      union all
      ----на балансе.реестр--------------
      select 'На балансе\Рестр' status_zap, oks_from_k_id, k_kontract_no, k_disb_date, k_stop_perc,
             cast(kont.k_percent_effect_fifd as decimal(15,3)) as cred_full_cost, k_from_pr_id
      from oksummary oks, kontract kont
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks.oks_from_k_id = kont.k_id and
            oks_close_state = 0 and
            k_disb_date > '30.09.2019'
      into :status_zap, k_id, k_kontract_no, k_disb_date, k_stop_perc, cred_full_cost, pr_id
  do
  begin
    k_percent_flat_annual = 0;
    select round(k_percent_per_day * 365.00, 3)
    from kontract kon
    where kon.k_id = :k_id
    into k_percent_flat_annual;

    percent_year_r = null;
    date_off = null;
    if (:status_zap = 'списан') then
    begin
      date_off = null;
      select ok_ol_date
      from operkred
      where ok_ol_is_active = 1 and
            ok_from_k_id = :k_id and
            ok_type = 5 and
            ok_os + ok_perc < 0
      order by ok_ol_date
      rows 1
      into date_off;

      select kd_percent_per_day * 365
      from kontdata
      where :date_off between kd_ol_date and kd_date_end and
            kd_ol_is_delete = 0 and
            kd_from_k_id = :k_id
      into percent_year_r;
    end
    else
    if (:status_zap = 'На балансе\Рестр') then
    begin
      select round(k_percent_per_day * 365.00, 3)
      from kontract kon
      where kon.k_id = :k_id
      into percent_year_r;
    end
    if (pr_id <> pr_id_old) then
    begin
      select pr_code, pr_name
      from product
      where pr_id = :pr_id
      into pr_code, pr_name;

      pr_id_old = pr_id;
    end

    for select kc_id, kc_from_c_id, kc_amount_approved
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into kc_id, c_id, kc_amount
    do
    begin

      c_name_fio = 'n' || :c_id;

      client_name_fio = null;
      c_birth_date = null;

      from clients c
      where c_id = :c_id
      into client_name_fio, c_birth_date;

      final_pdn = null;
      select kc_pdn_koef
      from knclient_pdn
      where kc_id = :kc_id
        and kcp_kind = 0
      into final_pdn;

      z_cost_likvid = 0;
      if (pr_code containing '.zalog_car.' or pr_code containing '.zalog.') then


        if (:status_zap = 'На балансе\Рестр') then
          z_cost_likvid = -f_get_acc_amount(:kc_id, :sab_id_zalog_cost, :rep_date_end, :rep_date_end, 'bal'); -- Залоговая стоимоть
      if (:status_zap = 'списан') then
        z_cost_likvid = -f_get_acc_amount(:kc_id, :sab_id_zalog_cost, :date_off, :date_off, 'bal'); -- Залоговая стоимоть

      cli_balance = null;
      cli_perc_saldo = null;
      select oks_kont_last_date, oks_last_sh_date, oks_percent_esp, oks_balance, oks_bal_perc, oks_is_reschedule_llr, oks_delinq_days

      from kn_oksummary
      where oks_from_kc_id = :kc_id and
            (case
               when :status_zap = 'На балансе\Рестр' and :rep_date_end between oks_date_begin and oks_date_end then 1
               when :status_zap = 'списан' and :rep_date_end - 1 between oks_date_begin and oks_date_end then 1
               else 0
             end) = 1 and
            (oks_close_state in (0, 1) or
            oks_bal_law <> 0 or
            oks_reserv_law <> 0)
      into oks_kont_last_date, oks_last_sh_date, -- дата погаш по актуальному графику

      --
      if (:status_zap = 'списан') then
      begin
        oks_delinq_days = null;

        select :date_off - oks_delinq_date_prv

        from kn_oksummary
        where oks_from_kc_id = :kc_id and
              :date_off between oks_date_begin and oks_date_end and
              (oks_close_state in (0, 1) or
              oks_bal_law <> 0 or
              oks_reserv_law <> 0)
        into oks_delinq_days;
      end

      --
      begin
        kc_anketa_income = null;
        kc_stat_avg_income = null;
        kc_final_income = null;
        if (coalesce(final_pdn, 0) <> 0) then
        begin
          select kc_anketa_income, kc_stat_avg_income, kc_final_income
          from aarep_knclient_one_pdnlist(:kc_id)
          where --:rep_only_summary = 0 or
                rec_no = -1
          into kc_anketa_income, kc_stat_avg_income, kc_final_income;
        end
      end
      if (:status_zap = 'На балансе\Рестр') then
      begin
        if ((cli_balance > 0 or cli_perc_saldo > 0) and
            (oks_is_reschedule_llr = 0)) then
          :status_zap = 'На балансе';
        if ((cli_balance > 0 or cli_perc_saldo > 0) and
            (oks_is_reschedule_llr = 1)) then
          :status_zap = 'Рестр';
      end

      term_in_days = 0;
      term_in_days = datediff(day, k_disb_date, oks_kont_last_date);

      if (char_length(extract(month from :k_disb_date)) = 1) then
        str_t1 = '0' || extract(month from :k_disb_date) || '.' || extract(year from :k_disb_date);
      if (char_length(extract(month from :k_disb_date)) = 2) then
        str_t1 = extract(month from :k_disb_date) || '.' || extract(year from :k_disb_date);

      str_t2 = '1 – для иных договоров потребительского займа';

      oks_is_zalog_llr = 0;
      if (pr_name = 'Микрозаем под залог') then
        oks_is_zalog_llr = 1;

      str_t3 = '';
      str_t3 = case
                 when oks_is_zalog_llr = 1 and kc_amount >= 10000 then '1.2 – для потребительских займов с обеспечением в виде залога (кроме POS-займов и потребительских займов, обязательства по которым обеспечены ипотекой) в сумме от 10 тысяч рублей включительно'
                 when oks_is_zalog_llr = 0 and kc_amount < 10000 and term_in_days <= 30 then '3.1.1 – для потребительских займов без обеспечения (кроме POS-займов) в сумме до 10 тысяч рублей на срок не более 30 дней'
                 when oks_is_zalog_llr = 0 and kc_amount >= 10000 and kc_amount <= 30000 and term_in_days <= 30 then '3.1.2 – для потребительских займов без обеспечения (кроме POS-займов) в сумме от 10 тысяч рублей включительно до 30 тысяч рублей включительно на срок не более 30 дней'
                 when oks_is_zalog_llr = 0 and kc_amount > 30000 and term_in_days <= 30 then '3.1.3 – для потребительских займов без обеспечения (кроме POS-займов) в сумме свыше 30 тысяч рублей на срок не более 30 дней'
                 when oks_is_zalog_llr = 0 and kc_amount < 10000 and term_in_days >= 31 and term_in_days <= 60 then '3.2.1 – для потребительских займов без обеспечения (кроме POS-займов) в сумме до 10 тысяч рублей на срок от 31 до 60 дней включительно'
                 when oks_is_zalog_llr = 0 and kc_amount >= 10000 and kc_amount <= 30000 and term_in_days >= 31 and term_in_days <= 60 then '3.2.2 – для потребительских займов без обеспечения (кроме POS-займов) в сумме от 10 тысяч рублей включительно до 30 тысяч рублей включительно на срок от 31 до 60 дней включительно'

                 when :cred_full_cost > 90 and :cred_full_cost <= 100 then '(90; 100]'

                 when :cred_full_cost > 220 and :cred_full_cost <= 230 then '(220; 230]'

                 when :cred_full_cost > 240 and :cred_full_cost <= 250 then '(240; 250]'
                 when :cred_full_cost > 250 and :cred_full_cost <= 260 then '(250; 260]'
                 when :cred_full_cost > 260 and :cred_full_cost <= 270 then '(260; 270]'
