CREATE PROCEDURE HP_EXTSYS_PROCESS
AS
declare variable esm_id d_integer;
declare variable err_text d_blob_text;
declare variable err_mailto d_str_255;
declare variable rep_project d_str_20;
declare variable db_id d_integer;
declare variable start_time d_datetime;
declare variable one_err_text d_str_8100;
begin
  -- Обработка платежных операций
  -- поступающие через платежные терминалы и банковские карточки

  start_time = cast('now' as timestamp);

  db_id = f_get_db_id();

  -- Общие настройки, зависящие от Страны
  err_mailto = 'finstart@mbfplanet.com';

  if (db_id = 81001) then
    err_mailto = err_mailto || ';sh.kirgizbaeva@molbulak.com;l.belousova@molbulak.com';

  err_text = ''; -- Список доступных проектов

  for select esm_id, esm_project
      from extsysmain
      left join extsyspro on esm_project = esp_project and
            esp_project_kind = 'payment' and -- только платежные системы
            esp_status = 1 -- только действующие
      where esm_action = 'pay' and
            esm_status = 0 and
            esm_object_id is not null and
            esm_id is not null
      into :esm_id, :rep_project
  do
  begin
    one_err_text = '';

    in autonomous transaction do
    begin
      select coalesce(err_text, '')
      from hp_extsys_process_one(:esm_id)
      into :one_err_text;
    end
    if (one_err_text <> '') then
      err_text = blob_append(err_text, one_err_text);

    -- Работаем не больше минуты
    if (datediff(second, start_time, cast('now' as timestamp)) > 50) then
      break;
  end -- конец обработки транзакций по проекту

  -- Если в ходе обработки появились ошибки, отправляем разработчикам и ответственным лицам
  if (err_text <> '') then
  begin
    insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body,
                                msg_lifetime_days, msg_from_un_id,
                                msg_command_kind)
    values (0, :err_mailto, :rep_project, :err_text, 30, 1000000000000,
            :rep_project);
  end

end
