CREATE PROCEDURE AAREP_SALARY_SUMMARY_REPORT(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REP_REGION_KIND VARCHAR(100),
    REP_REGION_NAME VARCHAR(100),
    REP_PERIOD VARCHAR(20),
    REP_JOB_KIND VARCHAR(100),
    REP_SALARY DECIMAL(18,4),
    REP_BONUS DECIMAL(18,4),
    REP_COMPENS DECIMAL(18,4),
    REP_SF DECIMAL(18,4),
    REP_BOLNICH DECIMAL(18,4),
    REP_OKRUG_NAME VARCHAR(100))
AS
declare variable reg_id integer;
begin
  for with tab
      as (select reg_code || ' ' || reg_name as reg_name,
                 iif(reg_unicode = 1090100, '1 ГО', iif(reg_unicode = 1060100, '2 БРУ', '3 Регионы')) as reg_kind, iif(extract(month from osm_ol_date) >= 10,
                 (extract(year from osm_ol_date) || '-' || extract(month from osm_ol_date)),
                 (extract(year from osm_ol_date) || '-0' || extract(month from osm_ol_date))) as rep_period,
                 case jk_system_code
                   when 'JobSystem.LO_Standart' then 'КС'
                   when 'JobSystem.LO_Partner' then 'КС'
                   when 'JobSystem.Region_KM' then 'КМ'
                   when 'JobSystem.Specialist_KOBD' then 'ОБД'
                   when 'JobSystem.LO_Senior' then 'СКС'
                   else 'Другие'
                 end as job_kind,
                 sum(iif(osd_from_sac_id = 9, osd_value, 0)) as total_calc,
                 sum(iif(osd_from_sac_id = 49, osd_value, 0)) as total_bonus,
                 sum(iif(osd_from_sac_id in (20, 34, 35, 36), osd_value, 0)) as total_bolnich,
                 sum(iif(osd_from_sac_id = 22, osd_value, 0)) as total_sf,
                 sum(iif(osd_from_sac_id = 51, osd_value, 0)) as other_compens, reg_unicode
          from opersalmain, opersaldetail, region, joblist, hr_pstatus, jobkind, region_list(:rep_region_id, 1, -1, 1)
          where osd_from_osm_id = osm_id and
                ps_from_reg_id = reg_unicode and
                reg_unicode = id_list and
                osm_ol_is_active = 1 and
                osm_ol_date between :rep_date_begin and :rep_date_end and
                osd_from_sac_id in (9, 20, 34, 49, 51, 22, 35, 36) and
                ps_from_jl_id = jl_id and
                ps_date_begin <= osm_ol_date and
                ps_date_end >= osm_ol_date and
                ps_status = 1 and
                ps_from_c_id = osm_from_c_id and
                jl_kind = jk_id
          group by 1, 2, 3, 4, reg_unicode)
      select reg_kind, reg_name, rep_period, job_kind, reg_unicode, sum(total_calc - total_bonus - other_compens),
             sum(total_bonus), sum(total_bolnich), sum(total_sf), sum(other_compens)
      from tab
      group by reg_kind, reg_name, rep_period, job_kind, reg_unicode
      into :rep_region_kind, :rep_region_name, :rep_period, :rep_job_kind, :reg_id, :rep_salary, :rep_bonus,
           :rep_bolnich, :rep_sf, :rep_compens
  do
  begin
    select aa_curr_level3_name
    from aarep_get_office_categ(:reg_id, :rep_date_end)
    into :rep_okrug_name;
    suspend;
  end
end
