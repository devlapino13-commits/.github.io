CREATE PROCEDURE HP_OPERKRED_DD_CHANGE(
    REPORT_REGION_ID D_INTEGER,
    REPORT_DATE_BEGIN D_DATE,
    REPORT_DATE_END D_DATE,
    REPORT_DATE_NEW D_DATE,
    REPORT_USER_ID D_ID,
    REPORT_ONE_K_ID D_ID = null)
AS
declare variable kol integer;
declare variable aarep_plan_os decimal(18,4);
declare variable aarep_plan_perc decimal(18,4);
declare variable aarep_plan_repay_no integer;
declare variable aarep_plan_repay_date date;
declare variable aarep_k_id decimal(18,0);
declare variable curr_reg integer;
declare variable new_ol_id decimal(18,0);
declare variable ss_memo varchar(200);
declare variable oks_date_begin d_date;
begin
-- operkred_fix

  curr_reg = f_hq_id();

  select kv
  from get_kv(:curr_reg, 5)
  into :new_ol_id;
  kol = 0;

  if (report_date_begin <> report_date_end) then
    ss_memo = 'Перевод даты погашения (' || :report_date_begin || ' - ' || :report_date_end || ') -> ' || :report_date_new;
  else
    ss_memo = 'Перевод даты погашения ' || :report_date_begin || ' -> ' || :report_date_new;
  insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                        ol_status, is_delete, ol_datetime, ol_user_id, ol_document_client, ol_from_mo_id,
                        ol_is_mo_modified, ol_document_office, ol_acc_no_client, ol_acc_no_office, ol_oper_subtype,
                        ol_date_oper_day, ol_pocket, ol_doc_client_name, ol_is_client, ol_year, ol_month)
  values (:new_ol_id, :report_region_id, 1, :report_date_begin, null, '', 3, :ss_memo, 0, 0, null, :report_user_id,
          null, null, 0, null, null, null, 9, :report_date_begin, null, null, 0, null, null);

  if (f_tmp_const('new_schedule_change_active') = 0) then  -- старый вариант переноса даты погашения по графику
  begin
    for select aarep_k_id, aarep_plan_os, aarep_plan_perc, aarep_plan_repay_no, aarep_plan_repay_date
        from aarep_kontract_id(:report_region_id, :report_date_begin, :report_date_end, 0, '---', 6)
        into :aarep_k_id, :aarep_plan_os, :aarep_plan_perc, :aarep_plan_repay_no, :aarep_plan_repay_date
    do
    begin
      kol = kol + 1;
      insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                            ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
      values (:new_ol_id, :kol, :aarep_k_id, :aarep_plan_repay_no, :aarep_plan_repay_date, 2, 0, 0, 0, 3, 0,
              -1 * :aarep_plan_os, -1 * :aarep_plan_perc, 0, 0);
  
      kol = kol + 1;
      insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                            ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
      values (:new_ol_id, :kol, :aarep_k_id, :aarep_plan_repay_no, :report_date_new, 2, 0, 0, 0, 3, 0, :aarep_plan_os,
              :aarep_plan_perc, 0, 0);
    end
  end else
  begin
    -- новый вариант переноса даты погашения по графику, просрочки не переносим
    for select oks_from_k_id, oks_date_begin, oks_delinq_date
    from oksummary
    where oks_date_end >= '31.12.2999' and
          oks_region_id = :report_region_id and
          oks_delinq_date between :report_date_begin and :report_date_end and
          (:report_one_k_id is null or oks_from_k_id = :report_one_k_id) and
          oks_close_state = 0
    into :aarep_k_id, :oks_date_begin, aarep_plan_repay_date
    do
      execute procedure hp_operkred_dd_one_change(:aarep_k_id, :new_ol_id, :report_date_begin, :report_date_end, :report_date_new);

    -- переносим зависшие долги по закрытым договорам за указанным период
    for select ks_from_k_id, ks_date_begin
    from kontstate
    where ks_date_begin >= :report_date_begin and
          ks_region_id = :report_region_id and
          ks_state = 2 and
          ks_ol_is_delete = 0 and
          (:report_one_k_id is null or ks_from_k_id = :report_one_k_id)
    into :aarep_k_id, :oks_date_begin
    do
    begin
      for select ok_sh_id, ok_sh_date, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
      from operkred
      where ok_from_k_id = :aarep_k_id and
            ok_ol_is_active = 1 and
            ok_type <= 6 and
            ok_sh_date between :report_date_begin and :report_date_end
      group by ok_sh_id, ok_sh_date
      having sum(ok_os + ok_sh_os) > 0 or sum(ok_perc + ok_sh_perc) > 0
      into aarep_plan_repay_no, aarep_plan_repay_date, aarep_plan_os, aarep_plan_perc
      do
      begin
        kol = kol + 1;
        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                              ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
        values (:new_ol_id, :kol, :aarep_k_id, :aarep_plan_repay_no, :aarep_plan_repay_date, 2, 0, 0, 0, 3, 0,
                -1 * :aarep_plan_os, -1 * :aarep_plan_perc, 0, 0);
    
        kol = kol + 1;
        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                              ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
        values (:new_ol_id, :kol, :aarep_k_id, :aarep_plan_repay_no, :report_date_new, 2, 0, 0, 0, 3, 0, :aarep_plan_os,
                :aarep_plan_perc, 0, 0);
      end
    end

    -- заменяем плановую дату по уже проведенным операциям.
    -- данный код актуален когда перевод даты производится задним числом
--    update operkred
--    set ok_sh_date = :report_date_new
--    where ok_ol_date >= :report_date_begin and
--          ok_sh_date between :report_date_begin and :report_date_end and
--          ok_ol_date > ok_sh_date and -- только по платежам просрочек
--          ok_ol_is_active = 1 and
--          ok_ol_reg_id = :report_region_id and
--          ok_type <> 2 and
--          (:report_one_k_id is null or ok_from_k_id = :report_one_k_id);
  end
end
