CREATE PROCEDURE AAREP_EMP_DATA_OK(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(255),
    NAME_FIO VARCHAR(255),
    JL_NAME VARCHAR(255),
    BIRTH_DATE VARCHAR(10),
    SEX VARCHAR(10),
    PASS_NO VARCHAR(10),
    ADDRESS_LEGAL VARCHAR(10),
    ADDRESS_FACT VARCHAR(10),
    EMAIL VARCHAR(10),
    SKYPE VARCHAR(10),
    PHOTO VARCHAR(10),
    PLASTCARD_NO VARCHAR(10),
    PASS_DATE VARCHAR(10),
    PASS1_FILE VARCHAR(10),
    PASS2_FILE VARCHAR(10),
    VERIFY_STATUS VARCHAR(50))
AS
begin
  for select r1.reg_name, cl.c_name_fio, jl_name,
             (
             case
               when c_birth_date is null then 'Нет данных'
               else 'Ok'
             end) birth_date,
             (
             case
               when c_sex is null then 'Нет данных'
               else 'Ok'
             end) c_sex,
             (
             case
               when coalesce(cl.c_pass_no, '') = '' then 'Нет данных'
               else 'Ok'
             end) pass_no,
             (
             case
               when cl.c_pass_date is null then 'Нет данных'
               else 'Ok'
             end) pass_date,
             (
             case
               when f_cont_info('clients', c_id, 'cell_main', 'ok') = '' then 'Нет данных'
               else 'Ok'
             end) plastcard_no,

             (
             case
               when lower(adl_legal.adl_address_str) = 'не указан' then 'Нет данных'
               else 'Ok'
             end) as address_legal,
             (
             case
               when lower(adl_fact.adl_address_str) = 'не указан' then 'Нет данных'
               else 'Ok'
             end) as address_fact,
             (
             case
               when f_cont_info('clients', c_id, 'email', 'ok') = '' then 'Нет данных'
               else 'Ok'
             end) c_email,

             (
             case
               when f_cont_info('clients', c_id, 'skype', 'last') = '' then 'Нет данных'
               else 'Ok'
             end) c_skype,

             (
             case
               when coalesce(cl.c_blob_photo_id, 0) = 0 then 'Нет данных'
               else 'Ok'
             end) photo_id,
             (
             case
               when coalesce(cl.c_blob_pass1_id, 0) = 0 then 'Нет данных'
               else 'Ok'
             end) pass1_id,
             (
             case
               when coalesce(cl.c_blob_pass2_id, 0) = 0 then 'Нет данных'
               else 'Ok'
             end) pass2_id,
             case c_verify_status
               when -2 then 'подтверждена: данные не верные'
               when -1 then 'не требуется верификация (юр.лица)'
               when 0 then 'данные не полные'
               when 1 then 'готов для верификаци'
               when 2 then 'подтверждена: верные данные'
             end verify_status
      from hr_pstatus, hr_orders, region_list(:report_region_id, 1, -1, 1), joblist, region r1, clients cl, region br, addrlist adl_legal, addrclient adcl_legal, jobkind, addrlist adl_fact, addrclient adcl_fact
      where ho_id = ps_from_ho_id and
            (ps_from_reg_id = id_list) and
            ps_date_begin <= :report_date_end and
            ps_date_end >= :report_date_end and
            ps_from_reg_id = r1.reg_unicode and
            br.reg_unicode = r1.parent and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            ps_is_working = 1 and
            ps_from_c_id = c_id

            and
            adcl_legal.adc_address_type = 1 and
            adcl_legal.adc_from_c_id = c_id and
            adcl_legal.adc_from_adl_id = adl_legal.adl_id

            and
            adcl_fact.adc_address_type = 2 and
            adcl_fact.adc_from_c_id = c_id and
            adcl_fact.adc_from_adl_id = adl_fact.adl_id
      into :reg_name, :name_fio, :jl_name, :birth_date, :sex, :pass_no, :pass_date, :plastcard_no, :address_legal,
           :address_fact, :email, :skype, :photo, :pass1_file, :pass2_file, :verify_status
  do
  begin
    if ((birth_date = 'Нет данных') or (sex = 'Нет данных') or (pass_no = 'Нет данных') or (pass_date = 'Нет данных') or (plastcard_no = 'Нет данных') or (address_legal = 'Нет данных') or (address_fact = 'Нет данных') or (email = 'Нет данных') or (skype = 'Нет данных') or (pass1_file = 'Нет данных') or (pass2_file = 'Нет данных') or (verify_status = 'подтверждена: верные данные') or (photo = 'Нет данных')) then
      suspend;
  end
end
