CREATE PROCEDURE HP_ACCESSCASHROOM_VIEW(
    OFFICE_ID D_INTEGER,
    DT_BEG D_DATE,
    DT_END D_DATE,
    RECEIVER_ID D_ID = null)
RETURNS (
    ACR_ID D_INTEGER,
    ACR_DATE D_DATE,
    ACR_OFFICE_ID D_INTEGER,
    OFFICE_NAME D_STR_CODE,
    ACR_REG_ID D_INTEGER,
    REG_NAME D_STR_CODE,
    ACR_START_DATE D_DATE,
    ACR_END_DATE D_DATE,
    ACR_BASIS_ID D_INTEGER,
    BASIS D_STR_255,
    ACR_EDL_ID D_ID,
    ACR_STATUS D_SMALLINT,
    STATUS D_STR_CODE,
    RECEIVER_LST D_STR_LONG,
    ACR_ISSUER_ID D_ID,
    ISSUER_NAME D_STR_100,
    ACR_CASHIER_ID D_ID,
    CASHIER_NAME D_STR_100,
    ACR_PERSON_ID D_ID,
    PERSON_NAME D_STR_100,
    ACR_DIRECTOR_ID D_ID,
    DIRECTOR_NAME D_STR_100,
    ACR_ISSUER_DATESIGN D_DATETIME,
    ACR_CASHIER_DATESIGN D_DATETIME,
    ACR_PERSON_DATESIGN D_DATETIME,
    ACR_DIRECTOR_DATESIGN D_DATETIME)
AS
begin
  for
    select
      acr_id,
      acr_date,
      acr_office_id,
      rg1.reg_name,
      acr_reg_id,
      rg2.reg_name,
      acr_start_date,
      acr_end_date,
      acr_basis_id,
      rd_name,
      acr_edl_id,
      acr_status,
      case acr_status
        when -1 then 'аннулирован'
        when 0 then 'создан'
        when 1 then 'завизирован'
        when 2 then 'частично подписан'
        when 3 then 'подписан'
      end,
      acr_issuer_id,
      cl1.c_name_fio,
      acr_cashier_id,
      cl2.c_name_fio,
      acr_person_id,
      cl3.c_name_fio,
      acr_director_id,
      cl4.c_name_fio,
      acr_issuer_datesign,
      acr_cashier_datesign,
      acr_person_datesign,
      acr_director_datesign
    from accesscashroom
      inner join region as rg1
        on rg1.reg_unicode = acr_office_id
      inner join region as rg2
        on rg2.reg_unicode = acr_reg_id
      inner join refdata
        on rd_id = acr_basis_id
      inner join clients cl1
        on cl1.c_id = acr_issuer_id
      inner join clients cl2
        on cl2.c_id = acr_cashier_id
      inner join clients cl3
        on cl3.c_id = acr_person_id
      inner join clients cl4
        on cl4.c_id = acr_director_id
    where exists(
        select * from region_list(:office_id,1,3,1)
        where id_list = acr_office_id)
      and acr_date between :dt_beg and :dt_end
  into
    :acr_id,
    :acr_date,
    :acr_office_id,
    :office_name,
    :acr_reg_id,
    :reg_name,
    :acr_start_date,
    :acr_end_date,
    :acr_basis_id,
    :basis,
    :acr_edl_id,
    :acr_status,
    :status,
    :acr_issuer_id,
    :issuer_name,
    :acr_cashier_id,
    :cashier_name,
    :acr_person_id,
    :person_name,
    :acr_director_id,
    :director_name,
    :acr_issuer_datesign,
    :acr_cashier_datesign,
    :acr_person_datesign,
    :acr_director_datesign
  do
  begin
    if (receiver_id > 0 and not exists(
      select * from accesscashroomd
      where acd_acr_id = :acr_id
        and acd_receiver_id = :receiver_id))
    then
      continue;

    receiver_lst = (
      select list(c_name_fio, '; ')
      from accesscashroomd
        inner join clients
          on c_id = acd_receiver_id
      where acd_acr_id = :acr_id);

    suspend;
  end
end
