CREATE PROCEDURE AAREP_KS_ANALYSIS_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_CURRENCY VARCHAR(3))
RETURNS (
    REG_NAME VARCHAR(50),
    PARENT_NAME VARCHAR(50),
    KAL_NEW_KF_OFFICER_OUT DECIMAL(18,0),
    KF_OFFICER_NAME_OUT VARCHAR(100),
    KAL_NEW_KF_OFFICER DECIMAL(18,0),
    KF_OFFICER_NAME VARCHAR(100),
    K_ID DECIMAL(18,0),
    KS_DATE_BEGIN DATE,
    KS_DATE_END DATE,
    K_DISB_DATE DATE,
    K_KONTRACT_NO VARCHAR(50),
    K_KOLICH_CLIENT_APPROVED INTEGER,
    K_INDEX_CURR_RATE DECIMAL(18,4),
    K_AMOUNT_APPROVED DECIMAL(18,4),
    K_KONTRACT_BALANCE DECIMAL(18,4),
    K_KOLICH_CLIENT_ACTIVE INTEGER,
    K_INDEX_CURR_CODE VARCHAR(3),
    CR_RATE DECIMAL(18,4),
    K_DELIQ_DATE_BEGIN DATE,
    K_DELIQ_DAY INTEGER,
    K_DELIQ_BALANCE DECIMAL(18,4),
    K_DELIQ_BALANCE7 DECIMAL(18,4),
    K_DELIQ_BALANCE30 DECIMAL(18,4),
    K_DATE_FACT_END DATE,
    K_SCHEDULE_DATE_END DATE,
    K_DISB_AMOUNT DECIMAL(18,4),
    K_DISB_KOLICH_CLIENT INTEGER)
AS
declare variable st_staff integer;
declare variable st_client integer;
declare variable curr_reg_id integer;
declare variable curr_reg_id_prior integer;
declare variable kal_new_kf_officer_out_prior decimal(18,0);
declare variable kal_new_kf_officer_prior decimal(18,0);
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  if (:rep_currency is null) then
    rep_currency = '000';
  curr_reg_id_prior = 0;
  kal_new_kf_officer_out_prior = 0;
  kal_new_kf_officer_prior = 0;

  for select kk_region_id, kk_id, kk_disb_date, kk_close_date, kk_disb_date, kk_kontract_no, kk_kolich_client,
             kk_disb_curr_rate, kk_currency, kk_amount_approved, kk_disb_kc_id, kk_kc_id, kk_curr_rate, kk_balance,
             kk_delinq_date, kk_delinq_balance
      from aarep_active_kont_id_in_period(:rep_region_id, :rep_date_begin, :rep_date_end, 1, 0, 1, 0, 0)
      into :curr_reg_id, :k_id, :ks_date_begin, :ks_date_end, :k_disb_date, :k_kontract_no, :k_kolich_client_approved,
           :k_index_curr_rate, :k_index_curr_code, :k_amount_approved, :kal_new_kf_officer_out, :kal_new_kf_officer,
           :cr_rate, :k_kontract_balance, :k_deliq_date_begin, :k_deliq_balance
  do
  begin
    if (curr_reg_id <> curr_reg_id_prior) then
    begin
      select r1.reg_code || ' ' || r1.reg_name, r2.reg_name
      from region r1, region r2
      where r1.reg_unicode = :curr_reg_id and
            r1.parent = r2.reg_unicode
      into :reg_name, :parent_name;
      curr_reg_id_prior = curr_reg_id;
    end
    if (kal_new_kf_officer_out <> kal_new_kf_officer_out_prior) then
    begin
      select c_name_fio
      from clients
      where c_id = :kal_new_kf_officer_out
      into :kf_officer_name_out;
      kal_new_kf_officer_out_prior = kal_new_kf_officer_out;
    end
    if (kal_new_kf_officer <> kal_new_kf_officer_prior) then
    begin
      select c_name_fio
      from clients
      where c_id = :kal_new_kf_officer
      into :kf_officer_name;
      kal_new_kf_officer_prior = kal_new_kf_officer;
    end

    if (:rep_date_end > :ks_date_end) then
      k_date_fact_end = :ks_date_end + 1;
    else
      k_date_fact_end = '31.12.9999';

    k_deliq_balance7 = 0;
    k_deliq_balance30 = 0;
    k_deliq_day = 0;
    k_schedule_date_end = null;
    k_disb_amount = 0;
    k_disb_kolich_client = 0;
    k_kolich_client_active = 0;

    if (rep_currency = '000') then
    begin
      k_kontract_balance = k_kontract_balance * cr_rate;
      k_deliq_balance = k_deliq_balance * cr_rate;
    end
    if (:k_kontract_balance > 0) then
      k_kolich_client_active = :k_kolich_client_approved;
    if (k_deliq_date_begin is not null) then
    begin
      k_deliq_day = cast((:rep_date_end - :k_deliq_date_begin) as integer);
      if (:k_deliq_day > 7) then
        k_deliq_balance7 = :k_kontract_balance;
      if (:k_deliq_day > 30) then
        k_deliq_balance30 = :k_kontract_balance;
    end

    if (:ks_date_begin between :rep_date_begin and rep_date_end) then
    /*Выданные кредиты за указанный период*/
    begin
      if (rep_currency = '000') then
        k_disb_amount = k_amount_approved * k_index_curr_rate;
      k_disb_kolich_client = :k_kolich_client_approved;
    end
    suspend;
  end

end
