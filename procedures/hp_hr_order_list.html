CREATE PROCEDURE HP_HR_ORDER_LIST(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_LINE_ID D_INTEGER,
    RES_OBJECT_KIND D_INTEGER,
    RES_OBJECT_ID D_ID,
    RES_DATE D_DATE,
    RES_ORDER_NO D_STR_20,
    RES_ORDER_TYPE D_INTEGER,
    RES_ORDER_TYPE_STR D_STR_100,
    RES_OFFICE D_STR_100,
    RES_STAFF_LIST D_STR_4096)
AS
declare variable reg_id d_integer;
begin
  res_object_kind = f_objkind('hr_orders', 'id');

  res_line_id = 0;

  for select ho_id, reg_id, staff_list
  from hp_hr_order_tmp(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_c_id, :un_id) p
  into :res_object_id, :reg_id, :res_staff_list
  do
  begin
    for select ho_date, ho_order_no, ho_order_type, sawt_name, reg_code || ' ' || reg_name
    from hr_orders, sa_worktype, region
    where ho_id = :res_object_id and
          reg_unicode = coalesce(:reg_id, ho_from_reg_id) and
          ho_order_type = sawt_id
    into res_date, res_order_no, res_order_type, res_order_type_str, res_office
    do
    begin
      res_line_id = res_line_id + 1;
      suspend;
    end
  end
end
