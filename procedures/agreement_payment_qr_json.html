CREATE PROCEDURE AGREEMENT_PAYMENT_QR_JSON(
    AG_ID D_INTEGER)
RETURNS (
    APQ_ID D_INTEGER,
    URL_DATA D_STR_8100)
AS
begin
  select apq.apq_id,
         apq.apq_url_data
  from agreement_payment_qr apq
  where apq.apq_from_ag_id = :ag_id
  and apq.apq_is_last = 1
  and apq.apq_status = f_enum_id('agreement_payment_qr_status', 'assigned')
  into :apq_id,
       :url_data;
  suspend;
end
