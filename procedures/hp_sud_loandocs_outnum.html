CREATE PROCEDURE HP_SUD_LOANDOCS_OUTNUM(
    SMD_ID D_ID,
    DOC_KIND D_INTEGER,
    DOC_DATETIME D_DATETIME)
RETURNS (
    RES_TEXT D_STR_CODE,
    RES_IS_ERROR D_BOOL)
AS
declare variable str_date d_str_20;
declare variable k_kontract_no d_str_20;
declare variable doc_code d_str_code;
declare variable num d_integer;
begin
  -- возвращает исходящий номер документа судебной заявки (RFCRU-4792)

  res_is_error = 1;
  res_text = 'Не найден входящий параметр doc_code';

  if (not exists(select *
                 from sud_loanclient
                 where slc_from_smr_id = :smd_id)) then
  begin
    res_text = 'Клиент судебной заявки не найден';
    suspend;
    exit;
  end

  if (doc_datetime is null) then
  begin
    res_text = 'Дата документа судебной заявки не указана';
    suspend;
    exit;
  end

  select scd_code
  from sud_cfgdockind
  where scd_id = :doc_kind
  into doc_code;

  if (coalesce(doc_code, '') = '') then
  begin
    res_text = 'Вид документа судебной заявки не найден';
    suspend;
    exit;
  end

  if (doc_code = 'pretrial_claim') then -- претензия досудебная
  begin
    select k_kontract_no
    from kontract
    inner join knclient on kc_from_k_id = k_id and kc_amount_approved > 0
    inner join sud_loanclient on slc_from_kc_id = kc_id
    where slc_from_smr_id = :smd_id -- smr_id и smd_id один и тот же код, т.е. smr_id = smd_id
          rows 1
    into k_kontract_no;

    select count(*) + 1
    from edoc_list
    where edl_from_edc_id = f_edoc_kind(:doc_code, 'id') and
          edl_object_kind = f_objkind('sud_mainreq', 'id') and
          edl_object_id = :smd_id and
          cast(edl_create_datetime as date) = cast(:doc_datetime as date)
    into num; -- порядковый номер в течение дня (на случай, если в один день отправлено более одной претензии)

    str_date = f_datetostr(doc_datetime, 'ddmmyyyy');
    res_text = left(str_date, 4) || right(str_date, 2);
    res_text = res_text || k_kontract_no || num;
    res_is_error = 0;
  end

  suspend;
end
