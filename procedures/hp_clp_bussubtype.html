CREATE PROCEDURE HP_CLP_BUSSUBTYPE(
    BS_ID D_INTEGER)
RETURNS (
    CBS_ID D_INTEGER,
    CBS_BUSSUBTYPE_ID D_INTEGER,
    CBS_LOAN_TARGET_ID D_INTEGER,
    TARGET_NAME D_STR_255,
    CBS_IS_DELETE D_BOOL,
    CBS_FROM_UN_ID D_ID,
    UN_FIO D_STR_100,
    CBS_DATETIME D_DATETIME)
AS
begin
  for
    select
      cbs_id,
      cbs_bussubtype_id,
      cbs_loan_target_id,
      rd_name,
      cbs_is_delete,
      cbs_from_un_id,
      un_fio,
      cbs_datetime
    from clp_bussubtype
      inner join refdata
        on rd_id = cbs_loan_target_id
      inner join username
        on un_id = cbs_from_un_id
    where cbs_bussubtype_id = :bs_id
  into
    :cbs_id,
    :cbs_bussubtype_id,
    :cbs_loan_target_id,
    :target_name,
    :cbs_is_delete,
    :cbs_from_un_id,
    :un_fio,
    :cbs_datetime
  do
  begin
    suspend;
  end
end
