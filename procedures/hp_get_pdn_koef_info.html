CREATE PROCEDURE HP_GET_PDN_KOEF_INFO(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    DISB_DATE D_DATE,
    PDN_CATEGORY D_INTEGER,
    DISB_AMOUNT D_CURRENCY)
AS
begin
  disb_date = rep_date_begin;

  while (disb_date <= rep_date_end) do
  begin
    for select iif(coalesce(pdn.kc_pdn_koef, 0) <= 50, 0, iif(coalesce(pdn.kc_pdn_koef, 0) <= 80, 1, 2)),
               sum(kc_amount_approved)
    from kontstate
         left join knclient kc on (kc.kc_from_k_id = ks_from_k_id and kc.kc_amount_approved > 0)
         left join knclient_pdn pdn on (kc.kc_id = pdn.kc_id and kcp_kind = 0)
    where ks_state = 1 and
          ks_ol_is_delete = 0 and
          ks_date_begin = :disb_date
    group by 1
    into pdn_category, disb_amount
    do
    begin
      if (disb_amount is null) then
        disb_amount = 0;

      suspend;
    end

    disb_date = disb_date + 1;
  end
end
