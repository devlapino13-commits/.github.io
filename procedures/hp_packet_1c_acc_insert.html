CREATE PROCEDURE HP_PACKET_1C_ACC_INSERT(
    PCA_GUID VARCHAR(50),
    PCA_CODE VARCHAR(50),
    PCA_NAME VARCHAR(255),
    PCA_OFFICE_NAME VARCHAR(255),
    PCA_CURRENCY_NAME VARCHAR(255),
    PCA_DATE_OPEN DATE,
    PCA_DATE_CLOSE DATE,
    PCA_BANK_ACCOUNT VARCHAR(50),
    PCA_BANK_CODE VARCHAR(50),
    PCA_CLIENT_TYPE VARCHAR(10),
    PCA_NON_RESIDENT VARCHAR(10),
    PCA_CLIENT_NAME VARCHAR(1024),
    PCA_CLIENT_BIRTH_DATE DATE,
    PCA_CLIENT_INN VARCHAR(50),
    PCA_CLIENT_PASS_NO VARCHAR(50),
    PCA_CLIENT_GUID VARCHAR(50))
AS
begin
  insert into packet_1c_acc (pca_guid, pca_code, pca_name, pca_office_name, pca_currency_name, pca_date_open,
                             pca_date_close, pca_bank_account, pca_bank_code, pca_client_type, pca_non_resident,
                             pca_client_name, pca_client_birth_date, pca_client_inn, pca_client_pass_no,
                             pca_client_guid)
  values (:pca_guid, :pca_code, :pca_name, :pca_office_name, :pca_currency_name, :pca_date_open, :pca_date_close,
          :pca_bank_account, :pca_bank_code, :pca_client_type, :pca_non_resident, :pca_client_name,
          :pca_client_birth_date, :pca_client_inn, :pca_client_pass_no, :pca_client_guid);
end
