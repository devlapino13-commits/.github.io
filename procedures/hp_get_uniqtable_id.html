CREATE PROCEDURE HP_GET_UNIQTABLE_ID(
    UKT_NAME VARCHAR(100))
RETURNS (
    UKT_ID INTEGER)
AS
begin
  /* -- Старый вариант, до 20.03.2023
  ukt_id = f_objkind(ukt_name, 'id');
  ukt_id = ukt_id - 3;
  */

  -- Новый вариант от 21.03.2023
  if (coalesce(ukt_name, '') = '') then
    exception e_default 'Empty ukt_name';

  -- Обозначие типов объектов в таблице Permissions
  ukt_id = case ukt_name
    when 'username' then 0
    when 'rolename' then 1
    when 'menulist' then 2
    when 'permactions' then 3
    when 'region' then 4
    when 'uxalias' then 5
  end;

  if (ukt_id is null) then
    exception e_default 'Undefined ukt_name: ' || ukt_name;

  suspend;
end
