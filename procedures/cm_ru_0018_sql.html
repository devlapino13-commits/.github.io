CREATE PROCEDURE CM_RU_0018_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare kf_client_pass_no varchar(30);
declare variable is_email smallint;
declare variable new_cl_count integer;
declare variable total_cl_count integer;
declare variable dopusk smallint;
declare variable last_k_id_prior id_type;
declare variable dostup_temp smallint;
declare variable kol_magration_card integer;
declare variable kol_cli integer;
declare variable kol_cli_last integer;
declare variable kc_from_c_id id_type;
declare variable kc_circle integer;
declare variable c_email varchar(100);
declare variable C_Birth_Country varchar(100);
declare variable C_Birth_City varchar(100);
declare variable C_CITIZEN integer;
declare variable CDS_DOC_DATE_END date;
declare variable is_patent smallint;
declare variable magration_card_begin date;
declare variable LAST_K_ID id_type;
declare variable max_delinq_day integer;
declare variable ks_date_end date;
declare variable declk_disb_date_last date;
declare variable k_disb_date_last date;
declare variable kol_delinq integer;
declare variable par_delinq_day integer;
declare variable pr_code varchar(20);
declare variable k_circle integer;
declare variable kf_region_id integer;
declare variable K_KOLICH_REPAY integer;
declare variable k_percent_pr perc_type;
declare variable k_percent PERC_TYPE;
declare variable k_kolich_lgota integer;

begin
-- Проверка каждого клиента в группе
  is_email = 0;
  new_cl_count = 0;
  total_cl_count = 0;
  dopusk = 0;  -- добавлено по приказу №04-02-15о/д от 2го февраля
               /*
                Проверить наличие следующих документов каждого клиента данного контракта, и если все соответствует,
                то пареметр нужно сделать dopusk = 1, тем самым в следующий раз не будет проверять.
                2.1.    Клиенты могут получить микрозаем при условии соответствия одному (любому) из следующих критериев: 
                a)    клиенты въехали в РФ после 1 января 2015г. и находятся в РФ не более 20 календарных дней на момент обращения за микрозаймом;
                b)    клиенты имеют один из следующих документов, необходимых для получения нового патента (на работу):
                      сертификат, подтверждающий знания клиентом русского языка, истории и основ законодательства России,
                      либо медицинскую справку (выданную не позднее 25 дней на момент обращения за займом) либо действующий
                      Полис (договор) добровольного медицинского страхования.
                */
  last_k_id_prior = 0; -- для вычисления, клиенты в том же составе группы берут или нет
  dostup_temp = 1;
  kol_magration_card = 0;
  kol_cli = 0;
  kol_cli_last = -1;
  for
  select kc_from_c_id, kc_circle, kc_amount_approved,
         c_name_fio, c_pass_no, --coalesce(c_email,''),
         coalesce(C_Birth_Country,''), coalesce(C_Birth_City,''), C_CITIZEN,
         pr_code, k_disb_date, k_circle, k_last_reg_id, k_kolich_repay, k_percent, k_kolich_lgota
  from knclient, clients, kontract, product
  where kc_from_k_id = :k_id
  and kc_amount_approved > 0
  and kc_from_c_id = c_id
  and kc_from_k_id = k_id
  and k_from_pr_id = pr_id
  and pr_product_kind = 1 -- проверка только групповых кредитов
  into kc_from_c_id, kc_circle, kc_amount,
       res_info_text, KF_CLIENT_PASS_NO, --c_email,
       C_Birth_Country, C_Birth_City, C_CITIZEN,
       pr_code, kf_disb_date, k_circle, kf_region_id, k_kolich_repay, k_percent, k_kolich_lgota
  do
  begin
    c_email = null;
    select c_email
    from AAREP_GET_CONTINFO('clients', :kc_from_c_id)
    into c_email;
    if (K_KOLICH_REPAY is null) then K_KOLICH_REPAY = 0;
    if (k_kolich_lgota is null) then k_kolich_lgota = 0;
    K_KOLICH_REPAY = K_KOLICH_REPAY - k_kolich_lgota;
    kol_cli = kol_cli + 1;
    if (dopusk = 0) then -- добавлено от 06.02.2015 Улан
    begin
      if (C_CITIZEN = 13) then
      begin
        dopusk = 1;
      end else
      begin
        CDS_DOC_DATE_END = null;
        is_patent = 0;
        select 
--              max(CDS_DOC_DATE_END)
                max(iif(pdt_code in ('work_permit', 'registration_rvp', 'registration_vnj'), CDS_DOC_DATE_END, null)), -- для проверки по пункту 2,2 №04-02-15
                max(iif(pdt_code = 'magration_card', CDS_DOC_DATE_BEGIN, null)), -- дата миграционной карты
                max(iif(pdt_code = 'work_patent',1,null))
        from CLIENTDOCS, PASSKIND, PASSDOCTYPE
        where CDS_FROM_C_ID = :kc_from_c_id
        and cds_from_pk_id = pk_id
        and pk_from_pdt_id = pdt_id
        and cds_is_delete = 0
        and :kf_disb_date between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END
        and (pdt_code in ('work_permit', -- Разрешение на работу
                          'registration_rvp', -- Разрешение на временное проживание (РВП)
                          'registration_vnj' , -- Вид на жительство (ВНЖ)
                          'magration_card'  -- Миграционная карта
                          )
             or (pdt_code = 'work_patent'  -- Патент выданный после 1 января
                 and CDS_DOC_DATE_BEGIN >= '01.01.2015')
             )
        into CDS_DOC_DATE_END, magration_card_begin, is_patent;
        if (CDS_DOC_DATE_END is null) then CDS_DOC_DATE_END = '01.01.1900';
        if (magration_card_begin is null) then magration_card_begin = '01.01.1900';
        if (is_patent is null) then is_patent = 0;
--        if (CDS_DOC_DATE_END >= K_KREDLINE_END) then
        if (CDS_DOC_DATE_END >= dateadd(month, 2, kf_disb_date) or is_patent = 1) then
        begin
          dopusk = 1;
        end
        if (magration_card_begin > '01.01.2015' and :kf_disb_date - magration_card_begin <= 20) then
        begin
          kol_magration_card = kol_magration_card + 1;
        end 
      end
    end 
    if (c_email is not null) then is_email = 1;
    if (KF_DISB_DATE >= '01.07.2014') then
    begin
        if (C_Birth_Country = '') then
        begin
          res_error_is_critical = 2;
          res_error_text = 'Страна рождения клиента не указан';
          suspend;
        end 
        if (C_CITIZEN = 13 and C_Birth_City = '') then -- для РФ гражданинов обязательно заполнять город/село рождения
        begin
          res_error_is_critical = 2;
          res_error_text = 'Город/Село рождения клиента не указан';
          suspend;
        end
       if (not exists(
            select * from addrclient, addrlist
            where adc_from_c_id = :kc_from_c_id
            and adc_from_adl_id = adl_id
            and adc_address_type in (1)
            and coalesce(adl_post_index, 0) <> 0)) then
       begin
         res_error_is_critical = 2;
         res_error_text = 'Почтовый индекс клиента не указан (для юридического адреса)';
         suspend;
       end 
       if (not exists(
            select * from addrclient, addrlist
            where adc_from_c_id = :kc_from_c_id
            and adc_from_adl_id = adl_id
            and adc_address_type in (2)
            and coalesce(adl_post_index, 0) <> 0)) then
       begin
         res_error_is_critical = 2;
         res_error_text = 'Почтовый индекс клиента не указан (для фактического адреса)';
         suspend;
       end
    end
    if (kc_amount < 10000) then
    begin
      res_error_is_critical = 1;
      res_error_text = 'Минимальная сумма займа одному клиенту - 10000,00 руб';
      suspend;
    end else
    if (kc_amount > 60000) then
    begin
      res_error_is_critical = 1;
      res_error_text = 'Максимальная сумма займа одному клиенту - 60000,00 руб';
      suspend;
    end
--  проверка последнего цикла клиента
    LAST_K_ID = null;
    max_delinq_day = 0;
    for
    select ks_from_k_id, ks_date_end, ks_date_begin
    from kontstate, knclient
    where ks_ol_is_delete = 0
    and ks_state = 1
    and ks_from_k_id = kc_from_k_id
    and kc_from_c_id = :KC_FROM_C_ID
    and kc_amount_approved > 0
    order by ks_date_begin desc
    rows 1
    into LAST_K_ID, ks_date_end, k_disb_date_last
    do
    begin
        max_delinq_day = 0;
        kol_delinq = null;
        select max(ok_ol_date - ok_sh_date), count(*) from operkred
        where ok_ol_is_active = 1
        and ok_ol_date <= :KF_DISB_DATE
        and ok_ol_date > ok_sh_date
        and ok_type < 7
        and ok_from_k_id = :LAST_K_ID
        and ok_os + ok_perc < 0
        and ok_from_c_id = 0
        into max_delinq_day, kol_delinq;
        if (max_delinq_day is null) then max_delinq_day = 0;
        if (kol_delinq is null) then kol_delinq = 0;
        if ((select rep_delinq_days from aarep_kont_delinq_status_new(:last_k_id,:KF_DISB_DATE)) > 1) then
        begin
          res_error_is_critical = 1;
          res_error_text = 'У клиента имеется просроченная задолженнсть!';
          suspend;
        end
    end
    if (LAST_K_ID is null) then LAST_K_ID = 0;  -- если LAST_K_ID > 0 то повторный клиент
    if (k_disb_date_last is null) then k_disb_date_last = current_date;
    if (k_disb_date_last < '21.04.2014') then par_delinq_day = 7;
    else par_delinq_day = 0;
--  если клиент не кредитовался более 6 мес. и на последнем цикле были просрочки более 14 денй то клиент будет считатся новым
    if (LAST_K_ID <> 0 and max_delinq_day > 14 and dateadd(month, -6, kf_disb_date) - 1 > ks_date_end) then
    begin
      LAST_K_ID = 0;
    end 
--  если группа не меняет состав то может получить без дополнительных справок
    if (last_k_id_prior = 0 and LAST_K_ID <> 0) then
    begin
      last_k_id_prior = LAST_K_ID;
    end 
    if (dostup_temp = 1) then
    begin
        if (LAST_K_ID = 0) then -- клиент новый и не может быть в том же составе группы
        begin
          dostup_temp = 0;
        end else
        if (last_k_id_prior <> LAST_K_ID) then -- если клиенты из разных групп то тоже не будет в том же составе
        begin
          dostup_temp = 0;
        end 
    end
    -- << igusev 12.03.2015 ----------------------------------------------------
    if (kol_cli_last = -1) then
    begin
      kol_cli_last = null;
      select count(*) from knclient
      where kc_from_k_id = :LAST_K_ID
      and kc_amount_approved > 0
      into kol_cli_last;
      if (kol_cli_last is null) then kol_cli_last = 0;
--    Приказ 04-02-15 о/д от 02.02.2015 пукнт 2.
      if (kol_cli_last > 0) then
      begin
        if (max_delinq_day > 14 or dateadd(month, -3, kf_disb_date) - 1 > ks_date_end) then
        begin
          dostup_temp = 0;
        end 
      end
    end
-- Тариф "Ассалам"
    if (pr_code containing '.ASALAM.') then
    begin
      if (LAST_K_ID = 0) then
      begin
        if (k_circle <= 1 and kc_amount > 30000) then -- если новая группа и клиент новый то клиент не может получать более 30 тыс.
        begin
          if (kc_amount <= 40000
              and exists(select * from region
                         where REG_LOGIC_LABEL containing '.mega_office.'
                         and reg_unicode = :kf_region_id)) then
          begin
              if (C_CITIZEN <> 13) then
              begin
                CDS_DOC_DATE_END = null;
                is_patent = 0;
                select max(CDS_DOC_DATE_END),
                       max(iif(pdt_code = 'work_patent',1,null))
                from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                where CDS_FROM_C_ID = :kc_from_c_id
                and cds_from_pk_id = pk_id
                and pk_from_pdt_id = pdt_id
                and cds_is_delete = 0
                and :kf_disb_date between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END
                and (pdt_code in ('work_permit', -- Разрешение на работу
                                  'registration_rvp', -- Разрешение на временное проживание (РВП)
                                  'registration_vnj' -- Вид на жительство (ВНЖ)
                                  )
                     or (pdt_code = 'work_patent'  -- Патент выданный после 1 января
                         and CDS_DOC_DATE_BEGIN >= '01.01.2015')
                     )
                into CDS_DOC_DATE_END, is_patent;
                if (CDS_DOC_DATE_END is null) then CDS_DOC_DATE_END = '01.01.1900';
                if (is_patent is null) then is_patent = 0;
                if (CDS_DOC_DATE_END < dateadd(month, 2, kf_disb_date) and is_patent = 0) then
                begin
                  res_error_is_critical = 1;
                  res_error_text = 'Клиент не может получить более 40тыс. не подлежит к условиям приказа №05-02-15о/д пукнт 1.';
                  suspend;
                end
              end
          end else
          begin
              res_error_is_critical = 1;
              res_error_text = 'Клиент новый и не имеет получать заем более 30000,00 руб';
              suspend;
          end
        end
      end
      k_percent_pr = case
                       when K_KOLICH_REPAY = 3 and (kc_amount = 10000 or kc_amount = 20000) then 8
                       when K_KOLICH_REPAY = 6 and (kc_amount = 20000 or kc_amount = 30000 or kc_amount = 40000) then 5.8
                       when K_KOLICH_REPAY = 9 and (kc_amount = 30000 or kc_amount = 40000) then 5.5
                       when K_KOLICH_REPAY = 12 and kc_amount = 40000 then 5
                       else 0
                     end;
      if (k_percent <> k_percent_pr or k_percent_pr = cast(0 as perc_type)) then
      begin
        res_error_is_critical = 1;
        res_error_text = 'Ежемесячная выплата клиента не соответствует к тарифу "Ассалам"';
        suspend;
      end 
    end
-- если клиент новый а группа повторная то клиент не может более 40 тыс получать
-- добавлено по письму Зары от 23.03.2015 : База пропускает нового клиента с 60 тыс. рублей. А должно быть максимум 40 тыс. в составе 
    if (LAST_K_ID = 0 and kc_amount > 40000) then
    begin
      res_error_is_critical = 1;
      res_error_text = 'Клиент новый и не имеет получать заем более 40000,00 руб';
      suspend;
    end
    total_cl_count=total_cl_count+1;
    if (Last_K_ID = 0) then
    begin
      new_cl_count = new_cl_count + 1;
    end
--  Кел кел
    if (pr_code containing '.KEL.') then
    begin
      if (max_delinq_day > 3) then  -- было if (max_delinq_day > par_delinq_day) then
      begin
        res_error_is_critical = 1;
        res_error_text = 'В предыдущем цикле клиента имеются просроченные дни, не может получить по тарифу "Кел кел"';
        suspend;
      end                   
      k_percent_pr = case
                       when K_KOLICH_REPAY = 3 and (kc_amount = 10000 or kc_amount = 20000) then 6.7
                       when K_KOLICH_REPAY = 6 and (kc_amount = 20000 or kc_amount = 30000 or kc_amount = 40000) then 4.9
                       when K_KOLICH_REPAY = 9 and (kc_amount = 30000 or kc_amount = 40000 or kc_amount = 50000 or kc_amount = 60000) then 3.8
                       when K_KOLICH_REPAY = 12 and (kc_amount = 40000 or kc_amount = 50000 or kc_amount = 60000) then 3.7
                       else 0
                     end;
      if (k_percent <> k_percent_pr or k_percent_pr = cast(0 as perc_type)) then
      begin
        res_error_is_critical = 1;
        res_error_text = 'Ежемесячная выплата клиента не соответствует к тарифу "Кел кел"';
        suspend;
      end
    end 
--  Тариф "Партнет"
    /* igusev 22.01.2015
       Для тарифа Партнер, был изменена проверка согласно письма от Зары от 21.01.2015
       >>
       Обращаюсь с просьбой изменить настройки Базы в отношении повторных клиентов, которые хотят получить заем в составе повторной (не своей) группы.
       Сейчас База настроена следующим образом: если группа проходит по всем параметрам по тарифу «Партнер», но присоединяет к себе клиента, который ранее получал в Компании, но не платил заранее и потому не проходит по тарифу «Партнер», то База не дает такой группе пройти по тарифу «Партнер».
       Повторные клиенты при получении займа в составе повторной (не своей) группы могут получить заем по тарифу этой группы – если у данных клиентов положительная кредитная история (нулевая просрочка на последнем предыдущем цикле);
       т.е. теперь реализована следующая проверка: В предыдущем цикле клиент получал заем в рамках текущей группы или нет.
       Если в рамках текущей, то для него действует проверка на досрочные погашения, если кредит был в рамках другой группы, то достаточно чтобы в предыдущем цикле у клиента было число дней просрочки  = 0
    */
    if (pr_code containing '.PARTNER.') then -- замена к предыдущей строке, изменен от 17.03.2015 Улан
    begin
      if (max_delinq_day > 0) then
      begin
        res_error_is_critical = 1;
        res_error_text = 'В предыдущем цикле клиента имеются просроченные дни, не может получить по тарифу "Партнер"';
        suspend;
      end 
      k_percent_pr = case
                       when K_KOLICH_REPAY = 3 and kc_amount = 20000 then 4
                       when K_KOLICH_REPAY = 6 and (kc_amount = 20000 or kc_amount = 30000 or kc_amount = 40000) then 3.5
                       when K_KOLICH_REPAY = 9 and (kc_amount = 30000 or kc_amount = 40000 or kc_amount = 50000 or kc_amount = 60000) then 3.2
                       when K_KOLICH_REPAY = 12 and (kc_amount = 40000 or kc_amount = 50000 or kc_amount = 60000) then 2.9
                       else 0
                     end;
      if (k_percent <> k_percent_pr or k_percent_pr = cast(0 as perc_type)) then
      begin
        res_error_is_critical = 1;
        res_error_text = 'Ежемесячная выплата клиента не соответствует к тарифу "Партнер"';
        suspend;
      end
    end 
  end
-- если кол-во соответствующих миграционок совпадает с кол-вом клентов то допускается к выдаче
-- или повторная группа в том же составе
  if (kol_magration_card = kol_cli) /* or kol_cli = kol_cli_last)*/ then  -- убран or kol_cli = kol_cli_last от 12.03.2015
  begin
    dopusk = 1;
  end
  if (dopusk = 0 and dostup_temp = 0) then -- добавлено от 06.02.2015 Улан
  begin
    res_error_is_critical = 0;
    res_error_text = '2. Отсутствуют необходимые документы. Приказ от 07.02.2015 N№04-02-15о/д, пункт 2.2!';
    suspend;
  end 
--  Если большинство клиентов в группе новые, они не могут получить по Кел-кел и Партнер
  if ((pr_code similar to '%(.KEL.|.PARTNER.)%') and (new_cl_count > (total_cl_count/2))) then -- замена к предыдущей строке, изменен от 17.03.2015 Улан
  begin
    res_error_is_critical = 1;
    res_error_text = 'Новая группа не может получать по этому продукту';
    suspend;
  end
  if (is_email = 0 and KF_DISB_DATE >= '01.07.2014') then
  begin
    res_error_is_critical = 1;
    res_error_text = 'В данной группе ни у одного клиента не имеются электронной почты';
    suspend;
  end 
end
