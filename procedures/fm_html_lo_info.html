    UN_ID DECIMAL(18,0),

    RES_HTML BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable RES_DATE_BEGIN date;
declare variable RES_PERIOD_ID integer;
declare variable AD_ACTIVE_CLIENT integer;
declare variable AD_ACTIVE_BALANCE decimal(18,4) = 0;
declare variable LO_FIO varchar(100);
declare variable LINE_STR varchar(1024);
declare variable LO_ID decimal(18,0);
declare variable N integer;
declare variable DELINQ_LIMIT integer;
declare variable DELIMITER_KIND integer;
declare variable DB_ID integer;
declare variable AD_BEGIN_DELINQ_LOAN integer;
declare variable AD_BEGIN_DELINQ_CLIENT integer;
declare variable AD_BEGIN_DELINQ_PAR decimal(18,4) = 0;
declare variable AD_BEGIN_DELINQ_LOAN30 integer;
declare variable AD_BEGIN_DELINQ_CLIENT30 integer;
declare variable AD_BEGIN_DELINQ_PAR30 decimal(18,4) = 0;
declare variable AD_ACTIVE_LOAN integer;
declare variable AD_DELINQ_LOAN2 integer;
declare variable AD_DELINQ_CLIENT2 integer;
declare variable AD_DELINQ_BALANCE_END2 decimal(18,4) = 0;
declare variable AD_DELINQ_BALANCE_DIFF2 decimal(18,4) = 0;
declare variable AD_DELINQ_PAR_END2 decimal(18,4) = 0;
declare variable AD_DELINQ_LOAN3 integer;
declare variable AD_DELINQ_CLIENT3 integer;
declare variable AD_DELINQ_BALANCE_END3 decimal(18,4) = 0;
declare variable AD_DELINQ_BALANCE_DIFF3 decimal(18,4) = 0;
declare variable AD_DELINQ_PAR_END3 decimal(18,4) = 0;
declare variable AD_DELINQ_LOAN4 integer;
declare variable AD_DELINQ_CLIENT4 integer;
declare variable AD_DELINQ_BALANCE_END4 decimal(18,4) = 0;
declare variable AD_DELINQ_BALANCE_DIFF4 decimal(18,4) = 0;
declare variable AD_DELINQ_PAR_END4 decimal(18,4) = 0;
declare variable AD_DISB_LOAN integer;
declare variable AD_DISB_CLIENT integer;
declare variable AD_DISB_AMOUNT decimal(18,4) = 0;
declare variable AD_DISB_NEW_LOAN integer;
declare variable AD_DISB_NEW_CLIENT integer;
declare variable AD_DISB_NEW_AMOUNT decimal(18,4) = 0;
declare variable AD_RETURN_OS_KOEF decimal(18,4) = 0;
declare variable AD_CLOSED_LOAN integer;
declare variable AD_CLOSED_CLIENT integer;
declare variable AD_DIFF_BALANCE decimal(18,4) = 0;
declare variable AD_DIFF_LOAN integer;
declare variable AD_DIFF_CLIENT integer;
declare variable AD7_BEGIN_DELINQ_LOAN integer;
declare variable AD7_BEGIN_DELINQ_CLIENT integer;
declare variable AD7_BEGIN_DELINQ_PAR decimal(18,4) = 0;
declare variable AD7_BEGIN_DELINQ_LOAN30 integer;
declare variable AD7_BEGIN_DELINQ_CLIENT30 integer;
declare variable AD7_BEGIN_DELINQ_PAR30 decimal(18,4) = 0;
declare variable AD7_DELINQ_BALANCE_DIFF2 decimal(18,4) = 0;
declare variable AD7_DELINQ_BALANCE_DIFF3 decimal(18,4) = 0;
declare variable AD7_DELINQ_BALANCE_DIFF4 decimal(18,4) = 0;
declare variable AD7_DISB_LOAN integer;
declare variable AD7_DISB_CLIENT integer;
declare variable AD7_DISB_AMOUNT decimal(18,4) = 0;
declare variable AD7_DISB_NEW_LOAN integer;
declare variable AD7_DISB_NEW_CLIENT integer;
declare variable AD7_DISB_NEW_AMOUNT decimal(18,4) = 0;
declare variable AD7_RETURN_OS_KOEF decimal(18,4) = 0;
declare variable AD7_CLOSED_LOAN integer;
declare variable AD7_CLOSED_CLIENT integer;
declare variable AD7_DIFF_BALANCE decimal(18,4) = 0;
declare variable AD7_DIFF_LOAN integer;
declare variable AD7_DIFF_CLIENT integer;
declare variable AD_RETENTION decimal(18,4) = 0;
declare variable AD7_RETENTION integer;
declare variable GROUP_KIND varchar(100);
declare variable RES_DATE_BEGINWEEK date;
declare variable KOL_KRED_BEGINWEEK integer;
declare variable SUM_KRED_BEGINWEEK CURR_TYPE;
declare variable KOL_KRED integer;
declare variable SUM_KRED CURR_TYPE;
declare variable AD7_DISB_LOAN_IPF integer;
declare variable AD7_DISB_CLIENT_IPF integer;
declare variable AD7_DISB_AMOUNT_IPF CURR_TYPE;
declare variable AD_DISB_LOAN_IPF integer;
declare variable AD_DISB_CLIENT_IPF integer;
declare variable AD_DISB_AMOUNT_IPF CURR_TYPE;
declare variable AD7_DISB_NEW_LOAN_IPF integer;
declare variable AD7_DISB_NEW_CLIENT_IPF integer;
declare variable AD7_DISB_NEW_AMOUNT_IPF CURR_TYPE;
declare variable AD_DISB_NEW_LOAN_IPF integer;
declare variable AD_DISB_NEW_CLIENT_IPF integer;
declare variable AD_DISB_NEW_AMOUNT_IPF CURR_TYPE;
declare variable CLI_OF_AGENT_MONTH integer;
declare variable CLI_OF_AGENT_DECADA integer;
declare variable AD_ACTIVE_BALANCE_IPF decimal(18,4) = 0;
declare variable AD_DIFF_BALANCE_IPF decimal(18,4) = 0;
declare variable AD7_DIFF_BALANCE_IPF decimal(18,4) = 0;
declare variable HTML_TEMPLATE D_BLOB_TEXT;
declare variable UN_NAME D_STR_CODE;
declare variable office_title d_str_255;
begin
  -- Расчет показателей портфеля СВЗ или МО

  -- A.S. Коллеги, максимально используйте конкатенацию при сборке res_html, это предотвращает распухание базы.
  -- Т.к. каждое присваивание res_html рождает новый временный блоб.

  db_id = f_get_db_id();

  if ((db_id in (39801, 41701)) and (rep_date < '01.01.2022')) then
  begin
    res_html = '<body></body>';
    suspend;
    exit;
  end

  delinq_limit = 999999;
  delimiter_kind = 0;

  if (db_id = 81001) then
  begin
    delinq_limit = 360;
    delimiter_kind = -1;
  end

  -- RFCRU-4360 Будет убрано после тестирования
  select un_name
  from username
  where un_id = :un_id
  into :un_name;

  if (coalesce(:reg_id, 0) = 0) then
  begin
    res_html = BLOB_APPEND(res_html, 'Регион не указан, обратитесь к тех. поддержке' || '<br>');
    suspend;
    exit;
  end

  office_title = (select reg_code || ' ' || reg_name from region where reg_unicode = :reg_id);

  if (un_name = 'fmtest') then
  begin
    select html_content
    from apple_test_data
    where report_type = 'lo_info'
    into :html_template;

    if (html_template is not null) then
    begin
      line_str = 'дата : ' || right('0' || extract(day from rep_date), 2) || '.' || right('0' || extract(month from rep_date), 2) || '.' || extract(year from rep_date);
      line_str = '<font color="black">' || :line_str || '</font>';
      res_html = blob_append(res_html, line_str || '<br>');
      res_html = blob_append(res_html, html_template);
      suspend;
      exit;
    end
    else
    begin
      res_html = blob_append(res_html, '<font color="red">Тестовый шаблон не найден в таблице apple_test_data</font><br>');
      suspend;
      exit;
    end
  end

  res_html = (select res_style from fm_html_styles) || '<body>';

  select un_from_c_id, c_name_fio
  from clients, username
  where un_id = :un_id and
        un_from_c_id = c_id
  into :lo_id, :lo_fio;

  if (lo_fio is null) then
    lo_fio = '';

  if (not exists(select ps_from_reg_id
                 from hr_pstatus, joblist, jobkind
                 where ps_from_c_id = :lo_id and
                       ps_date_begin <= :rep_date and
                       ps_date_end >= :rep_date and
                       ps_is_working = 1 and
                       ps_status = 1 and
                       ps_from_jl_id = jl_id and
                       jl_kind = jk_id and
                       jk_system_code = 'JobSystem.Region_KM')) then
  begin
    reg_id = 0;
    group_kind = 'lo_id';
  end
  else
  begin
    lo_id = 0;
    group_kind = 'reg_id';
  end

  line_str = f_format('<h1 class="green_text">%1</h1>', iif(db_id <> 94401, 'Показатели портфеля', 'Portfel göstəricisi'));
  res_html = BLOB_APPEND(res_html, line_str);
  if (db_id = 81001) then
    res_html = BLOB_APPEND(res_html, 'При расчете не учитываются просрочки свыше 360 дней<br>');
  res_html = BLOB_APPEND(res_html, '<br>');

  line_str = iif(db_id <> 94401, 'Дата', 'Tarix') || ': ' || iif(rep_date is not null, f_datetostr(rep_date), iif(db_id <> 94401, 'Не указана', 'Qeyd olunmayıb'));
  line_str = '<font color="black">' || :line_str || '</font>';
  res_html = BLOB_APPEND(res_html, line_str || '<br>');

  line_str = iif(db_id <> 94401, 'Офис', 'Ofis') || ': ' || office_title;
  line_str = '<font color="black">' || :line_str || '</font>';
  res_html = BLOB_APPEND(res_html, line_str || '<br>');

  res_html = BLOB_APPEND(res_html, lo_fio || '<br><br>');
  if (lo_fio = '') then
    exit;

  res_html = BLOB_APPEND(res_html, '<table width="100%"><col width="25%"><col width="25%"><col width="25%"><col width="25%"><tr>' ||
    f_format('<th>%1</th>', iif(db_id <> 94401, 'Показатели', 'Nəticələr')) ||
    f_format('<th>%1</th>', iif(db_id <> 94401, 'Данные на дату', 'Tarix üçün göstəricilər')) ||
    f_format('<th>%1</th>', iif(db_id <> 94401, 'С начала месяца', 'Ayın əvvəlindən')) || '<th>' ||
    iif(db_id <> 81001, iif(db_id <> 94401, 'С начала недели', 'Həftənin əvvəlindən'), 'С начала декады') || ' </th></tr>');

  -- Готовим данные с начала месяца
  res_date_begin = rep_date - extract(day from rep_date) + 1;
  execute procedure hp_analitica_create_period(:reg_id, :res_date_begin, :rep_date, 1, null, null, 30, :lo_id,
      'standart')
      returning_values :res_period_id;

  select ad_begin_delinq_loan, ad_begin_delinq_client, ad_begin_delinq_par, ad_begin_delinq_loan30,
         ad_begin_delinq_client30, ad_begin_delinq_par30, ad_active_loan, ad_active_client, ad_active_balance,
         ad_delinq_loan2, ad_delinq_client2, ad_delinq_balance_end2, ad_delinq_balance_diff2, ad_delinq_par_end2,
         ad_delinq_loan3, ad_delinq_client3, ad_delinq_balance_end3, ad_delinq_balance_diff3, ad_delinq_par_end3,
         ad_delinq_loan4, ad_delinq_client4, ad_delinq_balance_end4, ad_delinq_balance_diff4, ad_delinq_par_end4,
         ad_disb_loan, ad_disb_client, ad_disb_amount, ad_disb_new_loan, ad_disb_new_client, ad_disb_new_amount,
         ad_return_os_koef, ad_closed_loan, ad_closed_client, ad_diff_balance, ad_diff_loan, ad_diff_client
  from hp_analitica_report(:group_kind, 'tree_group', 8, :delinq_limit, 0, :delinq_limit, 8, :delinq_limit,
                           31, :delinq_limit, 1, :delinq_limit, :delinq_limit, :delimiter_kind, 'standart')
  where (res_field_1 = :lo_id or
        res_field_1 = :reg_id)
  into ad_begin_delinq_loan, ad_begin_delinq_client, ad_begin_delinq_par, ad_begin_delinq_loan30,
       ad_begin_delinq_client30, ad_begin_delinq_par30, ad_active_loan, ad_active_client, ad_active_balance,
       ad_delinq_loan2, ad_delinq_client2, ad_delinq_balance_end2, ad_delinq_balance_diff2, ad_delinq_par_end2,
       ad_delinq_loan3, ad_delinq_client3, ad_delinq_balance_end3, ad_delinq_balance_diff3, ad_delinq_par_end3,
       ad_delinq_loan4, ad_delinq_client4, ad_delinq_balance_end4, ad_delinq_balance_diff4, ad_delinq_par_end4,
       ad_disb_loan, ad_disb_client, ad_disb_amount, ad_disb_new_loan, ad_disb_new_client, ad_disb_new_amount,
       ad_return_os_koef, ad_closed_loan, ad_closed_client, ad_diff_balance, ad_diff_loan, ad_diff_client;

  if (db_id in (39801, 41701)) then
  begin
    select ad_disb_loan, ad_disb_client, ad_disb_amount, ad_disb_new_loan, ad_disb_new_client, ad_disb_new_amount
    from hp_analitica_report(:group_kind || ', kind_of_lending', 'tree_group', 8, :delinq_limit, 0, :delinq_limit,
                             8, :delinq_limit, 31, :delinq_limit, 1, :delinq_limit, :delinq_limit, :delimiter_kind, 'standart')
    where (res_field_1 = :lo_id or
          res_field_1 = :reg_id) and
          res_field_2 = 'ИПФ'
    into ad_disb_loan_ipf, ad_disb_client_ipf, ad_disb_amount_ipf, ad_disb_new_loan_ipf, ad_disb_new_client_ipf,
         ad_disb_new_amount_ipf;

    select ad_active_balance,  ad_diff_balance
    from hp_analitica_report(:group_kind || ', kind_of_lending', 'tree_group', 8, :delinq_limit, 0, :delinq_limit,
                             8, :delinq_limit, 31, :delinq_limit, 1, :delinq_limit, :delinq_limit, :delimiter_kind, 'standart')
    where (res_field_1 = :lo_id or
          res_field_1 = :reg_id) and
          res_field_2 = 'ИПФ'
    into:ad_active_balance_ipf,ad_diff_balance_ipf;

  end

  if (db_id = 81001) then
  begin
    cli_of_agent_month = null;
    select count(ks.ks_from_k_id)
    from kontstate ks
    inner join operagr o on ogr_from_ol_id = ks_from_ol_id
    inner join agreements ag on ag.ag_id = o.ogr_from_ag_id
    inner join refdata r on r.rd_id = ag.ag_from_rd_id
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          ogr_sign = 1 and
          ks_date_begin between :res_date_begin and :rep_date and
          r.rd_code = 'AgreementKinds.loyal_agreement' and
          ks.ks_region_id = :reg_id
    into :cli_of_agent_month;
  end

  ad_retention = 100.0;
  if (ad_closed_client <> 0) then
    ad_retention = (ad_disb_client - ad_disb_new_client) * 100.0 / ad_closed_client;

  -- Очищаем данные из памяти
  delete from temp_analitica_period;

  -- Готовим данные с начала недели
  if (db_id <> 81001) then
  begin
    n = extract(weekday from rep_date);
    if (n = 0) then
      n = 7;
    res_date_begin = rep_date - n + 1;
  end
  else -- RF
  begin
    -- Готовим данные с начала декады
    n = extract(day from rep_date);
    if (n <= 10) then
      res_date_begin = replace('01.' || iif(extract(month from rep_date) < 10, '0', '') || extract(month from rep_date) || '.' || extract(year from rep_date), ' ', '');
    else
    if (n between 11 and 20) then
      res_date_begin = replace('11.' || iif(extract(month from rep_date) < 10, '0', '') || extract(month from rep_date) || '.' || extract(year from rep_date), ' ', '');
    else
      res_date_begin = replace('21.' || iif(extract(month from rep_date) < 10, '0', '') || extract(month from rep_date) || '.' || extract(year from rep_date), ' ', '');
  end

  execute procedure hp_analitica_create_period(:reg_id, :res_date_begin, :rep_date, 1, null, null, 30, :lo_id,
      'standart')
      returning_values :res_period_id;

  select ad_begin_delinq_loan, ad_begin_delinq_client, ad_begin_delinq_par, ad_begin_delinq_loan30,
         ad_begin_delinq_client30, ad_begin_delinq_par30, ad_delinq_balance_diff2, ad_delinq_balance_diff3,
         ad_delinq_balance_diff4, ad_disb_loan, ad_disb_client, ad_disb_amount, ad_disb_new_loan, ad_disb_new_client,
         ad_disb_new_amount, ad_return_os_koef, ad_closed_loan, ad_closed_client, ad_diff_balance, ad_diff_loan, ad_diff_client

  from hp_analitica_report(:group_kind, 'tree_group', 8, :delinq_limit, 0, :delinq_limit, 8, :delinq_limit, 31, :delinq_limit, 1, :delinq_limit, :delinq_limit, :delimiter_kind, 'standart')
  where (res_field_1 = :lo_id or
        res_field_1 = :reg_id)
  into ad7_begin_delinq_loan, ad7_begin_delinq_client, ad7_begin_delinq_par, ad7_begin_delinq_loan30,
       ad7_begin_delinq_client30, ad7_begin_delinq_par30, ad7_delinq_balance_diff2, ad7_delinq_balance_diff3,
       ad7_delinq_balance_diff4, ad7_disb_loan, ad7_disb_client, ad7_disb_amount, ad7_disb_new_loan,
       ad7_disb_new_client, ad7_disb_new_amount, ad7_return_os_koef, ad7_closed_loan, ad7_closed_client,
       ad7_diff_balance, ad7_diff_loan, ad7_diff_client;

  if (db_id in (39801, 41701)) then
  begin
    select ad_disb_loan, ad_disb_client, ad_disb_amount, ad_disb_new_loan, ad_disb_new_client, ad_disb_new_amount,ad_diff_balance
    from hp_analitica_report(:group_kind || ', kind_of_lending', 'tree_group', 8, :delinq_limit, 0, :delinq_limit, 8, :delinq_limit, 31, :delinq_limit, 1, :delinq_limit, :delinq_limit, :delimiter_kind, 'standart')
    where (res_field_1 = :lo_id or
          res_field_1 = :reg_id) and
          res_field_2 = 'ИПФ'
    into ad7_disb_loan_ipf, ad7_disb_client_ipf, ad7_disb_amount_ipf, ad7_disb_new_loan_ipf, ad7_disb_new_client_ipf,
         ad7_disb_new_amount_ipf, ad7_diff_balance_ipf;
  end

  if (db_id = 81001) then
  begin
    cli_of_agent_decada = null;
    select count(ks.ks_from_k_id)
    from kontstate ks
    inner join operagr o on ogr_from_ol_id = ks_from_ol_id
    inner join agreements ag on ag.ag_id = o.ogr_from_ag_id
    inner join refdata r on r.rd_id = ag.ag_from_rd_id
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          ogr_sign = 1 and
          ks_date_begin between :res_date_begin and :rep_date and
          r.rd_code = 'AgreementKinds.loyal_agreement' and
          ks.ks_region_id = :reg_id
    into :cli_of_agent_decada;
  end

  ad7_retention = 100.0;
  if (ad7_closed_client <> 0) then
    ad7_retention = (ad7_disb_client - ad7_disb_new_client) * 100.0 / ad7_closed_client;

  res_html = BLOB_APPEND(res_html,
    f_html_table_row(f_format('<b>%1</b>', iif(:db_id <> 94401, 'Активные займы', 'Aktiv kreditlər')), '', '', '', -1, 1, 1, 1) ||
    f_html_table_row(iif(:db_id <> 94401, 'Клиенты', 'Müştərilər'), :ad_active_client, :ad_diff_client, :ad7_diff_client, -1, 1, 1, 1) ||
    f_html_table_row(iif(:db_id <> 94401, 'Кредиты', 'Kreditlər'), :ad_active_loan, :ad_diff_loan, :ad7_diff_loan, -1, 1, 1, 1) ||
    f_html_table_row(iif(:db_id <> 94401, 'Баланс', 'Balans'), round(:ad_active_balance, 0), round(:ad_diff_balance, 0), round(:ad7_diff_balance, 0), -1, 1, 1, 1) ||
    iif(:db_id in (39801, 41701), f_html_table_row('Баланс классика', round(:ad_active_balance - :ad_active_balance_ipf, 0), round(:ad_diff_balance - :ad_diff_balance_ipf, 0), round(:ad7_diff_balance -:ad7_diff_balance_ipf, 0), -1, 1, 1, 1), '') ||
    iif(:db_id in (39801, 41701), f_html_table_row('Баланс ИПФ', round(:ad_active_balance_ipf, 0), round(:ad_diff_balance_ipf, 0), round(:ad7_diff_balance_ipf, 0), -1, 1, 1, 1),'') ||
    iif(db_id <> 94401, f_html_table_row('PAR 1', round(:ad_delinq_par_end4, 2) || '%', round(:ad_delinq_par_end4 - :ad_begin_delinq_par, 2) || '%', round(:ad_delinq_par_end4 - :ad7_begin_delinq_par, 2) || '%', -1, 1, 1, 1), '') ||
    f_html_table_row('PAR 7', round(:ad_delinq_par_end2, 2) || '%', round(:ad_delinq_par_end2 - :ad_begin_delinq_par, 2) || '%', round(:ad_delinq_par_end2 - :ad7_begin_delinq_par, 2) || '%', -1, 1, 1, 1) ||
    f_html_table_row('PAR 30', round(:ad_delinq_par_end3, 2) || '%', round(:ad_delinq_par_end3 - :ad_begin_delinq_par30, 2) || '%', round(:ad_delinq_par_end3 - :ad7_begin_delinq_par30, 2) || '%', -1, 1, 1, 1) ||
    f_html_table_row(f_format('%1<wbr>%2', trim(iif(:db_id <> 94401, 'Возврат', 'Əsas məbl')), trim(iif(:db_id <> 94401, 'ность ОС', 'qayıdışı'))), round(:ad_return_os_koef, 2) || '%', round(:ad_return_os_koef, 2) || '%', round(:ad7_return_os_koef, 2) || '%', -1, 1, 1, 1) ||
