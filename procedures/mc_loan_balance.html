CREATE PROCEDURE MC_LOAN_BALANCE(
    MCL_ID ID_TYPE,
    K_ID ID_TYPE)
RETURNS (
    SH_DATE D_STR_10,
    DEBT_TOTAL D_INTEGER,
    IS_DELINQ D_SMALLINT)
AS
declare variable sh_date_tmp d_date;
begin
  for select sh_date, round(zad_os + sh_calc_perc + sh_calc_fine)
      from aarep_make_zad_list(:k_id, current_date, 1)
      into :sh_date_tmp, :debt_total
  do
  begin
    is_delinq = 0;
    if (coalesce(sh_date_tmp, '01.01.1999') < current_date) then
      is_delinq = 0;
    sh_date = f_datetostr(sh_date_tmp);

    suspend;
  end
end
