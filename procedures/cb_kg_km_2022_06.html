CREATE PROCEDURE CB_KG_KM_2022_06(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE,
    REP_C_ID D_BLOB_TEXT = 0)
RETURNS (
    DELINQ_BALANCE_7 CURR_TYPE,
    BALANCE_BEGIN CURR_TYPE,
    ZNACH_PAR7 CURR_TYPE,
    BONUS_FACTOR CURR_TYPE,
    KOL_SSVZ INTEGER,
    FIO_SSVZ VARCHAR(1024),
    REG_NAME VARCHAR(100),
    BONUS_KM INTEGER,
    BONUS_KM_COMMENT CURR_TYPE,
    SUM_VOZNOG_OFFICE CURR_TYPE,
    REPAY_SUM_NOT_OS CURR_TYPE,
    DELINQ_OS_PERIOD CURR_TYPE,
    REPAY_KOMMIS_IPF CURR_TYPE,
    CONSULT_SUMMA CURR_TYPE,
    PLAN_SPF INTEGER,
    FACT_SPF INTEGER,
    AAREP_BONUS_MEMO VARCHAR(50),
    USING_SCORES D_INTEGER,
    C_ID D_ID)
AS
declare variable rep_date_begin d_date;
declare variable reg_id d_integer;
declare variable rep_period d_integer;
declare variable diff_spf d_integer;
declare variable bonus_memo d_str_255;
declare variable km_list d_str_long;
declare variable rd_id_ipf d_integer;
declare variable rd_id_cons d_integer;
declare variable tmp_sum d_currency;
declare variable rd_id d_integer;
declare variable sql_svz d_str_2048;
declare variable reg_km_jl_id_list d_str_long;
declare variable sawt_id d_integer;
declare variable stage d_integer;
declare variable first_work_date d_date;
begin

  rep_date_begin = rep_date_end - extract(day from :rep_date_end) + 1;
  rep_period = extract(year from rep_date_end) || right((extract(month from rep_date_end) + 100), 2);
  if (rep_c_id not containing '.') then
    rep_c_id = rep_c_id || '.';

  select rd_id
  from refdata
  where rd_code = 'cashtransactionincom.ipfcommission'
  into :rd_id_ipf;

  select rd_id
  from refdata
  where rd_code = 'cashtransactionincom.consultation'
  into :rd_id_cons;

  select '.' || list(ps.ps_from_c_id, '.') || '.'
  from hr_pstatus ps
  where ps.ps_is_working = 1 and
        ps.ps_from_jl_id = 140
  into :km_list;

  select list(jl.jl_id, ', ')
  from joblist jl
  inner join jobkind jk on jl.jl_kind = jk.jk_id and
        jk.jk_system_code = 'jobsystem.region_km'
  into :reg_km_jl_id_list;

  for select oks.oks_region_id, sum(iif(:rep_date_end - oks.oks_delinq_date > 7, oks.oks_balance, 0))
      from oksummary oks
      inner join region_list(:rep_region_id, 1, 3, 1) t on t.id_list = cast(oks.oks_region_id as d_integer)
      where :rep_date_end between oks.oks_date_begin and oks.oks_date_end and
            oks.oks_balance > 0 and
            oks.oks_close_state = 0 and
            :km_list not containing '.' || oks.oks_kc_id || '.'
      group by oks.oks_region_id
      into reg_id, delinq_balance_7
  do
  begin
    --  балан кп и баланс par7 на начало
    balance_begin = null;
    select sum(oks_balance)
    from oksummary
    where oks_region_id = :reg_id and
          :rep_date_begin - 1 between oks_date_begin and oks_date_end and
          oks_close_state = 0 and
          :km_list not containing '.' || oks_kc_id || '.'
    into balance_begin;
    balance_begin = coalesce(balance_begin, 0);
    --  значение par7* в % от кп
    if (balance_begin = 0 and
        delinq_balance_7 > 0) then
      znach_par7 = 100;
    else
    if (delinq_balance_7 = 0) then
      znach_par7 = 0;
    else
      znach_par7 = cast(delinq_balance_7 as double precision) / balance_begin * 100;
    znach_par7 = round(znach_par7, 2);

    --  суммы погашении процентов пеня и админ сбор
    repay_sum_not_os = null;
    select -sum(ok_perc + ok_fine + ok_kommis)
    from operkred op
    inner join kontract kt on kt.k_id = op.ok_from_k_id
    inner join knaktlist kal on kal.kal_from_kf_id = kt.k_from_kf_id
    where ok_ol_is_active = 1 and
          ok_type = 4 and
          ok_ol_reg_id = :reg_id and
          ok_ol_date between :rep_date_begin and :rep_date_end and
          kal.is_delete = 0 and
          :rep_date_end between kal.kal_date and kal.kal_date_end and
          :km_list not containing '.' || kal.kal_new_kf_officer || '.'
    into repay_sum_not_os;
    repay_sum_not_os = coalesce(repay_sum_not_os, 0);

    repay_kommis_ipf = 0;
    consult_summa = 0;
    tmp_sum = 0;

    for select sum(ok.ok_amount), ok_operation_type
        from operkontract ok
        inner join operlist ol on ol.ol_id = ok.ok_from_ol_id
        inner join kontract kon on kon.k_id = ok.ok_from_k_id
        inner join knaktlist kal on kal.kal_from_kf_id = kon.k_from_kf_id
        where ok_operation_type in (:rd_id_ipf, :rd_id_cons) and
              ol.is_delete = 0 and
              ol_from_reg_id = :reg_id and
              ol_date between :rep_date_begin and :rep_date_end and
              kal.is_delete = 0 and
              :rep_date_end between kal.kal_date and kal.kal_date_end and
              :km_list not containing '.' || kal.kal_new_kf_officer || '.'
        group by ok_operation_type
        into :tmp_sum, :rd_id

    do
    begin
      if (rd_id = :rd_id_ipf) then
        repay_kommis_ipf = :tmp_sum;
      if (rd_id = :rd_id_cons) then
        consult_summa = :tmp_sum;
    end

    --  сумма просроченной задолженности по ос за отчетный период
    delinq_os_period = null;
    select sum(ok_os + ok_sh_os)
    from oksummary, operkred
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_region_id = :reg_id and
          :rep_date_end >= oks_delinq_date and
          oks_from_k_id = ok_from_k_id and
          oks_close_state = 0 and
          ok_ol_is_active = 1 and
          ok_type < 7 and
          ok_sh_date between :rep_date_begin and :rep_date_end and
          ok_ol_date <= :rep_date_end and
          :km_list not containing '.' || oks_kc_id || '.'
    into delinq_os_period;
    delinq_os_period = coalesce(delinq_os_period, 0);
    --  сумма вознаграждения подотчетного офиса
    sum_voznog_office = (repay_sum_not_os + repay_kommis_ipf + consult_summa) * 10 / 100 - delinq_os_period * 50 / 100;

    --  бонус фактор
    if (rep_date_end <= '31.03.2019') then
    begin
      bonus_factor = case
                       when znach_par7 = 0.0 then 1.20
                       when znach_par7 <= 0.5 then 1.00
                       when znach_par7 <= 1.0 then 0.90
                       when znach_par7 <= 1.5 then 0.50
                       else 0
                     end;
    end
    else
    begin
      bonus_factor = case
                       when znach_par7 <= 0.2 then 1.20
                       when znach_par7 <= 0.6 then 1.00
                       when znach_par7 <= 0.9 then 0.90
                       when znach_par7 <= 1.3 then 0.70
                       when znach_par7 <= 1.7 then 0.50
                       when znach_par7 <= 2.1 then 0.35
                       when znach_par7 <= 2.5 then 0.20
                       else 0
                     end;
    end
    kol_ssvz = 0;
    fio_ssvz = '';

    sql_svz = f_format('select ps.ps_from_c_id, count(*)
                        from hr_pstatus ps
                        where ps.ps_status = 1 and
                              ps.ps_is_working = 1 and
                              ps.ps_from_jl_id in (%3) and
                              ''%1'' between ps.ps_date_begin and ps.ps_date_end and
                              ps.ps_from_reg_id = %2 and coalesce((select sawt_id
                                                                   from hr_voctravel hrv
                                                                   inner join hr_orders ho on hrv.hrv_from_ho_id = ho.ho_id
                                                                   inner join sa_worktype sawt on ho.ho_order_type = sawt.sawt_id
                                                                   where hrv.hrv_from_c_id = ps.ps_from_c_id and
                                                                   ''%1'' between hrv_date_begin and hrv_date_end_fact
                                                                   order by hrv_date_begin desc
                                                                   rows 1),0) not in (16, 17, 18, 8) %4
                        group by 1', :rep_date_end, :reg_id, :reg_km_jl_id_list, iif(:rep_c_id <> '0.', ' and ps.ps_from_c_id in (' ||(select list(value_out, ', ')
                                                                                                                                       from str_with_separator_parser(:rep_c_id, '.')) || ')', ''));
    for execute statement sql_svz
            into :c_id, :kol_ssvz
    do
    begin
      stage = 0;
      select c.c_name_fio
      from clients c
      where c.c_id = :c_id
      into :fio_ssvz;

      if (kol_ssvz is null) then
        kol_ssvz = 0;

      first_work_date = null;
      select ps.ps_date_begin
      from hr_pstatus ps
      where ps.ps_from_c_id = :c_id and
            ps.ps_is_working = 1 and
            ps.ps_status = 1
      order by ps.ps_date_begin
      rows 1
      into first_work_date;

      stage = datediff(month, first_work_date, :rep_date_end);
    end

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :reg_id
    into reg_name;

    --  план спф
    plan_spf = null;
    select prs_plan_spf
    from planregionspf
    where prs_period = :rep_period and
          prs_kind = 1 and
          prs_region = :reg_id
    into plan_spf;
    plan_spf = coalesce(plan_spf, 0);

    /*    --  факт спф
    fact_spf = null;
    select count(distinct oks_kc_id)
    from oksummary
    where oks_region_id = :reg_id and
          :rep_date_end between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_close_state = 0
    into fact_spf;
    fact_spf = coalesce(fact_spf, 0);   */

    fact_spf = 0;
    select count(*)
    from aarep_personal_list(:reg_id, :rep_date_end, 1)
    where jl_name in ('специалист по финансированию', 'старший специалист по финансированию') and
          ps_curr_status not in ('отпуск без содерж.по уходу за ребенком', 'отпуск по беременности и родам', 'трудовой отпуск декретницы')
    into fact_spf;
    fact_spf = coalesce(fact_spf, 1);

    diff_spf = plan_spf - fact_spf;
    if (diff_spf < 0) then
      diff_spf = 0;

    bonus_km = sum_voznog_office * 0.15 * bonus_factor;

    if (bonus_km < 0) then
      bonus_km = 0;

    --  штраф 10% с  бонуса к начислению за каждого недостающего спф относительно планового показателя количества спф на конец месяца.
    if (bonus_km > 0 and
        diff_spf > 0) then
    begin
      if (diff_spf > 10) then
        diff_spf = 10;
      if (rep_period <> 202004) then
        bonus_km = bonus_km - bonus_km * diff_spf * 10 / 100;
    end

    bonus_km_comment = 0;
    bonus_memo = '';

    select bonus_memo
    from hp_get_vigovor_koef(:c_id, :rep_date_end)
    where bonus_memo <> 'нет замечаний и выговоров'
    into bonus_memo;

    select bonus_fact, using_scores
    from hp_calc_rebuke_bonus(:bonus_km, :c_id, current_date)
    into :bonus_km_comment, :using_scores;

    aarep_bonus_memo = bonus_memo;

    if (kol_ssvz = 0) then
      using_scores = 0;

    if (stage <= 3) then
    begin
      bonus_km_comment = 0;
      aarep_bonus_memo = 'Стаж меньще 3х месяцев';
    end 
    /*
    if (bonus_km > 0 and
        bonus_memo <> '') then
    begin
      bonus_km_comment = bonus_km * bonus_koef / 100 - bonus_sum;
      if (bonus_km_comment < 0) then
        bonus_km_comment = 0;
    end
    aarep_bonus_memo = bonus_memo;
    if (bonus_memo = '') then
      bonus_km_comment = bonus_km;
    */
    suspend;
  end
end
