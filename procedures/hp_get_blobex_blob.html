CREATE PROCEDURE HP_GET_BLOBEX_BLOB(
    BLS_ID D_ID)
RETURNS (
    BLX_BLOB BLOB SUB_TYPE BINARY SEGMENT SIZE 80)
AS
declare variable blf_id d_integer;
declare variable blf_filename d_str_255;
declare variable is_actual d_smallint;
begin
  blx_blob = null;
  -- По переданному в процедуру bls_id выясняем где искать blob
  -- Сначала ищем в таблице архивных файлов blobfiles
  -- Если начнется перекрытие диапазонов, то будет ругаться на multiple rows
  -- Если будет ругаться, нужно скорректировать blf_first_id, blf_last_id

  select blf_id, blf_filename from blobfiles
  where :bls_id between blf_first_id and blf_last_id
  into blf_id, blf_filename;

  is_actual = 0;
  if (blf_id = 0 and exists(select * from hp_db_is_actual)) then
    is_actual = 1;

  if (blf_filename is not null) then
  begin
    -- Для всех blx, кроме текущей делаем запрос через указание ip-адреса сервера
    -- Исключения: КГ-база (были тормоза), тестовые базы, т.к. у них нет blx
    -- Внимание: Когда указан ip-адрес, execute statement не видит blobы вставленные в текущей транзакции
    if (blf_id <> 0 or f_get_db_id() in (39801, 41701) or is_actual = 0) then
      blf_filename = f_db_blob_name(bls_id);

    execute statement ('select bls_blob from blobextore where bls_id = :bls_id')
    (bls_id := :bls_id)
    on external data source blf_filename
    as user 'UBLOBSTORE' password 'qJdCv55GcYpnwqBf'
    into blx_blob;
  end

  -- Далее только для тестовой БД
  -- Тестовые блобы кладутся только в blobstore,
  -- поэтому в blobstore может оказаться блоб "новее"
  if (is_actual = 0) then
  begin
    select bls_blob from blobstore
    where bls_id = :bls_id
      and bls_blob is not null
    into blx_blob;
  end
  suspend;
end
