CREATE PROCEDURE AAREP_ANALITICA_PORT_OWNS(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_PROJECT VARCHAR(100),
    REP_PERIOD_KIND INTEGER)
RETURNS (
    AA_CURR_OFFICE_NAME VARCHAR(100),
    AA_CURR_BRANCH_NAME VARCHAR(100),
    AA_CURR_LEVEL_NAME VARCHAR(100),
    AA_CURR_KC_FIO VARCHAR(100),
    AA_KOLICH_OWN INTEGER,
    AA_AMOUNT_OWN DECIMAL(18,4),
    AA_KOLICH_OTHER INTEGER,
    AA_AMOUNT_OTHER DECIMAL(18,4),
    AA_KOLICH_ALL INTEGER,
    AA_AMOUNT_ALL DECIMAL(18,4))
AS
begin
  for select aa_curr_office_name, aa_curr_branch_name, aa_curr_level_name, aa_curr_kc_fio, sum(aa_kolich_own),
             sum(aa_amount_own), sum(aa_kolich_other), sum(aa_amount_other)
      from aarep_analitica_port_owns_tmp(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_project, :rep_period_kind)
      group by aa_curr_office_name, aa_curr_branch_name, aa_curr_level_name, aa_curr_kc_fio
      into :aa_curr_office_name, :aa_curr_branch_name, :aa_curr_level_name, :aa_curr_kc_fio, :aa_kolich_own,
           :aa_amount_own, :aa_kolich_other, :aa_amount_other
  do
  begin
    aa_kolich_all = aa_kolich_own + aa_kolich_other;
    aa_amount_all = aa_amount_own + aa_amount_other;
    suspend;
  end

end
