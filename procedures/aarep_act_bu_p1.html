CREATE PROCEDURE AAREP_ACT_BU_P1(
    REP_DATE_END DATE)
RETURNS (
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    ZAD_OS CURR_TYPE,
    SH_CALC_PERC CURR_TYPE,
    SH_DATE DATE,
    REPAY_DATE DATE,
    EXPIRATION_DATE DATE,
    IS_OPEREXTRA_STR VARCHAR(10))
AS
declare variable is_operextra smallint;
declare variable k_id decimal(18,0);
begin

  for select iif(k_base_k_id is null, k_id, k_base_k_id), ks_date_begin, max(iif(exists(select *
                                                                                        from operextra
                                                                                        where ox_object_kind = 9 and
                                                                                              ox_ol_is_active = 1 and
                                                                                              ox_amount_plus > 0 and
                                                                                              ox_ol_date <= :rep_date_end and
                                                                                              ox_from_rd_id = 6336 and
                                                                                              ox_object_id in (k_id, k_base_k_id)), 1, 0))
      from kontstate, kontract
      where ks_ol_is_delete = 0 and
            (:rep_date_end between ks_date_begin and ks_date_end or
            ks_date_end = :rep_date_end - 1) and
            ks_state = 1 and
            ks_date_begin < '28.02.2012' and
            ks_from_k_id = k_id
--  and k_id = 101010000046200101
      group by 1, 2
      into k_id, k_disb_date, is_operextra
  do
  begin
    select k_kontract_no
    from kontract
    where k_id = :k_id
    into k_kontract_no;
    /*
    if (exists(select * from operextra
               where ox_object_kind = 9
               and ox_ol_is_active = 1
               and ox_amount_plus > 0
               and ox_ol_date <= :rep_date_end
               and ox_from_rd_id = 6336
               and ox_object_id = :k_id
               )) then is_operextra = 1;
    else is_operextra = 0;
    */
    if (is_operextra = 1) then
      is_operextra_str = 'Да';
    else
      is_operextra_str = 'Нет';

    for select sh_date, sum(zad_os), sum(sh_calc_perc)
        from aarep_act_bu_p1_temp(:k_id, :rep_date_end)
        group by sh_date
        into sh_date, zad_os, sh_calc_perc
    do
    begin
      repay_date = null;
      select max(ok_ol_date)
      from operkred, kontract
      where ok_ol_is_active = 1 and
            ok_sh_date = :sh_date and
            ok_ol_date < :rep_date_end and
            (ok_os < 0 or
            ok_perc < 0) and
            ok_type = 4 and
            ok_from_k_id = k_id and
            (k_id = :k_id or
            k_base_k_id = :k_id)
      into repay_date;

      if (is_operextra = 1) then
      begin
        expiration_date = null;
      end
      else
      begin
        if (repay_date is not null) then
          expiration_date = dateadd(year, 3, repay_date);
        else
          expiration_date = dateadd(year, 3, sh_date);
      end

      suspend;
    end

  end
end
