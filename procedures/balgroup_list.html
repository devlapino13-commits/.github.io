CREATE PROCEDURE BALGROUP_LIST(
    ID VARCHAR(20),
    SELF_INCLUDE INTEGER,
    KIND INTEGER,
    IS_DOWN INTEGER)
RETURNS (
    ID_LIST VARCHAR(20))
AS
begin
  if (is_down = 1) then
  begin
    for with recursive bg_list
        as (select bg1.bg_id, bg1.parent, bg1.bg_status
            from balgroup bg1
            where bg1.bg_id = :id
            union all
            select bg2.bg_id, bg2.parent, bg2.bg_status
            from balgroup bg2
            join bg_list on (bg_list.bg_id = bg2.parent))
        select bg_id
        from bg_list
        where ((:self_include = 0 and
              bg_id <> :id) or
              (:self_include = 1)) and
              ((:kind = -1) or
              (:kind <> -1 and
              bg_status = :kind))
        into :id_list
    do
      suspend;
  end
  else
  begin
    for with recursive bg_list
        as (select bg1.bg_id, bg1.parent, bg1.bg_status
            from balgroup bg1
            where bg1.bg_id = :id
            union all
            select bg2.bg_id, bg2.parent, bg2.bg_status
            from balgroup bg2
            join bg_list on (bg_list.parent = bg2.bg_id))
        select bg_id
        from bg_list
        where ((:self_include = 0 and
              bg_id <> :id) or
              (:self_include = 1)) and
              ((:kind = -1) or
              (:kind <> -1 and
              bg_status = :kind))
        into :id_list
    do
      suspend;
  end
end
