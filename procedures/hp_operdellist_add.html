CREATE PROCEDURE HP_OPERDELLIST_ADD(
    ODM_ID D_INTEGER,
    K_ID D_ID,
    NEW_AMOUNT D_CURRENCY)
AS
declare variable MAIN_OL_ID D_ID;
declare variable OL_ID D_ID;
declare variable OL_DATE D_DATE;
declare variable TMP_LOAN_OL_ID D_ID;
declare variable TMP_MAIN_OL_ID D_ID;
declare variable TMP_ODM_ID D_INTEGER;
declare variable ODM_MAIN_OL_ID D_ID;
declare variable MAIN_K_ID D_ID;
declare variable MAIN_OPER_DATE D_DATE;
declare variable ODM_NEED_REVERSE D_BOOL;
declare variable IS_MAIN D_BOOL;
begin
  if (not exists(select * from operdelkont
  where odk_from_odm_id = :odm_id and
        odk_from_k_id = :k_id)) then
  begin
    select odm_id
    from operdelmain, operdelkont
    where odm_id = odk_from_odm_id and
          odm_state in (0, 1) and
          odk_from_k_id = :k_id
    rows 1
    into tmp_odm_id;

    if (tmp_odm_id is not null) then
      exception e_message 'Договор уже участвует в заявке на удаление №: ' || tmp_odm_id ||ascii_char(13) ||
                          'Сначала нужно закрыть эту заявку';

    insert into operdelkont (odk_from_odm_id, odk_from_k_id)
    values (:odm_id, :k_id);

    select odm_main_ol_id, odk_from_k_id
    from operdelmain, operdelkont
    where odm_id = :odm_id and
          odk_from_odm_id = odm_id and
          odk_loan_ol_id is not null
    into odm_main_ol_id, main_k_id;

    update operdellist
    set odl_new_amount = :new_amount
    where odl_from_odm_id = :odm_id and
          odl_from_k_id = :k_id and
          odl_main_ol_id = :odm_main_ol_id;

    update operdellist
    set odl_new_amount = odl_new_amount - :new_amount
    where odl_from_odm_id = :odm_id and
          odl_from_k_id = :main_k_id and
          odl_main_ol_id = :odm_main_ol_id;
  end else
  begin
    select odm_main_ol_id, odm_oper_date, odk_loan_ol_id, odm_need_reverse
    from operdelkont, operdelmain, operlist
    where odk_from_odm_id = :odm_id and
          odk_from_k_id = :k_id and
          odk_from_odm_id = odm_id and
          odm_main_ol_id = ol_id
    into main_ol_id, main_oper_date, ol_id, odm_need_reverse;
  
    is_main = 1;
    if (ol_id is null) then
    begin
      ol_id = main_ol_id;
      is_main = 0;
    end
  
    -- Временно отключаем ранее добавленные
    update operdellist
    set odl_state = -1
    where odl_from_odm_id = :odm_id and
          odl_from_k_id = :k_id;
  
    update or insert into operdellist (odl_from_odm_id, odl_from_k_id, odl_main_ol_id, odl_loan_ol_id, odl_state)
    values (:odm_id, :k_id, :main_ol_id, :ol_id, 0)
    matching (odl_from_odm_id, odl_from_k_id, odl_main_ol_id);
  
    -- добавляем все последующие операции контракта
    for select distinct ok_ol_date, ol_id, coalesce(ol_parent_id, ol_id)
    from operkred
         inner join operlist on (ok_from_ol_id = ol_id)
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          not(ok_type = 9 and ok_os > 0) and -- исключаем возврат переплаты
          (ok_ol_date > :main_oper_date or
          (ok_ol_date = :main_oper_date and ok_from_ol_id > :ol_id and :odm_need_reverse = 0 /* and :is_main = 1*/))
    into ol_date, tmp_loan_ol_id, tmp_main_ol_id
    do
    begin
      if (tmp_main_ol_id = main_ol_id or tmp_loan_ol_id = ol_id) then
        Continue;
  
      update or insert into operdellist (odl_from_odm_id, odl_from_k_id, odl_main_ol_id, odl_loan_ol_id, odl_state)
      values (:odm_id, :k_id, :tmp_main_ol_id, :tmp_loan_ol_id, 0)
      matching (odl_from_odm_id, odl_from_k_id, odl_main_ol_id);
    end
  
    -- Удаляем если неактуальные записи
    delete from operdellist
    where odl_from_odm_id = :odm_id and
          odl_from_k_id = :k_id and
          odl_state = -1;
  end
end
