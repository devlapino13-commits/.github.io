CREATE PROCEDURE HP_COMPETITION_DATA(
    CPP_ID INTEGER)
RETURNS (
    RES_DATE_BEGIN DATE,
    RES_DATE_END DATE,
    RES_PERIOD_NAME VARCHAR(255),
    RES_COMPET_NAME VARCHAR(255),
    RES_GROUP_NAME VARCHAR(100),
    RES_OBJECT_NAME VARCHAR(255),
    RES_ORDER_IN_GROUP INTEGER,
    RES_DETAIL_NAME VARCHAR(255),
    RES_COMMENT VARCHAR(255),
    CPD_FROM_CPR_ID INTEGER,
    CPD_POINTS_ALL DECIMAL(18,4),
    CPD_STAFF_KOLICH INTEGER,
    CPD_ID INTEGER,
    CPD_POINTS_1 DECIMAL(18,4),
    CPD_POINTS_2 DECIMAL(18,4),
    CPD_POINTS_3 DECIMAL(18,4),
    CPD_POINTS_4 DECIMAL(18,4),
    CPD_POINTS_5 DECIMAL(18,4),
    CPD_POINTS_6 DECIMAL(18,4),
    CPD_POINTS_7 DECIMAL(18,4),
    CPD_POINTS_8 DECIMAL(18,4),
    CPD_POINTS_9 DECIMAL(18,4),
    CPD_COMMENT D_STR_4096,
    CPD_ADD_DATA_1 DECIMAL(18,4),
    CPD_ADD_DATA_2 DECIMAL(18,4),
    CPD_ADD_DATA_3 DECIMAL(18,4),
    CPD_ADD_DATA_4 DECIMAL(18,4),
    CPD_ADD_DATA_5 DECIMAL(18,4),
    CPD_ADD_DATA_6 DECIMAL(18,4),
    CPD_ADD_DATA_7 DECIMAL(18,4),
    CPD_ADD_DATA_8 DECIMAL(18,4),
    CPD_ADD_DATA_9 DECIMAL(18,4),
    CPR_CALCED_POINTS DECIMAL(18,4),
    CPR_ADDITIONAL_POINTS DECIMAL(18,4),
    CPR_FINAL_POINTS DECIMAL(18,4),
    CPR_FINAL_ORDERNO INTEGER)
AS
declare variable cpd_object_kind integer;
declare variable cpd_object_id decimal(18,0);
declare variable cpr_object_kind integer;
declare variable cpr_object_id decimal(18,0);
begin
  select cpp_date_begin, cpp_date_end, cpp_period_name, rd_name
  from competperiod, refdata
  where cpp_id = :cpp_id and
        cpp_from_rd_id = rd_id
  into :res_date_begin, res_date_end, res_period_name, res_compet_name;
  if (res_date_begin is null) then
    exit;

  for select cpd_id, cpd_from_cpr_id, cpd_object_kind, cpd_object_id, cpd_points_all, cpd_staff_kolich, cpd_points_1,
             cpd_points_2, cpd_points_3, cpd_points_4, cpd_points_5, cpd_points_6, cpd_points_7, cpd_points_8,
             cpd_points_9, cpd_comment, cpd_add_data_1, cpd_add_data_2, cpd_add_data_3, cpd_add_data_4, cpd_add_data_5,
             cpd_add_data_6, cpd_add_data_7, cpd_add_data_8, cpd_add_data_9, cpr_group_name, cpr_object_kind,
             cpr_object_id, cpr_comment, cpr_order_in_group, cpr_calced_points, cpr_additional_points, cpr_final_points,
             cpr_final_orderno
      from competdetails, competresult
      where cpr_from_cpp_id = :cpp_id and
            cpr_id = cpd_from_cpr_id
      order by cpr_group_name, cpr_order_in_group, cpd_id
      into :cpd_id, :cpd_from_cpr_id, :cpd_object_kind, :cpd_object_id, :cpd_points_all, :cpd_staff_kolich,
           :cpd_points_1, :cpd_points_2, :cpd_points_3, :cpd_points_4, :cpd_points_5, :cpd_points_6, :cpd_points_7,
           :cpd_points_8, :cpd_points_9, :cpd_comment, :cpd_add_data_1, :cpd_add_data_2, :cpd_add_data_3,
           :cpd_add_data_4, :cpd_add_data_5, :cpd_add_data_6, :cpd_add_data_7, :cpd_add_data_8, :cpd_add_data_9,
           :res_group_name, :cpr_object_kind, :cpr_object_id, :res_comment, :res_order_in_group, :cpr_calced_points,
           :cpr_additional_points, :cpr_final_points, :cpr_final_orderno
  do
  begin
    res_object_name = '';
    res_detail_name = '';

    if (cpr_object_kind = -1) then
    begin
      select rd_name
      from refdata
      where rd_id = :cpr_object_id
      into :res_object_name;
    end
    if (cpr_object_kind = 0) then
    begin
      select c_name_fio
      from clients
      where c_id = :cpr_object_id
      into :res_object_name;
    end
    if (cpr_object_kind = 7) then
    begin
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :cpr_object_id
      into :res_object_name;
    end

    if (cpd_object_kind = -1) then
    begin
      select rd_name
      from refdata
      where rd_id = :cpd_object_id
      into :res_detail_name;
    end
    if (cpd_object_kind = 0) then
    begin
      select c_name_fio
      from clients
      where c_id = :cpd_object_id
      into :res_detail_name;
    end
    if (cpd_object_kind = 7) then
    begin
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :cpd_object_id
      into :res_detail_name;
    end

    cpd_comment = replace(cpd_comment, '<b>', '');
    cpd_comment = replace(cpd_comment, '</b>', '');
    cpd_comment = replace(cpd_comment, '<br>', '. ');
    suspend;
  end
end
