CREATE PROCEDURE HP_EXTINFOLIST_INN_UPDATE(
    EIL_ID D_ID)
AS
declare variable c_id d_id;
declare variable json_str d_str_255;
declare variable c_inn d_str_20;
declare variable eic_code d_str_code;
declare variable icq_id d_id;
declare variable inn_received d_bool;
declare variable eic_id d_id;
declare variable bls_id d_id;
declare variable eil_status d_integer;
begin
  select eil_from_bls_id, eil_from_c_id, eic_code, eic_id, eil_status
  from extinfolist, extinfocfg
  where eic_code in ('inn_nalog', 'inn_gnivc', 'inn_smev')
    and eil_from_eic_id = eic_id
    and eil_id = :eil_id
  into bls_id, c_id, eic_code, eic_id, eil_status;

  if (coalesce(eil_status, 0) in (0, -1)) then
    exit;

  -- RFCRU-4557 Перенес из триггера extinfolist_au0
  update inn_check_queries
  set icq_status = 1
  where icq_from_eil_id = :eil_id;

  select cast(blx_blob as d_blob_text)
  from hp_get_blobex_blob(:bls_id)
  into json_str;

  -- Если ИНН не получен, не обновляем
  if (coalesce(json_str, '') = '') then
    exit;

  if (eic_code = 'inn_smev') then  -- RFCRU-3525
    c_inn = f_json_str_get_value(json_str, 'ИННФЛ', null);
  else
     c_inn = iif(json_str starting with '{', f_json_str_get_value(json_str, 'inn', null), null);

  if (c_inn is not null) then
  begin
    if (eic_code in ('inn_gnivc', 'inn_smev')) then
    begin
        select icm.icm_inn_received
        from inn_check_main icm
        where icm.icm_from_c_id = :c_id
        into :inn_received;

        select icq.icq_id,
               icq.icq_from_c_id
        from inn_check_queries icq
        where icq.icq_from_eil_id = :eil_id
        into :icq_id,
             :c_id;

        update inn_check_queries icq
        set icq.icq_inn_received = 1
        where icq.icq_id = :icq_id;

        if (inn_received = 0) then
        begin
            update inn_check_main t
               set t.icm_from_icq_id = :icq_id,
                   t.icm_inn = :c_inn
             where t.icm_from_c_id = :c_id;
             when gdscode deadlock do
             begin
                insert into inn_update_error_log (iuel_from_eil_id, iuel_from_eic_id, iuel_from_c_id, iuel_message)
                values (:eil_id, :eic_id, :c_id, f_format('Обнаружен Deadlock (GDS %1, SQL %2, STATE %3) при обновлении', gdscode, sqlcode, sqlstate));
             end
             when any do
             begin
                insert into inn_update_error_log (iuel_from_eil_id, iuel_from_eic_id, iuel_from_c_id, iuel_message)
                values (:eil_id, :eic_id, :c_id, f_format('Ошибка (GDS %1, SQL %2, STATE %3) при обновлении', gdscode, sqlcode, sqlstate));
             end
        end
    end

    if (inn_received = 0) then
      execute procedure hp_inn_safe_update(c_id, c_inn);
  end
end
