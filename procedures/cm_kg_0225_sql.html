CREATE PROCEDURE CM_KG_0225_SQL(
    MA_ID D_INTEGER NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable C_ID D_ID;
declare variable C_FIO D_STR_100;
declare variable OBJ_KIND D_INTEGER;
begin
  if (ma_id = 4756346) then
    exit; -- RFCKG-3978

  if (current_date = cast('05.01.2025' as date)) then -- RFCKG-3440
    exit;
  
  if (exists(select *
             from miniapp ma
             inner join product pr on pr.pr_id = ma.ma_from_pr_id
             where ma.ma_id = :ma_id and
                   ma.ma_is_active = 1 and
                   ma.ma_date_end >= '31.12.2999' and
                   (pr.pr_code containing '.installment_goods.' or
                    pr.pr_code containing '.online_loan.'))) then

    exit;

  res_error_is_critical = 1;
  if (current_date = cast('24.11.2024' as date)) then res_error_is_critical = 0;

  obj_kind = f_objkind('miniapp', 'id');
  for select mac.mac_from_c_id, c.c_name_fio
      from miniappclient mac
      inner join clients c on c.c_id = mac.mac_from_c_id
      where mac.mac_from_ma_id = :ma_id and
            mac.mac_is_poruch in (0, -1)
      into :c_id, c_fio
  do
  begin
    if (not exists(select *
                   from extinfolist eil
                   inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
                   where eil.eil_from_c_id = :c_id and
                         eil.eil_object_kind = :obj_kind and
                         eil.eil_object_id = :ma_id and
                         eil.eil_status = 2 and
                         eic.eic_code = 'check_in_ru_db')) then
    begin
      res_error_text = c_fio || ' не был проверен на наличие проcроченного займа в РФ';
      suspend;
    end
    else
    begin
      if (exists(select *
                 from extinfolist eil
                 inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
                 where eil.eil_from_c_id = :c_id and
                       eil.eil_object_kind = :obj_kind and
                       eil.eil_object_id = :ma_id and
                       eil.eil_status = 2 and
                       eil.eil_kk_result = 1 and
                       eic.eic_code = 'check_in_ru_db')) then
      begin
        res_error_text = c_fio || ' имеет активную просрочку в РФ';
        suspend;
      end
    end
  end
end
