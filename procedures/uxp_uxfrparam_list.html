CREATE PROCEDURE UXP$UXFRPARAM_LIST(
    FR_UUID D_UUID,
    UFP_LOOKUP D_UUID,
    UFP_DOUBLE_FIELD D_STR_CODE = null,
    FR_UUID_CODE D_STR_CODE = null)
RETURNS (
    UFP_CODE D_STR_CODE,
    UFP_UUID D_UUID,
    UFP_UFR_UUID D_UUID,
    UFP_UF_UUID D_UUID,
    UF_IS_PARAM D_BOOL,
    UF_PK_FIELD D_SMALLINT,
    UF_POSITION D_INTEGER,
    UF_DOMAIN D_STR_CODE,
    UF_DATA_TYPE D_STR_CODE,
    UF_NOT_NULL D_BOOL,
    UF_DEFAULT D_STR_100,
    UF_NAME D_STR_100,
    UFP_REFRESH D_BOOL,
    UFP_NOT_NULL D_BOOL,
    UFP_EDITOR D_STR_CODE,
    UFP_DEFAULT D_STR_100,
    UFP_LOCKED D_BOOL,
    RES_FIELD_NAME D_STR_100,
    RES_IN_USE D_BOOL,
    RES_EDITOR_INFO D_STR_LONG,
    RES_CAPTION D_STR_100,
    RES_UCM_CODE D_STR_100,
    RES_UCM_IS_OWN D_BOOL,
    RES_BUTTONEDIT_UXA D_STR_100,
    RES_INDEX_TXT D_STR_CODE,
    RES_FIELD_KIND D_STR_10,
    RES_IN_UI D_SMALLINT,
    UEX_EXPRESSION D_STR_4096,
    RES_CATEGORY D_STR_255,
    RES_IS_FK D_BOOL,
    UFP_PUBLIC D_BOOL,
    RES_EDI_KIND D_SMALLINT,
    UF_EDI_UUID D_UUID,
    UMA_MASK D_STR_LONG,
    UMA_MESSAGE D_STR_255,
    RES_EXPR_KINDS D_SMALLINT,
    UFP_PARENT D_UUID)
AS
declare variable tmp_editor_info d_str_long;
declare variable ucm_source d_str_code;
declare variable ucm_field d_str_code;
declare variable enter_sym d_str_lang;
declare variable ube_frame_text_field d_str_code;
declare variable ube_find_text_field d_str_code;
declare variable uxa_name d_str_100;
declare variable ux_table_name d_str_code;
declare variable res_editor_warning d_bool;
declare variable ube_find_id_field d_str_code;
declare variable us_uuid d_uuid;
declare variable ucl_uf_uuid d_uuid;
declare variable orig_index d_integer;
declare variable tmp_us_uuid d_uuid;
declare variable error_text d_str_255;
declare variable us_code d_str_code;
declare variable ufp_lookup_code d_str_code;
declare variable res_fk_fields d_str_long;
declare variable ufl_list_kind d_smallint;
declare variable us_keyfields d_str_100;
declare variable ufc_hide_text d_bool;
declare variable uf_type d_smallint;
declare variable erl_kind d_smallint;
declare variable uxa_uuid d_uuid;
declare variable ufc_kind d_integer;
declare variable main_us_uuid d_uuid;
declare variable uf_owner d_uuid;
declare variable ufr_add_table_fields d_str_code;
declare variable ufm_uuid d_uuid;
begin
  -- Список параметров для добавления в uxFrItem.
  if (coalesce(fr_uuid, '') = '') then
  begin
    if (coalesce(fr_uuid_code, '') <> '') then
      fr_uuid = uxf$uniq(fr_uuid_code);
    else
      Exit;
  end

  if (ufp_double_field is null) then
    ufp_double_field = '';

  ufp_ufr_uuid = fr_uuid;
  enter_sym = ascii_char(13) || ascii_char(10);

  orig_index = null;
  ufp_lookup_code = '';
  main_us_uuid = null;
  if (ufp_lookup is null) then
  begin
    select ufr_us_uuid, ufr_owner, ufr_add_table_fields
    from ux$frame
    where ufr_uuid = :fr_uuid
    into us_uuid, ufm_uuid, ufr_add_table_fields;

    if (ufr_add_table_fields <> '') then
    begin
      select us_uuid
      from ux$source
      where us_code = :ufr_add_table_fields and
            coalesce(us_keyfields, '') <> ''
      into main_us_uuid;
    end
  end else
  begin
    select uf_name, uf_position
    from ux$frparam, ux$field
    where ufp_uuid = :ufp_lookup and
          ufp_uf_uuid = uf_uuid
    into ufp_lookup_code, orig_index;

    select ufl_us_uuid
    from ux$frlookup
    where ufl_ufp_uuid = :ufp_lookup
    into :us_uuid;
  end

  us_code = uxf$code(us_uuid, 'code');

  select list(uix_fields, ';')
  from ux$index
  where uix_source = :us_code and
        uix_fk = 1
  into res_fk_fields;

  if (res_fk_fields is null) then
    res_fk_fields = '';

  if (res_fk_fields <> '') then
    res_fk_fields = ';' || res_fk_fields || ';';

  if (main_us_uuid is null) then
    main_us_uuid = us_uuid;

  for select uf_uuid, uf_owner, uf_is_param, uf_pk_field, uf_position, uf_domain, uf_data_type, uf_not_null, uf_default,
             uf_field_name, uf_name, uf_edi_uuid, uf_type
  from ux$field
  where uf_owner in (:us_uuid, :main_us_uuid) and
        (:ufp_double_field = '' or uf_field_name = :ufp_double_field)
  order by uf_is_param desc, iif(uf_owner = :us_uuid, 1, 2), uf_position
  into :ufp_uf_uuid, :uf_owner, :uf_is_param, :uf_pk_field, :uf_position, :uf_domain, :uf_data_type, :uf_not_null, :uf_default,
       :res_field_name, :uf_name, uf_edi_uuid, uf_type
  do
  begin
    if (ufp_lookup is null and uf_owner = main_us_uuid and main_us_uuid <> us_uuid) then
    begin
      if (exists(select * from ux$frparam, ux$field
                 where ufp_ufr_uuid = :ufp_ufr_uuid and
                       ufp_uf_uuid <> :ufp_uf_uuid and
                       ufp_uf_uuid = uf_uuid and
                       uf_field_name = :res_field_name)) then
        Continue;
    end

    res_caption = uf_name;

    res_edi_kind = 0;
    if (uf_edi_uuid is not null) then
      res_edi_kind = -1;

    if (ufp_lookup is not null) then
    begin
      if (uf_is_param = 1) then
        res_category = 'Выборка ' || Upper(ufp_lookup_code) || ', параметры ' || upper(us_code);
      else
        Continue;
    end else
    begin
      if (uf_is_param = 1) then
        res_category = ' Параметры';
      else
        res_category = ' Поля';
    end

    res_is_fk = 0;
    ufp_public = 0;

    if (res_fk_fields containing ';' || res_field_name || ';') then
      res_is_fk = 1;

    res_editor_info = '';
    res_in_use = 0;
    ufp_uuid = null;
    ufp_parent = null;
    ufp_editor = '';
    ufp_refresh = 0;
    ufp_not_null = 0;
    ufp_default = '';
    ufp_locked = 0;

    res_index_txt = uf_position;
    if (orig_index is not null) then
      res_index_txt = orig_index || '.' || res_index_txt;

    select ufp_uuid, ufp_parent, ufp_editor, ufp_refresh, ufp_not_null, ufp_public
    from ux$frparam
    where ufp_ufr_uuid = :ufp_ufr_uuid and
          ufp_uf_uuid = :ufp_uf_uuid and
          ufp_parent is not distinct from :ufp_lookup
    into :ufp_uuid, :ufp_parent, :ufp_editor, :ufp_refresh, :ufp_not_null, ufp_public;

    if (row_count = 1) then
    begin
      res_in_use = 1;

      select res_def_value, res_locked
      from uxp$uxfrparam_value(:ufp_uuid, null, null)
      into :ufp_default, :ufp_locked;
    end

    res_field_kind = '';
    if (res_is_fk = 0) then
    begin
      if (uf_pk_field <> 0) then
        res_field_kind = 'Pk' || uf_pk_field;
    end else
    begin
      if (uf_pk_field = 0) then
        res_field_kind = 'Fk';
      else
        res_field_kind = 'FP' || uf_pk_field;
    end

    if (uf_type = 1) then
    begin
      if (res_field_kind = '') then
        res_field_kind = 'V';
      else
        res_field_kind = 'V ' || res_field_kind;
    end else
    if (uf_type = -1) then
    begin
      if (res_field_kind = '') then
        res_field_kind = 'X';
      else
        res_field_kind = 'X ' || res_field_kind;
    end

    res_ucm_code = '';
    res_ucm_is_own = 0;
    res_buttonedit_uxa = '';

    res_editor_info = '';
    tmp_editor_info = '';
    uxa_name = '';
    ube_frame_text_field = '';
    ube_find_text_field = '';
    ux_table_name = '';
    res_editor_warning = 0;
    uxa_uuid = null;

    res_in_ui = 0;

    if (exists(select * from ux$fritem
    where ufi_owner = :fr_uuid and
          ufi_uf_uuid = :ufp_uf_uuid)) then
      res_in_ui = 1;
    else
    if (exists(select * from ux$fritem
    where ufi_owner = :ufm_uuid and
          ufi_uf_uuid = :ufp_uf_uuid)) then
      res_in_ui = 1;


    if (exists(select * from ux$grid, ux$gritem
    where ugr_ufr_uuid = :fr_uuid and
          ugr_uuid = ugi_ugr_uuid and
          ugi_uf_uuid = :ufp_uf_uuid)) then
      res_in_ui = res_in_ui + 2;

    select 'Редактор: ' || editor_name || ', ' || editor_memo
    from uxp$uxfrparam_controls('editor')
    where editor_id = :ufp_editor
    into res_editor_info;

    if (ufp_editor in ('buttonedit', 'dialogedit')) then
    begin
      select ua_usr_text_field,
             ua_uxa_fcode,
             usr_text_field,
             usr_id_field,
             uxa_name,
             us_code,
             uxa_uuid
      from ux$frparam, ux$action, ux$search, ux$alias, ux$code, ux$source
      where ufp_uuid = :ufp_uuid and
            ufp_ufr_uuid = ua_owner and
            ua_usr_id_field = :res_field_name and
            ua_uxa_uuid = usr_uuid and
            usr_uuid = uxa_uuid and
            uxa_owner = ux_uuid and
            ux_us_uuid = us_uuid
      into :ube_frame_text_field, :res_buttonedit_uxa, :ube_find_text_field, :ube_find_id_field,
           :uxa_name, :ux_table_name, :uxa_uuid;

      if (row_count = 0) then
      begin
        res_editor_warning = 1;
        res_editor_info = res_editor_info || enter_sym || 'Отсутствует действие поисковика';
      end
      else
      begin
        if (ufp_editor = 'buttonedit') then
          res_editor_info = res_editor_info || enter_sym ||
                            f_format('Открывает окно: %1 (%2)', res_buttonedit_uxa, uxa_name);
        else
          res_editor_info = res_editor_info || enter_sym ||
                            f_format('Открывает диалоговое окно: %1 (%2)', res_buttonedit_uxa, uxa_name);
  
        res_editor_info = res_editor_info || enter_sym || f_format('Возвращает поле: %1', upper(ube_find_id_field));
        if (ube_find_text_field <> '') then
          res_editor_info = res_editor_info || ', поле текста: ' || upper(ube_find_text_field);
  
        if (ube_frame_text_field <> '') then
        begin
          if (ube_frame_text_field not containing '$') then
          begin
            res_editor_info = res_editor_info || enter_sym ||
                              'Текст вставляется в поле: ' || upper(ube_frame_text_field);
          end else
          begin
            res_editor_info = res_editor_info || enter_sym ||
                              'Мультивыбор, разделитель: ';
  
            res_editor_info = res_editor_info || trim(case ube_frame_text_field
              when '$delimiter_comma' then 'Запятая'
              when '$delimiter_semicolon' then 'Точка с запятой'
              when '$square_brackets' then 'Квадратные скобки - []'
              else ube_frame_text_field
            end);
          end
        end else
        begin
          tmp_editor_info = '';
  
          tmp_us_uuid = uxf$uniq(ux_table_name);
  
          select keyname_sql, error_text
          from uxp$uxsource_sql_prepare(:tmp_us_uuid, :res_field_name, uuid_to_char(:uxa_uuid))
          into :tmp_editor_info, error_text;
  
          if (tmp_editor_info <> '') then
          begin
            res_editor_info = res_editor_info || enter_sym ||
                              'SQL получения текста: ' || tmp_editor_info;
          end else
          begin
            res_editor_warning = 1;
            res_editor_info = res_editor_info || enter_sym ||
                              f_format('ВНИМАНИЕ, в таблице %1 отсутствует SQL получения текста', upper(ux_table_name));
          end
        end
      end
    end

    if (ufp_editor = 'lookup') then
    begin
      tmp_us_uuid = null;
      select ufl_us_uuid, ufl_list_kind, us_keyfields
      from ux$frlookup, ux$source
      where ufl_ufp_uuid = :ufp_uuid and
            ufl_us_uuid = us_uuid
      into tmp_us_uuid, ufl_list_kind, us_keyfields;

      tmp_editor_info = '';
      error_text = '';

      if (tmp_us_uuid is not null) then
      begin
        select lookup_sql, error_text
        from uxp$uxsource_sql_prepare(:tmp_us_uuid, '')
        into :tmp_editor_info, error_text;

        if (tmp_editor_info <> '') then
        begin
          res_editor_info = res_editor_info || enter_sym || f_format('SQL списка: %1', tmp_editor_info);

          if (error_text = '' and ufl_list_kind = 0 and us_keyfields = '') then
          begin
            res_editor_warning = 1;
            res_editor_info = res_editor_info || enter_sym || 'ВНИМАНИЕ, ключевое поле не указано.';
          end
        end
        else
        begin
          if (error_text <> '') then
            res_editor_info = res_editor_info || enter_sym || f_format('SQL списка пуст: %1', error_text);
          else
            res_editor_info = res_editor_info || enter_sym || 'SQL списка пуст.';
        end
      end
      else
      begin
        res_editor_warning = 1;
        res_editor_info = res_editor_info || enter_sym || 'ВНИМАНИЕ, настройки не найдены.';
      end
    end

    ucl_uf_uuid = null;
    res_ucm_is_own = 0;
    if (ufp_editor = 'combobox') then
    begin
      ufc_hide_text = 0;
      select ufc_combo_source, ufc_combo_field, uf_uuid, ufc_hide_text, ufc_kind
      from ux$frcombo, ux$source, ux$field
      where ufc_ufp_uuid = :ufp_uuid and
            ufc_combo_source = us_code and
            uf_owner = us_uuid and
            uf_field_name = ufc_combo_field
      into ucm_source, ucm_field, ucl_uf_uuid, ufc_hide_text, ufc_kind;

      if (uxf$owner(ufp_uf_uuid, 'code') = ucm_source and ucm_field = res_field_name) then
        res_ucm_is_own = 1; -- Свой список, значит доступно для редактирования

      if (ufc_kind <> 0) then
      begin
        if (ufc_kind >= 1000) then
          res_editor_info = res_editor_info || ' (Радиокнопки, горизонтальное расположение)';
        else
          res_editor_info = res_editor_info || ' (Радиокнопки, ' || ufc_kind ||' колон.)';
      end
      else
      if (ufc_hide_text = 1) then
        res_editor_info = res_editor_info || ' без текста, только иконки';

      res_editor_info = res_editor_info || enter_sym ||
                        f_format('Создан для %1', upper(ucm_source || '.' || ucm_field));

      tmp_editor_info = '';
      select right(list(ucl_id || '-' || ucl_text, ', '), 1000)
      from ux$combolist
      where ucl_uf_uuid = :ucl_uf_uuid
      into tmp_editor_info;

      if (tmp_editor_info is null) then
        tmp_editor_info = '';

      if (tmp_editor_info <> '') then
      begin
        if (char_length(res_editor_info || tmp_editor_info) > 1000) then
          tmp_editor_info = left(tmp_editor_info, 1000 - char_length(res_editor_info));

        res_editor_info = res_editor_info || enter_sym || 'Значения: ' || tmp_editor_info;
      end else
      begin
        res_editor_warning = 1;
        res_editor_info = res_editor_info || enter_sym || 'ВНИМАНИЕ, значения не заполнены.';
      end
    end

    uma_mask = '';
    uma_message = '';

    if (ufp_editor = 'mask') then
    begin
      select uma_mask, uma_message
      from ux$frmask
      where uma_ux_uuid = :ufp_uuid
      into uma_mask, uma_message;

      res_editor_info = res_editor_info || 'Маска ввода: ' || uma_mask || enter_sym;
      res_editor_info = res_editor_info || 'Текст несоответствия: ' || uma_message;
    end

    if (ufp_editor = 'currfixed') then
    begin
      if (not exists(select * from ux$frparam, ux$field
                     where ufp_ufr_uuid = :fr_uuid and
                           uf_is_param = :uf_is_param and
                           uf_field_name = :res_field_name || '$bond')) then
      begin
        res_editor_warning = 1;
        res_editor_info = res_editor_info || enter_sym || 'ВНИМАНИЕ, отсутствует поле для передачи купюр в БД, необходимо добавить реальное или вычисляемое поле с названием: ' || upper(res_field_name) || '$BOND с типом данных: D_STR_LONG';
      end
    end

    if (ufp_editor in ('enum', 'reflist')) then
    begin
      erl_kind = null;
      res_ucm_code = '';

      select erl_code, erl_kind
      from ux$frereflist, ux$ereflist
      where fer_ufp_uuid = :ufp_uuid and
            fer_erl_uuid = erl_uuid
      into res_ucm_code, erl_kind;

      if (erl_kind is null) then
      begin
        res_editor_warning = 1;
        res_editor_info = res_editor_info || enter_sym || 'ВНИМАНИЕ, справочник отсутстует.';
      end else
      begin
        if (erl_kind = 0) then
          res_editor_info = res_editor_info || enter_sym || 'Код списка Enum: ' || res_ucm_code;
        else
          res_editor_info = res_editor_info || enter_sym || 'Код справочника RefData: ' || res_ucm_code;
      end
    end

    uex_expression = '';

    select exp_expression_all, exp_kinds
    from uxp$uxexpression_text(:ufp_uuid)
    into :uex_expression, res_expr_kinds;

    if (uex_expression <> '') then
      res_editor_info = res_editor_info || enter_sym || uex_expression;

    if (uf_pk_field = 1) then
      res_caption = res_caption || '   (pk' || uf_pk_field || ')';

    if (res_editor_warning = 1) then
      res_in_ui = -1;

    ufp_code = uuid_to_char(ufp_uuid);
    suspend;

    if (ufp_editor = 'lookup' and ufp_double_field = '' and ufp_lookup is null) then
    begin
      ufp_lookup = ufp_uuid;
      ufp_public = 0;

      res_editor_info = '';
      res_in_use = 0;
      ufp_editor = '';
      ufp_refresh = 0;
      ufp_not_null = 0;
      ufp_default = '';
      ufp_locked = 0;

      res_ucm_code = '';
      res_ucm_is_own = 0;
      res_buttonedit_uxa = '';
  
      res_editor_info = '';
      tmp_editor_info = '';
      uxa_name = '';
      ube_frame_text_field = '';
      ube_find_text_field = '';
      ux_table_name = '';
      res_editor_warning = 0;
  
      res_in_ui = 0;

      uex_expression = '';
      uma_mask = '';
      uma_message = '';
      res_expr_kinds = -1;

      for select ufp_uuid, ufp_code, ufp_ufr_uuid, ufp_uf_uuid, uf_is_param, uf_pk_field, uf_position, uf_domain, uf_data_type, uf_not_null,
                 uf_default, ufp_refresh, ufp_not_null, ufp_editor, ufp_default, ufp_locked, res_field_name, res_in_use,
                 res_editor_info, res_caption, res_ucm_code, res_ucm_is_own, res_buttonedit_uxa, res_index_txt,
                 res_field_kind, res_in_ui, uex_expression, res_category, res_is_fk, uf_name,
                 res_edi_kind, uf_edi_uuid, uma_mask, uma_message, res_expr_kinds, ufp_parent
          from uxp$uxfrparam_list(:fr_uuid, :ufp_lookup)
          into :ufp_uuid, :ufp_code, :ufp_ufr_uuid, :ufp_uf_uuid, :uf_is_param, :uf_pk_field, :uf_position, :uf_domain, :uf_data_type, :uf_not_null,
