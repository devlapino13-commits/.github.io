CREATE PROCEDURE FSROBOT_MAILING_FP_ANK_UPDATE(
    REP_PARAMS VARCHAR(255),
    REP_EMAIL VARCHAR(255),
    USER_FIO VARCHAR(100))
RETURNS (
    RES_SUBJECT VARCHAR(255),
    RES_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_MAILLIST VARCHAR(255),
    RES_C_ID DECIMAL(18,0))
AS
declare variable reg_name varchar(100);
declare variable c_id decimal(18,0);
declare variable date_format_str varchar(20);
declare variable k_id id_type;
declare variable svz_id id_type;
declare variable k_kontract_no str_100;
declare variable disb_date date;
declare variable reg_email_manager str_255;
declare variable svz_fio str_100;
declare variable c_fio str_100;
declare variable message_type str_100;
begin
-- Процедура вызывается из FSROBOT_ON_MAILING
-- Процедура готовит список уведомлений для отправки роботом по почте или sms
-- REP_COMMAND_KIND. Настройки сообщения хранятся в таблице RefData:
-- справочник NotifyMail || '.' || Rep_Command_Kind - строка настроек email
-- справочник NotifySMS  || '.' || Rep_Command_Kind - строка настроек SMS

-- RefData.RD_Memo    - Список получателей
-- RefData.RD_Tag     - Срок жизни сообщения
-- RefData.RD_STR_ID  - Название хранимой процедуры, которая готовит текст сообщения

-- Rep_Params - входящий параметры в зависимости от команды
-- Rep_UN_ID - пользователь
  res_c_id = 0;

--ok;109010027753700101;1090100681604;1090100229945
  message_type = substring(rep_params from 1 for position(';' in rep_params) - 1);
  rep_params = substring(rep_params from position(';' in rep_params) + 1);
  k_id = substring(rep_params from 1 for position(';' in rep_params) - 1);
  rep_params = substring(rep_params from position(';' in rep_params) + 1);
  c_id = substring(rep_params from 1 for position(';' in rep_params) - 1);
  svz_id = substring(rep_params from position(';' in rep_params) + 1);

  select kont.k_kontract_no, kont.k_fact_disb_date, reg.reg_name, reg_email_manager
  from kontract kont
  inner join region reg on kont.k_last_reg_id = reg.reg_unicode
  where kont.k_id = :k_id
  into :k_kontract_no, :disb_date, :reg_name, :reg_email_manager;

  reg_email_manager = coalesce(reg_email_manager, '');

  select cl.c_name_fio
  from clients cl
  where cl.c_id = :svz_id
  into :svz_fio;

  select cl.c_name_fio
  from clients cl
  where cl.c_id = :c_id
  into :c_fio;

  date_format_str = right('0' || extract(day from disb_date), 2) || '.' || right('0' || extract(month from disb_date), 2) || '.' || extract(year from disb_date);

  if (reg_email_manager = '') then
    res_maillist = 'compliance@mbf.kg';
  else
    res_maillist = :reg_email_manager || ';compliance@mbf.kg';

  if (message_type = 'ok') then
  begin
    res_subject = 'Внимание! Требуется обновить анкету ФТ';

    res_body = '<b>Требуется обновить Анкету ФТ для клиента с высоким уровнем риска</b><br>';
  end
  else
  begin
    res_subject = 'Внимание! Задача по обновлению Анкеты ФТ просрочена';

    res_body = '<b>Срочно обновите Анкету ФТ для клиента с высоким уровнем риска. <br>';
    res_body = BLOB_APPEND(res_body, 'С момента поступления задачи прошло 5 дней</b> <br>');
  end

  res_body = BLOB_APPEND(res_body, '<br>');
  res_body = BLOB_APPEND(res_body, 'Офис..................' || reg_name || '<br>');
  res_body = BLOB_APPEND(res_body, 'Номер договора........' || k_kontract_no || '<br>');
  res_body = BLOB_APPEND(res_body, 'Дата выдачи...........' || date_format_str || '<br>');
  res_body = BLOB_APPEND(res_body, 'СПФ...................' || svz_fio || '<br>');
  res_body = BLOB_APPEND(res_body, 'Клиент................' || c_fio || '<br>');
  res_body = BLOB_APPEND(res_body, 'Почта менеджера офиса.' || reg_email_manager || '<br>');
  res_body = BLOB_APPEND(res_body, '<br>');

  suspend;
end
