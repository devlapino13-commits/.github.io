CREATE PROCEDURE FM_CLP_ANK_ACTION(
    MAC_ID D_INTEGER,
    UN_ID D_ID,
    DO_ACTION D_STR_CODE,
    QUESTION_ID D_INTEGER,
    CLARIFICATION_ID D_INTEGER,
    TARGET_ID D_INTEGER)
RETURNS (
    RES_HTML D_BLOB_TEXT)
AS
declare variable lang d_str_code;
declare variable c_id d_id;
declare variable ma_id d_id;
declare variable c_name_fio d_str_255;
declare variable target_name d_str_255;
declare variable cq_name d_str_255;
declare variable cc_name d_str_255;
declare variable bustype_id d_id;
declare variable bs_name d_str_255;
declare variable cc_subtype_id d_id;
declare variable kc_id d_id;
declare variable kc_business_type d_integer;
begin
  res_html = '';

  select mac.mac_from_c_id, mac.mac_from_ma_id
  from miniappclient mac
  where mac.mac_id = :mac_id and
        mac.mac_amount > 0
  into c_id, ma_id;

  if (coalesce(ma_id, 0) = 0) then
     exception e_message 'Заявка не определена!';

  if (do_action = 'preview') then
  begin
    select c.c_name_fio, c.c_blank_lang
    from clients c
    where c.c_id = :c_id
    into c_name_fio, lang;

    res_html = (select res_style
                from fm_html_styles) || f_format('<body><table width="100%"><col width="40%"><col width="60%">');
    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(:lang, 'ФИО клиента') || '</b>', :c_name_fio, '', '', -1, -1, -1, -1, 2, ''));

    select rd.rd_name
    from refdata rd
    where rd.rd_id = :target_id
    into target_name;

    select cq.cq_name
    from clp_questions cq
    where cq.cq_id = :question_id
    into cq_name;

    select cc.cc_name, cc.cc_subtype_id
    from clp_clarification cc
    where cc.cc_id = :clarification_id
    into cc_name, cc_subtype_id;

    select bs.bs_name
    from bussubtype bs
    where bs.bs_id = :cc_subtype_id
    into bs_name;

    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(:lang, 'Цель') || '</b>', :target_name, '', '', -1, -1, -1, -1, 2, ''));
    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(:lang, 'Отрасль кредитование / финансирование') || '</b>', :bs_name, '', '', -1, -1, -1, -1, 2, ''));
    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(:lang, 'На что клиент хочет потратить деньги?') || '</b>', :cq_name, '', '', -1, -1, -1, -1, 2, ''));
    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(:lang, 'Уточнение по цели') || '</b>', :cc_name, '', '', -1, -1, -1, -1, 2, ''));
    res_html = blob_append(res_html, '</table><br><br><br>');
  end
  if (do_action = 'save') then
  begin
    update or insert into clp_miniappclient (cmc_from_ma_id, cmc_from_c_id, cmc_questions_id, cmc_clarification_id,
                                             cmc_from_un_id)
    values (:ma_id, :c_id, :question_id, :clarification_id, :un_id)
    matching (cmc_from_ma_id, cmc_from_c_id);

    select cc.cc_subtype_id
    from clp_clarification cc
    where cc.cc_id = :clarification_id
    into bustype_id;

    update miniappclient mac
    set mac.mac_loan_target = :target_id,
        mac.mac_business_subtype = :bustype_id
    where mac.mac_id = :mac_id and
          mac.mac_from_c_id = :c_id;

    if (exists(select *
               from kontract k where k.k_from_ma_app_id = :ma_id)) then
    begin
      for select kc.kc_id
          from knclient kc
          inner join kontract k on k.k_id = kc.kc_from_k_id and
                k.k_from_ma_app_id = :ma_id
          where kc.kc_amount_approved > 0
          into kc_id
      do
      begin
         select bs_from_bt_id
         from bussubtype
         where bs_id = :bustype_id
          into kc_business_type;
        update knclient kc
        set kc.kc_loan_target = :target_id,
            kc.kc_busuness_subtype = :bustype_id,
            kc.kc_business_type = :kc_business_type
        where kc.kc_id = :kc_id;
      end
    end
  end
  suspend;
end
