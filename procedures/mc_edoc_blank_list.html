CREATE PROCEDURE MC_EDOC_BLANK_LIST(
    MCL_ID D_ID,
    EDS_ID D_ID)
RETURNS (
    EDB_ID D_ID,
    EDB_NAME D_STR_100,
    EDB_VISITED D_BOOL,
    EDS_UUID D_STR_CODE,
    EDF_IS_ONLINE D_BOOL,
    EDB_OPTIONAL D_BOOL,
    EDB_CHECKED D_BOOL)
AS
declare variable def_checked d_bool;
begin
  def_checked = 1; -- По-умолчанию для каждого бланка выставлена галочка для подписания

  if (eds_id < 0) then
  begin
    for
    select edb_id, edf_print_name, edf_optional_sign, iif(edl_status = 1, edb_sign_status, :def_checked)
    from edoc_list, edoc_blank, edoc_frx
    where edl_id = -:eds_id
    and edl_id = edb_from_edl_id
    and edb_from_edf_id = edf_id
    into edb_id, edb_name, edb_optional, edb_checked
    do
    begin
      edb_visited = 1;
      suspend;
    end
  end 
  else

  eds_uuid = null;
  select uuid_to_char(udl.udl_uuid)
  from uuidlist udl
  where udl.udl_object_kind = f_objkind('edoc_sign', 'id') and
        udl.udl_object_id = :eds_id
  into eds_uuid;

  for select edb_id, edf_print_name, edf_optional_sign, edf_is_online,
    iif(eds_status = 1, iif(eds_from_c_id is null, eds_status, edb_sign_status), :def_checked)
  from edoc_blank, edoc_frx, edoc_sign, edoc_list
  where eds_id = :eds_id and  
        eds_from_edl_id = edl_id and
        eds_from_edl_id =  edb_from_edl_id and
        edb_from_edf_id = edf_id and
        (edb_from_c_id is null or edb_from_c_id = eds_from_c_id) -- выдаем общие бланки и бланки текущего подписанта
  into edb_id, edb_name, edb_optional, edf_is_online, edb_checked
  do
  begin
    edb_visited = 1;
    if (exists(
    select * from edoc_open
    where edo_from_eds_id = :eds_id and
          edo_from_edb_id = :edb_id and
          (edo_before_sign_datetime is not null or
          edo_after_sign_datetime is not null))) then edb_visited = 0;
    suspend;
  end
end
