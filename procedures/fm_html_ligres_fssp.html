CREATE PROCEDURE FM_HTML_LIGRES_FSSP(
    C_ID D_ID)
RETURNS (
    RES_HTML D_BLOB_TEXT)
AS
declare variable c_name_fio d_str_255;
declare variable eil_date d_date;
declare variable ip_num d_str_100;
declare variable debt_type d_str_255;
declare variable ip_start d_date;
declare variable debt_sum d_currency;
declare variable subject_name d_str_255;
declare variable fssp_dep d_str_255;
declare variable end_section d_str_255;
declare variable end_reason d_str_long;
declare variable active_ip d_integer;
declare variable closed_ip d_integer;
declare variable row_no d_integer;
declare variable fio_pristav d_str_255;
declare variable bls_id d_id;
begin
  select c_name_fio
  from clients
  where c_id = :c_id
  into c_name_fio;

  res_html = (select res_style from fm_html_styles) || '<body>';
  res_html = blob_append(res_html, f_format('<h1>%1</h1>', c_name_fio));

  select cast(eil_status_datetime as d_date), eil.eil_from_bls_id
  from extinfolist eil
  inner join extinfocfg eic on eil_from_eic_id = eic_id
  where eil.eil_from_c_id = :c_id
    and eil_is_json = 1
    and eil_status = 2
    and eic_code = 'ligres_fssp_cli_check'
  order by eil.eil_id desc
  rows 1
  into :eil_date, bls_id;

  if (row_count = 0) then
  begin
    res_html = blob_append(res_html, '<br><br>Информация не получена</body>');
    suspend;
    exit;
  end

  select sum(iif(is_closed = 0, 1, 0)),
         sum(iif(is_closed = 1, 1, 0))
  from hp_get_ligres_content(:bls_id)
  into active_ip, closed_ip;

  if (coalesce(active_ip, 0) = 0 and coalesce(closed_ip, 0) = 0) then
  begin
    res_html = blob_append(res_html, '<br><br>Информация отсутствует</body>');
    suspend;
    exit;
  end

  res_html = blob_append(res_html, '<br>Дата получения информации: ' || f_datetostr(eil_date));
  res_html = blob_append(res_html, '<br>Всего ИП: ' || (coalesce(active_ip, 0) + coalesce(closed_ip, 0)));
  res_html = blob_append(res_html, '<br>Активных ИП ' || active_ip);
  res_html = blob_append(res_html, '<br>Закрытых ИП ' || closed_ip);

  res_html = blob_append(res_html, '<h3>Активные ИП</h3>');
  row_no = 0;
  for select ip_num, debt_type, ip_start, debt_sum, subject_name, fssp_dep, fio_pristav
  from hp_get_ligres_content(:bls_id)
  where is_closed = 0
  into ip_num, debt_type, ip_start, debt_sum, subject_name, fssp_dep, fio_pristav
  do
  begin
    row_no = row_no + 1;
    res_html = blob_append(res_html, '<table width="100%"><col width="60%"><col width="40%"><tr><th>ИП №' || row_no || '</th><th>Активное ИП</th></tr>');
    res_html = blob_append(res_html, '<tr><td>Номер ИП</td><td>' || ip_num || '</td></tr>');  
    res_html = blob_append(res_html, '<tr><td>Дата начала ИП</td><td>' || f_datetostr(ip_start) || '</td></tr>'); 
    res_html = blob_append(res_html, '<tr><td>Сумма задолженности</td><td>' || round(debt_sum, 2) || '</td></tr>');
    res_html = blob_append(res_html, '<tr><td>Тип задолженности</td><td>' || debt_type || '</td></tr>');
    res_html = blob_append(res_html, '<tr><td>Предмет исполнения</td><td>' || subject_name || '</td></tr>');
    res_html = blob_append(res_html, '<tr><td>Отдел ФССП</td><td>' || fssp_dep || '</td></tr>');
    res_html = blob_append(res_html, '<tr><td>Судебный пристав-исполнитель</td><td>' || fio_pristav || '</td></tr></table>');
  end

  row_no = 0; 
  res_html = blob_append(res_html, '<h3>Приостановленное ИП</h3>');
  for select ip_num, debt_type, ip_start, debt_sum, subject_name, fssp_dep, end_section, end_reason
  from hp_get_ligres_content(:bls_id)
  where is_closed = 1
  into ip_num, debt_type, ip_start, debt_sum, subject_name, fssp_dep, end_section, end_reason
  do
  begin
    row_no = row_no + 1;
    res_html = blob_append(res_html, '<table width="100%"><col width="60%"><col width="40%"><tr><th>ИП №' || row_no || '</th><th>Приостановленное ИП</th></tr>');
    res_html = blob_append(res_html, '<tr><td>Номер ИП</td><td>' || ip_num || '</td></tr>');    
    res_html = blob_append(res_html, '<tr><td>Дата начала ИП</td><td>' || f_datetostr(ip_start) || '</td></tr>');  
    res_html = blob_append(res_html, '<tr><td>Сумма задолженности</td><td>' || round(debt_sum, 2) || '</td></tr>');
    res_html = blob_append(res_html, '<tr><td>Тип задолженности</td><td>' || debt_type || '</td></tr>');
    res_html = blob_append(res_html, '<tr><td>Предмет исполнения</td><td>' || subject_name || '</td></tr>');
    res_html = blob_append(res_html, '<tr><td>Отдел ФССП</td><td>' || fssp_dep || '</td></tr>');  
    res_html = blob_append(res_html, '<tr><td>Судебный пристав-исполнитель</td><td>' || fio_pristav || '</td></tr>');
    res_html = blob_append(res_html, '<tr><td>Статья окончания</td><td>' || end_section || '</td></tr>');
    res_html = blob_append(res_html, '<tr><td>Причина окончания</td><td>' || end_reason || '</td></tr></table>');
  end

  res_html = blob_append(res_html, '<br></body>');
  suspend;
end
