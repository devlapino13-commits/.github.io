CREATE PROCEDURE AAREP_KZ_FRD_MFO(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    PERIOD_STR D_STR_CODE,
    KOL_DISB_ONLINE D_INTEGER,
    SUM_DISB_ONLINE D_CURRENCY,
    ZAD_ALL_ONLINE D_CURRENCY)
AS
declare variable K_ID D_ID;
declare variable OKS_BALANCE D_CURRENCY;
declare variable PERC_FINE D_CURRENCY;
begin
  period_str = 'Отчетный период: c '||f_datetostr(:rep_date_begin)||' по '||f_datetostr(:rep_date_end);

  kol_disb_online = null;
  sum_disb_online = null;
  select count(*), sum(k_amount_approved)
  from kontstate, region_list(:rep_region_id, 1, 3, 1), kontract
  where ks_ol_is_delete = 0
  and ks_state = 1
  and ks_date_begin between :rep_date_begin and :rep_date_end
  and ks_region_id = id_list
  and ks_from_k_id = k_id
  and k_is_online = 1
  into kol_disb_online, sum_disb_online;
  kol_disb_online = coalesce(kol_disb_online, 0);
  sum_disb_online = coalesce(sum_disb_online, 0);

  zad_all_online = 0;
  for
  select k_id, oks_balance from oksummary, region_list(:rep_region_id, 1, 3, 1), kontract
  where oks_region_id = id_list
  and oks_close_state = 0
  and :rep_date_end between oks_date_begin and oks_date_end
  and oks_from_k_id = k_id
  and k_is_online = 1
  into k_id, oks_balance
  do
  begin


    perc_fine = null;
    select sum(sh_calc_perc + sh_calc_fine)
    from aarep_make_zad_list(:k_id, :rep_date_end, 1)
    into perc_fine;

    zad_all_online = zad_all_online + coalesce(oks_balance,0) + coalesce(perc_fine, 0);
  end 


  suspend;
end
