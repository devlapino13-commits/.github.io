CREATE PROCEDURE HP_DO_ACTIONSAVE(
    ACL_ID D_ID,
    QUESTION_ID D_INTEGER,
    RD_ID D_INTEGER,
    STATE D_INTEGER,
    STR_VALUE D_STR_2048,
    UN_ID D_ID,
    ACL_DATE D_DATE = null)
RETURNS (
    ERR_TEXT D_STR_LONG)
AS
declare variable RES_ACL_ID D_ID;
declare variable RES_QUESTION_ID D_INTEGER;
declare variable RES_RD_ID D_INTEGER;
declare variable RES_STATE D_INTEGER;
declare variable RES_STR_VALUE D_STR_255;
declare variable ACL_OBJECT_KIND D_INTEGER;
declare variable ACL_OBJECT_ID D_ID;
declare variable TRACK_RESULT D_INTEGER;
declare variable ORIG_STATE D_INTEGER;
declare variable RES_ID_NOANSWER D_INTEGER;
declare variable Q_STATE D_INTEGER;
declare variable NOANSWER D_INTEGER;
declare variable NEED_CLOSE D_INTEGER;
declare variable RD_STR_ID D_STR_100;
declare variable ISVISITCHECK D_INTEGER;
declare variable NO_ANSWER_VALUE D_INTEGER;
declare variable RD_CODE D_STR_255;
declare variable QUESTION_CODE D_STR_100;
declare variable QUESTION_NAME D_STR_255;
declare variable RD_ID_HOME D_INTEGER;
declare variable RD_ID_BUSINESS D_INTEGER;
declare variable NEW_ACL_ID D_ID;
declare variable OPR_ID D_ID;
declare variable OVS_ID D_INTEGER;
declare variable OVS_ID1 D_INTEGER;
declare variable RES_ID_NONE D_INTEGER;
declare variable KOL D_INTEGER;
declare variable RES_ID_ANSWERYES D_INTEGER;
declare variable DB_ID D_INTEGER;
declare variable RD_STR_VALUE D_INTEGER;
declare variable ACL_ID1 D_ID;
declare variable C_ID ID_TYPE;
declare variable RVD_MIN D_CURRENCY;
declare variable RVD_MAX D_CURRENCY;
declare variable MONTHLY_INCOME_FACT D_CURRENCY;
declare variable PR_CODE STR_100;
declare variable MA_ID D_ID;
declare variable K_ID D_ID;
declare variable RES_ORIG_STATE D_INTEGER;
begin
  db_id = f_get_db_id();
  acl_date = coalesce(acl_date, current_date);

  select acl_object_id
  from actionlist
  where acl_id = :acl_id
  into c_id;

  if (db_id = 81001 and
      question_id in (f_get_rd_id('AnketaQuestions.CanBeSurety'), f_get_rd_id('AnketaQuestions.Nation'))) then
  begin
    ma_id = null;
    pr_code = '';

    select ma_id, pr_code
    from miniapp
    inner join miniappclient on mac_from_ma_id = ma_id
    inner join product on pr_id = ma_from_pr_id
    where mac_from_c_id = :c_id and
          ma_is_active = 1 and
          ma_date_end = '31.12.9999'
    order by ma_id desc rows 1
    into ma_id, pr_code;

    k_id = null;

    select k_id
    from kontract
    inner join kontstate on ks_from_k_id = k_id
    where k_from_ma_app_id = :ma_id and
          ks_ol_is_delete = 0 and
          current_date between ks_date_begin and ks_date_end and
          ks_state = 0
    into k_id;
  end

  -- процедура сохранения анкеты
  err_text = '';
  str_value = trim(str_value);
  rd_str_value = f_get_rd_id('Results.StrValue');

  -- Проверка числовых значений полей
  if (question_id <> 0) then
  begin
    select
      rd_code,
      rd_name
    from refdata
    where rd_id = :question_id
    into
      :question_code,
      :question_name;
    if (question_code in (
        'AnketaQuestions.living_expenses', -- Расход на проживание (в месяц)
        'AnketaQuestions.transfer_to_family', --  Переводы на Родину (в месяц)
        'AnketaQuestions.legal_docs_expenses', -- Расходы на патент и др. (в месяц)
        'AnketaQuestions.communal_expenses', -- Расход на жилье и ком.платежи (в месяц)
        'AnketaQuestions.CommitmentsSumMonth', -- Сумма платежей по обязательствам (в месяц)
        'work_monthly_income', -- Ежемесячный доход
        'work_extra_amount', -- Сумма доп. дохода (в месяц)
        -- КГ
        'monthly_income', --
        'AnketaQuestions.communal_expenses', --
        'living_expenses',
        'AnketaQuestions.repayment_obligations', --для тушения 099 блока
        'AnketaQuestions.unforeseen_expenses', --
        'AnketaQuestions.legal_docs_expenses', --
        'AnketaQuestions.Other_business_expenses', --
        'extra_amount', --
        'net_profit', --
        'amount_monthly_payments')) then
    begin
      if (str_value <> '' and
          str_value not similar to '[-]?[[:DIGIT:]]+') then
        begin
          err_text = 'Ответ на вопрос "' || trim(question_name) || '" должен быть числом; ';
          suspend;
          exit;
        end
    end

    ----проверка на заполненность в анкете целового мониторинга
    if (db_id = 41701 and rd_id = rd_str_value) then
    begin
      if (question_code in ('AnketaQuestions.descr_of_loan', 'monitoring.memo')
        and char_length(str_value) between 1 and 29) then
      begin
        err_text = 'Ответ на вопрос "' || trim(question_name) || '" Текст должен быть не меньше 30 символов; ';
        suspend;
        exit;
      end
      if (question_code = 'monitoring.memo'
        and str_value = '') then
      begin
        err_text = 'Необходимо заполнить "'|| trim(question_name) || '"; ';
        suspend;
        exit;
      end
    end

    if (db_id = 81001 and
        coalesce(k_id, ma_id) is not null and
        (pr_code containing '.universal.' or pr_code containing '.oivo_pack.') and
        question_id in (f_get_rd_id('AnketaQuestions.CanBeSurety'), f_get_rd_id('AnketaQuestions.Nation'))) then
    begin

      if (question_code = 'AnketaQuestions.Nation' and
          rd_id = f_get_rd_id('Nation.RU') and
          state = 0) then
        state = 1;
      if (question_code = 'AnketaQuestions.Nation' and
          rd_id <> f_get_rd_id('Nation.RU') and
          state = 1) then
        state = 0;
      if (question_code = 'AnketaQuestions.CanBeSurety' and
          rd_id = f_get_rd_id('AnketaAnswers.6719') and
          state = 0) then
        state = 1;
      if (question_code = 'AnketaQuestions.CanBeSurety' and
          rd_id <> f_get_rd_id('AnketaAnswers.6719') and
          state = 1) then
        state = 0;
    end

    if (db_id = 81001 and rd_id = rd_str_value) then
    begin
      if (question_code = 'work_monthly_income'
        and f_strtonum(str_value) < 10000) then
      begin
        rdb$set_context('USER_TRANSACTION', 'anketa_error', 1);
        err_text = trim(question_name) || ' должен быть не ниже 10000 руб.; ';
        suspend;
        exit;
      end
      if (question_code in ('work_monthly_income', 'work_extra_amount')
        and f_strtonum(str_value) > 500000 ) then
      begin
        rdb$set_context('USER_TRANSACTION', 'anketa_error', 1);
        err_text = trim(question_name) || ' свыше 500000 руб., проверьте цифры.; ';
        suspend;
        exit;
      end
    end
  end

  if (question_id <> 0) then
  begin
    -- сохранение строки
    update or insert into actionsave (acl_id, question_id, rd_id, state, str_value, orig_state, orig_str_value)
    values (:acl_id, :question_id, :rd_id, :state, :str_value, :state, :str_value)
    matching (acl_id, question_id, rd_id);
    exit;
  end

  -- последняя строка создана, обрабатываем всю анкету
  -- получаем последние значения из бд для сравнения измнений
  err_text = '';
  res_acl_id = acl_id;
  select
    acl_object_kind,
    acl_object_id,
    rd_code
  from actionlist
    inner join refdata
      on rd_id = acl_from_rd_id
  where acl_id = :acl_id
  into
    :acl_object_kind,
    :acl_object_id,
    :rd_code;

  -- Запрет на редактирование анкеты-опросника, если в этот день выдан кредит
  -- Для того, чтобы не менялся ПДН, и не подтирали значения на основании
  -- которых принимается решение выдать займ
  if (db_id = 81001 and rd_code = 'Anketa.Main') then
  begin
    if (coalesce(rdb$get_context('USER_TRANSACTION', 'anketa_error'), 0) <> 1) then
    begin
      select sum(f_strtonum(coalesce(str_value, 0)))
      from actionsave
      where acl_id = :acl_id and
            rd_id = f_get_rd_id('Results.StrValue') and
            question_id in (f_get_rd_id('work_monthly_income'), f_get_rd_id('work_extra_amount'))
      into monthly_income_fact;

      select rvd_min, rvd_max
      from hp_pdn_rvd_amount('clients', :c_id)
      into rvd_min, rvd_max;
  
      if (monthly_income_fact > rvd_max) then
      begin
        err_text = 'Ежемесячный доход в анкете ' || monthly_income_fact || ' руб. больше максимального ежемесячного дохода для получения займа - ' || rvd_max || ' руб.';
        suspend;
        exit;
      end
      else if (monthly_income_fact < rvd_min) then
      begin
        err_text = 'Ежемесячный доход в анкете ' || monthly_income_fact || ' руб. меньше минимального ежемесячного дохода для получения займа - ' || rvd_min || ' руб.';
        suspend;
        exit;
      end
    end
    if (exists(
      select *
      from knclient
        inner join kontract
          on k_id = kc_from_k_id
            and k_disb_date >= current_date
            and k_status = 1
      where kc_from_c_id = :acl_object_id
        and kc_amount_approved > 0)) then
    err_text = 'Редактирование личного листа с момента утверждения до дня выдачи запрещено';

    if (err_text <> '') then
    begin
      -- если в один день 2 выдачи то на второй контракт тоже даем редактироват анкуту
      if (exists(select * from kontstate, knclient, kontract
                where ks_ol_is_delete = 0
                and ks_state = 0
                and current_date between ks_date_begin and ks_date_end
                and ks_from_k_id = kc_from_k_id
                and kc_amount_approved > 0
                and kc_from_c_id = :acl_object_id
                and ks_from_k_id = k_id
                and k_status = 0
                and k_disb_date >= current_date))
      then
        err_text = '';
    end
  end

  if (err_text = '') then
  begin
    res_id_noanswer = (select rd_id from refdata where rd_code = 'Results.NoAnswer');
    res_id_none = (select rd_id from refdata where rd_code = 'Results.None');
    res_id_answeryes = (select rd_id from refdata where rd_code = 'Results.Yes');

    no_answer_value = 0;
    rd_id_home = 0;
    rd_id_business = 0;
  
    for
      select
        res_question_id,
        res_rd_id,
        res_state,
        res_str_value,
        track_result,
        orig_state
      from hp_actions_short_detail_tmp(
        :acl_object_kind,
        :acl_object_id,
        :rd_code,
        :acl_date)
      order by rd_orderno
    into
      :res_question_id,
      :res_rd_id,
      :res_state,
      :res_str_value,
      :track_result,
      :res_orig_state
    do
    begin
    if (db_id = 81001 and
          res_question_id in (f_get_rd_id('AnketaQuestions.CanBeSurety'), f_get_rd_id('AnketaQuestions.Nation'))) then
        update actionsave set
          orig_state = :res_orig_state,
          orig_str_value = :res_str_value,
          track_result = :track_result
        where acl_id = :res_acl_id
          and question_id = :res_question_id
          and rd_id = :res_rd_id;
      else
        update actionsave set
          orig_state = :res_state,
          orig_str_value = :res_str_value,
          track_result = :track_result
        where acl_id = :res_acl_id
          and question_id = :res_question_id
          and rd_id = :res_rd_id;
    end
  
    -- получаем параметры анкеты
    isvisitcheck = 0;
    select
      acl_object_kind,
      acl_object_id,
      rd_str_id
    from actionlist
      inner join refdata
        on rd_id = acl_from_rd_id
    where acl_id = :acl_id
    into
      :acl_object_kind,
      :acl_object_id,
      :rd_str_id;
    if (rd_str_id <> 'Anketa') then
      isvisitcheck = 1;
  
    need_close = 1; -- Можно закрыть задачу
    if (isvisitcheck = 0) then -- Если это анкета
    begin
      if (exists(
        select *
        from actionsave
        where acl_id = :acl_id
          and rd_id = :res_id_noanswer
          and state = 1)) then
      need_close = 0; -- Так как имеются не отвечанные вопросы, отменяем закрытие задачи

    end
    else -- если это визит осмотр
    begin
      if (exists(
        select *
        from actionsave
        where acl_id = :acl_id
          and state = 0
          and question_id = rd_id)) then
      need_close = 0; -- Так как имеются не отвечанные вопросы, отменяем закрытие задачи
    end
  
    -- Пробег по вопросам и удаление не измененных
    for
      select
        question_id
      from actionsave
      where acl_id = :acl_id
        and question_id = rd_id -- Признак вопроса
    into
      :res_question_id
    do
    begin
      if (not exists
        (
          select *
          from actionsave
          where acl_id = :acl_id
            and question_id = :res_question_id
            and (orig_state <> state or orig_str_value <> str_value)
        )
      ) then
      begin
        -- RFCKG-4126
        if (:db_id in (41701, 39801) and :rd_code = 'Anketa.TargetUseLoan') then
        begin
          -- Данный момент был затёрт, просто вернул как было
        end
        else
          delete from actionsave
          where acl_id = :acl_id
            and question_id = :res_question_id;
      end
    end
  
    -- Пробег по вопросам
    for
      select
        question_id,
        track_result,
        state
      from actionsave
      where acl_id = :acl_id
        and question_id = rd_id -- Признак вопроса
    into
      :res_question_id,
      :track_result,
      :q_state
    do
    begin
      -- RFCKG-4545
      if (db_id = 41701
        and f_get_rd_code(:res_question_id) in ('AnketaQuestions.Family', 'AnketaQuestions.Family_tunduk')
        and exists(
          select *
          from objectprops
            inner join objectvalues
              on ovs_from_opr_id = opr_id
                and ovs_id = opr_actual_ovs_id
            inner join objectvalres
              on ovr_from_opr_id = opr_id
                and ovr_from_ovs_id = ovs_id
                and ovr_result_rd_id <> f_get_rd_id('None') -- RFCKG-4632
          where opr_object_kind = :acl_object_kind
            and opr_object_id = :acl_object_id
            and opr_subject_kind = -1
            and opr_subject_id = :res_question_id)) then
      begin
        delete from actionsave
        where acl_id = :res_acl_id
          and question_id = :res_question_id;
        continue;
      end

      if (isvisitcheck = 0) then
      begin
        if (exists(
          select *
          from actionsave
          where acl_id = :acl_id
            and question_id = :res_question_id
            and question_id <> rd_id
            and state <> orig_state))
        then q_state = no_answer_value;
      end
      noanswer = 0;
  
      if (isvisitcheck = 0) then -- Если это анкета
      begin
        if (exists(
          select *
          from actionsave
          where acl_id = :acl_id
            and question_id = :res_question_id
            and rd_id = :res_id_noanswer
            and state = 1))
        then noanswer = 1;
      end

      -- RFCKZ-716
      if (:db_id = 39801 and noanswer = 1) then err_text = 'Все ответы должны быть заполнены';


      if (noanswer = 1) then -- Вопрос еще не отвечан
      begin
        if (exists(
          select *
          from objectprops
            inner join objectvalues
              on ovs_from_opr_id = opr_id
                and ovs_id = opr_actual_ovs_id
            inner join objectvalres
              on ovr_from_opr_id = opr_id
                and ovr_from_ovs_id = ovs_id
          where opr_object_kind = :acl_object_kind
            and opr_object_id = :acl_object_id
            and opr_subject_kind = -1
            and opr_subject_id = :res_question_id)) then

        -- Если ранее отвечан, то удаляем создав новую запись
        execute procedure hp_objectprops_set(
          :acl_object_kind,
          :acl_object_id,
          -1,
          :res_question_id,
          :acl_date,
          '',
          :no_answer_value,
          :un_id);
      end
      else
      begin
        -- Сохраняем новое значение
        execute procedure hp_objectprops_set(
          :acl_object_kind,
          :acl_object_id,
          -1,
          :res_question_id,
          :acl_date,
          '',
          :q_state,
          :un_id);

        select
          opr_id,
          opr_actual_ovs_id
        from objectprops
        where opr_object_kind = :acl_object_kind
          and opr_object_id = :acl_object_id
          and opr_subject_kind = -1
          and opr_subject_id = :res_question_id
        into
          :opr_id,
          :ovs_id;
        ovs_id1 = (select max(ovs_id) from objectvalues where ovs_from_opr_id = :opr_id);

        kol = 0;
  
        -- находим записи где старые и новые значения отличаются
        for
          select
            a.rd_id,
            state,
            orig_state,
            str_value,
            rq.rd_code
          from actionsave a
            inner join refdata rq
              on rq.rd_id = question_id
          where acl_id = :acl_id
            and question_id = :res_question_id
            and question_id <> a.rd_id
            and a.rd_id <> :res_id_noanswer
            and state = 1
        into
          :res_rd_id,
          :res_state,
          :orig_state,
          :res_str_value,
          :question_code
        do
        begin
          kol = kol + 1;
          delete from objectvalres
          where ovr_from_opr_id = :opr_id
            and ovr_from_ovs_id = :ovs_id
            and ovr_result_rd_id = :res_id_none;
  
          insert into objectvalres (ovr_from_opr_id, ovr_from_ovs_id, ovr_result_rd_id, ovr_state, ovr_value)
          values (:opr_id, :ovs_id1, :res_rd_id, 1, :res_str_value);
  
          if (orig_state <> res_state) then
          begin
            if (question_code = 'AnketaQuestions.VisitHomeNeeded'
              and res_rd_id = res_id_answeryes) then
            begin
              if (rd_id_home = 0) then
                rd_id_home = (
                  select
                    rd_id
                  from refdata
                  where rd_code = 'CheckAnketa.Home'
                    and rd_is_active = 1);
  
              if (rd_id_home <> 0) then
              begin
                select
                  acl_id
                from fm_anketa_add(
                  :acl_object_kind,
                  :acl_object_id,
                  :rd_id_home,
                  :un_id)
                into
                  :new_acl_id;
              end
            end
            if (question_code = 'AnketaQuestions.VisitBusinessNeeded'
              and res_rd_id = res_id_answeryes) then
            begin
              if (rd_id_business = 0) then
                rd_id_business = (
                  select
                    rd_id
                  from refdata
                  where rd_code = 'CheckAnketa.Business'
                    and rd_is_active = 1);
  
              if (rd_id_business <> 0) then
              begin
                select
                  acl_id
                from fm_anketa_add(
                  :acl_object_kind,
                  :acl_object_id,
                  :rd_id_business,
                  :un_id)
                into
                  :new_acl_id;
              end
            end
  
            if (question_code in (
              'AnketaQuestions.Family',
              'AnketaQuestions.Education',
              'AnketaQuestions.Nation',
              'AnketaQuestions.InfoSource',
              'AnketaQuestions.OKTELL.BirthDate',
              'AnketaQuestions.OKTELL.AddCellNo')) then
              execute procedure hp_do_anketasync(
                :question_code,
                :acl_object_kind,
                :acl_object_id,
                :res_rd_id,
                :res_str_value,
                :un_id);

            if (question_code = 'AnketaQuestions.LoanUseFact') then
              execute procedure hp_loan_target_check(
                :acl_id,
                :acl_object_kind,
                :acl_object_id,
                :res_rd_id,
                :un_id);
          end
        end
        if (kol = 0) then
          insert into objectvalres (ovr_from_opr_id, ovr_from_ovs_id, ovr_result_rd_id, ovr_state, ovr_value)
          values (:opr_id, :ovs_id1, :res_id_none, 1, '');
      end
    end
  
    if (need_close = 1) then -- если все вопросы с ответами, закрываем задачу
      update actionlist set
        acl_status = 2,
        acl_from_un_id = :un_id
      where acl_id = :acl_id;
  end

  -- После сохранения, очишаем данные из промежуточной таблицы
  delete from actionsave
  where acl_id = :res_acl_id;

  begin
    for
      select
        distinct acl.acl_id
      from actionsave acs
        inner join actionlist acl
          on acl.acl_id = acs.acl_id
            and acl.acl_date < current_date
    into
      :acl_id1
    do delete from actionsave where acl_id = :acl_id1;

    when any do
    begin
