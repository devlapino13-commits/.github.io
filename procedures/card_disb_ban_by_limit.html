CREATE PROCEDURE CARD_DISB_BAN_BY_LIMIT(
    REG_ID D_INTEGER,
    REP_DATE D_DATE)
RETURNS (
    IS_BAN D_BOOL,
    REASON D_STR_100)
AS
declare variable ucf_id d_integer;
declare variable ucf_limit_amount d_currency;
declare variable ucf_date_begin d_date;
declare variable ucf_date_end d_date;
declare variable mac_amount d_currency;
begin
  is_ban = 1;
  reason = 'Для текущего офиса выдача займов на карту запрещена';

  ucf_id = 0;
  ucf_limit_amount = 0;
  mac_amount = 0;

  select ucf_id, ucf_limit_amount, ucf_date_begin, ucf_date_end
  from underwritingcertif ucf
  where ucf_from_reg_id = :reg_id
  and :rep_date between ucf_date_begin and coalesce(ucf_date_end, '31.12.9999') -- период, на который установлен лимит по офису
  and ucf_active = 1 -- разрешено офису выдавать на карты МБулак
  and ucf_system = f_enum_id('access_sytem', 'card_disb_allow')
  into ucf_id,
       ucf_limit_amount, -- установленный лимит на период по офису
       ucf_date_begin, -- дата начала периода лимита
       ucf_date_end; -- дата конца периода лимита

  if (ucf_id > 0) then
  begin
    -- выдача займов на карту разрешена

    -- общая сумма займов, выданных и планиуремых к выдаче за период, на который установлен лимит
    select sum(mac.mac_amount)
    from miniapp ma
    inner join miniappclient mac on mac.mac_from_ma_id = ma.ma_id
    where ma.ma_office_id = :reg_id and
          ma.ma_date_oper between :ucf_date_begin and :ucf_date_end and -- фактическая или планируемая дата выдачи внутри периода лимита
          ma.ma_disb_kind = 6 and -- выдача на карту МБулак
          ma.ma_is_active = 1 and -- активные заявки
          ma.ma_date_end = '31.12.9999' -- актуальные заявки, последней версии, т.е.
                                        -- после возврата из договора в процессе или туда еще не попавшие
    into mac_amount;

    -- проверка на превышение установленного лимита
    if (mac_amount > ucf_limit_amount) then
    begin
      reason = 'Превышен лимит выдачи на карты МБулак!';
      suspend;
      exit;
    end

    -- лимит не превышен
    is_ban = 0;
    reason = '';
  end
  suspend;
end
