CREATE PROCEDURE AAREP_ANALITICA_PORT_LO_OWN_PAR(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_PROJECT VARCHAR(100),
    REP_PERIOD_KIND INTEGER)
RETURNS (
    AA_DISB_LO_ID DECIMAL(18,0),
    AA_LO_REGION_ID INTEGER,
    AA_LO_OFFICE_NAME VARCHAR(100),
    AA_LO_BRANCH_NAME VARCHAR(100),
    AA_LO_LEVEL_NAME VARCHAR(100),
    AA_LO_LEVEL2_NAME VARCHAR(100),
    AA_LO_LEVEL3_NAME VARCHAR(100),
    AA_DISB_LO_FIO VARCHAR(100),
    AA_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_KOLICH_CL_OWN_END INTEGER,
    AA_KOLICH_CL_ACTIVE_END INTEGER,
    AA_DELIQ_CL_KOLICH INTEGER,
    AA_DELIQ_CL_KOLICH7 INTEGER,
    AA_DELIQ_CL_KOLICH30 INTEGER,
    AA_DELIQ_CL_KOLICH60 INTEGER,
    AA_DELIQ_CL_KOLICH90 INTEGER,
    AA_DELIQ_CL_KOLICH180 INTEGER,
    AA_DELIQ_CL_KOLICH360 INTEGER,
    AA_DELIQ_BALANCE DECIMAL(18,4),
    AA_DELIQ_BALANCE7 DECIMAL(18,4),
    AA_DELIQ_BALANCE30 DECIMAL(18,4),
    AA_DELIQ_BALANCE60 DECIMAL(18,4),
    AA_DELIQ_BALANCE90 DECIMAL(18,4),
    AA_DELIQ_BALANCE180 DECIMAL(18,4),
    AA_DELIQ_BALANCE360 DECIMAL(18,4),
    AA_DELIQ_OS_ALL DECIMAL(18,4),
    AA_DELIQ_OS7 DECIMAL(18,4),
    AA_DELIQ_OS30 DECIMAL(18,4),
    AA_DELIQ_OS60 DECIMAL(18,4),
    AA_DELIQ_OS90 DECIMAL(18,4),
    AA_DELIQ_OS180 DECIMAL(18,4),
    AA_DELIQ_OS360 DECIMAL(18,4),
    AA_DELIQ_PERC_ALL DECIMAL(18,4),
    AA_DELIQ_PERC7 DECIMAL(18,4),
    AA_DELIQ_PERC30 DECIMAL(18,4),
    AA_DELIQ_PERC60 DECIMAL(18,4),
    AA_DELIQ_PERC90 DECIMAL(18,4),
    AA_DELIQ_PERC180 DECIMAL(18,4),
    AA_DELIQ_PERC360 DECIMAL(18,4),
    RC_VALUE_01 VARCHAR(100),
    RC_VALUE_02 VARCHAR(100),
    RC_VALUE_03 VARCHAR(100),
    RC_VALUE_04 VARCHAR(100),
    RC_VALUE_05 VARCHAR(100),
    RC_VALUE_06 VARCHAR(100),
    RC_VALUE_07 VARCHAR(100),
    RC_VALUE_08 VARCHAR(100),
    RC_VALUE_09 VARCHAR(100),
    RC_VALUE_10 VARCHAR(100),
    AA_LO_STAGE_BEGIN DATE,
    AA_LO_JOBNAME VARCHAR(100),
    PS_IS_ON_STAFF_STR VARCHAR(100))
AS
declare variable ps_is_on_staff integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select sum(aa_amount_active_end), sum(aa_kolich_cl_active_end), sum(aa_deliq_cl_kolich), sum(aa_deliq_cl_kolich7),
             sum(aa_deliq_cl_kolich30), sum(aa_deliq_cl_kolich60), sum(aa_deliq_cl_kolich90),
             sum(aa_deliq_cl_kolich180), sum(aa_deliq_cl_kolich360), sum(aa_deliq_balance), sum(aa_deliq_balance7),
             sum(aa_deliq_balance30), sum(aa_deliq_balance60), sum(aa_deliq_balance90), sum(aa_deliq_balance180),
             sum(aa_deliq_balance360), sum(aa_deliq_os_all), sum(aa_deliq_os7), sum(aa_deliq_os30), sum(aa_deliq_os60),
             sum(aa_deliq_os90), sum(aa_deliq_os180), sum(aa_deliq_os360), sum(aa_deliq_perc_all), sum(aa_deliq_perc7),
             sum(aa_deliq_perc30), sum(aa_deliq_perc60), sum(aa_deliq_perc90), sum(aa_deliq_perc180),
             sum(aa_deliq_perc360), aa_disb_kc_id, sum(aa_is_own * aa_kolich_cl_active_end)
      from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_project, :rep_period_kind, 0, 0, 0, 0, 0, 0)
      where aa_amount_active_end <> 0
      group by aa_disb_kc_id
      into :aa_amount_active_end, :aa_kolich_cl_active_end, :aa_deliq_cl_kolich, :aa_deliq_cl_kolich7,
           :aa_deliq_cl_kolich30, :aa_deliq_cl_kolich60, :aa_deliq_cl_kolich90, :aa_deliq_cl_kolich180,
           :aa_deliq_cl_kolich360, :aa_deliq_balance, :aa_deliq_balance7, :aa_deliq_balance30, :aa_deliq_balance60,
           :aa_deliq_balance90, :aa_deliq_balance180, :aa_deliq_balance360, :aa_deliq_os_all, :aa_deliq_os7,
           :aa_deliq_os30, :aa_deliq_os60, :aa_deliq_os90, :aa_deliq_os180, :aa_deliq_os360, :aa_deliq_perc_all,
           :aa_deliq_perc7, :aa_deliq_perc30, :aa_deliq_perc60, :aa_deliq_perc90, :aa_deliq_perc180, :aa_deliq_perc360,
           :aa_disb_lo_id, aa_kolich_cl_own_end
  do
  begin
    select c_name_fio
    from clients
    where c_id = :aa_disb_lo_id
    into :aa_disb_lo_fio;

    aa_lo_region_id = null;
    aa_lo_office_name = 'Нет приказа';
    aa_lo_jobname = 'Нет приказа';
    ps_is_on_staff = 0;
    aa_lo_stage_begin = null;
    aa_disb_lo_fio = upper(aa_disb_lo_fio);

    select reg_unicode, reg_code || ' ' || reg_name, jl_name, ps_is_on_staff
    from region, hr_pstatus, joblist, clients
    where reg_unicode = ps_from_reg_id and
          ps_from_jl_id = jl_id and
          ps_from_c_id = c_id and
          ps_date_end >= :rep_date_end and
          ps_date_begin <= :rep_date_end
--    and c_index_c_name_fio=:AA_DISB_LO_FIO
          and
          c_name_fio = :aa_disb_lo_fio
    order by ps_is_working desc
    rows 1
    into :aa_lo_region_id, :aa_lo_office_name, :aa_lo_jobname, :ps_is_on_staff;

    if (ps_is_on_staff is null) then
      ps_is_on_staff_str = 'Нет приказа';
    if (ps_is_on_staff = 0) then
      ps_is_on_staff_str = 'Внештатный';
    if (ps_is_on_staff = 1) then
      ps_is_on_staff_str = 'Штатный';

    select min(ps_date_begin)
    from hr_pstatus
    where ps_from_c_id = :aa_disb_lo_id and
          ps_status = 1
    into :aa_lo_stage_begin;

    if (aa_lo_region_id is not null) then
    begin
      select rc_value_01, rc_value_02, rc_value_03, rc_value_04, rc_value_05, rc_value_06, rc_value_07, rc_value_08,
             rc_value_09, rc_value_10, aa_curr_branch_name, aa_curr_level2_name, aa_curr_level3_name, aa_curr_level4_name

      from aarep_get_office_categ(:aa_lo_region_id, :rep_date_end)
      into :rc_value_01, :rc_value_02, :rc_value_03, :rc_value_04, :rc_value_05, :rc_value_06, :rc_value_07,
           :rc_value_08, :rc_value_09, :rc_value_10, :aa_lo_branch_name, aa_lo_level_name, aa_lo_level2_name,
           aa_lo_level3_name;

    end
    else
    begin
      aa_lo_branch_name = 'Не определен';
      aa_lo_level_name = 'Не определен';
      aa_lo_level2_name = 'Не определен';
      aa_lo_level3_name = 'Не определен';
    end
    suspend;
  end

end
