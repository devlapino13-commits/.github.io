CREATE PROCEDURE HP_DO_REPAYMENT(
    K_ID DECIMAL(18,0),
    C_ID DECIMAL(18,0),
    REP_DATE DATE,
    REPAY_AMOUNT DECIMAL(18,4),
    REPAY_TO_OS INTEGER,
    REPAY_TO_PERC INTEGER,
    REPAY_TO_FINE INTEGER,
    REPAY_TO_KOMMIS INTEGER,
    NEED_SAVE INTEGER,
    IS_FULL_REPAY INTEGER,
    IS_WRITEOFF INTEGER,
    REPAY_FACT_OFFICE_ID INTEGER,
    REPAY_DT_ACCOUNT_KIND INTEGER,
    REPAY_DT_ACCOUNT_NO VARCHAR(20),
    REPAY_CURR_RATE DECIMAL(18,4),
    REPAY_USER_ID DECIMAL(18,0),
    REPAY_DOCUMENT_CLIENT DECIMAL(18,0),
    REPAY_DOC_CLIENT_NAME_REASON VARCHAR(255),
    REPAY_MANAGER_C_ID DECIMAL(18,0),
    REPAY_FROM_RD_ID INTEGER,
    REPAY_CONTROL_AMOUNT DECIMAL(18,4) = 0,
    NEED_WRITEOFF_OPEREXTRA INTEGER = 0)
RETURNS (
    OL_ID DECIMAL(18,0),
    LOAN_OL_ID DECIMAL(18,0),
    OL_REPAY_AMOUNT DECIMAL(18,4),
    ERROR_TEXT VARCHAR(255),
    OL_ID_STR VARCHAR(255),
    IM_ID D_ID)
AS
declare variable s_running_total decimal(18,4);
declare variable res_amount decimal(18,4);
declare variable ol_doc_kind integer;
declare variable reg_code varchar(20);
declare variable kroff_rd_tag integer;
declare variable db_id integer;
declare variable last_kn_date d_date;
declare variable show_zero_line d_smallint;
declare variable rd_code str_255;
declare variable tmp_amount d_currency;
declare variable tmp_kommis d_currency;
declare variable operrepay_ol_id d_id;
declare variable operrepay_main_ol_id d_id;
declare variable date_plan d_date;
declare variable amount_plan d_currency;
declare variable kd_percent_effect d_percent;
declare variable tmp_reg_id d_integer;
declare variable k_disb_kind d_integer;
declare variable pr_id d_id;
declare variable k_disb_date d_date;
declare variable k_kolich_client_approved d_integer;
declare variable cs_id d_integer;
declare variable k_kontract_no d_str_20;
declare variable c_name_fio d_str_100;
declare variable need_psk d_bool;
declare variable resh_kind d_integer;
declare variable zad_os d_currency;
declare variable pr_code d_str_100;
declare variable easy_start d_boolean;
declare variable payed_os d_currency;
declare variable payed_perc d_currency;
declare variable k_amount_approved d_currency;
declare variable k_stop_perc d_date;
declare variable pant_id d_id;
begin
  if (repay_dt_account_kind = -3) then
    repay_dt_account_kind = 3;

  -- Проверка на предмет перепринятия погашения
  select topr_from_ol_id, topr_fact_reg_id
  from temp_operrepay
  where topr_from_k_id = :k_id
  into operrepay_ol_id, tmp_reg_id;

  if (operrepay_ol_id is not null) then
  begin
    -- Если имеется запись во временной таблице temp_operrepay, значит это не новое погашение,
    -- а перепринятие существующего, привзяка будет на существующую операцию

    select iif(ol_parent_id is not null, ol_parent_id, ol_id)
    from operlist
    where ol_id = :operrepay_ol_id
    into operrepay_main_ol_id;

    select iif(:tmp_reg_id is not null, :tmp_reg_id, ol_from_reg_id), ol_user_id, ol_document_client,
           ol_doc_client_name, ol_manager_c_id, ol_from_rd_id
    from operlist
    where ol_id = :operrepay_main_ol_id
    into repay_fact_office_id, repay_user_id, repay_document_client, repay_doc_client_name_reason, repay_manager_c_id,
         repay_from_rd_id;

    select opl_acc_kind, opl_from_al_id
    from operloan
    where opl_from_ol_id = :operrepay_main_ol_id
    into :repay_dt_account_kind, :repay_dt_account_no;

    repay_curr_rate = 1.0000;
    repay_control_amount = 0;
    need_writeoff_operextra = 0;
  end

  need_writeoff_operextra = coalesce(need_writeoff_operextra, 0);

  if ((need_writeoff_operextra = 1) and
      (is_writeoff <> 1)) then
    need_writeoff_operextra = 0;

  error_text = '';
  if (coalesce(k_id, 0) = 0) then
    exception e_default 'input parameter K_ID is null';

  -- Repay_Amount=-777 означает прием полной задолженности

  -- REPAY_CONTROL_AMOUNT
  -- если значение <>0 и при сохранении не совпадет с текущим общий итогом (s_running_total),
  -- то не будет сохраняться, возможно другой пользователь уже принял погашение
  -- OL_ID вернет -1.

  if (c_id is null) then
    c_id = 0;
  if (is_writeoff is null) then
    is_writeoff = 0;

  if (repay_from_rd_id is null) then
    repay_from_rd_id = 0;
  if (repay_doc_client_name_reason is null) then
    repay_doc_client_name_reason = '';
  if (repay_curr_rate is null) then
    repay_curr_rate = 0;

  if (repay_to_os is null) then
    repay_to_os = 1;
  if (repay_to_perc is null) then
    repay_to_perc = 1;
  if (repay_to_fine is null) then
    repay_to_fine = 1;
  if (repay_to_kommis is null) then
    repay_to_kommis = 1;

  if (repay_control_amount is null) then
    repay_control_amount = 0;

  db_id = f_get_db_id();
  im_id = 0;
  show_zero_line = 1;

  select k_disb_kind, k_from_pr_id, k_disb_date, k_kolich_client_approved, k_kontract_no, k_amount_approved,
         k_stop_perc, pr_code
  from kontract kont, product
  where k_id = :k_id and
        k_from_pr_id = pr_id
  into :k_disb_kind, pr_id, k_disb_date, k_kolich_client_approved, k_kontract_no, k_amount_approved,
       :k_stop_perc, :pr_code;

  --
  if (db_id in (39801, 41701) and k_disb_kind = 6 and rep_date > current_date and need_save <> 0) then
  begin
    error_text = 'Погашение карточного кредита будущей датой недопустимо.';
    suspend;
  end

  -- если в группе всего один участник, тогда погашения доли участиника быть не может
  if (k_kolich_client_approved = 1) then
    c_id = 0;

  -- Заполняем задолженность на дату
  -- если кредит списывается, то ставку НсП делаем 0.
  select res_amount, s_running_total
  from hp_repay_fill(:k_id, :c_id, :rep_date, :is_writeoff, :repay_to_os, :repay_to_perc, :repay_to_fine, :repay_to_kommis, :show_zero_line)
  into :res_amount, -- сумма по на текущую дату
       :s_running_total; -- сумма для полного погашения

  if (is_writeoff = 1 and
      coalesce(repay_from_rd_id, 0) <> 0) then
  begin
    -- Если rd_tag причины списания равно 1,
    -- значит после списания исключаем (списываем) из внесистемного учета
    select coalesce(rd_tag, 0), rd_code
    from refdata
    where rd_id = :repay_from_rd_id
    into :kroff_rd_tag, rd_code;

    update temp_params
    set rd_exclude_from_operkroff = :kroff_rd_tag,
        rd_code_temp = :rd_code
    where 1 = 1;
  end

  ol_repay_amount = repay_amount;
  if (need_save = 0) then -- если это не погашение, а предварительный просмотр
  begin
    if (is_full_repay = 1) then
      ol_repay_amount = s_running_total; -- берем полную сумму для закрытия
    else
    if (repay_amount < 0) then
      ol_repay_amount = :res_amount; -- берем сумму на день оплаты
    if (ol_repay_amount < 0) then
      ol_repay_amount = 0;
  end
  else
  begin
    if (repay_amount = -777) then
      ol_repay_amount = s_running_total;

    if (repay_control_amount > 0 and
        s_running_total <> repay_control_amount) then
    begin
      ol_id = -1;
      error_text = 'Сумма задолженности изменилась.' || ascii_char(13) || ascii_char(10);
      error_text = error_text || 'Возможно другой пользователь уже принял погашение.' || ascii_char(13) || ascii_char(10);
      error_text = error_text || 'Попробуйте повторить операцию заново.';
      suspend;

      delete from temp_operrepay
      where topr_from_k_id = :k_id;

      exit;
    end
  end

  if (need_save = 1 and
      db_id = 81001 and
      rep_date >= '01.01.2018') then
  begin
    select min(kc_last_ol_date)
    from knclient
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0
    into last_kn_date;
  end

  -- RFCRU-5204 "Лёгкий старт"
  easy_start = False;
  if (db_id = 81001 and is_writeoff = 0 and (rep_date - k_disb_date) <= 15 and pr_code containing '.easy_start.') then
  begin
    select -sum(ok_os), -sum(ok_perc)
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_type = 4 and
          ok_ol_date <= :rep_date
    into :payed_os, :payed_perc;

    payed_os = coalesce(payed_os, 0);
    payed_perc = coalesce(payed_perc, 0);
    easy_start = (ol_repay_amount + payed_os + payed_perc) >= k_amount_approved;
  end

  -- Распределить погашение
  if (easy_start) then
    execute procedure hp_repay_alloc_lite(:rep_date, :k_id, :ol_repay_amount, :payed_perc);
  else
    execute procedure hp_repay_alloc(:rep_date, :k_id, :ol_repay_amount, :repay_to_os, :repay_to_perc, :repay_to_fine,
                                     :repay_to_kommis, :is_writeoff);

  -- если это кредит по картам МБулак --------- >
  if ((db_id in (39801, 41701)) and
      (k_disb_kind = 6)) then
  begin
    update temp_vrepay tvr
    set tvr.s_all = tvr.pay_perc + tvr.pay_tax,
        tvr.pay_perc = tvr.pay_perc + tvr.pay_tax,
        tvr.graph_diff_perc = 0,
        tvr.pay_tax = 0,
        tvr.s_running_total = tvr.pay_perc + tvr.pay_tax
    where tvr.s_rec_no = 1 and
          tvr.is_extra = 1 and
          tvr.zad_os = 0 and
          tvr.zad_perc > 0 and
          tvr.zad_fine = 0;
  end
  --------------------------------------------- <

  -- если это отлож. коммисия ИПФ --------- >
  if ((db_id in (39801, 41701)) and
      :k_disb_date > cast('07.12.2021' as date) and
      exists(select *
             from product pr
             where pr.pr_id = :pr_id and
                   pr.pr_code containing '.ipf.' and
                   (coalesce(pr.pr_ipf_commission, 0) <> 0))) then
  begin
    update temp_vrepay tvr
    set tvr.s_all = tvr.pay_perc + tvr.pay_tax,
        tvr.pay_perc = tvr.pay_perc + tvr.pay_tax,
        tvr.graph_diff_perc = 0,
        tvr.pay_tax = 0,
        tvr.s_running_total = tvr.pay_perc + tvr.pay_tax
    where tvr.s_rec_no = 1 and
          tvr.is_extra = 1 and
          tvr.zad_os = 0 and
          tvr.zad_perc > 0 and
          tvr.zad_fine = 0;
  end
  --------------------------------------------- <
  select sum(zad_os + calcing_perc + zad_fine - pay_os - pay_perc - pay_fine), sum(zad_kommis - pay_kommis)
  from temp_vrepay
  into tmp_amount, tmp_kommis;
  if (tmp_amount is null) then
    tmp_amount = 0;
  if (tmp_kommis is null) then
    tmp_kommis = 0;

  if (repay_dt_account_kind = 1 and
      tmp_amount <= 0 and
      tmp_kommis > 0) then
  begin
    -- при погашении через кассу, не даем закрыть кредит без комиссии за досрочное
    ol_id = -1;
    error_text = 'Комиссия за досрочное погашение не оплачена.' || ascii_char(13) || ascii_char(10);
    error_text = error_text || 'Выберите "Полное закрытие долга по договору", или укажите сумму с учетом комиссии.';
    suspend;

    delete from temp_operrepay
    where topr_from_k_id = :k_id;

    exit;
  end

  if (repay_dt_account_kind = 1 and
      db_id = 81001 and
      exists(select * from product where pr_id = :pr_id
      and pr_code containing '.reliable_emp.')
      and rep_date <= '31.08.2022') then
  begin
    -- при погашении через кассу по надженому сотруднику не даем схранить
    ol_id = -1;
    error_text = 'По продукту "Надежный сотрудник" погашении через кассу запрещено!' || ascii_char(13) || ascii_char(10);
    suspend;

    delete from temp_operrepay
    where topr_from_k_id = :k_id;

    exit;
  end

  if (repay_dt_account_kind = 1 and
      db_id = 81001 and
      pr_code containing '.business.') then
  begin
    -- при погашении через кассу по ИП не даем схранить
    ol_id = -1;
    error_text = 'По продукту "Бизнес" погашение через кассу запрещено!' || ascii_char(13) || ascii_char(10);
    suspend;

    delete from temp_operrepay
    where topr_from_k_id = :k_id;

    exit;
  end

  /*
  if (repay_dt_account_kind = 1 and
      db_id = 81001) then 
  begin
    if (coalesce(repay_user_id, 0) not in (1010100000529, 1010100002814)) then
    begin
      select res_text
      from hp_check_kont_refund(:k_id, :rep_date)
      into error_text;

      if (error_text <> '') then
      begin
        ol_id = -1;
        suspend;

        delete from temp_operrepay
        where topr_from_k_id = :k_id;

        exit;
      end
    end
  end       */

  if (repay_dt_account_kind = 1 and
      operrepay_ol_id is null) then -- если наличная оплата, проверяем возможность оплаты
  begin
    select res_error_text
    from hp_do_repayment_check(:k_id, :rep_date, :repay_amount)
    into error_text;

    if (error_text <> '') then
    begin
      ol_id = -1;
      suspend;

      delete from temp_operrepay
      where topr_from_k_id = :k_id;

      exit;
    end
  end

  -- Автореструктуризация графика
  need_psk = 0;
  resh_kind = 0;

  if (f_ipf_newrepmode(rep_date, k_id, is_writeoff) = 1) then
    resh_kind = 777;

  select sum(pay_os)
  from temp_vrepay
  where s_date is not null and
        is_vac_perc = 1
  into zad_os;

  if (db_id = 39801 and k_stop_perc is not null) then
  begin
    update temp_vrepay
    set sh_new_os = zad_os - pay_os,
        sh_new_perc = zad_perc - pay_perc
    where s_date is not null and
          is_extra = 0;
  end ---else  RFCKZ-747 --закомментировала из-за *else не вызывался hp_repay_reschedule, который обнуляет sh_new_perc будущих периодов
  if (not easy_start) then
  begin
    select need_psk
    from hp_repay_reschedule(:rep_date, :resh_kind, :repay_dt_account_kind, :k_id)
    into need_psk; -- Требуется ли уведомление клиента
  end

  if (need_save = 1) then -- Если необходимо сохранить операцию
  begin
    if (repay_fact_office_id is null) then
      repay_fact_office_id = 0;

    if (repay_fact_office_id = 0) then
    begin
      select kal_new_reg_id, reg_code
      from knaktlist, kontract, region
      where k_id = :k_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            kal_date <= :rep_date and
            kal_date_end >= :rep_date and
            reg_unicode = kal_new_reg_id
      into :repay_fact_office_id, reg_code;
      if (coalesce(repay_dt_account_no, '') <> '') then
      begin
        if (reg_code <> left(repay_dt_account_no, 4)) then
          repay_dt_account_no = reg_code || substring(repay_dt_account_no from 5);
      end
    end

    -- ВС о досрочном погашении от третьего лица
    if (db_id = 81001 and
        repay_dt_account_kind = 1 and
        is_writeoff = 0 and
        repay_doc_client_name_reason <> '') then
    begin
      select iif(oks_delinq_date > :rep_date, oks_delinq_date, :rep_date),
             iif(oks_delinq_os + oks_delinq_perc = 0, oks_curr_sh_os + oks_curr_sh_perc, oks_delinq_os + oks_delinq_perc)
      from oksummary
      where oks_from_k_id = :k_id and
            :rep_date between oks_date_begin and oks_date_end
      into :date_plan, :amount_plan;
    end

    -- Создание операции и сохранение Operbook
    select ol_id, ol_main_id, ol_document_kind
    from hp_repay_save_operbook(:k_id, :rep_date, :is_writeoff, :repay_fact_office_id, :repay_dt_account_kind, :repay_dt_account_no, :repay_curr_rate, :repay_user_id, :repay_document_client, :repay_doc_client_name_reason, :repay_manager_c_id, :repay_from_rd_id, :repay_to_os, :repay_to_perc, :repay_to_fine, :need_writeoff_operextra)
    into :loan_ol_id, ol_id, ol_doc_kind;

    -- если операция успешно создана, сохранение Operkred в связке с OL_ID
    if (loan_ol_id is not null) then
    begin
      -- проверка баланса по карте МБулак(МПЦ)
      if (db_id in (39801, 41701) and k_disb_kind = 6) then
        execute procedure card_check_balance_new('kontract', :k_id) returning_values :cs_id;

      if (repay_to_kommis > 1 and  uxf$flag('operdel_executing') = 1) then
      begin
        -- значит передана запись по перепринятию погашения из OperDelList

        update operdellist
        set odl_new_ol_id = :ol_id
        where odl_id = :repay_to_kommis;

        if (row_count = 0) then
          exception e_message 'Неверный ODL_ID: ' || repay_to_kommis;
      end

      if (easy_start) then
        execute procedure hp_repay_save_operkred(:k_id, :c_id, :loan_ol_id, :rep_date, -1,
                                                :repay_dt_account_kind, :need_writeoff_operextra);
      else
        execute procedure hp_repay_save_operkred(:k_id, :c_id, :loan_ol_id, :rep_date, :is_writeoff,
                                                :repay_dt_account_kind, :need_writeoff_operextra);

      if (amount_plan > 0 and
          operrepay_ol_id is null) then
        im_id = (select im_id
                 from hp_imessage_create(:ol_id, :date_plan, :amount_plan));

      if (db_id = 81001 and
          need_psk = 1 and
          operrepay_ol_id is null) then
      begin
        -- Формируем дополнительный СМС о досрочно-частичном погашении.
        select kps_psk
        from hp_edoc_ins_kontpsk(:k_id, :loan_ol_id, :repay_user_id)
        into :kd_percent_effect;
      end

      -- операция погашения по карте МБулак (МПЦ)
      execute procedure hp_mbulak_card_payment(:k_id, :loan_ol_id);
      -- Мурабаха аннуитет
      if (is_writeoff = 0) then
        execute procedure add_markup_discount(:k_id, :loan_ol_id);

      -- По просьбе комплаенс офицера отправлять письмо о погашении данных кредитов
      if (db_id in (39801, 41701) and repay_amount > 0 and k_id in (109010176108200101, 109010178450000101)) then
      begin
        --Кредиты индивидуальные, поэтому можно не бояться multiple rows в нижестоящем запросе
        select cl.c_name_fio
        from knclient kc
        inner join clients cl on cl.c_id = kc.kc_from_c_id
        where kc.kc_from_k_id = :k_id and
              kc.kc_amount_approved > 0
        into c_name_fio;

        execute procedure hp_send_quick_message('email', 'compliance@mbulak.kg', 'Уведомление о погашении',
                                                                                   'ФИО Клиента: ' || :c_name_fio || ', ' ||
                                                                                   'Номер договора: ' || :k_kontract_no,
                                                '', 1, 1000000000000);

      end 
    end
  end
  else
  begin
    if (repay_amount <= 0 and
        c_id <> 0) then
    begin
      select sum(pay_os + pay_perc + pay_fine + pay_tax + pay_kommis + pay_pereplat + pay_extra)
      from temp_vrepay
      into :ol_repay_amount;
    end
  end
  if (ol_id is null) then
    ol_id = 0; -- OL_ID основной операции (приходник)
  if (loan_ol_id is null) then
    loan_ol_id = 0; -- OL_ID связанный с кредитной операцией

  if (ol_id > 0) then
    ol_id_str = ol_id || '-' || :ol_doc_kind;
  else
    ol_id_str = '';

  -- формируем проводки по ЕПС
  if (need_save = 1 and
      loan_ol_id <> 0 and
      db_id = 81001 and
      rep_date >= '01.01.2018') then
  begin
    if (last_kn_date is not null and
        last_kn_date = rep_date) then
    begin
      update knclient
      set kc_last_ol_date = :rep_date
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0;
    end

    execute procedure hp_do_kn_operbook_one_online(:k_id, :rep_date, :loan_ol_id, null);
  end

  delete from temp_operrepay
  where topr_from_k_id = :k_id;

  if (need_save = 1) then
  begin
    delete from pantera_preexec ppe
    where ppe.ppe_from_k_id = :k_id;
  end

  if (need_save = 1 and db_id = 81001) then   -- RFCRU-5953
  begin
    select oal.oal_from_pant_id
    from oktell_api_pant_list oal
    where oal.oal_from_k_id = :k_id
    into pant_id;

    if (coalesce(pant_id, 0) <> 0) then
    begin
      insert into oktell_api_pant_for_del (oad_from_pant_id)
      values (:pant_id);

      post_event 'oktell_api_delete_subscribers_by_pant_id';
    end
  end

  suspend;
end
