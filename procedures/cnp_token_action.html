CREATE PROCEDURE CNP_TOKEN_ACTION(
    C_ID D_ID,
    ACTION_CODE D_STR_10,
    TOKEN_JSON D_BLOB_TEXT)
AS
declare variable TOKEN D_STR_100;
declare variable CARDNUMBER D_STR_100;
declare variable MERCHANT_NUMBER D_STR_CODE;
declare variable TRANSACTION_NUMBER D_STR_100;
declare variable ET_ID ID_TYPE;
begin
    -- Процедура для обработки запросов по токенам для Processing.kz (для сервиса Володи)
    if (action_code in ('add', 'update')) then
    begin
        token = f_json_str_get_value(:token_json, 'token', '');
        cardnumber = f_json_str_get_value(:token_json, 'cardnumber', '');
        merchant_number = f_json_str_get_value(:token_json, 'merchant_number', '');
        transaction_number = f_json_str_get_value(:token_json, 'transaction_number', '');        

        insert into extsys_token (et_from_c_id, et_token, et_is_actual, et_cardnumber)
        values (:c_id, :token, 1, :cardnumber)
        returning et_id
        into :et_id;

        insert into extsys_token_add (eta_from_et_id, eta_token_id, eta_name_on_card, eta_state)
        values (:et_id, :merchant_number, :transaction_number, 1);
    end
    else if (action_code = 'delete') then
    begin
        update extsys_token et
        set et_is_actual = 0
        where et.et_from_c_id = :c_id and
              et.et_is_actual = 1;
    end
end
