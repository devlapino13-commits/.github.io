CREATE PROCEDURE HP_DO_EXPRESSAPP_ACTION_CHECK(
    EXA_ID D_INTEGER)
RETURNS (
    RES_STATUS D_SMALLINT)
AS
declare variable c_id id_type;
declare variable delinq_day d_integer;
declare variable lgc_calc d_integer;
declare variable asp_status d_str_code;
declare variable k_id d_id;
declare variable db_id d_integer;
declare variable point d_integer;
declare variable max_balance curr_type;
declare variable cli_balance curr_type;
begin
  db_id = f_get_db_id();

  res_status = 0;
  c_id = null;
  select exa_from_c_id, exa_base_k_id
  from expressapp
  where exa_id = :exa_id
  into c_id, k_id;

  if (row_count = 0) then
  begin
    -- не имеется запись для данного клиента в таблице expressapp
    res_status = -1;
    suspend;
    exit;
  end

  if (db_id = 81001) then
  begin
    /*
    if (res_status = 0) then
    begin
      -- проверка на легитимный платеж
      lgc_calc = null;
      select sum(lgc_calc)
      from legitimcalc
      where lgc_from_c_id = :c_id
      and lgc_from_k_id = :k_id
      and lgc_date <= current_date
      into lgc_calc;
      lgc_calc = coalesce(lgc_calc,0);
      if (lgc_calc <= 0) then
      begin
        res_status = -1;
      end
    end
    */

    if (res_status = 0) then
    begin
      -- проверка на действующий догоовр АСП
      asp_status = f_edoc_status('asp.agreement', :c_id, current_date);
      if (asp_status <> 'signed') then
      begin
        res_status = -1;
      end
    end
    -- проверка на сотрудника
    if (res_status = 0) then
    begin
      if (exists(select 1 from hr_pstatus  -- сотрудников исключаем
      where ps_from_c_id = :c_id
      and current_date between ps_date_begin and ps_date_end
      and ps_status = 1
      and ps_is_working = 1)) then
      begin
        res_status = -1;
      end 
    end 
    -- проверка имеется ли МП у клиента
    if (res_status = 0) then
    begin
      if (f_fsm_client_status(c_id) = 0) then
      begin
        res_status = -1;
      end 
    end
    -- проверка суммы
    if (res_status = 0) then
    begin
      select sum(oks_balance),
             max(current_date - oks_delinq_date)
      from knclient, oksummary
      where kc_from_c_id = :c_id
      and kc_amount_approved > 0
      and kc_from_k_id = oks_from_k_id
      and current_date between oks_date_begin and oks_date_end
      and oks_close_state = 0
      into cli_balance, delinq_day;

      cli_balance = coalesce(cli_balance, 0);
      delinq_day = coalesce(delinq_day,0);
      if (delinq_day < 0) then delinq_day = 0;

      -- если имеется просрочка то исключаем
      if (delinq_day > 0 ) then
      begin
        res_status = -1;
      end

      -- если легитимный платеж менее 2х исключаем
      if (res_status = 0) then
      begin
        select sum(lgc_calc) from legitimcalc
        where lgc_from_k_id = :k_id
        into lgc_calc;
        lgc_calc = coalesce(lgc_calc,0);
  
        if (lgc_calc < 2) then
        begin
          res_status = -1;
        end 
      end
      -- проверка на доступную сумму
      if (res_status = 0) then
      begin
        point = f_point(c_id, current_date);
  
        max_balance = case
                        when point >= 24 then 150000
                        when point >= 18 then 120000
                        when point >= 15 then 100000
                        when point >= 12 then 80000
                        when point >= 9 then 60000
                        when point >= 6 then 50000
                        when point >= 3 then 40000
                        else 35000
                      end;
         if (max_balance - cli_balance < 35000) then
         begin
           res_status = -1;
         end 
      end
    end
  end

  if (db_id in (39801, 41701)) then
  begin
    if (res_status = 0) then
    begin
      point = f_point(c_id, current_date);
      -- балл клиента должно быть не менее 10ти баллов
      if (point < 10) then
        res_status = -1;
    end 
  end 

  if (res_status = 0) then
  begin
    -- проверка на активную просрочку
    delinq_day = null;
    select max(current_date - oks_delinq_date) from knclient, oksummary
    where kc_from_c_id = :c_id
    and kc_amount_approved > 0
    and kc_from_k_id = oks_from_k_id
    and current_date between oks_date_begin and oks_date_end
    and current_date > oks_delinq_date
    into delinq_day;
    delinq_day = coalesce(delinq_day,-1);
    if (delinq_day > 0) then res_status = -1;
  end

  if (res_status = -1) then
  begin
    update expressapp
    set exa_final_status = -1
    where exa_id = :exa_id;
  end 

  suspend;
end
