CREATE PROCEDURE ELCARD_TRANSFER_STATUS_LIST
RETURNS (
    ESM_ID D_ID,
    ORDER_ID D_STR_CODE)
AS
begin
    -- процедура возвращает список операций, по которым нам нужно получить статус пеервода денег на сторонний Элкарт
    for select em.esm_id,
               em.esm_account
    from extsysmain em
    where em.esm_project = 'elcard_disb' and
    em.esm_action = 'check_transfer_status' and
    em.esm_status = 0
    into :esm_id,
         :order_id
    do
        suspend;
end
