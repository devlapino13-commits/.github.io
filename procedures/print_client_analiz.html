CREATE PROCEDURE PRINT_CLIENT_ANALIZ(
    KC_ID D_ID)
RETURNS (
    CLIENT_FIO D_STR_100,
    MA_NO D_INTEGER,
    PRODUCT_NAME D_STR_100,
    FOLDER_NAME D_STR_100,
    LOAN_AMOUNT D_CURRENCY,
    LOAN_TARGET D_STR_100,
    MONTHLY_INCOME DECIMAL(15,2),
    EXTRA_AMOUNT DECIMAL(15,2),
    COMMUNAL_EXPENSES DECIMAL(15,2),
    LIVING_EXPENSES DECIMAL(15,2),
    LEGAL_DOCS_EXPENSES DECIMAL(15,2),
    OTHER_BUSINESS_EXPENSES DECIMAL(15,2),
    REPAYMENT_OBLIGATIONS DECIMAL(15,2),
    NET_PROFIT DECIMAL(15,2),
    AMOUNT_MONTHLY_PAYMENTS DECIMAL(15,2),
    KPD DECIMAL(15,2),
    SVZ_FIO D_STR_100,
    APPROVED_FIO D_STR_100,
    SVZ_MEMO D_STR_2048,
    WORK_PLACE_NAME D_STR_LONG,
    DESCR_OF_LOAN D_STR_LONG,
    WORK_ADDRESS_ORIENTIR D_STR_LONG,
    LASTCH_DATE D_DATE,
    DATA_SOURCE D_STR_255)
AS
declare variable C_ID D_ID;
declare variable KF_ID D_ID;
declare variable MA_ID D_INTEGER;
declare variable K_ID D_ID;
declare variable KL_DATE D_DATE;
declare variable REP_DATE D_DATE;
declare variable OBJ_NAME D_STR_20;
declare variable OBJ_ID D_ID;
declare variable MAC_ID D_INTEGER;
declare variable IS_APPL D_BOOL;
declare variable PR_CODE D_STR_100;
declare variable K_DISB_DATE D_DATE;
declare variable LANG D_STR_10;
declare variable CA_ID D_ID;
begin
  data_source = '';

  select kont.k_id, kont.k_from_kf_id, kont.k_from_ma_app_id, cl.c_id, cl.c_name_fio, kncl.kc_amount_approved,
         pr.pr_name, clapp.c_name_fio, pr.pr_code, kont.k_fact_disb_date, cl.c_blank_lang
  from knclient kncl
  inner join kontract kont on kont.k_id = kncl.kc_from_k_id
  inner join clients cl on cl.c_id = kncl.kc_from_c_id
  inner join product pr on pr.pr_id = kont.k_from_pr_id
  left join clients clapp on clapp.c_id = kont.k_approved_manager
  where kncl.kc_id = :kc_id
  into k_id, kf_id, ma_id, c_id, client_fio, loan_amount, product_name, approved_fio, pr_code, k_disb_date, lang;

  loan_amount = round(loan_amount, 2);

  select max(kl.kl_datetime)
  from kontlog kl
  where kl.kl_from_k_id = :k_id and
        kl.kl_state = 1
  into kl_date;

  svz_fio = '';
  select kal.kal_new_kf_name, cl.c_name_fio
  from knaktlist kal
  inner join clients cl on kal.kal_new_kf_officer = cl.c_id
  where kal.is_delete = 0 and
        kal.kal_from_kf_id = :kf_id and
        :kl_date between kal.kal_date and kal.kal_date_end
  into folder_name, svz_fio;

  select maa.ma_no
  from miniapp maa
  where maa.ma_id = :ma_id
  order by maa.ma_id desc
  rows 1
  into ma_no;

  select macl.mac_id, rd.rd_name
  from miniappclient macl
  left join refdata rd on rd.rd_id = mac_loan_target
  where macl.mac_from_ma_id = :ma_id and
        macl.mac_from_c_id = :c_id
  into mac_id, loan_target;

  obj_name = 'MiniAppClient';
  obj_id = mac_id;
  rep_date = current_date;
  if (not exists(select *
                 from actionlist
                 where acl_object_kind = f_objkind(:obj_name, 'id') and
                       acl_object_id = :obj_id and
                       acl_from_rd_id = f_get_rd_id('anketa.solvency'))) then
  begin
    obj_name = 'Clients';
    obj_id = c_id;
    rep_date = coalesce(kl_date, current_date);
  end

  if (exists(select *
             from cfp_anketa ca
             where ca.ca_object_kind = f_objkind('miniappclient', 'id') and
                   ca.ca_object_id = :mac_id and
                   ca.ca_from_c_id = :c_id)) then
  begin
    select ca.ca_id, iif(:lang = 'kg', ca.ca_income_txt_kg, ca.ca_income_txt_ru), ca.ca_avr_income, ca.ca_date
    from cfp_anketa ca
    where ca.ca_object_kind = f_objkind('miniappclient', 'id') and
          ca.ca_object_id = :mac_id and
          ca.ca_from_c_id = :c_id
    into ca_id, work_place_name, monthly_income, lastch_date;

    select kpd, extra_amount, communal_expenses, living_expenses, legal_docs_expenses,
           other_business_expenses, repayment_obligations, net_profit, amount_monthly_payments, data_source
    from get_cfp_kpd(:mac_id)
    into :kpd, :extra_amount, :communal_expenses, :living_expenses, :legal_docs_expenses,
         :other_business_expenses, :repayment_obligations, :net_profit, :amount_monthly_payments , :data_source;
  end
  else
  begin

    select monthly_income, extra_amount, communal_expenses, living_expenses, legal_docs_expenses,
           other_business_expenses, repayment_obligations, net_profit, amount_monthly_payments, kpd
    from hp_get_kpd(:kc_id, :rep_date)
    into monthly_income, extra_amount, communal_expenses, living_expenses, legal_docs_expenses, other_business_expenses,
         repayment_obligations, net_profit, amount_monthly_payments, kpd;

    select cast(acl_last_datetime as date)
    from actionlist
    where acl_from_rd_id = f_get_rd_id('anketa.main') and
          acl_object_kind = 0 and
          acl_object_id = :c_id and
          acl_status = 2 and
          acl_date <= :k_disb_date
    order by acl_date desc
    rows 1
    into lastch_date; -- rfckg-1792

    select res_str_values
    from hp_objectprops_anketa_get(:obj_name, :obj_id, -1, 'work_place_name', :rep_date, null)
    into work_place_name;
  end
  if (pr_code containing '.express_similar.') then
  begin
    lastch_date = kl_date;
  end

  svz_memo = '';
  select left(obm_memo, 2048)
  from hp_objectprops_memo_get('Clients', :c_id, 'Refdata', 'ObjectProps.Client_memo', 0)
  where cast(obm_datetime as date) <= :rep_date rows 1
  into svz_memo;

  if (svz_memo is null) then
    svz_memo = '';

  select res_str_values
  from hp_objectprops_anketa_get(:obj_name, :obj_id, -1, 'work_address_orientir', :rep_date, null)
  into work_address_orientir;

    -- RFCKG-2805
    select loan_description
    from hp_get_loan_description(:c_id, :ma_id)
    into :descr_of_loan;


  -- Замена текста для старых контрактов завязанных на c_id которые были выданы в один день (НБКР)
  if (kc_id = 3849508) then
    descr_of_loan = 'Клиент приобрел соковыжималку для дома';
  if (kc_id = 3849546) then
  begin
    net_profit = 4601;
    repayment_obligations = 12000;
    kpd = 922.04;
  end
  if (kc_id = 3929460) then
  begin
    repayment_obligations = 0;

    net_profit = 17600;
    amount_monthly_payments = 3200;
    kpd = 550;
  end
  if (kc_id = 3929614) then
    descr_of_loan = 'клиент оформляет Смарт телевизор для дома';

  -------- RFCKG-3089 --------------
  if (kc_id = 4639911) then
    begin
      repayment_obligations = 16500;
      net_profit = 16999;
      kpd = 190.04;
    end
  if (kc_id = 4650428) then
    begin
      repayment_obligations = 36663;
      net_profit = 7292;
      kpd = 240;
    end
  ----------------------------------

  ---------- RFCKG-3216 ------------
  if (kc_id = 4349829) then
    descr_of_loan = replace(descr_of_loan, 'кредита ', '');
  if (kc_id = 4768633) then
    descr_of_loan = 'Товарооборот бизнеса контракта';
  if (kc_id = 4786816) then
    descr_of_loan = 'Покупка автомобиля сыну для личного пользования';
  if (kc_id = 4575850) then
    descr_of_loan = 'Покупка Redmi Note13 8GB RAM 256GB ROM цвет Ice Blue производство Китай';
  ----------------------------------
  suspend;
end
