CREATE PROCEDURE CM_RU_0040_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare kf_client_pass_no varchar(30);
declare variable kc_from_c_id id_type;
declare variable c_citizen integer;
declare variable rd_code varchar(255);
declare variable C_PASS_DATE date;
declare variable C_PASS_DATE_EXPIRE date;
declare variable k_percent_for_days integer;
declare variable pr_percent_for_days integer;
declare variable db_id integer;
declare variable k_kolich_repay integer;
declare variable kc_circle integer;
declare variable CDS_DOC_DATE_BEGIN date;
declare variable CDS_DOC_DATE_END date;
declare variable pdt_code varchar(100);
declare variable is_work_patent smallint;
declare variable is_kvit_patent smallint;
declare variable can_next smallint;
begin
  select k_disb_date
  from kontract
  where k_id = :k_id
  into KF_DISB_DATE;
-- проверка по пункту 4,2
  can_next = 1;
  for
  select kc_from_c_id,
         c_citizen
  from knclient, clients
  where kc_from_k_id = :k_id
  and kc_amount_approved > 0
  and kc_from_c_id = c_id
  into kc_from_c_id,
       c_citizen
  do
  begin
    if (can_next = 1) then
    begin
      select rd_code from refdata
      where rd_id = :c_citizen
      into rd_code;
      if (rd_code in ('Citizen.RU')) then
      begin
        can_next = 0;
      end else
      begin
        if (exists(select *
                   from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                   where CDS_FROM_C_ID = :kc_from_c_id
                   and cds_from_pk_id = pk_id
                   and pk_from_pdt_id = pdt_id
                   and cds_is_delete = 0
                   and :kf_disb_date between CDS_DOC_DATE_BEGIN and coalesce(CDS_DOC_DATE_END,'31.12.9999')
                   and pdt_code in ('registration_vnj',
                                    'registration_rvp',
                                    'work_permit',
                                    'kvit_patent',
                                    'work_patent', --Добавил 010416 Музаффар
                                    'certificate_approving_doc'
                                    )
                   )) then
        begin
          can_next = 0;
        end 
      end
    end 
  end
  if (can_next = 1) then
  for
  select c_name_fio,
         c_pass_no,
         kc_amount_approved,
         c_citizen,
         kc_from_c_id
  from knclient, clients
  where kc_from_k_id = :k_id
  and kc_amount_approved > 0
  and kc_from_c_id = c_id
  into res_info_text,
       kf_client_pass_no,
       kc_amount,
       c_citizen,
       kc_from_c_id
  do
  begin
    select rd_code from refdata
    where rd_id = :c_citizen
    into rd_code;
    if (rd_code in ('Citizen.TJ','Citizen.UZ')) then
    begin
      cds_doc_date_begin = null;
      cds_doc_date_end = null;
      select CDS_DOC_DATE_BEGIN, CDS_DOC_DATE_END
      from CLIENTDOCS, PASSKIND, PASSDOCTYPE
      where CDS_FROM_C_ID = :kc_from_c_id
      and cds_from_pk_id = pk_id
      and pk_from_pdt_id = pdt_id
      and cds_is_delete = 0
      and :kf_disb_date between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END
      and pdt_code = 'magration_card'
      order by CDS_DATETIME desc -- будем брать последний регистрированную миграционку
      rows 1
      into cds_doc_date_begin, cds_doc_date_end;
      if (cds_doc_date_begin is null) then
      begin
        cds_doc_date_begin = current_date;
      end
      if (kf_disb_date - cds_doc_date_begin > 90) then
      begin
--      проверка на наличие патента и квитанции патента
        is_work_patent = 0;
        is_kvit_patent = 0;
        for
        select pdt_code --   ,cds_doc_date_begin, cds_doc_date_end
        from CLIENTDOCS, PASSKIND, PASSDOCTYPE
        where CDS_FROM_C_ID = :kc_from_c_id
        and cds_from_pk_id = pk_id
        and pk_from_pdt_id = pdt_id
        and cds_is_delete = 0
        and :kf_disb_date between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END
        and pdt_code in ('work_patent', 'kvit_patent')  -- дополнительно нужно добавить по пункту 4,1 - по 2му и 3му подпунктам
        into pdt_code
        do
        begin
          if (pdt_code = 'work_patent') then is_work_patent = 1;
          if (pdt_code = 'kvit_patent') then is_kvit_patent = 1;
        end
        if (is_work_patent = 0) then
        begin
          res_error_text = 'Срок пребывания по мигр.карте превышает 90 дней, необходимо патент.';
          res_error_is_critical = 1;
          suspend;
        end 
        if (is_kvit_patent = 0) then
        begin
          res_error_text = 'Срок пребывания по мигр.карте превышает 90 дней, необходимо квитанция патента.';
          res_error_is_critical = 1;
          suspend;
        end
        
        
        /*Если патент скопирован не с оригинала то проверям:
        в этом случае клиент дополнительно должен предоставить оригинал справки по форме 2-НДФЛ или выписку из трудовой книжки,
        заверенную работодателем клиента.
        Справка 2-НДФЛ или выписка из трудовой книжки действительны в течение 1 месяца с даты их выдачи (заверения)
        -- так как нет возможности проверить отсканирован документ с оригинала или с копии проверка будет по все клиентам,
        сообщение будет не критичным - Музаффар 04 апреля 16 года.*/
        if (not exists (select pdt_code
                        from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                        where CDS_FROM_C_ID = :kc_from_c_id
                        and cds_from_pk_id = pk_id
                        and pk_from_pdt_id = pdt_id
                        and cds_is_delete = 0
--                            and :kf_disb_date > CDS_DOC_DATE_BEGIN -29
                        and pdt_code in ('certificate_2NDFL')
                    )) then
        begin
          res_error_text = 'Срок пребывания по мигр.карте превышает 90 дней, необходимо Справка 2-НДФЛ.';
          res_error_is_critical = 0;
          suspend;
        end
        if (not exists (select pdt_code
                        from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                        where CDS_FROM_C_ID = :kc_from_c_id
                        and cds_from_pk_id = pk_id
                        and pk_from_pdt_id = pdt_id
                        and cds_is_delete = 0
                        and :kf_disb_date > CDS_DOC_DATE_BEGIN -29
                        and pdt_code in ('certificate_2NDFL')
                    )) then
        begin
          res_error_text = 'Срок пребывания по мигр.карте превышает 90 дней, необходимо выписка из трудовой книжки.';
          res_error_is_critical = 0;
          suspend;
        end
        /*Конец условии по 2НДФЛ и выписки из трудовой книжки - Музаффар 04 апреля 16-года*/
      end
    end
  end
end
