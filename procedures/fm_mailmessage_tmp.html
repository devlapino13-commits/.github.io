CREATE PROCEDURE FM_MAILMESSAGE_TMP(
    UN_ID DECIMAL(18,0),
    REP_ID VARCHAR(255))
RETURNS (
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0),
    RES_SENDER_FIO VARCHAR(100),
    RES_SUBJECT VARCHAR(512),
    RES_DATETIME TIMESTAMP,
    RES_KIND VARCHAR(20))
AS
declare variable C_ID decimal(18,0);
declare variable CEO_UN_ID decimal(18,0);
declare variable SENDER_UN_ID decimal(18,0);
declare variable SENDER_UN_ID_PRIOR decimal(18,0);
declare variable MSG_TYPE integer;
declare variable S_KIND integer;
declare variable S_ID decimal(18,0);
declare variable UN_NAME D_STR_100;
begin
  select un_name
  from username
  where un_id = :un_id
  into :un_name;

  if (:un_name = 'fmtest') then
  begin
    if (rep_id = 'mailmessages_all' or rep_id = 'mailmessages_ceo') then
    begin
      OBJ_KIND = 'fsrobot_mailmain_ceo';
      OBJ_ID = -1;
      RES_SENDER_FIO = 'Тестовый пользователь (Ген директор)';
      RES_SUBJECT = 'Тестовое сообщение от Ген директора';
      RES_DATETIME = current_timestamp;
      RES_KIND = 'mail';
      suspend;
    end
    if (rep_id = 'mailmessages_all' or rep_id = 'mailmessages_other') then
    begin
      OBJ_KIND = 'fsrobot_mailmain_other';
      OBJ_ID = -2;
      RES_SENDER_FIO = 'Тестовый пользователь';
      RES_SUBJECT = 'Тестовое сообщение';
      RES_DATETIME = current_timestamp;
      RES_KIND = 'mail';
      suspend;
    end
    exit;
  end

  select un_from_c_id
  from username
  where un_id = :un_id
  into :c_id;
  if (c_id is null) then
    exit;

  select un_id
  from username
  where un_name = 'btolbaev'
  into :ceo_un_id;

  res_sender_fio = '';
  res_subject = '';
  res_datetime = null;

  if (rep_id = 'mailmessages_sent') then
  begin
-- Сообщения от ГД
    sender_un_id_prior = 0;
    res_kind = 'mail';
    obj_kind = 'fsrobot_mailmain';
    res_sender_fio = '';
    for select fsm_id, fsm_subject, fsm_datetime, fsm_from_un_id, fsm_mail_to
        from fsrobot_mailmain
        where fsm_from_un_id = :un_id and
              fsm_is_delete = 0
        into :obj_id, :res_subject, :res_datetime, sender_un_id, res_sender_fio
    do
    begin
      suspend;
    end
    exit;
  end -- конец fsrobot_mailmain (от CEO)

-- Выполняем на сотрудника и по каждому топику где он подписан
  for select 0, :c_id
      from company
      union
      select -1, mg_id
      from fm_user_topics(:un_id)
      into :s_kind, :s_id
  do
  begin
    if (rep_id = 'mailmessages_all' or rep_id = 'mailmessages_ceo') then
    begin
-- Сообщения от ГД
      sender_un_id_prior = 0;
      res_kind = 'mail';
      obj_kind = 'fsrobot_mailmain';
      res_sender_fio = '';
      for select fsm_id, fsm_subject, fsm_datetime, fsm_from_un_id
          from fsrobot_mailmain, fsrobot_mailusers
          where fsm_from_un_id = :ceo_un_id and
                fsm_is_delete = 0 and
                fsu_object_kind = :s_kind and
                fsu_object_id = :s_id and
                fsu_from_fsm_id = fsm_id
          into :obj_id, :res_subject, :res_datetime, sender_un_id
      do
      begin
        if (sender_un_id <> sender_un_id_prior) then
        begin
          sender_un_id_prior = sender_un_id;
          select c_name_fio
          from username, clients
          where un_id = :sender_un_id and
                un_from_c_id = c_id
          into :res_sender_fio;

          select name_surname
          from hp_fio_clear(:res_sender_fio)
          into :res_sender_fio;
        end
        suspend;
      end
    end -- конец fsrobot_mailmain (от CEO)

    if (rep_id = 'mailmessages_all' or rep_id = 'mailmessages_other') then
    begin
-- Другие входящие сообщения
      sender_un_id_prior = 0;
      res_kind = 'mail';
      obj_kind = 'fsrobot_mailmain';
      res_sender_fio = '';
      for select fsm_id, fsm_subject, fsm_datetime, fsm_from_un_id
          from fsrobot_mailmain, fsrobot_mailusers
          where fsm_from_un_id <> :ceo_un_id and
                fsm_is_delete = 0 and
                fsu_object_kind = :s_kind and
                fsu_object_id = :s_id and
                fsu_from_fsm_id = fsm_id
          into :obj_id, :res_subject, :res_datetime, :sender_un_id
      do
      begin
        if (sender_un_id <> sender_un_id_prior) then
        begin
          sender_un_id_prior = sender_un_id;
          select c_name_fio
          from username, clients
          where un_id = :sender_un_id and
                un_from_c_id = c_id
          into :res_sender_fio;

          select name_surname
          from hp_fio_clear(:res_sender_fio)
          into :res_sender_fio;
        end
        suspend;
      end
    end -- конец fsrobot_mailmain (кроме CEO)

    if (rep_id = 'mailmessages_push') then
    begin
-- История уведомлений
      sender_un_id_prior = 0;
      res_kind = 'mail';
      obj_kind = 'fsrobot_msgbox';
      res_sender_fio = '';
      for select msg_id, msg_type, iif(msg_type = 1, cast(msg_body as varchar(512)), msg_subject), msg_datetime,
                 msg_from_un_id
          from fsrobot_msgbox
          where msg_object_kind = :s_kind and
                msg_object_id = :s_id and
                msg_status = 1
          into :obj_id, :msg_type, :res_subject, :res_datetime, :sender_un_id
      do
      begin
        if (sender_un_id <> sender_un_id_prior) then
        begin
          sender_un_id_prior = sender_un_id;
          select c_name_fio
          from username, clients
          where un_id = :sender_un_id and
                un_from_c_id = c_id
          into :res_sender_fio;

          select name_surname
          from hp_fio_clear(:res_sender_fio)
          into :res_sender_fio;
        end
        if (msg_type = 0) then
          res_kind = 'mail';
        if (msg_type = 1) then
          res_kind = 'sms';
        if (msg_type = 2) then
          res_kind = 'push';
        suspend;
      end
    end -- конец fsrobot_msgbox
  end
end
