CREATE PROCEDURE HP_CONTVERI_SET_OK(
    CDA_ID D_ID,
    UN_ID D_ID,
    VERIFY_DATETIME D_DATETIME = null)
AS
declare variable c_id d_id;
declare variable cda_status d_smallint;
declare variable cda_date d_date;
declare variable cda_cont_kind d_integer;
declare variable cda_is_main d_smallint;
declare variable cda_is_delete d_smallint;
declare variable tmp_cda_id d_id;
begin
  -- Процедура устанавливает статус "Верифицирован" к контактной информации.

  if (cda_id is null) then
    exit;

  if (verify_datetime is null) then
    verify_datetime = current_timestamp;

  if (un_id is null) then
    un_id = 1000000000000; -- user: fsrobot

  select cda_from_c_id, cda_status, cda_date, cda_cont_kind, cda_is_main, cda_is_delete
  from contdata
  where cda_id = :cda_id
  into c_id, cda_status, cda_date, cda_cont_kind, cda_is_main, cda_is_delete;

  if (row_count = 0) then
    exception e_default 'cda_id not found: ' || cda_id;

  if (cda_date > cast(:verify_datetime as date)) then Exit;

  -- если он еще не верифицирован, меняем статус
  if (cda_status <> 1) then
  begin
    update contdata
    set cda_status = 1,
        cda_date = maxvalue(cda_date, cast(:verify_datetime as date)),
        cda_from_un_id = :un_id
    where cda_id = :cda_id;

    -- так как на уровне триггера могут быть внесены изменения в другие записи этого клиента,
    -- обновляем данные пользователя и время изменений
    for select cda_id
    from contdata
    where cda_from_c_id = :c_id and
          cda_cont_kind = :cda_cont_kind
    into tmp_cda_id
    do
    begin
      update contlog
      set clg_from_un_id = :un_id,
          clg_datetime = :verify_datetime
      where clg_from_cda_id = :tmp_cda_id and
            clg_from_un_id is null;
    end

    if (cda_is_main = 1 and cda_is_delete = 0 and f_enum('cont_kind', cda_cont_kind, 'code') = 'cell') then
      execute procedure hp_contdata_delete_cascade(:cda_id, :un_id);
  end else
  begin
    -- иначе просто обновляем дату последней верификации
    -- изменений даты не приводит к появлению логирований
    if (cast(verify_datetime as date) > cda_date) then
    begin
      update contdata
      set cda_date = cast(:verify_datetime as date),
          cda_from_un_id = :un_id
      where cda_id = :cda_id;
    end
  end
end
