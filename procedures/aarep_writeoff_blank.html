Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE AAREP_WRITEOFF_BLANK(
    REP_REGION_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    WRITEOFF_DATE DATE,
    K_KONTRACT_NO VARCHAR(50),
    K_DISB_DATE DATE,
    WRITEOFF_OS CURR_TYPE,
    WRITEOFF_PERC CURR_TYPE,
    WRITEOFF_TOTAL CURR_TYPE,
    ZAD_OS CURR_TYPE,
    ZAD_PERC CURR_TYPE)
AS
begin
  writeoff_date = rep_date;

  for select k_kontract_no, k_disb_date, sum(writeoff_os), sum(writeoff_perc), sum(writeoff_total), sum(zad_os),
             sum(zad_perc)
      from aarep_writeoff_blank_tmp(:rep_region_id, :rep_date)
      group by k_kontract_no, k_disb_date
      into :k_kontract_no, k_disb_date, :writeoff_os, :writeoff_perc, :writeoff_total, :zad_os, :zad_perc
  do
  begin
    suspend;
  end

end
