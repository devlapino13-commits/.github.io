CREATE PROCEDURE OKT_SET_RESULT_PATH(
    P_ID D_ID,
    P_KIND D_INTEGER,
    RES_PATH D_STR_255)
AS
declare variable KIND D_STR_100;
begin
  kind = f_objkind(:p_kind, 'name');

  if (kind = 'pantera_contacts') then
  begin
    update pantera_contacts pcon
    set pcon.pcon_res_dialog = :res_path
    where pcon.pcon_id = :p_id;
  end
  else if (kind = 'pantera_clients') then
  begin
    update pantera_clients pcl
    set pcl.pcl_res_dialog = :res_path
    where pcl.pcl_id = :p_id;
  end
end
