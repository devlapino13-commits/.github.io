CREATE PROCEDURE HP_PETTRANS_VIEW(
    IS_ACTIVE D_BOOL = 1,
    IS_LEASE D_BOOL = 1)
RETURNS (
    PET_ID D_INTEGER,
    PET_NAME D_STR_255,
    PET_LABEL D_STR_CODE,
    PET_IS_RENT D_INTEGER,
    PED_IS_ACTIVE D_INTEGER,
    PET_NORMA_YEDIZ D_INTEGER,
    PET_NORMA_KM D_INTEGER,
    PET_BILL_ID D_ID,
    PET_BILL_LINE_NO D_INTEGER)
AS
declare variable is_act boolean;
begin
  for
    select
      pet_id,
      pet_name,
      pet_label,
      pet_is_rent,
      pet_norma_yediz,
      pet_norma_km,
      pet_bill_id,
      pet_bill_line_no
    from pettrans
    where pet_is_rent = coalesce(:is_lease, pet_is_rent)
  into
    :pet_id,
    :pet_name,
    :pet_label,
    :pet_is_rent,
    :pet_norma_yediz,
    :pet_norma_km,
    :pet_bill_id,
    :pet_bill_line_no
  do
  begin
    if (pet_bill_id > 0) then
      select first 1
        ml_name || iif(ml_model in (ml_name, ''), '', ' ' || ml_model),
        ob_invent_str
      from operbill
        inner join matlist
          on ml_id = ob_from_ml_id
      where ob_bill_id = :pet_bill_id
        and ob_bill_line_no = :pet_bill_line_no
        and ob_is_delete = 0
      into
        :pet_name,
        :pet_label;

    ped_is_active = iif(
      exists(
        select * from petdata
        where ped_from_pet_id = :pet_id
          and ped_is_active = 1), 1, 0);
    if ((is_active is null) or (is_active = ped_is_active ))
    then
      suspend;
  end
end
