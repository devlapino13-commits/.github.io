CREATE PROCEDURE HP_CLIENTS_VERIFY_PDL
RETURNS (
    C_ID D_ID,
    C_INN D_STR_20,
    C_NAME_FIO D_STR_100,
    OFFICE_NAME D_STR_100,
    KAL_NEW_KF_NAME D_STR_100)
AS
declare variable k_fact_disb_date d_date;
declare variable kf_id d_id;
declare variable scan1 d_id;
declare variable scan2 d_id;
begin
  -- возвращает клиентов в кредитных заявках, которые находятся в статусе заявка или договор в процессе,
  -- при этом в clientdocs один из документов "Проверка на ПДЛ (является)" или "Проверка на ПДЛ (не является)"
  -- не имеет скана обоих страниц, т.е поля "Скан1" или "Скан2" пустые (RFCKG-4553)

  for select mac_from_c_id,
             ma_from_kf_id
      from miniappclient
      inner join miniapp on ma_id = mac_from_ma_id
      left join kontract on k_from_ma_app_id = ma_id
      where k_id is null and
            ma_is_active = 1 and
            ma_date_end = '31.12.9999' and
            (mac_amount > 0 or mac_is_poruch = 1)
      union
      select kc_from_c_id,
             k_from_kf_id
      from kontract
      inner join kontstate on ks_from_k_id = k_id
      inner join knclient on kc_from_k_id = k_id
      where k_status = 1 and
            ks_state = 0 and
            ks_ol_is_delete = 0 and
            current_date between ks_date_begin and ks_date_end and
            kc_amount_approved > 0
      into c_id,
           kf_id
  do
  begin
    k_fact_disb_date = null;
    select max(k_fact_disb_date)
    from kontract
    inner join kontstate on ks_from_k_id = k_id
    inner join knclient on kc_from_k_id = k_id
    where kc_from_c_id = :c_id and
          k_status = 1 and
          ks_state in (1, 2) and
          ks_ol_is_delete = 0 and
          current_date between ks_date_begin and ks_date_end and
          kc_amount_approved > 0
    into k_fact_disb_date; -- дата выдачи предыдущего договора

    scan1 = 0;
    scan2 = 0;
    select cds_doc_blob_1,
           cds_doc_blob_2
    from clientdocs
    inner join passkind on pk_id = cds_from_pk_id
    inner join passdoctype on pdt_id = pk_from_pdt_id
    where cds_from_c_id = :c_id and
          cds_is_delete = 0 and
          pdt_code = 'proverka_na_pdl' and
          (cds_doc_date_begin > :k_fact_disb_date or :k_fact_disb_date is null) -- дата документа после даты выдачи предыдущего договора (если есть предыдущий договор)
    into scan1,
         scan2;

    if (coalesce(scan1, 0) > 0 and coalesce(scan2, 0) > 0) then
      continue; -- есть документ "Проверка на ПДЛ" и оба его скана, клиента не отображаем

    c_name_fio = '';
    c_inn = '';
    select c_name_fio,
           c_inn
    from clients
    where c_id = :c_id
    into c_name_fio,
         c_inn;

    office_name = '';
    kal_new_kf_name = '';
    select reg_code || ' ' || reg_name,
           kal_new_kf_name
    from knaktlist
    inner join region on reg_unicode = kal_new_reg_id
    where kal_from_kf_id = :kf_id
    into office_name,
         kal_new_kf_name;

    suspend;
  end
end
