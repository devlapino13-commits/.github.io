CREATE PROCEDURE HP_SIGNDOC_SENDSMS(
    SGN_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    EDL_ID D_ID,
    ERR_TXT D_STR_255)
AS
declare variable sgn_bls_id d_id;
declare variable sgn_edl_id d_id;
declare variable sgn_status d_smallint;
begin
  err_txt = '';
  for
    select
      sgn_bls_id,
      sgn_edl_id,
      sgn_status
    from signdoc
    where sgn_id = :sgn_id
  into
    :sgn_bls_id,
    :sgn_edl_id,
    :sgn_status
  do
  begin
    if (sgn_status <> 0) then
      err_txt = f_format('Статус запроса на подписание ЭД №%1 не соответствует возможности отправки SMS на подписание', sgn_id);
    else
    if (coalesce(sgn_bls_id, 0) = 0) then
      err_txt = 'ЭД не прикреплен';
    else
    if (sgn_edl_id > 0) then
      err_txt = f_format('SMS на подписание ЭД №%1 уже отправлялось', sgn_id);
    else
      select
        edl_id,
        res_text
      from hp_edoc_ins(
        'hr.general_doc',
        f_objkind('signdoc', 'id'),
        :sgn_id,
        :un_id)
      into
        :sgn_edl_id,
        :err_txt;
    if (err_txt = '') then
    begin
      update signdoc set
        sgn_edl_id = :sgn_edl_id,
        sgn_status = 1
      where sgn_id = :sgn_id;

      update edoc_blank set
        edb_from_bls_id = :sgn_bls_id
      where edb_from_edl_id = :sgn_edl_id;
    end
  end
  suspend;
end
