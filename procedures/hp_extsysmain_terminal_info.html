CREATE PROCEDURE HP_EXTSYSMAIN_TERMINAL_INFO(
    TERMINAL_ID D_INTEGER NOT NULL,
    DATETIME_START D_DATETIME NOT NULL,
    DATETIME_FINISH D_DATETIME NOT NULL)
RETURNS (
    REP_LAST_ESM_ID D_ID,
    REP_AMOUNT_ALL D_CURRENCY,
    REP_AMOUNT_CLS D_CURRENCY,
    REP_AMOUNT_IPF D_CURRENCY,
    REP_PERIOD_INFO D_STR_255)
AS
declare variable project_code d_str_20;
declare variable terminal_code d_str_100;
declare variable filter_text d_str_100;
declare variable date_begin d_date;
declare variable date_end d_date;
declare variable last_esm_id d_id;
declare variable time_str d_str_100;
declare variable tm_date_begin d_date;
declare variable tm_date_end d_date;
declare variable last_datetime d_datetime;
begin
  rep_last_esm_id = null;
  rep_amount_all = 0;
  rep_amount_cls = 0;
  rep_amount_ipf = 0;
  rep_period_info = '';

  last_esm_id = null;

  select tm_terminal_id, esp_project, tm_date_begin, tm_date_end
  from terminals, extsyspro
  where tm_id = :terminal_id and
        tm_project = esp_id
  into terminal_code, project_code, tm_date_begin, tm_date_end;

  if (project_code = 'DosCredo') then
    project_code = 'PaySoft';

  if (row_count = 0) then
  begin
    suspend;
    exit;
  end

  if (tm_date_begin > cast(datetime_finish as date)) then
  begin
    suspend;
    exit;
  end

  if (tm_date_end < cast(datetime_start as date)) then
  begin
    suspend;
    exit;
  end

  select tml_last_datetime, tml_last_esm_id
  from termilast
  where tml_from_tm_id = :terminal_id
  into last_datetime, last_esm_id;

  if (last_datetime is null or datediff(second, last_datetime, datetime_start) > 60) then
    last_esm_id = 0;

  rep_period_info = 'Период:';
  filter_text = 'terminal_number=' || terminal_code;

  date_begin = cast(datetime_start as date);
  date_end = cast(datetime_finish as date);

  select max(esm_id),
         sum(esm_amount),
         sum(iif(pr_code containing '.ipf.', esm_amount, 0.00))
  from extsysmain, kontract, product
  where esm_date between :date_begin and :date_end and
        esm_project = :project_code and
        esm_action = 'pay' and
        esm_status = 1 and
        esm_datetime between :datetime_start and :datetime_finish and
        (esm_datetime > :datetime_start or
         (esm_datetime = :datetime_start and esm_id > :last_esm_id)) and
        esm_full_source containing :filter_text and
        esm_from_k_id = k_id and
        k_from_pr_id = pr_id
  into rep_last_esm_id, rep_amount_all, rep_amount_ipf;

  if (rep_amount_all is null) then
    rep_amount_all = 0;

  if (rep_amount_ipf is null) then
    rep_amount_ipf = 0;

  rep_amount_cls = rep_amount_all - rep_amount_ipf;

  time_str = f_datetimetostr(datetime_start);
  time_str = time_str || ':' || right(100 + extract(second from datetime_start), 2);
  rep_period_info = rep_period_info || ' ' ||time_str;

  time_str = f_datetimetostr(datetime_finish);
  time_str = time_str || ':' || right(100 + extract(second from datetime_finish), 2);
  rep_period_info = rep_period_info || ' - ' ||time_str;

  suspend;
end
