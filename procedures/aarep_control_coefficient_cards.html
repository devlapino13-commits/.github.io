CREATE PROCEDURE AAREP_CONTROL_COEFFICIENT_CARDS(
    REP_REGION_ID D_ID,
    REP_DATE_END D_DATE)
RETURNS (
    PARENT_NAME D_STR_100,
    REG_NAME D_STR_100,
    COUNT_CARDS_SUM CURR_TYPE,
    COUNT_CARDS_KREDIT D_INTEGER,
    SUM_ALL_ISSUED_CARD D_INTEGER,
    SUM_ISSUED_MONTH D_INTEGER,
    SUM_ISSUED_KREDIT D_INTEGER,
    RESULT_DATE_BEGIN D_DATE,
    RESULT_DATE_END D_DATE)
AS
declare variable reg_unicode d_id;
begin
  result_date_end = rep_date_end - 1;
  result_date_begin = cast('01.' || extract(month from result_date_end) || '.' || extract(year from result_date_end) as date);
  for select reg_unicode, reg_name, parent_name
      from region, region_list(:rep_region_id, 1, 3, 1) rl
      where reg_unicode = rl.id_list and
            coalesce(reg_date_close, cast('31.12.9999' as date)) >= :result_date_end and
            reg_unicode <> 1090100 and
            reg_unicode <> 1060100
      order by reg_name
      into reg_unicode,
           reg_name,
           parent_name
  do
  begin

    count_cards_sum = 0;
    sum_issued_kredit = 0;
    sum_all_issued_card = 0;
    sum_issued_month = 0;
    count_cards_kredit = 0;

    select count(kt.k_id),
           sum(kt.k_amount_approved)
    from kontract kt
    inner join kontstate ks on ks.ks_from_k_id = kt.k_id
    where ks.ks_state = 1 and
          ks.ks_date_begin between :result_date_begin and :result_date_end and
          ks.ks_ol_is_delete = 0 and
          ks.ks_region_id = :reg_unicode and
          kt.k_disb_kind = 6
    into :count_cards_kredit,
         :count_cards_sum;

    select count(kt.k_id)
    from kontract kt
    inner join kontstate ks on ks.ks_from_k_id = kt.k_id
    where ks.ks_state = 1 and
          ks.ks_date_begin between :result_date_begin and :result_date_end and
          ks.ks_ol_is_delete = 0 and
          ks.ks_region_id = :reg_unicode
    into :sum_issued_kredit;

    select count(cdt.cac_from_k_id)
    from cardaktclient cdt
    where cdt.cac_date between '01.04.2021' and :result_date_end and
          cdt.cac_from_reg_id = :reg_unicode and
          cdt.cac_status >= 2
    into sum_all_issued_card;

    select count(cdt.cac_from_k_id)
    from cardaktclient cdt
    where cdt.cac_date between :result_date_begin and :result_date_end and
          cdt.cac_from_reg_id = :reg_unicode and
          cdt.cac_status = 4
    into sum_issued_month;
    suspend;
  end
end
