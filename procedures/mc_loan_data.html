CREATE PROCEDURE MC_LOAN_DATA(
    MCL_ID D_ID,
    K_ID D_ID,
    C_ID D_ID)
RETURNS (
    KONTRACT_NO D_STR_CODE,
    ACTIVE_DEBT D_INTEGER,
    NEAREST_PAYMENT_DATE D_STR_10,
    NEAREST_NEXT_PAYMENT_AMOUNT D_INTEGER,
    DELINQ_DEBT D_INTEGER,
    LOAN_STATUS D_STR_100,
    AMOUNT_APPROVED D_INTEGER,
    AMOUNT_APPROVED_PERSONAL D_INTEGER)
AS
declare variable sh_date date;
declare variable zad_os integer;
declare variable sh_calc_perc integer;
declare variable sh_calc_fine integer;
declare variable ks_state integer;
begin
  /*
   "МП для клиентов" Главное окно

   иноформация по кредиту заемщика.
   Отображаю только если в статусе: активный, закрытый

   kontract_no - договор займа
   active_debt - текущий баланс
   nearest_payment_date - ближайший патеж
   nearest_next_payment_amount - сумма погашения
   delinq_debt - просроченная задолженность
  */
  for select kont.k_kontract_no || ' от ' || f_datetostr(kont.k_disb_date), ks.ks_state,
             round(kont.k_amount_approved), round(kncl.kc_amount_approved)
      from knclient kncl
      inner join kontstate ks on ks.ks_from_k_id = kncl.kc_from_k_id and
            ks.ks_ol_is_delete = 0 and
            current_date between ks.ks_date_begin and ks.ks_date_end and
            ks.ks_state in (1, 2)
      inner join kontract kont on kont.k_id = kncl.kc_from_k_id
      where kncl.kc_from_c_id = :c_id and
            kncl.kc_from_k_id = :k_id and
            kncl.kc_amount_approved > 0
      order by ks.ks_date_begin
      into :kontract_no, :ks_state, :amount_approved, :amount_approved_personal
  do
  begin
    nearest_payment_date = null;
    active_debt = 0;
    nearest_next_payment_amount = 0;
    delinq_debt = 0;
    loan_status = iif(ks_state = 1, 'Активный', 'Неактивный');

    for select sh_date, zad_os, sh_calc_perc, sh_calc_fine
        from aarep_make_zad_list(:k_id, current_date, 1)
        into :sh_date, :zad_os, sh_calc_perc, sh_calc_fine
    do
    begin
      if ((coalesce(sh_date, '01.01.1999') >= current_date) and
          (nearest_payment_date is null)) then
      begin
        nearest_payment_date = f_datetostr(sh_date);
        nearest_next_payment_amount = round((zad_os + sh_calc_perc + sh_calc_fine), 0);

      end
      if (coalesce(sh_date, '01.01.1999') < current_date) then
        delinq_debt = delinq_debt + round((zad_os + sh_calc_perc + sh_calc_fine), 0);

      active_debt = active_debt + round((zad_os + sh_calc_perc + sh_calc_fine), 0);
    end

    suspend;
  end
end
