CREATE PROCEDURE FM_SCHEDULE_LIST(
    OBJECT_KIND INTEGER,
    OBJECT_ID DECIMAL(18,0))
RETURNS (
    SH_FROM_K_ID DECIMAL(18,0),
    SH_ID INTEGER,
    SH_DATE DATE,
    SH_OS DECIMAL(18,4),
    SH_PERC DECIMAL(18,4),
    SH_KOMMIS DECIMAL(18,4),
    SH_GRACE INTEGER,
    SH_PERC_TAX DECIMAL(18,4),
    SH_DISB_AMOUNT DECIMAL(18,4),
    SH_ALL DECIMAL(18,4),
    SH_ALL_WITH_TAX DECIMAL(18,4))
AS
declare variable k_nsp_percent decimal(18,7);
declare variable obj_id decimal(18,0);
declare variable db_id integer;
declare variable k_disb_date date;
begin

  db_id = f_get_db_id();

  if (object_kind = 9) then
    obj_id = object_id;
  else
  if (object_kind = 11) then
  begin
    select k_id
    from kontract
    where k_from_ma_app_id = :object_id
    order by k_disb_date desc
    rows 1
    into obj_id;
    if (obj_id is null) then
      obj_id = -:object_id;
  end
  else
    exit;

  if (object_kind = 9) then
  begin
    select nsd_percent, k_disb_date
    from nspconfig_data, kontract
    where nsd_from_nsm_id = -1 -- нсп от процента
          and
          nsd_date_begin <= k_disb_date and
          nsd_date_end >= k_disb_date and
          k_id = :object_id
    into :k_nsp_percent, k_disb_date;
  end
  else
  if (object_kind = 11) then
  begin
    select nsd_percent
    from nspconfig_data, miniapp
    where nsd_from_nsm_id = -1 -- нсп от процента
          and
          nsd_date_begin <= ma_date_oper and
          nsd_date_end >= ma_date_oper and
          ma_id = :object_id and
          ma_date_end >= '31.12.2999'
    into :k_nsp_percent;
  end
  else
    exit;
  if (k_nsp_percent is null) then
    k_nsp_percent = 0;

  for select sh_from_k_id, sh_id, sh_date, sh_os, sh_perc, sh_kommis, sh_grace, sh_disb_amount
      from schedule
      where sh_from_k_id = :obj_id
      order by sh_from_k_id, sh_id
      into :sh_from_k_id, :sh_id, :sh_date, :sh_os, :sh_perc, :sh_kommis, :sh_grace, :sh_disb_amount
  do
  begin
--    sh_perc_tax=Round((sh_perc+sh_kommis)*k_nsp_percent/100.0);
    if (db_id in (39801, 41701) and
        k_disb_date <= '28.06.2018') then
      sh_perc_tax = (select res
                     from fhp_round_bank_old(((:sh_perc + :sh_kommis) * :k_nsp_percent / 100.0), 0));
    else
      select res
      from fhp_round_bank(((:sh_perc + :sh_kommis) * :k_nsp_percent / 100.0), 0)
      into :sh_perc_tax;

    sh_all = sh_os + sh_perc;
    sh_all_with_tax = sh_os + sh_perc + sh_kommis + sh_perc_tax;
    suspend;
  end
end
