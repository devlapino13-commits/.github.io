CREATE PROCEDURE HP_APP_STATUS_UPDATE(
    C_ID D_ID,
    CDS_ID D_INTEGER,
    UN_ID D_ID,
    K_NO D_STR_20)
RETURNS (
    RES_TXT D_STR_100)
AS
declare variable arc_id d_integer;
declare variable k_id d_id;
begin

  select res_txt, k_id
  from hp_check_save_app_photo(:k_no, :c_id)
  into :res_txt, :k_id;

  if (res_txt = '') then
  begin
    select arc.arc_id
    from app_return_consult arc
    where arc.arc_from_c_id = :c_id and
          arc.arc_from_k_id = :k_id
    into :arc_id;

    update app_return_consult
    set arc_status = 1,
        arc_status_un_id = :un_id,
        arc_status_datetime = current_timestamp
    where (arc_id = :arc_id);

    update app_return_consult_info
    set arci_from_cds_id = :cds_id
    where (arci_id = :arc_id);
  end
  suspend;
end
