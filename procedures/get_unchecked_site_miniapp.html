CREATE PROCEDURE GET_UNCHECKED_SITE_MINIAPP(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    SITE_MA_QTY INTEGER)
AS
declare variable esm_id id_type;
declare variable phone_no str_100;
declare variable old_phone_no str_100;
begin

  site_ma_qty = 0;
  old_phone_no = '';

-- вытаскиваю заявки с сайта за период
  for select esm_id, phone_no
      from aarep_website_miniapp_analiz(:rep_date_begin, :rep_date_end, 1, 1, 1)
      where (trim(coalesce(phone_no, '')) <> '')
      order by phone_no, esm_date, esm_id
      into :esm_id, :phone_no
  do
  begin
    if (phone_no <> old_phone_no) then -- убираю дубли, когда клиент дважды в течении дня отправил заявку
    begin
      old_phone_no = phone_no;

-- если не было попыток звонка
      if (not exists(select *
                     from cc_check_knclient chk
                     where chk.ck_type = 6 and
                           chk.ck_from_esm_id = :esm_id)) then
      begin
        site_ma_qty = site_ma_qty + 1;
      end
    end
  end
  suspend;
end
