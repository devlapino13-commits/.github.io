CREATE PROCEDURE FSROBOT_FINPOLIS_APP_REGISTERED(
    REP_PARAMS VARCHAR(255),
    REP_EMAIL VARCHAR(255),
    USER_FIO VARCHAR(100))
RETURNS (
    RES_SUBJECT VARCHAR(255),
    RES_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_MAILLIST VARCHAR(255),
    RES_C_ID DECIMAL(18,0))
AS
declare variable c_name_fio varchar(100);
declare variable res_tmp_bode blob sub_type 1 segment size 80;
declare variable ma_no integer;
declare variable reg_name varchar(100);
declare variable c_id decimal(18,0);
declare variable c_pass_no varchar(30);
declare variable c_inn varchar(30);
declare variable ma_date_save timestamp;
declare variable un_id decimal(18,0);
declare variable kf_id decimal(18,0);
declare variable lo_fio varchar(100);
declare variable c_cell_no varchar(20);
begin
  res_c_id = 0;
-- Процедура вызывается из FSROBOT_ON_MAILING
-- Процедура готовит список уведомлений для отправки роботом по почте или sms
-- REP_COMMAND_KIND. Настройки сообщения хранятся в таблице RefData:
-- справочник NotifyMail || '.' || Rep_Command_Kind - строка настроек email
-- справочник NotifySMS  || '.' || Rep_Command_Kind - строка настроек SMS

-- RefData.RD_Memo    - Список получателей
-- RefData.RD_Tag     - Срок жизни сообщения
-- RefData.RD_STR_ID  - Название хранимой процедуры, которая готовит текст сообщения

-- Rep_Params - входящий параметры в зависимости от команды, ID-СВЗ
-- Rep_UN_ID - пользователь

-- FINPOLIS_APP_REGISTERED - найдено совпадение с ПФТОД

  res_tmp_bode = '';

  user_fio = 'Finstart';

  for select c_id, c_name_fio, ma_no, reg_code || ' ' || reg_name, coalesce(c_pass_no, 'XXX'), coalesce(c_inn, 'XXX'),
             ma_date_save, ma_from_un_id, ma_from_kf_id, r.reg_email_operator || ';' || reg_email_manager
      from miniappclient, clients, miniapp, region r
      where mac_from_c_id = c_id and
            ((coalesce(mac_amount, 0) > 0) or
            (mac_is_poruch = 1)) and
            mac_from_ma_id = :rep_params and
            mac_from_ma_id = ma_id and
            ma_date_end >= '31.12.2999' and
            ma_office_id = reg_unicode
      into c_id, c_name_fio, ma_no, reg_name, c_pass_no, c_inn, ma_date_save, un_id, kf_id, res_maillist
  do
  begin
    if (exists(select * from aarep_check_finpolis_tmp(:c_id))) then
    begin
      res_tmp_bode = BLOB_APPEND(res_tmp_bode, 'Клиент: <b>' || c_name_fio || '</b>, паспорт: <b>' || c_pass_no || '</b>, ИНН: <b>' || c_inn || '</b><br>');

      select c_name_fio
      from clients, knaktlist
      where kal_from_kf_id = :kf_id and
            kal_date <= current_date and
            kal_date_end >= current_date and
            knaktlist.is_delete = 0 and
            kal_new_kf_officer = c_id
      into :lo_fio;

      res_tmp_bode = BLOB_APPEND(res_tmp_bode, '<br>');
      res_tmp_bode = BLOB_APPEND(res_tmp_bode, 'КС: ' || lo_fio || '<br>');

      select c_name_fio,
             (select cont_value
              from hp_contdata_get_cell_main(0, c_id))
      from clients, username
      where un_id = :un_id and
            un_from_c_id = c_id
      into :lo_fio, :c_cell_no;

      if (c_cell_no = '') then
        c_cell_no = 'XXX';

      res_tmp_bode = BLOB_APPEND(res_tmp_bode, 'Пользователь: ' || lo_fio || '<br>');
      res_tmp_bode = BLOB_APPEND(res_tmp_bode, 'Сотовый номер: ' || c_cell_no || '<br>');
      res_tmp_bode = BLOB_APPEND(res_tmp_bode, 'Дата и время ввода в БД: ' || right('00' || extract(day from ma_date_save), 2) || '.' || right('00' || extract(month from ma_date_save), 2) || '.' || extract(year from ma_date_save) || ' ' || right('00' || extract(hour from ma_date_save), 2) || ':' || right('00' || extract(minute from ma_date_save), 2) || '<br>');
    end
  end

  res_subject = 'Внимание, совпадение ПФТОД в заявке';

  res_body = '<b>Внимание !</b><br>';
  res_body = BLOB_APPEND(res_body, '<br>');
  res_body = BLOB_APPEND(res_body, '<b>При регистрации заявки выявлено нечеткое совпадение с лицами из списка ПФТОД</b><br>');
  res_body = BLOB_APPEND(res_body, 'Номер заявки: ' || ma_no || ', офис: ' || reg_name || '<br>');
  res_body = BLOB_APPEND(res_body, '<br>');

  res_body = BLOB_APPEND(res_body, res_tmp_bode);

  res_body = BLOB_APPEND(res_body, '<br>');
  res_body = BLOB_APPEND(res_body, '<b>Срочно примите необходимые действия!</b>');
  res_body = BLOB_APPEND(res_body, '<br>');

  res_maillist = rep_email || ';' || res_maillist;

  suspend;
end
