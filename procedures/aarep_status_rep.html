CREATE PROCEDURE AAREP_STATUS_REP(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_BONUS_TEXT_ON_DISB VARCHAR(100),
    ACTIVE_KOL INTEGER,
    ACTIVE_BALANCE DECIMAL(18,4),
    DISB_KOL INTEGER,
    DISB_AMOUNT DECIMAL(18,4),
    PR_LOAN_KIND VARCHAR(100))
AS
declare variable k_id decimal(18,0);
declare variable c_last_bonus_status integer;
declare variable c_curr_bonus_status integer;
declare variable k_disb_date date;
declare variable pr_category integer;
begin
  active_kol = 0;
  active_balance = 0;
  disb_kol = 0;
  disb_amount = 0;

  for select ks_from_k_id, k_kolich_client_approved, pr_category
      from kontstate, kontract, product
      where ks_date_begin <= :rep_date_end and
            ks_date_end >= :rep_date_end and
            ks_ol_is_delete = 0 and
            k_id = ks_from_k_id and
            ks_state = 1 and
            k_from_pr_id = pr_id
      into :k_id, :active_kol, :pr_category
  do
  begin
    if (pr_category = 1) then
      pr_loan_kind = 'Параллельный';
    else
      pr_loan_kind = 'Основной';

    select c_last_bonus_status, c_curr_bonus_status, c_last_bonus_status_str
    from hp_kontstate_loan_status(:k_id, 0, 0, :rep_date_end, '')
    into :c_last_bonus_status, :c_curr_bonus_status, :k_bonus_text_on_disb;

    active_balance = 0;
    select oks_balance
    from oksummary
    where oks_from_k_id = :k_id and
          oks_date_begin <= :rep_date_end and
          oks_date_end >= :rep_date_end and
          oks_close_state = 0
    into :active_balance;
    if (active_balance < 0) then
      active_balance = 0;

    suspend;
  end

  active_kol = 0;
  active_balance = 0;
  disb_kol = 0;
  disb_amount = 0;

  for select ks_from_k_id, k_kolich_client_approved, ks_date_begin, k_amount_approved, pr_category
      from kontstate, kontract, product
      where ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_ol_is_delete = 0 and
            k_id = ks_from_k_id and
            ks_state = 1 and
            k_from_pr_id = pr_id
      into :k_id, :disb_kol, :k_disb_date, :disb_amount, :pr_category
  do
  begin
    if (pr_category = 1) then
      pr_loan_kind = 'Параллельный';
    else
      pr_loan_kind = 'Основной';

    select c_last_bonus_status, c_curr_bonus_status, c_last_bonus_status_str
    from hp_kontstate_loan_status(:k_id, 0, 0, :k_disb_date, '')
    into :c_last_bonus_status, :c_curr_bonus_status, :k_bonus_text_on_disb;

    suspend;
  end

end
