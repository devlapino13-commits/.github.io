CREATE PROCEDURE AAREP_ORIV_MOVEMENT(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_SBL_ID INTEGER)
RETURNS (
    SUM_ORIV_DATE CURR_TYPE,
    BALANCE_BEGIN CURR_TYPE,
    BALANCE_END CURR_TYPE,
    ENROLLED_SUMM CURR_TYPE,
    REPAY_SUM_PERIOD CURR_TYPE,
    K_KONTRACT_NO D_STR_20,
    SBL_NAME D_STR_100,
    SBA_DATE_BEGIN D_DATE,
    C_NAME_FIO D_STR_100,
    REP_LASTPAY_DATE D_DATE,
    OKO_OS CURR_TYPE,
    SUMMA_OKO_OS CURR_TYPE,
    OKO_SH_OS CURR_TYPE,
    K_STATUS D_STR_20,
    DATE_TO_ORIV D_DATE,
    IS_COURT SMALLINT,
    SBL_ID D_ID,
    ORIV_MEMO D_STR_255,
    ORIV_COMMENT D_STR_255,
    ORIV_MEMO_DATE D_DATE,
    ORIV_COMMENT_DATE D_DATE,
    K_DISB_DATE D_DATE,
    REG_NAME D_STR_100,
    KF_OFFICE_NAME D_STR_100,
    IS_ORIV D_STR_100,
    DEPARTMENT_NAME D_STR_100,
    QTY_DELINQ_DAYS D_INTEGER,
    NOTIFICATION D_STR_10)
AS
declare variable k_id d_id;
declare variable c_id d_id;
declare variable c_dolya curr_type;
declare variable date_writeoff d_date;
declare variable reg_id d_id;
declare variable reg_id_old d_id;
declare variable kf_id d_id;
begin
  rep_date_begin = rep_date_begin - 1;
  reg_id_old = 0;
  -- По просьбе Нуржан контракты берутся с конца прошлого месяца до конца следующего месяца
  --далее погашения берутся с 1 числа месяца до его конца  (пример: строка 143)
  for select distinct cli.c_name_fio, sba.sba_from_k_id,
                      case oks.oks_close_state
                        when 1 then 'спец. учёт'
                        when 0 then 'активный'
                        when 2 then 'закрытый'
                        else 'ошибка'
                      end,
                      sbl.sbl_id, sba.sba_date_begin, t.id_list
                      -- тянем из oksummary, потому что в отчёте нужны активные и спец. учёты
      from oksummary oks
      inner join sb_akt sba on sba.sba_from_k_id = oks.oks_from_k_id
      inner join sb_list sbl on sbl.sbl_id = sba.sba_from_sbl_id
      inner join clients cli on cli.c_id = sbl.sbl_from_c_id
      inner join region_list(:rep_region_id, 1, 3, 1) t on t.id_list = cast(oks.oks_region_id as d_integer)
      where oks.oks_close_state in (0, 1) and
            --oks.oks_date_begin between :rep_date_begin and :rep_date_end and
            sba.sba_status = 2 and
            sba.sba_is_delete = 0 and
            coalesce(sbl.sbl_date_end, '31.12.9999') > current_date and -- активные сб
            sba.sba_date_begin <= :rep_date_end and
            sba.sba_date_end >= :rep_date_begin and
            --:rep_date_end between sba.sba_date_begin and sba.sba_date_end and
            oks.oks_date_begin <= :rep_date_end and
            oks.oks_date_end >= :rep_date_begin and
            iif(coalesce(:rep_sbl_id, 0) = 0, true, iif(:rep_sbl_id = 1, sbl.sbl_id not in (55, 60, 0), sbl.sbl_id = :rep_sbl_id))
      order by sbl.sbl_name
      into sbl_name, k_id, k_status, sbl_id, sba_date_begin, reg_id
  do
  begin
    notification = 'нет';
    if (exists(select *
               from printcert prc
               where prc.prc_from_k_id = :k_id and
                     prc.prc_date between :rep_date_begin and :rep_date_end and
                     prc.prc_kind = 1 and
                     prc.prc_from_bln_id in (1558, 1559, 1581, 1582))) then
      notification = 'да';
    qty_delinq_days = 0;
    select iif((cast(:rep_date_end as date) - oks.oks_delinq_date < 0), 0, cast(:rep_date_end as date) - oks.oks_delinq_date) -- числоднейактивнойпросрочки
    from oksummary oks
    where :rep_date_end between oks.oks_date_begin and oks.oks_date_end and
          oks.oks_from_k_id = :k_id and
          oks_close_state = 0
    into qty_delinq_days;
    for select distinct kc.kc_from_c_id, kt.k_kontract_no,
                        (kc.kc_amount_approved / kt.k_amount_approved), kt.k_disb_date, kt.k_from_kf_id
        from kontract kt
        inner join knclient kc on kc.kc_from_k_id = kt.k_id
        where kt.k_id = :k_id and
              kc.kc_amount_approved > 0
        into c_id, k_kontract_no, c_dolya, k_disb_date, kf_id
    do
    begin
      if (reg_id_old <> reg_id) then
      begin
        reg_name = null;
        select rg.reg_name
        from region rg
        where rg.reg_unicode = :reg_id
        into reg_name;
      end
      --По просьбе Байан Мантис 411 -----------------------
      department_name = case
                          when sbl_id = 0 then 'Офис'
                          when sbl_id = 55 then 'Пантера'
                          when sbl_id = 60 then 'Юридический отдел'
                          else 'ОРиВ'
                        end;
      -----------------------------------------------------
      kf_office_name = '';
      is_oriv = '';
      select kal.kal_new_kf_name, cl.c_name_fio
      from knaktlist kal
      inner join clients cl on cl.c_id = kal.kal_new_kf_officer
      where kal.kal_from_kf_id = :kf_id and
            :rep_date_end between kal.kal_date and kal.kal_date_end and
            kal.is_delete = 0
      into kf_office_name, is_oriv;
      c_name_fio = null;
      select cl.c_name_fio
      from clients cl
      where cl.c_id = :c_id
      group by c_name_fio
      into c_name_fio;
      date_to_oriv = null;
      select distinct s.sba_date_begin
      from sb_akt s
      where s.sba_from_k_id = :k_id
      order by s.sba_date_begin
      rows 1
      into date_to_oriv;
      oriv_comment = null;
      oriv_comment_date = null;
      select obm_memo, cast(obm_datetime as date)
      from hp_objectprops_memo_get('clients', :c_id, 'Refdata', 'ObjectProps.oriv_comment', 1)
      into oriv_comment, oriv_comment_date;
      oriv_memo = null;
      oriv_memo_date = null;
      select obm_memo, cast(obm_datetime as date)
      from hp_objectprops_memo_get('clients', :c_id, 'Refdata', 'ObjectProps.oriv_memo', 1)
      into oriv_memo, oriv_memo_date;
      if (:k_id in (109010107649000201, 109010114207900201, 109010114207900301, 109010111341700101, 109010111337500201, 109010072951100201, 109010072951100301, 109010072951100401, 109010093060500301, 109010115136100101,
                    109010116846500101, 109010108614000101)) then
        continue;
      --------------------------------спец учёт--------------------------------------------------------
      if (k_status = 'спец. учёт') then
      begin
        date_writeoff = null;
        select max(oko1.oko_ol_date)
        from operkroff oko1
        inner join operlist ol1 on ol1.ol_id = oko1.oko_from_ol_id
        where oko1.oko_from_k_id = :k_id and
              ol1.is_delete = 0 and
              oko1.oko_from_c_id = :c_id and
              oko1.oko_type = 1
        into date_writeoff; --дата списания
        select -sum(coalesce(oko_os, 0) + coalesce(oko_perc, 0) + coalesce(oko_other, 0))
        from operkroff, operlist
        where oko_from_k_id = :k_id and
              oko_from_ol_id = ol_id and
              oko_from_c_id = :c_id and
              is_delete = 0 and
              oko_ol_date > :date_to_oriv
        into summa_oko_os; --сумма всех погашений
        summa_oko_os = coalesce(summa_oko_os, 0);
        --по просьбе байан нужно было отразить сумму всех погашений после передачи в орив
        rep_lastpay_date = null;
        oko_sh_os = 0;
        select max(ol_date_oper_day)
        from operkroff okf, operlist ol
        where oko_from_k_id = :k_id and
              oko_from_ol_id = ol_id and
              oko_from_c_id = :c_id and
              ol.is_delete = 0
        into :rep_lastpay_date; -- дата последнего погашения
        oko_os = 0;
        select -(coalesce(okf.oko_os, 0) + coalesce(okf.oko_perc, 0) + coalesce(oko_other, 0))    -- RFCKG-3465
        from operkroff okf, operlist ol
        where oko_from_k_id = :k_id and
              oko_from_ol_id = ol_id and
              oko_from_c_id = :c_id and
              okf.oko_ol_date > :date_to_oriv and
              ol.is_delete = 0 and
              okf.oko_type = 4
        order by ol.ol_datetime desc
        rows 1
        into :oko_os; --последнее погашение
        oko_os = coalesce(oko_os, 0);
        enrolled_summ = 0;
        if (:date_to_oriv between :rep_date_begin and :rep_date_end) then
        begin
          select sum(coalesce(zad_os, 0) + coalesce(sh_calc_perc, 0))
          from aarep_make_zad_list(:k_id, :rep_date_begin + 1, 1)
          where sh_date is not null
          into enrolled_summ;
          enrolled_summ = coalesce(enrolled_summ, 0);
          enrolled_summ = enrolled_summ * c_dolya;
        end
        balance_begin = 0;
        select sum(coalesce(op.oko_os, 0) + coalesce(op.oko_perc, 0) + coalesce(op.oko_other, 0) + coalesce(op.oko_sh_other, 0) + coalesce(op.oko_sh_os, 0) + coalesce(op.oko_sh_perc, 0))
        from operkroff op, operlist ol
        where oko_from_k_id = :k_id and
              oko_from_ol_id = ol_id and
              oko_from_c_id = :c_id and
              oko_ol_date <= :rep_date_begin + 1 and
              ol.is_delete = 0
        into :balance_begin; -- баланс на начало
        balance_begin = coalesce(balance_begin, 0);
        balance_end = 0;
        select sum(coalesce(op.oko_os, 0) + coalesce(op.oko_perc, 0) + coalesce(op.oko_other, 0) + coalesce(op.oko_sh_other, 0) + coalesce(op.oko_sh_os, 0) + coalesce(op.oko_sh_perc, 0))
        from operkroff op, operlist ol
        where oko_from_k_id = :k_id and
              oko_from_ol_id = ol_id and
              oko_from_c_id = :c_id and
              oko_ol_date <= :rep_date_end and
              ol.is_delete = 0
        into :balance_end; -- баланс на конец
        balance_end = coalesce(balance_end, 0);
        sum_oriv_date = 0;
        select sum(zad_os + sh_calc_perc)
        from aarep_make_zad_list(:k_id, :date_writeoff - 1, 1)
        where sh_date is not null
        into sum_oriv_date; -- сумма долга на момент передачи в орив
        sum_oriv_date = coalesce(sum_oriv_date, 0);
        sum_oriv_date = sum_oriv_date * c_dolya;
        repay_sum_period = 0;
        select -sum(coalesce(oko_sh_os, 0) + coalesce(oko_sh_perc, 0) + coalesce(oko_sh_other, 0) + coalesce(oko_os, 0) + coalesce(oko_perc, 0) + coalesce(oko_other, 0))
        from operkroff
        inner join sb_akt sba on sba.sba_from_k_id = oko_from_k_id and
                          sba.sba_status = 2 and
                          sba.sba_is_delete = 0 and
                          iif(coalesce(:rep_sbl_id, 0) = 0, true, iif(:rep_sbl_id = 1, sba.sba_from_sbl_id not in (55, 60, 0), sba.sba_from_sbl_id = :rep_sbl_id))
        where oko_from_k_id = :k_id and
              oko_ol_is_active = 1 and
              oko_from_c_id = :c_id and
              oko_ol_date between :rep_date_begin + 1 and :rep_date_end and
              oko_ol_date between sba.sba_date_begin and sba.sba_date_end and
              oko_type = 4
        into repay_sum_period; --сумма погашений в орив за период
        repay_sum_period = coalesce(repay_sum_period, 0);
        oko_sh_os = sum_oriv_date - summa_oko_os;
        is_court = 0;
        if (exists(select *
                   from lawmain, lawkredit
                   where lam_is_active = 1 and
                         lam_id = lak_from_lam_id and
                         lak_from_k_id = :k_id)) then
          is_court = 1;
      end
      --------------------------------конец спец учёт-------------------------------------------------
      --------------------------------активные--------------------------------------------------------
      if (k_status = 'активный') then
      begin
        --здесь нужна переменная c_dolya т.к. в таблице operkred нет данных в поле ok_from_c_id
        rep_lastpay_date = null;
        select max(oke.ok_ol_date)
        from operkred oke
        inner join operlist t on t.ol_id = oke.ok_from_ol_id
        where oke.ok_from_k_id = :k_id and
              t.is_delete = 0 and
              oke.ok_type = 4 rows 1
        into rep_lastpay_date; -- дата последнего погашения
        oko_os = 0;
        select -(coalesce(ok.ok_os, 0) + coalesce(ok.ok_perc, 0) + coalesce(ok.ok_fine, 0))
        from operkred ok
        inner join operlist olo on olo.ol_id = ok.ok_from_ol_id
        where ok.ok_from_k_id = :k_id and
              olo.ol_date > :date_to_oriv and -- :rep_date_end and --погашения после попадания в орив
              olo.is_delete = 0 and
              ok.ok_type = 4
        order by olo.ol_date desc
        rows 1
        into :oko_os; -- сумма последнего погашений после орив
        oko_os = coalesce(oko_os, 0);
        oko_os = oko_os * c_dolya;
        summa_oko_os = 0;
        select -sum(coalesce(okr.ok_os, 0) + coalesce(okr.ok_perc, 0) + coalesce(okr.ok_fine, 0))
        from operkred okr
        inner join operlist ol1 on ol1.ol_id = okr.ok_from_ol_id
        where okr.ok_from_k_id = :k_id and
              ol1.is_delete = 0 and
              okr.ok_ol_date > :date_to_oriv and
              okr.ok_type = 4 rows 1
        into :summa_oko_os; --сумма всех погашений после орив
        summa_oko_os = coalesce(summa_oko_os, 0);
        summa_oko_os = summa_oko_os * c_dolya;
        --по просьбе байан нужно было отразить сумму всех погашений после передачи в орив
        balance_begin = 0;
        select sum(zad_os + sh_calc_perc)
        from aarep_make_zad_list(:k_id, :rep_date_begin + 1, 1)
        where sh_date is not null
        into balance_begin; --баланс активного и внесистемного портфеля на конец периода
        balance_begin = coalesce(balance_begin, 0);
        balance_begin = balance_begin * c_dolya;
        balance_end = 0;
        select sum(zad_os + sh_calc_perc)
        from aarep_make_zad_list(:k_id, :rep_date_end, 1)
        where sh_date is not null
        into balance_end; --баланс активного и внесистемного портфеля на конец периода
        balance_end = coalesce(balance_end, 0);
        balance_end = balance_end * c_dolya;
        enrolled_summ = 0;
        if (:date_to_oriv between :rep_date_begin and :rep_date_end) then
        begin
          select sum(coalesce(zad_os, 0) + coalesce(sh_calc_perc, 0))
          from aarep_make_zad_list(:k_id, :rep_date_begin + 1, 1)
          where sh_date is not null
          into enrolled_summ;
          enrolled_summ = coalesce(enrolled_summ, 0);
          enrolled_summ = enrolled_summ * c_dolya;
        end
        sum_oriv_date = 0;
        select sum(zad_os + sh_calc_perc)
        from aarep_make_zad_list(:k_id, :date_to_oriv - 1, 1)
        where sh_date is not null
        into sum_oriv_date; -- сумма долга на момент передачи в орив
        sum_oriv_date = coalesce(sum_oriv_date, 0);
        sum_oriv_date = sum_oriv_date * c_dolya;
        repay_sum_period = 0;
        select -sum(coalesce(okr.ok_os, 0) + coalesce(okr.ok_perc, 0))
        from operkred okr
        inner join sb_akt sba on sba.sba_from_k_id = okr.ok_from_k_id and
                                 sba.sba_status = 2 and
                                 sba.sba_is_delete = 0 and
                                 iif(coalesce(:rep_sbl_id, 0) = 0, true, iif(:rep_sbl_id = 1, sba.sba_from_sbl_id not in (55, 60, 0), sba.sba_from_sbl_id = :rep_sbl_id))
        where okr.ok_from_k_id = :k_id and
              okr.ok_ol_is_active = 1 and
              okr.ok_ol_date between :rep_date_begin + 1 and :rep_date_end and
              okr.ok_ol_date between sba.sba_date_begin and sba.sba_date_end and
              okr.ok_type = 4 rows 1
        into :repay_sum_period; --сумма погашений в орив за период
        repay_sum_period = coalesce(repay_sum_period, 0);
        repay_sum_period = repay_sum_period * c_dolya;
        oko_sh_os = sum_oriv_date - summa_oko_os;
        is_court = 0;
        if (exists(select *
                   from lawmain, lawkredit
                   where lam_is_active = 1 and
                         lam_id = lak_from_lam_id and
                         lak_from_k_id = :k_id)) then
          is_court = 1;
      end
      --------------------------------конец активные----------------------------------------------------
      suspend;
    end
  end
end
