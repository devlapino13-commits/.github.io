CREATE PROCEDURE HP_PDN_ZEROPERC_CHECK(
    PCG_ID D_INTEGER)
RETURNS (
    REC_NO D_INTEGER,
    PZP_DISB_PERIOD D_INTEGER,
    PZP_DAY D_INTEGER,
    PZP_AMOUNT D_CURRENCY,
    PZP_IS_ZALOG D_SMALLINT,
    PZP_IS_PORUCH D_SMALLINT,
    PCG_NAME D_STR_100,
    RES_TEXT D_STR_100)
AS
declare variable group_id d_integer;
declare variable max_val d_currency;
declare variable pct_id d_integer;
declare variable check_date d_date;
declare variable res_perc d_percent;
declare variable calc_step_day d_integer;
declare variable pzp_day_orig d_integer;
declare variable pzp_amount_orig d_currency;
declare variable calc_step_amount d_currency;
declare variable pzp_id d_integer;
declare variable kc_disb_date d_date;
declare variable is_new_format d_bool;
begin
  max_val = 999999999;
  rec_no = 0;


  if (pcg_id is null) then
    pcg_id = 0;

  for select pcg_id, pcg_name
  from pdn_credgroup
  where (:pcg_id = 0 or pcg_id = :pcg_id)
  into group_id, pcg_name
  do
  begin
    is_new_format = iif(pcg_id > 8, 1, 0);

    if (is_new_format = 0) then
      select pct_id
      from pdn_credtype
      where pct_from_pcg_id = :group_id
      rows 1
      into :pct_id;
    else
      select pct_id
      from pdn_credtype2
      where pct_from_pcg_id = :group_id
      rows 1
      into :pct_id;
  
    update pdn_zeroperc
    set pzp_check_state = 0
    where pzp_from_pcg_id = :group_id and
          pzp_check_state <> 0;
  
    for select distinct pzp_disb_period
        from pdn_zeroperc
        where pzp_from_pcg_id = :group_id
        order by pzp_disb_period desc
        into pzp_disb_period
    do
    begin
      check_date = cast('01.' || right(pzp_disb_period, 2) || '.' || left(pzp_disb_period, 4) as date);
      kc_disb_date = dateadd(month, 1, check_date);
      if (kc_disb_date < '01.09.2020') then
        kc_disb_date = '01.09.2020';

      for select distinct pzp_day_min
          from pdn_zeroperc
          where pzp_from_pcg_id = :group_id and
                pzp_disb_period = :pzp_disb_period
          union
          select distinct pzp_day_max
          from pdn_zeroperc
          where pzp_from_pcg_id = :group_id and
                pzp_disb_period = :pzp_disb_period
          order by 1
          into pzp_day_orig
      do
      begin
        calc_step_day = -1;
  
        while (calc_step_day <= 1) do
        begin
          pzp_day = pzp_day_orig + calc_step_day;
  
          if (pzp_day < 0 or
              pzp_day > max_val) then
          begin
            calc_step_day = calc_step_day + 2;
            continue;
          end
  
          for select distinct pzp_amount_min
              from pdn_zeroperc
              where pzp_from_pcg_id = :group_id and
                    pzp_disb_period = :pzp_disb_period
              union
              select distinct pzp_amount_max
              from pdn_zeroperc
              where pzp_from_pcg_id = :group_id and
                    pzp_disb_period = :pzp_disb_period
              order by 1
              into pzp_amount_orig
          do
          begin
            calc_step_amount = -1;
            while (calc_step_amount <= 1) do
            begin
              pzp_amount = pzp_amount_orig + calc_step_amount;
  
              if (pzp_amount < 0 or
                  pzp_amount > max_val) then
              begin
                calc_step_amount = calc_step_amount + 2;
                continue;
              end
  
              for select res_str
                  from hp_split_str('0,1', ',')
                  into :pzp_is_zalog
              do
              begin
                for select res_str
                    from hp_split_str('0,1', ',')
                    into :pzp_is_poruch
                do
                begin
                  res_text = '';
                  res_perc = f_pdn_zero(pct_id, kc_disb_date, check_date, pzp_amount, pzp_day, pzp_is_zalog, pzp_is_poruch, is_new_format);
  
                  if (res_perc = -1) then
                    res_text = 'Не найдена соответствующая запись';
  
                  if (res_perc = -2) then
                  begin
                    for select pzp_id
                    from pdn_zeroperc
                    where pzp_from_pcg_id = :group_id and
                          pzp_disb_period = :pzp_disb_period and
                          :pzp_day between pzp_day_min and pzp_day_max and
                          :pzp_amount between pzp_amount_min and pzp_amount_max and
                          (pzp_zalog_kind = -1 or pzp_zalog_kind = :pzp_is_zalog) and
                          (pzp_poruch_kind = -1 or pzp_poruch_kind = :pzp_is_poruch)
                    into pzp_id
                    do
                    begin
                      update pdn_zeroperc
                      set pzp_check_state = 1
                      where pzp_id = :pzp_id and
                            pzp_check_state <> 1;
                    end
                  end
  
                  if (res_text <> '') then
                  begin
                    rec_no = rec_no + 1;
                    suspend;
                  end
                end
              end
              calc_step_amount = calc_step_amount + 2;
            end
          end
          calc_step_day = calc_step_day + 2;
        end
      end
    end
  end
end
