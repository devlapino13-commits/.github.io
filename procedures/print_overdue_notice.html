CREATE PROCEDURE PRINT_OVERDUE_NOTICE(
    K_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    EDL_ID D_ID,
    K_KONTRACT_DATE D_DATE,
    C_NAME_FIO D_STR_100,
    K_KONTRACT_NO D_STR_20,
    K_FACT_DISB_DATE D_DATE,
    EDL_DATE D_DATE,
    DELINQ_ZAD D_CURRENCY,
    ZAD_OS D_CURRENCY,
    ZAD_PERC D_CURRENCY,
    ZAD_FINE D_CURRENCY,
    ZAD_ALL D_CURRENCY,
    SIGNMEN_PHON D_STR_255,
    DAYS_OVERDUE D_INTEGER,
    MONTHLY_PAYMENT D_CURRENCY,
    REPAY_DAY D_INTEGER,
    NOTICE_DATE D_DATE,
    ZAD_OS_ALL D_CURRENCY)
AS
declare variable REG_ID D_INTEGER;
declare variable MGR_ID D_ID;
begin
  edl_date = current_date;
  if (coalesce(add_params, '') <> '') then
  begin
    edl_id = f_json_str_get_value(add_params, 'edl_id', null);
    
    select cast(edl.edl_create_datetime as date)
    from edoc_list edl
    where edl.edl_id = :edl_id
    into :edl_date;
  end

  for select k.k_kontract_date,
             k.k_kontract_no,
             k.k_fact_disb_date,
             c.c_name_fio,
             k.k_fact_disb_reg_id
      from kontract k
      join knclient kc on kc.kc_from_k_id = k.k_id
      join clients c on c.c_id = kc.kc_from_c_id
      where k.k_id = :k_id
      into :k_kontract_date,
           :k_kontract_no,
           :k_fact_disb_date,
           :c_name_fio,
           :reg_id
  do
  begin
      select ps.ps_from_c_id
      from hr_pstatus ps
      join joblist jl on jl.jl_id = ps.ps_from_jl_id and jl.jl_name = 'Менеджер офиса'
      where ps.ps_is_working = 1
            and current_date between ps.ps_date_begin and ps.ps_date_end
            and ps.ps_from_reg_id = :reg_id
      into :mgr_id;

      signmen_phon = f_cont_info('clients', :mgr_id, 'cell', 'ok');

      select sum(iif(sh_date < current_date, zad_os, 0)) + sum(iif(sh_date < current_date, sh_calc_perc, 0)),
             sum(iif(sh_date < current_date, zad_os, 0)),
             sum(iif(sh_date < current_date, sh_calc_perc, 0)),
             sum(sh_calc_fine),
             sum(iif(sh_date < current_date, zad_os, 0)) + sum(iif(sh_date < current_date, sh_calc_perc, 0)) + sum(iif(sh_date < current_date, sh_calc_fine, 0)),
             sum(zad_os)
      from aarep_make_zad_list(:k_id, current_date, 1)
      where sh_date is not null
      rows 1
      into :delinq_zad,
           :zad_os,
           :zad_perc,
           :zad_fine,
           :zad_all,
           :zad_os_all;

      select iif((current_date - oks.oks_delinq_date)<0, 0, (current_date - oks.oks_delinq_date))
      from oksummary oks
      where oks.oks_from_k_id = :k_id
        and current_date between oks.oks_date_begin and oks.oks_date_end
        and oks_close_state = 0
      into :days_overdue;

      select first 1 zad_os + zad_perc, extract(day from sh_date)
      from aarep_make_zad_list(:k_id, current_date, 0)
      into :monthly_payment, :repay_day;

      select cast(edl.edl_create_datetime as date)
      from edoc_list edl
      where edl.edl_object_kind = f_objkind('kontract', 'id') and
            edl.edl_object_id = :k_id and
            edl.edl_from_edc_id = 5 and -- Уведомление о просрочке
            edl.edl_status in (0,1)
      order by edl.edl_blank_date desc
      rows 1
      into notice_date;
      suspend;
  end
end
