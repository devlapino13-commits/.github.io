CREATE PROCEDURE HP_CONTVERI_SIGN_SYNC(
    OBJ_CODE D_STR_CODE,
    OBJ_ID D_ID,
    MSG_ID D_INTEGER = null,
    SIGN_KIND D_STR_CODE = 'online')
AS
declare variable cont_kind d_str_code;
declare variable cont_value d_str_100;
declare variable cont_c_id d_id;
declare variable cont_is_main d_smallint;
declare variable cont_clt_id d_integer;
declare variable cont_cda_id d_id;
declare variable cont_rd_id d_integer;
declare variable rd_code d_str_code;
declare variable un_id d_id;
declare variable create_datetime d_datetime;
declare variable sign_datetime d_datetime;
declare variable uuid_str d_str_code;
declare variable code d_str_100; /* Пин-код */
begin
  cont_kind = '';
  cont_value = '';
  cont_c_id = null;
  cont_is_main = 0;
  cont_cda_id = null;
  cont_rd_id = null;

  if (code is null) then code = '';
  if (sign_kind is null) then
    sign_kind = 'online';

  if (obj_code = 'edoc_sign') then
  begin
    -- Если подписан ASP.Agreement, то заносим записи подтвреждения номера в ContVeri и ContVerify

    select cda_from_c_id, cda_cont_kind, cda_is_main, cda_id,
           f_cont_code(cda_from_clt_id),
           edl_create_un_id, edl_create_datetime, eds_status_datetime
    from edoc_msg, contdata, edoc_sign, edoc_list, edoc_cfg
    where edm_from_eds_id = :obj_id and
          edm_from_msg_id = :msg_id and
          edm_from_cda_id = cda_id and
          edm_from_eds_id = eds_id and
          eds_from_edl_id = edl_id and
          edl_from_edc_id = edc_id and
          cda_status = 0 and
          edc_code in ('asp.agreement', 'cell.application', 'asp.agreement_employee') and
          edc_state in (2, 3) -- только в боевом режиме и режиме теста с регионам
    into cont_c_id, cont_kind, cont_is_main, cont_cda_id, cont_value,
         un_id, create_datetime, sign_datetime;

    if (cont_c_id is not null) then
    begin
      rd_code = f_enum('cont_kind', cont_kind, 'code');
      if (rd_code = 'cell') then
      begin
        if (cont_is_main = 1)
        then rd_code = 'cell_main';
        else rd_code = 'cell_add';
      end
      cont_rd_id = f_get_rd_id('ContactInfo.' || rd_code);

      uuid_str = f_uuid('edoc_sign', :obj_id);

      insert into contverify (cv_object_kind, cv_object_id, cv_from_rd_id, cv_contact_value, cv_password, cv_status,
                              cv_from_un_id, cv_datetime, cv_verify_un_id, cv_verify_datetime)
      values (f_objkind('clients', 'id'), :cont_c_id, :cont_rd_id, :cont_value, :uuid_str, 3,
              :un_id, :create_datetime, 1000000000000, :sign_datetime);

      insert into contveri (cvr_from_cda_id, cvr_hash_pin, cvr_status, cvr_sign_kind, cvr_from_un_id, cvr_datetime)
      values (:cont_cda_id, :uuid_str, 1, f_enum('edoc_sign_kind', 'edoc', 'id'), :un_id, :sign_datetime);
    end
  end

  if (obj_code = 'contveri') then
  begin
    select cda_from_c_id, f_enum('cont_kind', cda_cont_kind, 'code'), cda_is_main, f_cont_code(cda_from_clt_id), cvr_hash_pin
    from contdata, contveri
    where cvr_id = :obj_id and
          cda_id = cvr_from_cda_id
    into cont_c_id, cont_kind, cont_is_main, cont_value, code;

    if (cont_kind = 'cell') then
    begin
      if (cont_is_main = 1)
      then cont_kind = 'cell_main';
      else cont_kind = 'cell_add';
    end
    cont_rd_id = f_get_rd_id('ContactInfo.' || cont_kind);

    update contverify
    set cv_status = 3
    where cv_object_kind = f_objkind('clients', 'id') and
          cv_object_id = :cont_c_id and
          trim(cv_contact_value) = :cont_value and
          cv_from_rd_id = :cont_rd_id and
          cv_password = :code;
  end

  if (obj_code = 'contverify') then
  begin
    -- тут проверка правиильности подписи из ContVerify
    select trim(cv_contact_value), cv_object_id, f_get_rd_code(cv_from_rd_id), cv_password
    from contverify
    where cv_id = :obj_id
    into cont_value, cont_c_id, cont_kind, code;

    cont_kind = replace(cont_kind, 'ContactInfo.', '');
    if (cont_kind = 'cell_main') then
    begin
      cont_kind = 'cell';
      cont_is_main = 1;
    end else
    if (cont_kind = 'cell_add') then
    begin
      cont_kind = 'cell';
      cont_is_main = 0;
    end

    cont_clt_id = f_cont_id(cont_value, null);

    if (cont_clt_id is not null) then
    begin
      select cda_id
      from contdata
      where cda_from_c_id = :cont_c_id and
            cda_from_clt_id = :cont_clt_id and
            cda_cont_kind = f_enum('cont_kind', :cont_kind, 'id') and
            cda_is_main = :cont_is_main
      into cont_cda_id;

      if (cont_cda_id is not null) then
      begin
        update contveri
        set cvr_status = 1,
            cvr_sign_kind = f_enum('edoc_sign_kind', :sign_kind, 'id')
        where cvr_from_cda_id = :cont_cda_id and
              cvr_hash_pin = :code;
      end
    end
  end
end
