CREATE PROCEDURE OKT_DELINQ_LIST(
    REP_DATE DATE,
    IS_BLOCK SMALLINT,
    CALL_TYPE SMALLINT,
    CALLED_OPER STR_100,
    K_FOLDER_NAME STR_100,
    KONTRACT_NO VARCHAR(50),
    CITIZENSHIP INTEGER,
    DELINQ_DAYS_MIN INTEGER,
    DELINQ_DAYS_MAX INTEGER,
    MAX_REC_QTY INTEGER)
RETURNS (
    K_ID ID_TYPE,
    K_KONTRACT_NO VARCHAR(20),
    KF_NAME STR_100,
    REG_NAME STR_100,
    C_ID ID_TYPE,
    C_NAME_FIO STR_100,
    CITIZEN_STR STR_20,
    DELINQ_DAYS INTEGER,
    CL_TRY_QTY INTEGER,
    CL_CALL_QTY INTEGER,
    CL_LAST_CALL_DATE DATE,
    CL_LAST_TRY_DATE DATE,
    CALLED_OPER_STR STR_100,
    C_PLASTCARD_NO STR_20,
    PANT_ID ID_TYPE,
    IS_SIRENA SMALLINT,
    CON_TRY_QTY INTEGER,
    CON_CALL_QTY INTEGER,
    RESULT_STR VARCHAR(1000),
    NUM_ID INTEGER,
    NEXT_CALL TIMESTAMP,
    CAN_CALL SMALLINT)
AS
declare variable KF_ID ID_TYPE;
declare variable IS_OK smallint;
declare variable PANT_DATE_BEGIN date;
declare variable REC_QTY integer;
declare variable CL_CALL_RESULT STR_100;
declare variable CL_MEMO varchar(1000);
declare variable PANT_ID_CHECK ID_TYPE;
declare variable IS_FOUND smallint;
declare variable PA_COMMAND_NAME STR_100;
declare variable RD_ID integer;
declare variable DB_ID integer;
begin
-- CALL_TYPE 1 - КЦ Сирена, 2 - звонки прочрочникам
  rec_qty = 0;

  :db_id = f_get_db_id();
  if (citizenship is null) then
    citizenship = -1;
  if (citizenship = 0) then
    citizenship = -1;
  max_rec_qty = coalesce(max_rec_qty, 0);

  is_found = 0;

  if (db_id = 81001) then
  begin
    if (called_oper <> '') then
    begin
      for select pa_command_name
          from cc_permission_access(:called_oper, null)
          where pa_command_name starting 'OKTELL.Projects.'
          into :pa_command_name
      do
      begin
        if (is_found = 0) then
        begin
          if (pa_command_name in ('OKTELL.Projects.Sirena', 'OKTELL.Projects.Sirena1', 'OKTELL.Projects.Sirena2')) then
            is_found = 1;

        end
      end
    end
  end

  if (is_block = 1) then
    called_oper = '';

  if (db_id <> 81001) then
    pa_command_name = 'OKTELL.Projects.Sirena';

  if (db_id = 81001) then
  begin
    if ((is_found = 0) and
        (called_oper <> '')) then
      exit;
  end

  select rd.rd_id
  from refdata rd
  where rd.rd_code = :pa_command_name
  into :rd_id;

  for select k_id, kf_id, k_kontract_no, delinq_days
      from cc_delinq_list_tmp(:rep_date, :pa_command_name, :call_type, :called_oper, :k_folder_name, :kontract_no, :delinq_days_min, :delinq_days_max)
      into :k_id, :kf_id, :k_kontract_no, :delinq_days
  do
  begin
    is_ok = 1;

    if (exists(select *
               from sb_akt sba
               where sba.sba_from_k_id = :k_id and
                     sba.sba_from_sbl_id <> 0 and
                     sba.sba_is_delete = 0 and
                     :rep_date between sba.sba_date_begin and sba.sba_date_end)) then
      is_ok = 0;

    pant_id = 0;
    called_oper_str = '';
    pant_date_begin = null;

    if (is_ok = 1) then
    begin
      is_ok = 0;

      if (called_oper = '') then --  если это блокировка
      begin
        pant_id_check = null;

        select pantk.pant_oper_called, pantk.pant_date_begin, pantk.pant_id
        from pantera_kontracts pantk
        where --pantk.pant_project_type = 4 -- Просрочкики + КЦ Сирена
              pantk.pant_rd_project_type = :rd_id and
              :rep_date between pantk.pant_date_begin and pantk.pant_date_end and
              pantk.pant_from_k_id = :k_id
        into :called_oper_str, :pant_date_begin, :pant_id_check;

        if (pant_date_begin = :rep_date) then
          is_ok = 1;
        if (pant_date_begin is null) then
          is_ok = 1;

        if (called_oper_str = 'IGUSEV') then
          is_ok = 0;

        if ((pant_id_check is not null) and
            (is_ok = 1)) then
        begin
          if ((select pantk.pant_oper_called
               from pantera_kontracts pantk
               where pantk.pant_date_end <= :pant_date_begin and
                     pantk.pant_id <> :pant_id_check and
                     pantk.pant_from_k_id = :k_id
--and pantk.pant_project_type = 4 -- Просрочкики + КЦ Сирена
                     and
                     pantk.pant_rd_project_type = :rd_id
               order by pantk.pant_date_end desc
               rows 1) = 'IGUSEV') then
            is_ok = 0;
        end

-- если группа числитится за другим проектом ее мы не отображаем
        if (exists(select *
                   from pantera_kontracts pantk
                   where pantk.pant_rd_project_type = :rd_id and
                         :rep_date between pantk.pant_date_begin and pantk.pant_date_end and
                         pantk.pant_from_k_id = :k_id)) then
          is_ok = 0;
      end
      else
      begin
        select pantk.pant_id, pantk.pant_from_num_id
        from pantera_kontracts pantk
        where --pantk.pant_project_type = 4 -- Просрочкики + КЦ Сирена
              pantk.pant_rd_project_type = :rd_id and
              :rep_date between pantk.pant_date_begin and pantk.pant_date_end and
              pantk.pant_oper_called = :called_oper and
              pantk.pant_from_k_id = :k_id
        into :pant_id, :num_id;

        called_oper_str = called_oper;

        if (pant_id is null) then
          pant_id = 0;
        if (pant_id <> 0) then
          is_ok = 1;
      end

      if (is_ok = 1) then
      begin
        select kal.kal_new_kf_name, reg.reg_name
        from knaktlist kal
        inner join region reg on reg.reg_unicode = kal.kal_new_reg_id
        where kal.kal_from_kf_id = :kf_id and
              kal.is_delete = 0 and
              :rep_date between kal.kal_date and kal.kal_date_end
        into :kf_name, :reg_name;

        for select cl.c_id, cl.c_name_fio, rd.rd_name,
                   (select cont_value
                    from hp_contdata_get_cell_valid(0, kncl.kc_from_c_id))
            from knclient kncl
            inner join clients cl on cl.c_id = kncl.kc_from_c_id
            inner join refdata rd on rd.rd_id = cl.c_citizen
            where kncl.kc_from_k_id = :k_id and
                  kncl.kc_amount_approved > 0 and
                  ((cl.c_citizen = :citizenship) or
                  (:citizenship = -1))
            into :c_id, :c_name_fio, :citizen_str, :c_plastcard_no
        do
        begin
          cl_try_qty = 0;
          cl_call_qty = 0;
          cl_last_call_date = null;
          cl_last_try_date = null;
          con_try_qty = 0;
          con_call_qty = 0;
          next_call = null;
          can_call = 0;

          if (pant_id <> 0) then -- только для режима звонков показываю эти цифры
          begin
            select pds.pds_try_qty, pds.pds_call_qty, pds.pds_last_call_date, pds.pds_last_try_date
            from pantera_data_svod pds
            where pds.pds_from_pan_id = :pant_id and
                  pds.pds_from_c_id = :c_id
            into :cl_try_qty, :cl_call_qty, :cl_last_call_date, :cl_last_try_date;

            select sum(pds.pds_try_qty), sum(pds.pds_call_qty)
            from pantera_data_svod pds
            inner join contacts cont on cont.cc_id = pds.pds_from_cc_id
            where pds.pds_from_pan_id = :pant_id and
                  cont.cc_from_c_id = :c_id
            into :con_try_qty, :con_call_qty;

            cl_try_qty = coalesce(cl_try_qty, 0);
            cl_call_qty = coalesce(cl_call_qty, 0);
            con_try_qty = coalesce(con_try_qty, 0);
            con_call_qty = coalesce(con_call_qty, 0);

            result_str = '';
            cl_call_result = '';
            cl_memo = '';

            if (cl_try_qty > 0) then
            begin
              select rd.rd_name, pc.pcl_memo
              from pantera_clients pc
              inner join refdata rd on rd.rd_id = pc.pcl_rd_res
              where pc.pcl_from_pan_id = :pant_id and
                    pc.pcl_from_c_id = :c_id
              order by pc.pcl_called_date desc, pc.pcl_id desc
              rows 1
              into :cl_call_result, :cl_memo;

              result_str = 'Результат: ' || cl_call_result || '. Примечание: ' || cl_memo;
            end

-- получаю дату следующего звонка
-- получаю дату следующего звонка
            select next_call_date, can_call
            from hp_can_call_check(:k_id, 'Clients', :c_id, null)
            into :next_call, can_call;

            cl_call_result = coalesce(cl_call_result, '');
            cl_memo = coalesce(cl_memo, '');
          end
          suspend;
          rec_qty = rec_qty + 1;

          if ((rec_qty >= max_rec_qty) and
              (max_rec_qty <> 0)) then
            exit;
        end
      end
    end
  end
end
