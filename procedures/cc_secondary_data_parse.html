CREATE PROCEDURE CC_SECONDARY_DATA_PARSE(
    STR STR_255)
RETURNS (
    PASS_NO VARCHAR(30),
    CELL_NO VARCHAR(20))
AS
declare variable str_buf str_255;
begin
  str = trim(coalesce(str, ''));
  pass_no = '';
  cell_no = '';
  if (str <> '') then
  begin
    if (position('<pass>' in str) > 0) then
    begin
      str_buf = substring(str from (position('<pass>' in str)) + 6);
      pass_no = substring(str_buf from 1 for (position('</>', str_buf) - 1));
      pass_no = trim(pass_no);

      pass_no = replace(pass_no, 'А', 'A');
      pass_no = replace(pass_no, 'Н', 'H');
      pass_no = replace(pass_no, 'С', 'C');

      if (position('AA' in pass_no) > 0) then
      begin
        pass_no = 'AA ' || substring(pass_no from 3);
      end
    end
    if (position('<cell>' in str) > 0) then
    begin
      str_buf = substring(str from (position('<cell>' in str)) + 6);
      cell_no = substring(str_buf from 1 for (position('</>', str_buf) - 1));
      cell_no = replace(cell_no, ' ', '');
      cell_no = replace(cell_no, '_', '');
      cell_no = replace(cell_no, '-', '');
      cell_no = replace(cell_no, ')', '');
      cell_no = replace(cell_no, '(', '');
      if (position('+7' in cell_no) > 0) then
        cell_no = substring(cell_no from 2);
      if (position('89' in cell_no) > 0) then
        cell_no = '7' || substring(cell_no from 2);
      if ((position('9' in cell_no) = 1) and
          (char_length(cell_no) = 10)) then
        cell_no = '7' || cell_no;
      cell_no = replace(cell_no, '+', '');
    end
  end
  suspend;
end
