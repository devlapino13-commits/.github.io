CREATE PROCEDURE HP_OUTMAILING_KONT_INFO(
    IN_K_ID D_ID)
RETURNS (
    RES_OFFICE_ID D_INTEGER,
    RES_BLANK_DATE D_DATE,
    RES_DELINQ_DAYS D_INTEGER,
    RES_IS_CLOSED D_BOOL,
    RES_STATUS D_STR_CODE,
    RES_HQ_USER D_BOOL)
AS
declare variable oks_delinq_date d_date;
declare variable off_amount d_currency;
declare variable un_id d_id;
begin
  res_blank_date = current_date;
  res_is_closed = 0;
  res_status = '';
  res_hq_user = 0;

  un_id = uxf$usr_un();

  if (exists(select * from username
             where un_id = :un_id and
                   un_curr_reg_id = f_hq_id())) then
    res_hq_user = 1;

  select k_last_reg_id
  from kontract
  where k_id = :in_k_id
  into res_office_id;

  res_status = '';
  select oks_delinq_date, iif(oks_close_state = 0, 0, 1)
  from oksummary
  where oks_from_k_id = :in_k_id and
        oks_date_begin <= current_date
  order by oks_from_k_id desc, oks_date_begin desc
  rows 1
  into oks_delinq_date, res_is_closed;

  if (res_is_closed = 1) then
  begin
    res_status = 'Закрыт';

    select sum(oko_sh_os + oko_os + oko_sh_perc + oko_perc)
    from operkroff
    where oko_from_k_id = :in_k_id and
          oko_ol_is_active = 1
    into off_amount;

    if (off_amount is null) then
      off_amount = 0;

    if (off_amount > 0) then
    begin
      res_is_closed = 0;
      res_status = 'Внебалансовый';
    end
  end

  res_delinq_days = 0;
  if (oks_delinq_date <= current_date) then
    res_delinq_days = datediff(day, oks_delinq_date, current_date);

  if (res_status = '') then
  begin
    if (res_delinq_days > 0) then
      res_status = 'Просроченный';
    else
      res_status = 'Активный';
  end

  suspend;
end
