CREATE PROCEDURE UXP$UI_UXGRID(
    GRID_UUID D_UUID)
RETURNS (
    UGR_NAME D_STR_100,
    UGR_MEMO D_STR_255,
    UGR_IS_PRIVATE D_BOOL,
    UGR_HIDE_ON_GROUP D_BOOL,
    UGR_GROUP_PANEL D_BOOL,
    UGR_FIND_PANEL D_SMALLINT,
    UGR_HIDE_SUMMARY D_BOOL,
    UGR_UPDATED D_DATETIME,
    UGR_READONLY D_BOOL,
    UGR_CAPTION_ALIGN D_SMALLINT,
    UGR_HIDDEN D_BOOL,
    UGR_H_ALIGN D_SMALLINT,
    UGR_V_ALIGN D_SMALLINT,
    UGR_WIDTH D_INTEGER,
    UGR_HEIGHT D_INTEGER,
    UGR_DISABLE_EDIT D_BOOL,
    UGR_AUTOWIDTH D_SMALLINT,
    UGR_FILTER_UUID_CODE D_STR_CODE,
    UGR_IS_BANDED D_BOOL,
    UGR_HEADER_AUTOHEIGHT D_BOOL,
    UGR_SHOW_INFO_PANEL D_BOOL,
    UGR_HIDE_INPLACE_EDITOR D_BOOL,
    UGR_SHOW_EXCEL_IMPORT D_BOOL,
    UGR_OWNER_USER D_STR_CODE,
    UGR_EXCEL D_BLOB_BINARY,
    UGR_DS_CODE D_STR_CODE,
    UGR_GROUP_FIELD D_STR_255,
    UGR_FORMATTING D_STR_8100,
    UGR_TREE_CAN_MOVE D_BOOL,
    UGR_ALLOW_APPEND D_BOOL,
    UGR_UI_KIND D_STR_CODE,
    UGR_CALENDAR_FIELDS D_STR_4096)
AS
declare variable grid_found d_bool;
declare variable band_count d_smallint;
declare variable ugr_ufr_uuid d_uuid;
declare variable res_tree_move_sql d_str_8100;
begin
  ugr_readonly = 0;
  ugr_caption_align = -1;
  ugr_hidden = 0;
  ugr_h_align = 3; -- alClient
  ugr_v_align = 3; -- alClient
  ugr_width = 0;
  ugr_height = 0;
  ugr_is_banded = 0;
  ugr_owner_user = '';
  ugr_excel = null;
  ugr_ds_code = '';
  ugr_group_field = '';
  ugr_formatting = '';
  ugr_tree_can_move = 0;
  ugr_allow_append = 0;
  ugr_ui_kind = '';
  ugr_calendar_fields = '';

  grid_found = 0;
  ugr_filter_uuid_code = '';

  select count(*) from ux$band
  where bnd_grid_uuid = :grid_uuid
  into band_count;

  if (band_count > 1) then
    ugr_is_banded = 1;

  for select ugr_ufr_uuid, ugr_name, ugr_memo, ugr_is_private, ugr_hide_on_group,
             ugr_group_panel, ugr_find_panel, ugr_hide_summary, ugr_disable_edit,
             ugr_autowidth, ugr_header_autoheight, ugr_show_info_panel,
             ugr_hide_inplace_editor, ugr_show_excel_import, ugr_owner_user,
             ugr_excel, ugr_ds_code, ugr_group_field, ugr_formatting, ugr_tree_can_move,
             ugr_allow_append, ugr_ui_kind, ugr_calendar_fields
  from ux$grid
  where ugr_uuid = :grid_uuid
  into :ugr_ufr_uuid, :ugr_name, :ugr_memo, :ugr_is_private, :ugr_hide_on_group,
       :ugr_group_panel, :ugr_find_panel, :ugr_hide_summary, :ugr_disable_edit,
       :ugr_autowidth, :ugr_header_autoheight, :ugr_show_info_panel,
       :ugr_hide_inplace_editor, :ugr_show_excel_import, :ugr_owner_user,
       :ugr_excel, :ugr_ds_code, :ugr_group_field, :ugr_formatting, :ugr_tree_can_move,
       :ugr_allow_append, :ugr_ui_kind, :ugr_calendar_fields
  do
  begin
    grid_found = 1;

    ugr_name = uxf$lang(ugr_name);
    ugr_memo = uxf$lang(ugr_memo);

    if (ugr_tree_can_move = 1) then
    begin
      select res_tree_move_sql
      from uxp$uxframe_sql_tree_move(:ugr_ufr_uuid, false)
      into :res_tree_move_sql;

      if (res_tree_move_sql = '') then
        ugr_tree_can_move = 0;
    end

    if (ugr_formatting is null) then
      ugr_formatting = '';

    if (coalesce(ugr_ds_code, '') = '' and ugr_excel is not null) then
      ugr_ds_code = 'qRep';

    select ufi_readonly, ufi_caption_align, ufi_hidden, ufi_h_align, ufi_v_align, ufi_width, ufi_height
    from ux$fritem
    where ufi_uuid = :grid_uuid
    into :ugr_readonly, :ugr_caption_align, :ugr_hidden, :ugr_h_align, :ugr_v_align, :ugr_width, :ugr_height;

    select ufm_updated
    from ux$grid, ux$frame, ux$form
    where ugr_uuid = :grid_uuid and
          ugr_ufr_uuid = ufr_uuid and
          ufr_owner = ufm_uuid
    into ugr_updated;

    select uuid_to_char(ufl_uuid)
    from ux$filter
    where ufl_owner = :grid_uuid and
          ufl_kind = -1
    rows 1
    into ugr_filter_uuid_code;

    suspend;
  end

  if (grid_found = 0) then
  begin
    ugr_is_private = 0;
    ugr_hide_on_group = 0;
    ugr_group_panel = 1;
    ugr_find_panel = 2;
    ugr_hide_summary = 0;
    ugr_autowidth = -1;
    ugr_header_autoheight = 1;
    ugr_show_info_panel = 0;
    ugr_hide_inplace_editor = 0;
    ugr_show_excel_import = 0;

    ugr_readonly = 0;
    ugr_caption_align = -1;
    ugr_hidden = 0;
    ugr_h_align = 3; -- alClient
    ugr_v_align = 3; -- alClient
    ugr_width = 0;
    ugr_height = 0;

    for select ufi_name, ufr_name, ufm_updated, ufi_readonly,
               ufi_caption_align, ufi_hidden, ufi_h_align, ufi_v_align, ufi_width, ufi_height,
               ufr_hide_on_group, ufr_group_panel, ufr_find_panel, ufr_hide_summary, ufr_grid_disable_edit,
               ufr_autowidth
    from ux$fritem, ux$frame, ux$form
    where ufi_uuid = :grid_uuid and
          ufi_owner = ufr_uuid and
          ufr_owner = ufm_uuid
    into :ugr_name, :ugr_memo, :ugr_updated, :ugr_readonly,
         :ugr_caption_align, :ugr_hidden, :ugr_h_align, :ugr_v_align, :ugr_width, :ugr_height,
         :ugr_hide_on_group, :ugr_group_panel, :ugr_find_panel, :ugr_hide_summary, :ugr_disable_edit,
         :ugr_autowidth
    do
    begin
      ugr_name = uxf$lang(ugr_name);
      ugr_memo = uxf$lang(ugr_memo);

      select uuid_to_char(ufl_uuid)
      from ux$filter
      where ufl_owner = :grid_uuid and
            ufl_kind = -1
      rows 1
      into ugr_filter_uuid_code;

      grid_found = 1;
      suspend;
    end
  end

  if (grid_found = 0) then
  begin
    ugr_is_private = 0;
    ugr_hide_on_group = 0;
    ugr_group_panel = 1;
    ugr_find_panel = 2;
    ugr_hide_summary = 0;
    ugr_autowidth = -1;
    ugr_header_autoheight = 1;
    ugr_show_info_panel = 0;
    ugr_hide_inplace_editor = 0;
    ugr_show_excel_import = 0;

    ugr_readonly = 0;
    ugr_caption_align = -1;
    ugr_hidden = 0;
    ugr_h_align = 3; -- alClient
    ugr_v_align = 3; -- alClient
    ugr_width = 0;
    ugr_height = 0;
    ugr_updated = current_timestamp;

    for select iif(coalesce(uni_caption, '') <> '', uni_caption, uni_code),
               uni_name, uni_readonly,
               uni_caption_align, uni_hidden, uni_h_align, uni_v_align, uni_width, uni_height
    from ux$unititem
    where uni_uuid = :grid_uuid
    into :ugr_name,
         :ugr_memo, :ugr_readonly,
         :ugr_caption_align, :ugr_hidden, :ugr_h_align, :ugr_v_align, :ugr_width, :ugr_height
    do
    begin
      ugr_name = uxf$lang(ugr_name);
      ugr_memo = uxf$lang(ugr_memo);

      grid_found = 1;

      select uuid_to_char(ufl_uuid)
      from ux$filter
      where ufl_owner = :grid_uuid and
            ufl_kind = -1
      rows 1
      into ugr_filter_uuid_code;

      suspend;
    end
  end

  if (grid_found = 0) then
    exception e_default f_format('Grid not found. Grid uuid: %2', uuid_to_char(grid_uuid));
end
