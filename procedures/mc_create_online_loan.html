CREATE PROCEDURE MC_CREATE_ONLINE_LOAN(
    MCL_ID D_ID,
    MA_ID D_ID)
RETURNS (
    RES_CODE D_STR_CODE,
    RES_TEXT D_STR_255,
    K_ID D_ID)
AS
declare variable c_id d_id;
declare variable ma_reg_id d_integer;
declare variable ool_id d_id;
declare variable kc_id d_id;
begin

    res_code = 'success';
    res_text = '';

    c_id = (select mcl_from_c_id from mc_login where mcl_id = :mcl_id);
    if (coalesce(c_id, 0) = 0) then
    begin
        res_code = 'error';
        res_text = 'Не удалось загрузить данные профиля. Попробуйте войти в приложение заново';
        suspend;
        exit;
    end

    if (coalesce(ma_id, 0) = 0) then
    begin
        res_code = 'error';
        res_text = 'Не удалось загрузить информацию о заявке на кредит. Пожалуйста, попробуйте снова или обратитесь к технической поддержке';
        suspend;
        exit;
    end

    -- Не даем перевести заявку в договор в процессе, если она была создана не в рамках онлайн-кредитования OiVO
    select ool.ool_id
    from mc_oivo_online_loan ool
    where ool.ool_from_mcl_id = :mcl_id
        and ool.ool_from_ma_id = :ma_id
        and ool.ool_from_k_id is null
    into :ool_id;

    if (ool_id is null) then
    begin
        res_code = 'error';
        res_text = 'Произошла ошибка. Пожалуйста, обратитесь к технической поддержке';
        suspend;
        exit;
    end

    -- Проверяем, подписана ли кредитная заявка
    if (f_edoc_status('loan.application', :ma_id, current_date) <> 'signed') then
    begin
        res_code = 'error';
        res_text = 'Для продолжения подпишите электронный документ заявки на займ';
        suspend;
        exit;
    end

    select first 1 ma.ma_office_id
    from miniapp ma
    where ma.ma_id = :ma_id
        and ma.ma_date_end >= '31.12.2999'
        and ma.ma_is_active = 1
    order by ma.ma_date_app desc
    into :ma_reg_id;

    -- Переводим в договор в процессе. Процедура make_new_kontract здесь не используется, т.к. в ней
    -- выполняются проверки, которые на данном этапе не нужны. Все необходимые проверки в рамках онлайн-кредитования
    -- OiVO выполняются отдельно
    select k_id
    from insert_new_kontract(:ma_reg_id, :ma_id, 0, 1000000000000)
    into :k_id;

    if (coalesce(k_id, 0) = 0) then
    begin
        res_code = 'error';
        res_text = 'Не удалось создать кредитный договор. Пожалуйста, попробуйте снова или обратитесь к технической поддержке';
        suspend;
        exit;
    end

    update mc_oivo_online_loan ool
    set ool.ool_from_k_id = :k_id,
        ool.ool_stage = f_enum_id('oivo_online_loan_stage', 'kontract_created')
    where ool.ool_id = :ool_id;

    -- Делаем рассчет ПДН
    select kc.kc_id
    from knclient kc
    where kc.kc_from_k_id = :k_id
        and kc.kc_from_c_id = :c_id
    into :kc_id;

    execute procedure hp_knclient_pdn_update(:kc_id, 0, null, null, null, null, null, 'create_oivo_online_loan');
    execute procedure hp_antifrod_check(:k_id);


    suspend;

    when any do
    begin
        res_code = 'error';
        res_text = 'Что-то пошло не так. Повторите попытку позже или обратитесь к технической поддержке';
        suspend;
    end
end
