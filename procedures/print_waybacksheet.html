CREATE PROCEDURE PRINT_WAYBACKSHEET(
    HO_ID D_ID)
RETURNS (
    C_NAME_FIO D_STR_100,
    HRS_JOBNAME D_STR_100,
    HRS_OFFICE VARCHAR(50),
    HRS_DATE_UVOLEN D_DATE,
    SIGNER1_NAME D_STR_100,
    SIGNER2_NAME D_STR_100,
    SIGNER3_NAME D_STR_100,
    SIGNER4_NAME D_STR_100,
    SIGNER5_NAME D_STR_100,
    SIGNER6_NAME D_STR_100,
    SIGNER7_NAME D_STR_100,
    SIGNER8_NAME D_STR_100,
    RES1 D_STR_20,
    RES2 D_STR_20,
    RES3 D_STR_20,
    RES4 D_STR_20,
    RES5 D_STR_20,
    RES6 D_STR_20,
    RES7 D_STR_20,
    RES8 D_STR_20)
AS
declare variable un_id1 d_id;
declare variable un_id2 d_id;
declare variable un_id3 d_id;
declare variable un_id4 d_id;
declare variable un_id5 d_id;
declare variable un_id6 d_id;
declare variable un_id7 d_id;
declare variable un_id8 d_id;
declare variable signed d_str_20;
declare variable not_signed d_str_20;
begin
  signed = 'Подписано';
  not_signed = 'Не подписано';
  select c.c_name_fio, jl.jl_name, reg.reg_name,
         (ps.ps_date_begin - 1)
  from hr_pstatus ps
  inner join clients c on c.c_id = ps.ps_from_c_id
  inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
  inner join region reg on reg.reg_unicode = ps.ps_from_reg_id
  where ps.ps_from_ho_id = :ho_id
  into :c_name_fio, :hrs_jobname, :hrs_office, :hrs_date_uvolen;

  select wbs.wbs_signer1, wbs.wbs_signer2, wbs.wbs_signer3, wbs.wbs_signer4, wbs.wbs_signer5, wbs.wbs_signer6,
         wbs.wbs_signer7, wbs.wbs_signer8
  from waybacksheet wbs
  where wbs.wbs_ho_id = :ho_id
  into :un_id1, :un_id2, :un_id3, :un_id4, :un_id5, :un_id6, :un_id7, :un_id8;

  signer1_name = (select un_fio
                  from username
                  where un_id = :un_id1);
  res1 = iif(signer1_name is not null, signed, not_signed);
  signer2_name = (select un_fio
                  from username
                  where un_id = :un_id2);
  res2 = iif(signer2_name is not null, signed, not_signed);
  signer3_name = (select un_fio
                  from username
                  where un_id = :un_id3);
  res3 = iif(signer3_name is not null, signed, not_signed);
  signer4_name = (select un_fio
                  from username
                  where un_id = :un_id4);
  res4 = iif(signer4_name is not null, signed, not_signed);
  signer5_name = (select un_fio
                  from username
                  where un_id = :un_id5);
  res5 = iif(signer5_name is not null, signed, not_signed);
  signer6_name = (select un_fio
                  from username
                  where un_id = :un_id6);
  res6 = iif(signer6_name is not null, signed, not_signed);
  signer7_name = (select un_fio
                  from username
                  where un_id = :un_id7);
  res7 = iif(signer7_name is not null, signed, not_signed);
  signer8_name = (select un_fio
                  from username
                  where un_id = :un_id8);
  res8 = iif(signer8_name is not null, signed, not_signed);

  suspend;
end
