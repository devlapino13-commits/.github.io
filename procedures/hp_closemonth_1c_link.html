CREATE PROCEDURE HP_CLOSEMONTH_1C_LINK(
    PCB_ID D_INTEGER,
    OL_ID D_ID)
AS
declare variable pcb_date d_date;
declare variable pcb_amount_sum d_currency;
declare variable pcb_from_ol_id d_id;
declare variable ol_date d_date;
declare variable ob_summa d_currency;
declare variable ob_client_dt d_id;
declare variable ob_client_kt d_id;
declare variable kob_detail_id d_id;
begin
  if (ol_id is null) then
    exception e_message 'Операция не указана, OL_ID = Null';

  if (pcb_id is null) then
    exception e_message 'PCB_ID не указан, Null';

  select pcb_date, pcb_amount_sum, pcb_from_ol_id
  from packet_1c_imp_bank
  where pcb_id = :pcb_id
  into pcb_date, pcb_amount_sum, pcb_from_ol_id;

  if (pcb_from_ol_id is not null) then
    exception e_message f_format('Строка 1С уже закреплена к операции OL_ID : %1', pcb_from_ol_id);

  select ol_date
  from operlist
  where ol_id = :ol_id
  into ol_date;

  if (ol_date is null) then
    exception e_message f_format('Операция не найдена, OL_ID: %1', ol_id);

  if (pcb_date <> ol_date) then
    exception e_message f_format('Даты отличаются, ol_date: %1, 1c date: %2',
                                 f_datetostr(ol_date), f_datetostr(pcb_date));

  select ob_summa, ob_client_dt, ob_client_kt
  from operbook
  where ob_from_ol_id = :ol_id
  into ob_summa, ob_client_dt, ob_client_kt;

  if (ob_summa is null) then
    ob_summa = 0;

  if (pcb_amount_sum <> ob_summa) then
    exception e_message f_format('Суммы отличаются, ob_summa: %1, 1c amount: %2',
                                 ob_summa, pcb_amount_sum);

  kob_detail_id = F_KN_DETAIL_ID('Импорт из 1С');

  insert into kn_operbook (kob_from_ol_id, kob_currency, kob_acc_dt, kob_acc_kt, kob_summa, kob_summa_val,
                           kob_summa_tax_dt, kob_summa_tax_kt,
                           kob_detail_id, kob_client_dt, kob_client_kt)
  select :ol_id, pcb_from_cu_id, pcb_acc_dt_id, pcb_acc_kt_id, pcb_amount_sum, pcb_amount_val,
         pcb_amount_tax_dt, pcb_amount_tax_kt, :kob_detail_id, :ob_client_dt, :ob_client_kt
  from packet_1c_imp_bank
  where pcb_id = :pcb_id;

  update packet_1c_imp_bank
  set pcb_from_ol_id = :ol_id,
      pcb_link_kind = 1
  where pcb_id = :pcb_id;

  update kn_opermoney
  set kom_is_done = 1
  where kom_from_ol_id = :ol_id;
end
