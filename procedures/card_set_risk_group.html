CREATE PROCEDURE CARD_SET_RISK_GROUP(
    OBJECT_KIND D_STR_100 NOT NULL,
    OBJECT_ID D_ID NOT NULL)
AS
declare variable card_id d_integer;
declare variable cca_id d_integer;
declare variable request_json d_str_8100;
declare variable is_cashing d_smallint;
declare variable disb_ol_id d_id;
declare variable cs_id d_integer;
declare variable cca_code d_str_code;
declare variable object_kind_str d_str_100;
declare variable k_id d_id;
declare variable c_id d_id;
declare variable risk_group d_str_10;
declare variable risk_group_id d_integer;
declare variable k_last_reg_id d_integer;
declare variable un_id d_id;
begin
    object_kind_str = f_objkind(object_kind, 'name');

    if (object_kind_str = 'kontract') then
        k_id = object_id;
    else
        exception e_message 'object_kind = '||object_kind_str||' не реализован';

    select kont.k_fact_disb_ol_id,
           iif(pr.pr_mbulak_cards_kind = 1, 1, 0),
           kont.k_last_reg_id
    from kontract kont
    inner join product pr on pr.pr_id = kont.k_from_pr_id
    where kont.k_id = :k_id
    into :disb_ol_id,
         :is_cashing,
         :k_last_reg_id;

    select ol.ol_user_id from operlist ol where ol.ol_id = :disb_ol_id
    into :un_id;

    select
      kc_from_c_id
    from knclient
      inner join clients
        on c_id = kc_from_c_id
    where kc_from_k_id = :k_id
      and kc_amount_approved > 0
    into
      :c_id;

    select
      card_id,
      cca_id,
      cca_code
    from hp_client_bank_card_mb(:c_id, current_date)
    into
      :card_id,
      :cca_id,
      :cca_code;

    risk_group = upper(iif(is_cashing = 1, 'nocre', 'nctsp'));

    request_json = f_format('{"card_acct":"%1", "risk_group":"%2"}', :cca_code, :risk_group);

    insert into cardsync(
      cs_from_csc_id,
      cs_card_id,
      cs_from_cca_id,
      cs_object_kind,
      cs_object_id,
      cs_request_json,
      cs_office_id,
      cs_from_un_id,
      cs_from_k_id,
      cs_from_c_id
    )
    values(
      f_get_cardsynccfg('set_risk_group', 'id'),
      :card_id,
      :cca_id,
      f_objkind('kontract', 'id'),
      :k_id,
      :request_json,
      :k_last_reg_id,
      :un_id,
      :k_id, 
      :c_id
    )
    returning cs_id
    into :cs_id;

    -- обновляю риск-группу для карточного счета
    risk_group_id = (select crg.crg_id
                     from cardriskgroups crg
                     where crg.crg_code = :risk_group);

    update cardclientacc cca
    set cca.cca_risk_group = :risk_group_id
    where cca.cca_id = :cca_id;

    -- создаю запись в cardsyncopers для связки cardsync с operlist
    insert into cardsyncopers (cso_from_cs_id, cso_from_ol_id, cso_after_exec)
    values (:cs_id, :disb_ol_id, 0);
end
