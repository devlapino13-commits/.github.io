CREATE PROCEDURE FIND_US_REF_ACCOUNT(
    FIND_ID D_STR_CODE,
    FIND_TEXT D_STR_255,
    ADD_PARAMS D_STR_8100,
    UN_ID D_ID)
RETURNS (
    RES_ID D_ID,
    RES_CODE D_STR_CODE,
    RES_TEXT D_STR_LONG)
AS
declare variable num_1 d_str_255;
declare variable num_2 d_str_255;
declare variable num_3 d_str_255;
declare variable txt d_str_255;
declare variable inx_id d_id;
declare variable reg_id d_integer;
declare variable ab_id d_id;
declare variable max_row d_integer;
declare variable find_reg_id d_integer;
declare variable find_c_id d_str_code;
declare variable find_al_id d_str_code;
declare variable find_currency d_str_10;
declare variable find_with_local_currency d_smallint;
declare variable local_currency d_str_10;
declare variable sys_names d_str_2048;
declare variable is_include d_smallint;
begin
  find_reg_id = f_json_str_get_value(add_params, 'region_id', '0');
  find_currency = f_json_str_get_value(add_params, 'currency', '');
  find_with_local_currency = f_json_str_get_value(add_params, 'with_local_currency', '1');
  find_al_id = f_json_str_get_value(add_params, 'al_id', '');
  find_c_id = f_json_str_get_value(add_params, 'client_id', '0');
  sys_names = f_json_str_get_value(add_params, 'sys_names', '');
  is_include = f_json_str_get_value(add_params, 'is_include', '1');

  local_currency = '';
  if (find_with_local_currency = 1) then
  begin
    select cm_default_currency
    from company
    into local_currency;
  end

  if (coalesce(find_c_id, '') = '' or find_c_id = '0') then
    find_c_id = null;

  if (un_id is null) then
    un_id = 0;

  if (find_text is null) then
    find_text = '';

  if (find_id is null) then
    find_id = '';

  if (find_id <> '' and
      find_id <> '0' and
      find_al_id = '' and
      find_text = '') then
  begin
    select acc_id, f_fmt_acc(acc_id), f_objname('ref_account', acc_id)
    from ref_account
    where acc_id = :find_id
    into res_id, res_code, res_text;

    suspend;

    exit;
  end

  reg_id = find_reg_id;

  if (find_al_id <> '') then
  begin
    select al_from_acc_id
    from acc_list
    where al_id = :find_al_id
    into :res_id;

    if (res_id is not null) then
    begin
      res_code = f_fmt_acc(res_id);
      res_text = f_objname('ref_account', res_id);
      suspend;

      exit;
    end
  end

  max_row = 200;

  find_text = replace(find_text, '.', '');

  select num_1, num_2, num_3, txt
  from find_editor_parcer(:find_text)
  into :num_1, :num_2, :num_3, :txt;

  if (num_1 <> '') then
  begin
    if (char_length(num_1) = 5) then
    begin
      select ab_id
      from ref_accbalance
      where ab_code = :num_1
      into ab_id;
    end
    if (char_length(num_1) < 3) then
      exit;
  end

  if (num_1 <> '' and
      num_2 <> '' and
      num_3 <> '') then
  begin
    select inx_id
    from ref_income_expense
    where inx_code = :num_2
    rows 1
    into inx_id;
    if (inx_id is null) then
      exit;

    if (num_3 <> '') then
    begin
      select reg_unicode
      from region
      where reg_code = :num_3
      rows 1
      into reg_id;
      if (coalesce(reg_id, 0) = 0) then
        exit;
    end

    if (ab_id is not null) then
    begin
      for select acc_id
          from ref_account, sys_accbalance, ref_income_expense
          where acc_from_ab_id = :ab_id and
                acc_from_inx_id = inx_id and
                (:reg_id = 0 or
                sab_acc_mask not containing 'RRRR' or
                (sab_acc_mask containing 'RRRR' and
                acc_from_reg_id = :reg_id)) and
                inx_code starting :num_2 and
                acc_from_sab_id = sab_id and
                f_check_refaccount_kind(sab_full_code, :sys_names) = :is_include and
                (:find_currency = '' or
                acc_from_cu_id in (:find_currency, :local_currency)) and
                (:find_c_id is null or
                coalesce(acc_object_kind, 0) <> 0 or
                (acc_object_kind = 0 and
                acc_object_id + 0 = :find_c_id))
          rows :max_row
          into res_id
      do
      begin
        res_code = f_fmt_acc(res_id);
        res_text = f_objname('ref_account', res_id);
        suspend;
      end
    end
    else
    begin
      for select acc_id
          from ref_account, sys_accbalance, ref_income_expense
          where acc_code starting :num_1 and
                acc_from_inx_id = inx_id and
                (:reg_id = 0 or
                sab_acc_mask not containing 'RRRR' or
                (sab_acc_mask containing 'RRRR' and
                acc_from_reg_id = :reg_id)) and
                inx_code starting :num_2 and
                acc_from_sab_id = sab_id and
                f_check_refaccount_kind(sab_full_code, :sys_names) = :is_include and
                (:find_currency = '' or
                acc_from_cu_id in (:find_currency, :local_currency)) and
                (:find_c_id is null or
                coalesce(acc_object_kind, 0) <> 0 or
                (acc_object_kind = 0 and
                acc_object_id + 0 = :find_c_id))
          rows :max_row
          into res_id
      do
      begin
        res_code = f_fmt_acc(res_id);
        res_text = f_objname('ref_account', res_id);
        suspend;
      end
    end
  end
  else
  if (num_1 <> '' and
      num_2 <> '') then
  begin
    if (ab_id is not null) then
    begin
      for select acc_id
          from ref_account, sys_accbalance
          where acc_from_ab_id = :ab_id and
                acc_object_kind = 8 and
                acc_object_id = :num_2 and
                acc_from_sab_id = sab_id and
                f_check_refaccount_kind(sab_full_code, :sys_names) = :is_include and
                (:reg_id = 0 or
                sab_acc_mask not containing 'RRRR' or
                (sab_acc_mask containing 'RRRR' and
                acc_from_reg_id = :reg_id)) and
                (:find_currency = '' or
                acc_from_cu_id in (:find_currency, :local_currency)) and
                (:find_c_id is null or
                coalesce(acc_object_kind, 0) <> 0 or
                (acc_object_kind = 0 and
                acc_object_id + 0 = :find_c_id))
          rows :max_row
          into res_id
      do
      begin
        res_code = f_fmt_acc(res_id);
        res_text = f_objname('ref_account', res_id);
        suspend;
      end
    end
    else
    begin
      for select acc_id
          from ref_account, sys_accbalance
          where acc_code starting :num_1 and
                acc_object_kind = 8 and
                acc_from_sab_id = sab_id and
                f_check_refaccount_kind(sab_full_code, :sys_names) = :is_include and
                (:reg_id = 0 or
                sab_acc_mask not containing 'RRRR' or
                (sab_acc_mask containing 'RRRR' and
                acc_from_reg_id = :reg_id)) and
                (:find_currency = '' or
                acc_from_cu_id in (:find_currency, :local_currency)) and
                acc_object_id = :num_2 and
                (:find_c_id is null or
                coalesce(acc_object_kind, 0) <> 0 or
                (acc_object_kind = 0 and
                acc_object_id + 0 = :find_c_id))
          rows :max_row
          into res_id
      do
      begin
        res_code = f_fmt_acc(res_id);
        res_text = f_objname('ref_account', res_id);
        suspend;
      end
    end

    if (row_count > 0) then
      exit;

    if (ab_id is not null) then
    begin
      for select acc_id
          from ref_account
          inner join sys_accbalance on (acc_from_sab_id = sab_id and
                f_check_refaccount_kind(sab_full_code, :sys_names) = :is_include)
          left join ref_income_expense on (acc_from_inx_id = inx_id)
          left join region on (acc_from_reg_id = reg_unicode)
          where acc_from_ab_id = :ab_id and
                acc_name containing :txt and
                (:reg_id = 0 or
                sab_acc_mask not containing 'RRRR' or
                (sab_acc_mask containing 'RRRR' and
                acc_from_reg_id = :reg_id)) and
                (:find_currency = '' or
                acc_from_cu_id in (:find_currency, :local_currency)) and
                (acc_number = :num_2 or
                inx_code starting :num_2 or
                reg_code = :num_2) and
                (:find_c_id is null or
                coalesce(acc_object_kind, 0) <> 0 or
                (acc_object_kind = 0 and
                acc_object_id + 0 = :find_c_id))
          rows :max_row
          into res_id
      do
      begin
        res_code = f_fmt_acc(res_id);
        res_text = f_objname('ref_account', res_id);
        suspend;
      end
    end
    else
    begin
      for select acc_id
          from ref_account
          inner join sys_accbalance on (acc_from_sab_id = sab_id and
                f_check_refaccount_kind(sab_full_code, :sys_names) = :is_include)
          left join ref_income_expense on (acc_from_inx_id = inx_id)
          left join region on (acc_from_reg_id = reg_unicode)
          where acc_code starting :num_1 and
                acc_name containing :txt and
                (:reg_id = 0 or
                sab_acc_mask not containing 'RRRR' or
                (sab_acc_mask containing 'RRRR' and
                acc_from_reg_id = :reg_id)) and
                (:find_currency = '' or
                acc_from_cu_id in (:find_currency, :local_currency)) and
                (acc_number = :num_2 or
                inx_code = :num_2 or
                reg_code = :num_2) and
                (:find_c_id is null or
                coalesce(acc_object_kind, 0) <> 0 or
                (acc_object_kind = 0 and
                acc_object_id + 0 = :find_c_id))
          rows :max_row
          into res_id
      do
      begin
        res_code = f_fmt_acc(res_id);
        res_text = f_objname('ref_account', res_id);
        suspend;
      end
    end
  end
  else
  if (num_1 <> '') then
  begin
    if (ab_id is not null) then
    begin
      for select acc_id
          from ref_account, sys_accbalance
          where acc_from_ab_id = :ab_id and
                acc_name containing :txt and
                acc_from_sab_id = sab_id and
                f_check_refaccount_kind(sab_full_code, :sys_names) = :is_include and
                (:reg_id = 0 or
                sab_acc_mask not containing 'RRRR' or
                (sab_acc_mask containing 'RRRR' and
                acc_from_reg_id = :reg_id)) and
                (:find_currency = '' or
                acc_from_cu_id in (:find_currency, :local_currency)) and
                (:find_c_id is null or
                coalesce(acc_object_kind, 0) <> 0 or
                (acc_object_kind = 0 and
                acc_object_id + 0 = :find_c_id))
          rows :max_row
          into res_id
      do
      begin
        res_code = f_fmt_acc(res_id);
        res_text = f_objname('ref_account', res_id);
        suspend;
      end
    end
    else
    begin
      for select acc_id
          from ref_account, sys_accbalance
          where acc_code starting :num_1 and
                acc_name containing :txt and
                acc_from_sab_id = sab_id and
                f_check_refaccount_kind(sab_full_code, :sys_names) = :is_include and
                (:reg_id = 0 or
                sab_acc_mask not containing 'RRRR' or
                (sab_acc_mask containing 'RRRR' and
                acc_from_reg_id = :reg_id)) and
                (:find_currency = '' or
                acc_from_cu_id in (:find_currency, :local_currency)) and
                (:find_c_id is null or
                coalesce(acc_object_kind, 0) <> 0 or
                (acc_object_kind = 0 and
                acc_object_id + 0 = :find_c_id))
          rows :max_row
          into res_id
      do
      begin
        res_code = f_fmt_acc(res_id);
        res_text = f_objname('ref_account', res_id);
        suspend;
      end
    end
  end
  else
  if (sys_names <> '') then
  begin
    for select acc_id
        from ref_account, sys_accbalance
        where acc_from_sab_id = sab_id and
              f_check_refaccount_kind(sab_full_code, :sys_names) = :is_include and
              (:reg_id = 0 or
              sab_acc_mask not containing 'RRRR' or
              (sab_acc_mask containing 'RRRR' and
              acc_from_reg_id = :reg_id)) and
              (:find_currency = '' or
              acc_from_cu_id in (:find_currency, :local_currency)) and
              (:find_c_id is null or
              coalesce(acc_object_kind, 0) <> 0 or
              (acc_object_kind = 0 and
              acc_object_id + 0 = :find_c_id))
        rows :max_row
        into res_id
    do
    begin
      res_code = f_fmt_acc(res_id);
      res_text = f_objname('ref_account', res_id);
      suspend;
    end
  end
end
