CREATE PROCEDURE GET_ACBAL_PROFEXPENSE_LIST(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    ACC VARCHAR(20),
    ACC_NAME VARCHAR(255),
    SUM_DT DECIMAL(18,4),
    SUM_KT DECIMAL(18,4))
AS
declare variable bal decimal(18,4);
begin
  for select al_id, al_name
      from acc_list, balgroup
      where al_from_bg_id = bg_id and
            al_from_reg_id = :report_region_id and
            (bg_kind = 4 or
            bg_kind = 5)
      order by al_id
      into :acc, :acc_name
  do
  begin
    bal = f_al_bal(acc, report_date_end);

    sum_dt = 0;
    sum_kt = 0;
    if (bal >= 0) then
      sum_dt = bal;
    else
      sum_kt = -1 * bal;
    if (bal <> 0) then
      suspend;
  end
end
