CREATE PROCEDURE HP_DOCPKLGOTA_LIST(
    PK_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    CITIZEN_KIND_TEXT D_STR_CODE,
    DPG_FROM_PK_ID D_INTEGER,
    DPG_CITIZEN_KIND D_INTEGER,
    DPG_DATE_BEGIN D_DATE,
    DPG_DATE_END D_DATE,
    DPG_IS_DELETE D_BOOL)
AS
declare variable country_code d_str_10;
begin
  dpg_citizen_kind = 0;
  dpg_from_pk_id = pk_id;

  country_code = '';
  country_code = case f_get_db_id()
    when 81001 then 'РФ'
    when 41701 then 'КР'
    when 94401 then 'АЗ'
  end;

  while (dpg_citizen_kind <= 2) do
  begin
    citizen_kind_text = case dpg_citizen_kind
      when 0 then 'Гражданство ' || country_code
      when 1 then 'Страны ЕАЭС'
      when 2 then 'Другие страны'
    end;


    dpg_is_delete = 1;
    dpg_date_begin = null;
    dpg_date_end = null;

    select dpg_date_begin, dpg_date_end
    from docpklgota
    where dpg_from_pk_id = :pk_id and
          dpg_citizen_kind = :dpg_citizen_kind and
          dpg_is_delete = 0
    into dpg_date_begin, dpg_date_end;

    if (row_count <> 0) then
      dpg_is_delete = 0;

    suspend;

    dpg_citizen_kind = dpg_citizen_kind + 1;
  end
end
