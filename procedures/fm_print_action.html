CREATE PROCEDURE FM_PRINT_ACTION(
    UN_ID DECIMAL(18,0),
    ACTION_KIND VARCHAR(255),
    PARAMS BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
RETURNS (
    RES_ID INTEGER,
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable show_manager integer;
declare variable user_ip varchar(100);
declare variable json_tmp blob sub_type 1 segment size 80;
declare variable json_key varchar(255);
declare variable json_value varchar(255);
declare variable user_c_id decimal(18,0);
declare variable reg_id integer;
declare variable tmp_id varchar(255);
declare variable tmp_name varchar(255);
declare variable tmp_number integer;
declare variable def_printer varchar(255);
declare variable rep_date date;
declare variable rep_date_str varchar(20);
declare variable add_params varchar(3000);
declare variable v_delimiter_res d_str_10;
declare variable v_delimiter_tmp d_str_10;
begin
-- параметры печати
-- Action_kind='get_params'
  res_id = 1;
  res_text = '';
  v_delimiter_res = '';
  rep_date = current_date;

  for select json_key, json_value
      from fm_json_parse(:params, 0, '')
      into :json_key, :json_value
  do
  begin
-- Доп. ключи, которые можно использовать: obj_kind, obj_id, blank_id, user_id, add_params
    if (json_key = 'show_manager') then
      show_manager = json_value;
    if (json_key = 'user_ip') then
      user_ip = json_value;
    if (json_key = 'add_params') then
      add_params = json_value;
  end

  if (action_kind = 'get_params') then
  begin
    select un_from_c_id
    from username
    where un_id = :un_id
    into :user_c_id;

    if (show_manager = 1) then
    begin
-- Получаем регион пользователя
      for select json_key, json_value
          from fm_json_parse(:add_params, 0, '')
          into :json_key, :json_value
      do
        if (json_key = 'reg_id') then
          reg_id = json_value;

      if (reg_id is null) then
      begin
        select ps_from_reg_id
        from hr_pstatus
        where ps_from_c_id = :user_c_id and
              ps_date_begin <= current_date and
              ps_date_end >= current_date and
              ps_status = 1 and
              ps_is_working = 1
        into :reg_id;

        if (reg_id is null) then
        begin
          res_id = -1;
          res_text = 'Не найден действующий приказ на пользователя';
          suspend;
          exit;
        end
      end

      json_tmp = '';
      v_delimiter_tmp = '';
      for select distinct ksr_id, c_name_fio
          from region r, kontsignreg, clients, kontsignmen
          where reg_unicode = :reg_id and
                ksr_from_reg_id in (reg_unicode, r.parent) and
                ksm_id = ksr_from_ksm_id and
                ksm_from_c_id = c_id and
                ksm_is_active = 1 and
                coalesce(ksr_date_begin, '01.01.1990') <= current_date and
                coalesce(ksr_date_end, '31.12.9999') >= current_date
          into :tmp_id, :tmp_name
      do
      begin
        json_tmp = BLOB_APPEND(json_tmp, v_delimiter_tmp || '{"id":"' || tmp_id || '", "name":"' || tmp_name || '"}');
        v_delimiter_tmp = ',';
      end
      res_text = BLOB_APPEND(res_text, v_delimiter_res || '"managers":[' || json_tmp || ']');
      v_delimiter_res = ',';
      res_text = BLOB_APPEND(res_text, v_delimiter_res || '"def_manager":"' || tmp_id || '"');
    end

    json_tmp = '';
    v_delimiter_tmp = '';
    def_printer = '';
    for select rp_printer_code, rp_printer_name, rp_is_default
        from aarep_get_user_printers(:un_id, :user_ip)
        into :tmp_id, :tmp_name, :tmp_number
    do
    begin
      json_tmp = BLOB_APPEND(json_tmp, v_delimiter_tmp, '{"id":"', tmp_id, '", "name":"', tmp_name, '"}');
      v_delimiter_tmp = ',';
      if (tmp_number = 1) then
        def_printer = tmp_id;
    end
    res_text = BLOB_APPEND(res_text, v_delimiter_res || '"printers":[' || json_tmp || ']');
    v_delimiter_res = ',';
    if (def_printer <> '') then
      res_text = BLOB_APPEND(res_text, v_delimiter_res || '"def_printer":"' || def_printer || '"');

    rep_date_str = right('0' || extract(day from rep_date), 2) || '.' || right('0' || extract(month from rep_date), 2) || '.' || extract(year from rep_date);
    res_text = BLOB_APPEND(res_text, v_delimiter_res || '"def_date":"' || rep_date_str || '"');
    res_text = '{' || res_text || '}';

    suspend;
    exit;
  end

end
