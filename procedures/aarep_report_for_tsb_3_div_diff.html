CREATE PROCEDURE AAREP_REPORT_FOR_TSB_3_DIV_DIFF(
    REP_DATE_END DATE)
RETURNS (
    KC_ID ID_TYPE,
    K_KONTRACT_NO VARCHAR(20),
    C_NAME_FIO VARCHAR(100),
    K_DISB_DATE DATE,
    RES_CLIENT_IS_RESIDENT INTEGER,
    IS_SECURED INTEGER,
    OL_DATE DATE,
    IS_RESTRUCT INTEGER,
    RESTRUCT_DATE_LAST DATE,
    CLI_BALANCE CURR_TYPE,
    CLI_PERC_SALDO CURR_TYPE,
    DELINQ_DAY INTEGER,
    LK_PERCENT PERC_TYPE,
    CLI_RPPU_OS CURR_TYPE,
    CLI_RPPU_PERC CURR_TYPE,
    PO_PERCENT PERC_TYPE,
    CLI_PO_SUMM CURR_TYPE,
    TERM_IN_DAYS INTEGER,
    PERCENT_YEAR PERC_TYPE,
    PERCENT_EFFECT PERC_TYPE,
    LOAN_EMP SMALLINT,
    DELINQ_DAY_REST_DATE INTEGER,
    KC_AMOUNT DECIMAL(18,4),
    PR_NAME VARCHAR(50),
    CLI_PERCENT_SCHEDULE CURR_TYPE,
    CLI_IS_RESIDENT_ON_DISB INTEGER,
    CLI_BALANCE_ROUND CURR_TYPE,
    CLI_PERC_SALDO_ROUND CURR_TYPE,
    OKS_DATE_BEGIN DATE,
    OKS_RESTRUCT_DATE DATE,
    OKS_BALANCE CURR_TYPE,
    OKS_KONT_LAST_DATE DATE,
    OKS_LAST_SH_DATE DATE,
    OKS_BAL_PERC CURR_TYPE,
    OKS_IS_RESCHEDULE INTEGER,
    OKS_DELINQ_DATE DATE,
    OKS_DELINQ_DAYS INTEGER,
    OKS_BAL_ESP_END CURR_TYPE,
    OKS_LLR_KIND INTEGER,
    OKS_LLR_PERCENT PERC_TYPE,
    OKS_IS_KONT_DEVIDED INTEGER,
    OKS_FINAL_LLR_PERCENT PERC_TYPE,
    OKS_ECL_KIND INTEGER,
    OKS_ECL_PERCENT PERC_TYPE,
    OKS_ECL_PERIOD_KOEF PERC_TYPE,
    OKS_ECL_ACC_KIND INTEGER,
    OKS_LLR_OS CURR_TYPE,
    OKS_LLR_PERC CURR_TYPE,
    OKS_ECL_VBS CURR_TYPE,
    OKS_DIFF_ECL_LLR CURR_TYPE)
AS
begin
  for select kc_id, k_kontract_no, c_name_fio, k_disb_date, res_client_is_resident, is_secured, ol_date, is_restruct,
             restruct_date_last, cli_balance, cli_perc_saldo, delinq_day, lk_percent, cli_rppu_os, cli_rppu_perc,
             po_percent, cli_po_summ, term_in_days, percent_year, percent_effect, loan_emp, delinq_day_rest_date,
             kc_amount, pr_name, cli_percent_schedule, cli_is_resident_on_disb, cli_balance_round,
             cli_perc_saldo_round
      from aarep_report_for_tsb_3_div(:rep_date_end)
      into :kc_id, :k_kontract_no, :c_name_fio, :k_disb_date, :res_client_is_resident, :is_secured, :ol_date,
           :is_restruct, :restruct_date_last, :cli_balance, :cli_perc_saldo, :delinq_day, :lk_percent, :cli_rppu_os,
           :cli_rppu_perc, :po_percent, :cli_po_summ, :term_in_days, :percent_year, :percent_effect, :loan_emp,
           :delinq_day_rest_date, :kc_amount, :pr_name, :cli_percent_schedule, :cli_is_resident_on_disb,
           :cli_balance_round, :cli_perc_saldo_round
  do
  begin
    oks_date_begin = null;
    oks_restruct_date = null;
    oks_kont_last_date = null;
    oks_last_sh_date = null;

    if (cast(k_disb_date as date) < cast('28.02.2012' as date)) then
    begin
      cli_rppu_os = 0;
      cli_rppu_perc = 0;
    end

    oks_balance = 0;
    oks_bal_perc = 0;
    oks_llr_os = 0;
    oks_llr_perc = 0;
    oks_ecl_vbs = 0;
    oks_diff_ecl_llr = 0;

    select oks_date_begin, oks_restruct_date, oks_balance, oks_kont_last_date, oks_last_sh_date, oks_bal_perc,
           oks_is_reschedule, oks_delinq_date, oks_delinq_days, iif(oks_close_state <> 2, oks_bal_esp_end, 0),
           oks_llr_kind, oks_llr_percent, oks_is_kont_devided, oks_final_llr_percent, oks_ecl_kind, oks_ecl_percent,
           oks_ecl_period_koef, oks_ecl_acc_kind, oks_llr_os, oks_llr_perc, iif(oks_close_state = 0, oks_ecl_vbs, 0),
           iif(oks_close_state = 0, oks_diff_ecl_llr, 0)
    from kn_oksummary
    where oks_from_kc_id = :kc_id and
          oks_date_begin = :rep_date_end
    into oks_date_begin, oks_restruct_date, oks_balance, oks_kont_last_date, oks_last_sh_date, oks_bal_perc,
         oks_is_reschedule, oks_delinq_date, oks_delinq_days, oks_bal_esp_end, oks_llr_kind, oks_llr_percent,
         oks_is_kont_devided, oks_final_llr_percent, oks_ecl_kind, oks_ecl_percent, oks_ecl_period_koef,
         oks_ecl_acc_kind, oks_llr_os, oks_llr_perc, oks_ecl_vbs, oks_diff_ecl_llr;

    if ((oks_llr_os <> cli_rppu_os) or (oks_llr_perc <> cli_rppu_perc)) then
      suspend;
  end
end
