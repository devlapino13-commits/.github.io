CREATE PROCEDURE MC_LINK_CLIENT(
    MCL_ID D_ID,
    OBJECT_KIND D_STR_CODE,
    OBJECT_ID D_ID)
RETURNS (
    ERR_TEXT D_STR_255)
AS
declare variable C_ID D_ID;
declare variable BLS_ID D_INTEGER;
declare variable JSON_TEXT D_BLOB_TEXT;
declare variable ADL_ID D_ID;
declare variable UN_VIRT D_ID;
declare variable PARENT_ID D_ID;
declare variable PL_STREET_ID D_ID;
declare variable ADDRESS_ARRAY D_BLOB_TEXT;
declare variable POST D_INTEGER;
declare variable ADDRESS_STR D_BLOB_TEXT;
declare variable ADDRESS_BUILDING D_BLOB_TEXT;
declare variable ADDRESS_APARTMENT D_BLOB_TEXT;
declare variable SURNAME D_STR_255;
declare variable NAME D_STR_255;
declare variable PATRONYMIC D_STR_255;
declare variable FAMILY_STATUS D_INTEGER;
declare variable GENDER D_STR_SYM;
declare variable DATE_OF_BIRTH D_DATE;
declare variable PASSPORT_SERIES D_STR_10;
declare variable PASSPORT_NUMBER D_STR_20;
declare variable PIN D_STR_20;
declare variable ISSUED_DATE D_DATE;
declare variable PASSPORT_AUTHORITY D_STR_64;
declare variable PASSPORT_AUTHORITY_CODE D_STR_64;
declare variable EXPIRED_DATE D_DATE;
declare variable MCL_LOGIN D_STR_CODE;
declare variable VOID_STATUS D_STR_SYM;
declare variable RD_MARITAL_STATUS D_INTEGER;
begin
  -- процедура линкует

  if (object_kind = 'clients') then
  begin
    c_id = object_id;
  end
  else
  if (object_kind = 'extinfolist') then
  begin
    select eil.eil_from_bls_id, eil.eil_from_c_id
    from extinfolist eil
    where eil.eil_id = :object_id and
          eil.eil_from_eic_id = 2 and
          eil.eil_status = 2
    into bls_id, c_id;

    if (coalesce(bls_id, 0) = 0) then
    begin
      err_text = 'Информация от ГРС не получена по данному клиенту!';
      suspend;
      exit;
    end
    else
    if (coalesce(c_id, 0) = 0) then
    begin
      select blx_blob
      from hp_get_blobex_blob(:bls_id)
      into json_text;

      pin = f_json_str_get_value(:json_text, 'pin', ''); -- Персональный идентификационный номер
      if (coalesce(pin, '') <> '') then
      begin
        select c.c_id
        from clients c
        where c.c_inn = :pin
        into c_id;

        if (coalesce(c_id, 0) = 0) then
        begin
          un_virt = 1000000000000;

          select voidstatus, surname, name, patronymic, family_status, date_of_birth, passport_series, passport_number,
                 pin, issued_date, passport_authority, passport_authority_code, expired_date, gender, post, parent_id,
                 pl_street_id, address_str, address_building, address_apartment, address_array, rd_marital_status
          from hp_parse_json_grs(:json_text)
          into :void_status, :surname, :name, :patronymic, :family_status, :date_of_birth, :passport_series,
               :passport_number, :pin, :issued_date, :passport_authority, :passport_authority_code, :expired_date,
               :gender, :post, :parent_id, :pl_street_id, :address_str, :address_building, :address_apartment,
               :address_array, :rd_marital_status;

          select mcl.mcl_login
          from mc_login mcl
          where mcl.mcl_id = :mcl_id
          into mcl_login;

          if (void_status = 1) then
          begin
            err_text = 'Паспорт не действительный';
            suspend;
            exit;
          end

          execute procedure fm_add_client(f_translit_fio_grs('cyr', 'cyr', coalesce(:surname, '') || ' ' || coalesce(:name, '') || ' ' || coalesce(:patronymic, ''), 'kg'), --
              iif(:gender = 'M', 1, 0), --
              :date_of_birth, --
              null, --
              null, --
              null, --
              f_get_rd_id('Citizen.KG'), --
              iif(:passport_series containing 'AN', 8, 296), -- тип документа (8 - пластик КГ, 296 - биометрический паспорт)
              :passport_series || :passport_number, --
              :issued_date, --
              :pin, --
              1000000000000,
              :surname,
              :name,
              :patronymic,
               null,
               null,
               null)
              returning_values :c_id, :err_text;

          if (coalesce(err_text, '') <> '') then
          begin
            suspend;
            exit;
          end
          execute procedure hp_continfo_update(:c_id, 'cell', :mcl_login, 0, :un_virt);

          if (:c_id > 0) then
          begin
            if (address_array <> '' and
                address_str not containing 'ликвидирован') then
            begin
              insert into addrlist (adl_from_pl_id, adl_depart_no, adl_depart_pref, adl_flat_no, adl_building_no,
                                    adl_section_no, adl_cross_street_id, adl_address_str, adl_comment, adl_post_index,
                                    adl_is_edited, adl_from_un_id, adl_street_id, adl_is_without_street, adl_is_filled,
                                    adl_long, adl_lat, adl_geo_count, adl_place_pls_id, adl_street_pls_id,
                                    adl_lang_kind)
              values (:parent_id, 0, :address_building, :address_apartment, 0, 0, null, :address_str, null, :post, 0,
                      :un_virt, :pl_street_id, iif(:pl_street_id is null, 1, 0), 1, null, null, -1, null, null, 0)
              returning adl_id
              into :adl_id;
              update extinfolist eil
              set eil.eil_from_c_id = :c_id
              where eil.eil_from_mcl_id = :mcl_id and
                    eil.eil_status = 2;

              update grs_clients gc
              set gc.gc_from_c_id = :c_id
              where gc.gc_from_eil_id = :object_id and
                    gc.gc_from_mcl_id = :mcl_id;

              update or insert into addrclient (adc_from_c_id, adc_address_type, adc_from_adl_id, adc_from_un_id)
              values (:c_id, 1, :adl_id, :un_virt)
              matching (adc_from_c_id, adc_address_type);

              insert into grs_addrlist (ga_from_eil_id, ga_from_adl_id)
              values (:object_id, :adl_id);
            end
            -- address ----------------------------------------------------------- <
            if (address_array = '' or address_str containing 'ликвидирован') then
            begin
              update grs_clients gc
              set gc.gc_status = 3
              where gc.gc_from_c_id = :c_id and
                    gc.gc_status = 2 and
                    gc.gc_is_last = 1 and
                    gc.gc_from_eil_id = :object_id;

              update addrclient adc
              set adc.adc_from_adl_id = 41701000000000
              where adc.adc_address_type = 1 and
                    adc.adc_from_c_id = :c_id;
            end
            execute procedure hp_objectprops_set_with_rd(f_objkind('clients', 'id'), :c_id, -1,
                f_get_rd_id('AnketaQuestions.Family'), current_date, '', 0, :family_status, :un_virt);

            execute procedure hp_objectprops_set_with_rd(f_objkind('clients', 'id'), :c_id, -1,
                f_get_rd_id('AnketaQuestions.Family_tunduk'), current_date, '', 0, :rd_marital_status, :un_virt);
          end
        end
      end
    end
  end
  update mc_login ml
  set ml.mcl_from_c_id = :c_id
  where ml.mcl_id = :mcl_id;
end
