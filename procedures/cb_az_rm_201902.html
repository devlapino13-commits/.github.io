CREATE PROCEDURE CB_AZ_RM_201902(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(50),
    BONUS_LO_ALL CURR_TYPE,
    BONUS_LO_REGION CURR_TYPE,
    BALANCE_DELINQ7_BEGIN CURR_TYPE,
    BALANCE_DELINQ7_END CURR_TYPE,
    PLAN_SUM CURR_TYPE,
    FACT_KRED INTEGER,
    PLAN_KRED INTEGER,
    KPI1 CURR_TYPE,
    KPI2 CURR_TYPE,
    KPI3 CURR_TYPE,
    KPI1_FACT CURR_TYPE,
    KPI2_FACT CURR_TYPE,
    KPI3_FACT CURR_TYPE,
    BONUS_RM_BAZA CURR_TYPE,
    BONUS_RM_TOTAL CURR_TYPE)
AS
declare variable reg_id integer;
declare variable c_id id_type;
declare variable aarep_bonus_total curr_type;
declare variable rep_date_begin date;
declare variable rep_period integer;
declare variable rep_indicator integer;
declare variable rep_date_for_plan date;
begin
  rep_date_begin = rep_date_end - extract(day from rep_date_end);
  for select reg_id, c_id, aarep_bonus_total
      from cb_az_lo_201602_one_lo(1000000, 0, :rep_date_end, 0)
      into reg_id, c_id, aarep_bonus_total
  do
  begin
    insert into temp_bonusru_lo (bl_personal_id, bl_region_id, bl_bonus_total)
    values (:c_id, :reg_id, :aarep_bonus_total);
  end

--Сумма заработка всех КС Компании за месяц
  select sum(bl_bonus_total)
  from temp_bonusru_lo
  into bonus_lo_all;

--  формируем данные для план выдачи
  rep_period = 100 * extract(year from rep_date_end) + extract(month from rep_date_end);
  select rd_id
  from refdata
  where rd_code in ('planindicators.disb_amount')
  into rep_indicator;
  execute procedure hp_plandata_fill(1000000, :rep_date_end, :rep_indicator, :rep_period, 1, null);

  for select parent, sum(bl_bonus_total)
      from temp_bonusru_lo, region
      where bl_region_id = reg_unicode
      group by parent
      having sum(bl_bonus_total) <> 0
      into reg_id, bonus_lo_region
  do
  begin
    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :reg_id
    into reg_name;
--  Баланс просрочек ПАР7 на конец месяца
    balance_delinq7_end = null;
    select sum(oks_balance)
    from oksummary, region
    where :rep_date_end between oks_date_begin and oks_date_end and
          :rep_date_end - oks_delinq_date > 7 and
          oks_region_id = reg_unicode and
          oks_close_state = 0 and
          parent = :reg_id
    into balance_delinq7_end;
    if (balance_delinq7_end is null) then
      balance_delinq7_end = 0;

--  Баланс просрочек ПАР7 плановый
--  письмо от s.morozova@molbulak.kg 27.03.2019: FW: Плановый ПАР 7 для бонусов РМ
--  Оставляем как было, на 31 декабря 2018года будем брать.
    balance_delinq7_begin = null;
    rep_date_for_plan = '31.12.2018';
    select sum(oks_balance)
    from oksummary, region
    where :rep_date_for_plan between oks_date_begin and oks_date_end and
          :rep_date_for_plan - oks_delinq_date > 7 and
          oks_region_id = reg_unicode and
          oks_close_state = 0 and
          parent = :reg_id
    into balance_delinq7_begin;
    if (balance_delinq7_begin is null) then
      balance_delinq7_begin = 0;

    kpi1 = 1 - (balance_delinq7_end / 1000 - balance_delinq7_begin / 1000) * 3 / 100;
--  Фактическое кол-во выданных кредитов
    fact_kred = 0;
    select count(*)
    from kontstate, knclient, region
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_date_begin between :rep_date_begin and :rep_date_end and
          ks_region_id = reg_unicode and
          parent = :reg_id and
          ks_from_k_id = kc_from_k_id and
          kc_amount_approved > 0
    into fact_kred;
    fact_kred = coalesce(fact_kred, 0);

--  План выдачи
    plan_sum = 0;
    select sum(pd_data_1)
    from temp_plandata, region
    where pd_is_office = 2 and
          pd_data_1 > 0 and
          pd_from_reg_id = reg_unicode and
          parent = :reg_id
    into plan_sum;
    plan_sum = coalesce(plan_sum, 0);
--  чтобы вычислить план кредитов по бюджету на одного кредита заложено 1100 манат, и сумму выдачи делим на 1100
    plan_kred = cast(plan_sum as double precision) / 1100;

    kpi2 = 1 - (1 - cast(fact_kred as double precision) / plan_kred) * 3;

    kpi3 = 1;
--  KPI для расчета бонусов
    kpi1_fact = case
                  when kpi1 > 1.1 then 1.1
                  when kpi1 < 0 then 0
                  else kpi1
                end;

    kpi2_fact = case
                  when kpi2 > 1.1 then 1.1
                  when kpi2 < 0 then 0
                  else kpi2
                end;

    kpi3_fact = case
                  when kpi3 > 1 then 1.1
                  when kpi3 < 0 then 0
                  else kpi3
                end;

    bonus_rm_baza = (cast(bonus_lo_region as double precision) / bonus_lo_all) * 3600;

    bonus_rm_total = round((cast(bonus_rm_baza as double precision) * kpi1_fact * kpi2_fact * kpi3_fact), 0);
    bonus_rm_total = round(bonus_rm_total);

    suspend;
  end
end
