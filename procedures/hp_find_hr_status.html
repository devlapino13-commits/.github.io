CREATE PROCEDURE HP_FIND_HR_STATUS(
    ORDER_ID D_ID)
RETURNS (
    PS_ID D_INTEGER,
    PS_FROM_HO_ID D_ID,
    PS_DATE_BEGIN D_DATE,
    PS_DATE_END D_DATE,
    PS_PERIOD_MONTH D_INTEGER,
    PS_DATE_END_IN_ORDER D_DATE,
    PS_FROM_REG_ID D_INTEGER,
    PS_FROM_DEP_ID D_INTEGER,
    PS_FROM_JL_ID D_INTEGER,
    PS_IS_ON_STAFF D_SMALLINT,
    PS_IS_WORKING D_SMALLINT,
    PS_FROM_C_ID D_ID,
    PS_SALARY D_CURRENCY,
    PS_LOOK_PER_NAME D_STR_100,
    HM_EXP D_STR_20,
    REG_NAME D_STR_100,
    JL_NAME D_STR_100,
    DEP_NAME D_STR_100,
    PS_IS_MOUNT D_BOOL,
    PS_IS_PART D_BOOL,
    PS_IS_TIMELESS D_BOOL,
    RES_DATE_END_STR D_STR_CODE,
    RES_WORK_KIND D_STR_CODE)
AS
declare variable hm_mount d_integer;
begin
  for select ps.ps_id, ps.ps_from_ho_id, ps.ps_date_begin, ps.ps_date_end, ps.ps_period_month, ps.ps_date_end_in_order,
             ps.ps_from_reg_id, ps.ps_from_dep_id, ps.ps_from_jl_id, ps.ps_is_on_staff, ps.ps_is_working,
             ps.ps_from_c_id, ps.ps_salary, ps_is_mount, ps_is_part, ps_is_timeless,
             c.c_name_fio, hm.hm_mount,
             reg_code || ' ' || reg_name, jl_name, dep_name
      from hr_pstatus ps
      left join clients c on ps.ps_from_c_id = c.c_id
      left join hr_mount hm on hm.hm_ps_from_ho_id = ps.ps_from_ho_id and hm.hm_from_ps_id = ps.ps_id
      left join region on ps.ps_from_reg_id = reg_unicode
      left join joblist on ps.ps_from_jl_id = jl_id
      left join depart on ps.ps_from_dep_id = dep_id
      where ps_from_ho_id = :order_id
      order by ps.ps_id
      into :ps_id, :ps_from_ho_id, :ps_date_begin, :ps_date_end, :ps_period_month, :ps_date_end_in_order,
           :ps_from_reg_id, :ps_from_dep_id, :ps_from_jl_id, :ps_is_on_staff, :ps_is_working, :ps_from_c_id, :ps_salary,
           :ps_is_mount, :ps_is_part, :ps_is_timeless,
           :ps_look_per_name, :hm_mount,
           :reg_name, :jl_name, :dep_name
  do
  begin
    hm_exp = 'Не установлен';
    if (:hm_mount is not null) then
      hm_exp = (floor(:hm_mount / 12) || iif(floor(:hm_mount / 12) between 1 and 5, ' г. ', ' л. ') || mod(:hm_mount, 12) || ' м.');

    res_date_end_str = '';
    if (ps_is_timeless = 1) then
    begin
      res_date_end_str = 'бессрочный';
      ps_period_month = null;
    end
    else
      res_date_end_str = f_datetostr(ps_date_end_in_order);

    res_work_kind = 'Штатный';
    if (ps_is_part = 1) then
      res_work_kind = 'Совместитель';
    else
    if (ps_is_on_staff = 0) then
      res_work_kind = 'Внештатный';

    suspend;
  end
end
