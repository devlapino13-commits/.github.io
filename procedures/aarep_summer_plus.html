CREATE PROCEDURE AAREP_SUMMER_PLUS(
    REP_REGION_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    OFFICE D_STR_100,
    CL_FIO D_STR_100,
    CL_BALL D_INTEGER,
    K_KONTRACT_NO D_STR_20,
    DISB_DATE DATE,
    PR_NAME D_STR_100,
    K_REFUND D_STR_10,
    BALANCE_ON_DATE D_CURRENCY,
    SUM_OVER D_CURRENCY,
    SVZ_FIO D_STR_100)
AS
declare variable c_id d_id;
declare variable k_id d_id;
declare variable pr_code str_100;
declare variable is_refund d_integer;
declare variable counter d_integer;
begin
  for select knc.kc_from_c_id, sum(k_refund), count(kc_from_c_id)
      from kontstate ks
      inner join kontract k on k.k_id = ks.ks_from_k_id
      inner join knclient knc on k.k_id = knc.kc_from_k_id
      inner join product pr on pr.pr_id = k.k_from_pr_id
      inner join region_list(:rep_region_id, 1, -1, 1) on id_list = ks.ks_region_id
      where ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            :rep_date between ks.ks_date_begin and ks.ks_date_end and
            knc.kc_amount_approved > 0 and
            pr_code containing '.comfortable.' /*and
            k_refund = 1                         */
      group by 1
      into :c_id, is_refund, counter
  do
  begin
    cl_fio = null;
    select cl.c_name_fio
    from clients cl
    where cl.c_id = :c_id
    into cl_fio;
    balance_on_date = 0;

    for select k.k_kontract_no, r.reg_name, k.k_fact_disb_date, pr.pr_name, iif(k.k_refund = 1, 'ДА', 'НЕТ'),
               cl.c_name_fio, k.k_id, f_point(:c_id, k_fact_disb_date), pr_code
        from knclient knc
        inner join kontract k on k.k_id = knc.kc_from_k_id
        inner join product pr on pr.pr_id = k.k_from_pr_id
        inner join kontstate ks on ks.ks_from_k_id = knc.kc_from_k_id
        inner join region r on r.reg_unicode = k.k_last_reg_id
        inner join clients cl on cl.c_id = k.k_fact_disb_lo_id
        where knc.kc_from_c_id = :c_id and
              ks.ks_ol_is_delete = 0 and
              ks.ks_state = 1 and
              :rep_date between ks.ks_date_begin and ks.ks_date_end
        order by c_id, k_fact_disb_date
        into k_kontract_no, office, disb_date, pr_name, k_refund, svz_fio, k_id, cl_ball, pr_code
    do
    begin
      balance_on_date = balance_on_date + (select sum(zad_os)
                                           from aarep_make_zad_list(:k_id, :rep_date, 1)
                                           where sh_date is not null);

      if (balance_on_date is null) then
        balance_on_date = 0;

      sum_over = 0;
      if (pr_code containing '.comfortable.') then
      begin
        sum_over = 0;
        if (balance_on_date > 100000 and
            cl_ball < 7) then
        begin
          sum_over = balance_on_date - 100000;
          suspend;
          continue;
        end
        if (balance_on_date > 160000 and
            cl_ball < 12) then
        begin
          sum_over = balance_on_date - 160000;
          suspend;
          continue;
        end
        if (balance_on_date > 200000 and
            cl_ball >= 12) then
        begin
          sum_over = balance_on_date - 200000;
          suspend;
          continue;
        end
      end
    end
    if (counter > 1) then
    begin
      balance_on_date = 0;

      for select k.k_kontract_no, r.reg_name, k.k_fact_disb_date, pr.pr_name, iif(k.k_refund = 1, 'ДА', 'НЕТ'),
                 cl.c_name_fio, k.k_id, f_point(:c_id, k_fact_disb_date), pr_code
          from knclient knc
          inner join kontract k on k.k_id = knc.kc_from_k_id
          inner join product pr on pr.pr_id = k.k_from_pr_id
          inner join kontstate ks on ks.ks_from_k_id = knc.kc_from_k_id
          inner join region r on r.reg_unicode = k.k_last_reg_id
          inner join clients cl on cl.c_id = k.k_fact_disb_lo_id
          where knc.kc_from_c_id = :c_id and
                ks.ks_ol_is_delete = 0 and
                ks.ks_state = 1 and
                :rep_date between ks.ks_date_begin and ks.ks_date_end
          order by c_id, k_fact_disb_date
          into k_kontract_no, office, disb_date, pr_name, k_refund, svz_fio, k_id, cl_ball, pr_code
      do
      begin
        balance_on_date = balance_on_date + (select sum(zad_os)
                                             from aarep_make_zad_list(:k_id, current_date, 1)
                                             where sh_date is not null);

        if (balance_on_date is null) then
          balance_on_date = 0;
        suspend;
      end
    end

  end
end
