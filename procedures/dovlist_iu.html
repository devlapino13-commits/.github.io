CREATE PROCEDURE DOVLIST_IU(
    DL_ID INTEGER,
    DL_DATE_BEGIN DATE,
    DL_DATE_END DATE,
    DL_DATE_END_FACT DATE,
    DL_FROM_DC_ID INTEGER,
    DL_FROM_BLN_ID INTEGER,
    DL_DOV_NO_1 INTEGER,
    DL_DOV_NO_2 INTEGER,
    DL_FROM_C_ID DECIMAL(18,0),
    DL_MANAGER_ID DECIMAL(18,0),
    DL_FROM_REG_ID INTEGER,
    DL_MAIN_TEXT VARCHAR(255),
    DL_CITY VARCHAR(255),
    DL_MEMO_1 VARCHAR(255),
    DL_MEMO_2 VARCHAR(255),
    DL_MEMO_3 VARCHAR(255),
    DL_MEMO_4 VARCHAR(255),
    DL_MEMO_5 VARCHAR(255),
    DL_MEMO_6 VARCHAR(255),
    DL_MEMO_7 VARCHAR(255),
    DL_MEMO_8 VARCHAR(255),
    DL_MEMO_9 VARCHAR(255),
    DL_MEMO_10 VARCHAR(255),
    DL_STATUS INTEGER,
    DL_BLOB BLOB SUB_TYPE BINARY SEGMENT SIZE 80,
    DL_FROM_UN_ID DECIMAL(18,0),
    DL_DATETIME TIMESTAMP,
    DL_BLOB_NAME VARCHAR(255))
AS
begin
  if (exists(select dl_id
             from dovlist
             where (dl_id = :dl_id))) then
    update dovlist
    set dl_date_begin = :dl_date_begin,
        dl_date_end = :dl_date_end,
        dl_date_end_fact = :dl_date_end_fact,
        dl_from_dc_id = :dl_from_dc_id,
        dl_from_bln_id = :dl_from_bln_id,
        dl_dov_no_1 = :dl_dov_no_1,
        dl_dov_no_2 = :dl_dov_no_2,
        dl_from_c_id = :dl_from_c_id,
        dl_manager_id = :dl_manager_id,
        dl_from_reg_id = :dl_from_reg_id,
        dl_main_text = :dl_main_text,
        dl_city = :dl_city,
        dl_memo_1 = :dl_memo_1,
        dl_memo_2 = :dl_memo_2,
        dl_memo_3 = :dl_memo_3,
        dl_memo_4 = :dl_memo_4,
        dl_memo_5 = :dl_memo_5,
        dl_memo_6 = :dl_memo_6,
        dl_memo_7 = :dl_memo_7,
        dl_memo_8 = :dl_memo_8,
        dl_memo_9 = :dl_memo_9,
        dl_memo_10 = :dl_memo_10,
        dl_status = :dl_status,
        dl_blob = :dl_blob,
        dl_from_un_id = :dl_from_un_id,
        dl_datetime = :dl_datetime,
        dl_blob_name = :dl_blob_name
    where (dl_id = :dl_id);
  else
    insert into dovlist (dl_id, dl_date_begin, dl_date_end, dl_date_end_fact, dl_from_dc_id, dl_from_bln_id,
                         dl_dov_no_1, dl_dov_no_2, dl_from_c_id, dl_manager_id, dl_from_reg_id, dl_main_text, dl_city,
                         dl_memo_1, dl_memo_2, dl_memo_3, dl_memo_4, dl_memo_5, dl_memo_6, dl_memo_7, dl_memo_8,
                         dl_memo_9, dl_memo_10, dl_status, dl_blob, dl_from_un_id, dl_datetime, dl_blob_name)
    values (:dl_id, :dl_date_begin, :dl_date_end, :dl_date_end_fact, :dl_from_dc_id, :dl_from_bln_id, :dl_dov_no_1,
            :dl_dov_no_2, :dl_from_c_id, :dl_manager_id, :dl_from_reg_id, :dl_main_text, :dl_city, :dl_memo_1,
            :dl_memo_2, :dl_memo_3, :dl_memo_4, :dl_memo_5, :dl_memo_6, :dl_memo_7, :dl_memo_8, :dl_memo_9, :dl_memo_10,
            :dl_status, :dl_blob, :dl_from_un_id, :dl_datetime, :dl_blob_name);
end
