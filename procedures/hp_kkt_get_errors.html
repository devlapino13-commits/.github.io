CREATE PROCEDURE HP_KKT_GET_ERRORS
RETURNS (
    KKM_ID D_ID,
    OL_ID D_ID,
    K_ID D_ID,
    KKT_KIND D_INTEGER,
    STATUS D_STR_CODE,
    KKM_CREATE_DATETIME D_DATETIME,
    KKM_SEND_DATETIME D_DATETIME,
    K_KONTRACT_NO D_STR_CODE,
    AMOUNT D_CURRENCY,
    OL_MEMO D_STR_LONG,
    OL_DATE D_DATE,
    ERROR_TEXT D_STR_8100)
AS
declare variable kkm_status d_integer;
declare variable error_cnt d_integer;
declare variable kke_error_text d_str_long;
begin
  for select kkm_id, kkm_from_ol_id, kkm_from_k_id, kkm_kkt_kind, kkm_status, kkm_create_datetime, kkm_send_datetime, k_kontract_no
  from kkt_main kkm, kontract
  where k_id = kkm_from_k_id
    and kkm_status < 0
    and kkm_create_datetime < dateadd(minute, -20, current_timestamp)
  into kkm_id, ol_id, k_id, kkt_kind, kkm_status, kkm_create_datetime, kkm_send_datetime, k_kontract_no
  do
  begin
    status = case kkm_status
                when -2 then 'Ошибка'
                when -1 then 'Ошибка. Будет отправлен еще раз'
                when 0 then 'Новый'
                when 1 then 'Отправлен'
                when 2 then 'Фискализирован'
                else 'Неопределен'
             end;
  
    amount = 0;
    ol_memo = '';
    ol_date = null;
    error_text = '';
  
    select sum(kki_amount)
    from kkt_items
    where kki_from_kkm_id = :kkm_id
    into amount;
  
    select ol_memo, ol_date
    from operlist
    where ol_id = :ol_id
    into ol_memo, ol_date;
  
    for select f_memo_text(kke_error_id), count(*)
    from kkt_error
    where kke_from_kkm_id = :kkm_id
    group by 1
    into kke_error_text, error_cnt
    do
      error_text = error_text || iif(error_text = '', '', ', ') || f_format('%1 (%2)', kke_error_text, error_cnt);
  
    suspend;
  end
end
