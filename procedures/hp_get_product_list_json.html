CREATE PROCEDURE HP_GET_PRODUCT_LIST_JSON(
    LANG D_STR_20,
    MCL_PROJECT D_STR_CODE = 'kmp')
RETURNS (
    PRODUCT_LIST_JSON D_BLOB_TEXT)
AS
declare variable product_json d_blob_text; /* JSON одного конкретного продукта, аппендится к :product_list_json */
declare variable loan_min d_integer;
declare variable loan_max d_integer;
declare variable period_min d_integer;
declare variable period_max d_integer;
declare variable mcp_from_pr_id d_integer;
declare variable mcp_title d_str_255;
declare variable mcp_text_1 d_str_255;
declare variable mcp_text_2 d_str_255;
declare variable mcp_text_3 d_str_255;
declare variable mcp_text_4 d_str_255;
declare variable mcp_additional_info d_str_255;
declare variable mcp_loan_purpose d_str_255;
declare variable mcp_amount_step d_integer;
declare variable mcp_picture_url d_str_long;
declare variable mcp_small_picture_url d_str_long;
declare variable pr_code d_str_255;
declare variable is_extra_product d_boolean; /* Определяет, является ли продукт исключением (RFCKG-5024) */
begin
    for select mcp_from_pr_id, mcp_title, mcp_text_1, mcp_text_2, mcp_text_3, mcp_text_4,
           mcp_additional_info, mcp_loan_purpose, mcp_amount_step, mcp_picture_url, mcp_small_picture_url
    from mc_products
    where
        mcp_is_active = 1 and
        mcp_lang containing '.' || :lang || '.'
    order by mcp_serial_number
    into :mcp_from_pr_id, :mcp_title, :mcp_text_1, :mcp_text_2, :mcp_text_3, :mcp_text_4,
         :mcp_additional_info, :mcp_loan_purpose, :mcp_amount_step, :mcp_picture_url, :mcp_small_picture_url
    do
    begin
        pr_code = (select pr_code from product where pr_id = :mcp_from_pr_id);
    
        -- RFCRU-3766
        if (mcl_project = 'kmp' and pr_code <> '' and pr_code containing '.oivo_pack.') then
          continue;
    
        if (mcl_project = 'oivo' and pr_code <> '' and pr_code not containing '.oivo_pack.') then
          continue;

        is_extra_product = exists (
            select 1
            from mc_products_extra mpe
            where mpe.mpe_from_pr_id = :mcp_from_pr_id
        );

        select first 1
               prl.prl_amount_cl_min,
               prl.prl_amount_cl_max,
               prl.prl_repay_min,
               prl.prl_repay_max
        from prodlogic prl
        where prl.prl_from_pr_id = :mcp_from_pr_id
          and (
                :is_extra_product -- Для продуктов-исключений запись не обязательно должна быть актуальной
                or current_date between prl.prl_date_begin and prl.prl_date_end -- Для обычных продуктов запись должна быть актуальной
              )
        order by
            case
                when :is_extra_product then prl.prl_date_begin -- Для продуктов-исключений берем последнюю из имеющихся записей в prodlogic
                else prl.prl_date_end
            end desc
        into :loan_min, :loan_max, :period_min, :period_max;

        -- Если какой-то из этих параметров не указан, значит продукт не настроен, либо настроен некорректно
        if (loan_min is null or loan_max is null or period_min is null or period_max is null) then
            continue; -- Пропускаем данную запись

        product_json = BLOB_APPEND(product_json, iif(product_list_json is null, '{', ',{'));
        product_json = BLOB_APPEND(product_json,
                                   f_json_make_pair('pr_id', coalesce(mcp_from_pr_id, ''), 'str', 1) ||
                                   f_json_make_pair('pr_name', coalesce(mcp_title, ''), 'str', 1) ||
                                   f_json_make_pair('text_1', coalesce(mcp_text_1, ''), 'str', 1) ||
                                   f_json_make_pair('text_2', coalesce(mcp_text_2, ''), 'str', 1) ||
                                   f_json_make_pair('text_3', coalesce(mcp_text_3, ''), 'str', 1) ||
                                   f_json_make_pair('text_4', coalesce(mcp_text_4, ''), 'str', 1) ||
                                   f_json_make_pair('additional_info', coalesce(mcp_additional_info, ''), 'str', 1) ||
                                   f_json_make_pair('loan_purpose', coalesce(mcp_loan_purpose, ''), 'str', 1) ||
                                   f_json_make_pair('loan_min', coalesce(loan_min, ''), 'str', 1) ||
                                   f_json_make_pair('loan_max', coalesce(loan_max, ''), 'str', 1) ||
                                   f_json_make_pair('period_min', coalesce(period_min, ''), 'str', 1) ||
                                   f_json_make_pair('period_max', coalesce(period_max, ''), 'str', 1) ||
                                   f_json_make_pair('amount_step', coalesce(mcp_amount_step, ''), 'str', 1) ||
                                   f_json_make_pair('picture_url', coalesce(mcp_picture_url, ''), 'str', 1) ||
                                   f_json_make_pair('small_picture_url', coalesce(mcp_small_picture_url, ''), 'str', 0)
                                  );
        product_json = BLOB_APPEND(product_json, '}');

        -- Аппендим продукт к общему списку
        product_list_json = BLOB_APPEND(product_list_json, product_json);

        -- Обнуляем переменные
        product_json = '';
        loan_min = null;
        loan_max = null;
        period_min = null;
        period_max = null;
    end

    product_list_json = f_format('"product_list": [%1]', coalesce(:product_list_json, ''));

    suspend;
end
