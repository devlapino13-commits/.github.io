CREATE PROCEDURE AAREP_LO_PERSONAL_RATING_TMP(
    REP_C_ID DECIMAL(18,0))
RETURNS (
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
AS
declare variable dd1 date;
declare variable dd_end_prior date;
declare variable jk_system_code_prior varchar(255);
declare variable reg_id_prior integer;
declare variable reg_id integer;
declare variable jk_system_code varchar(255);
declare variable rep_date_end2 date;
begin
  dd_end_prior = null;
  reg_id_prior = 0;
  jk_system_code_prior = '';
  for select ps_from_reg_id, ps_date_begin, ps_date_end, jk_system_code
      from hr_pstatus, joblist, jobkind
      where ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_code in ('JobSystem.Region_KM', 'JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner', 'JobSystem.Specialist_Expert') and
            ps_from_c_id = :rep_c_id
      order by ps_from_reg_id, jk_system_code, ps_date_begin
      into reg_id, dd1, rep_date_end, jk_system_code
  do
  begin
    if (rep_date_begin is null) then
      rep_date_begin = dd1;
    if (reg_id <> reg_id_prior or jk_system_code <> jk_system_code_prior) then
      rep_date_begin = dd1;
    suspend;
    reg_id_prior = reg_id;
    jk_system_code_prior = jk_system_code;
  end

end
