CREATE PROCEDURE UXP$TRANSMAIN_LIST(
    LANG_CODE D_STR_LANG,
    LIST_KIND D_SMALLINT = 0,
    ONLY_IS_DELPHI D_BOOL = 0)
RETURNS (
    TRM_ID D_ID,
    TRM_ORIG_TEXT D_STR_255,
    TRM_IS_DELPHI D_BOOL,
    TRM_DONT_TRANSLATE D_BOOL,
    TRL_LANG_CODE D_STR_LANG,
    TRL_TRANSLATE_TEXT D_STR_255,
    TRL_ACTUAL_TEXT D_STR_255,
    TRL_CHECKED D_BOOL,
    TRM_DATETIME D_DATETIME,
    IS_TRANSLATED D_BOOL,
    LANG_ID D_INTEGER)
AS
begin
  if (coalesce(lang_code, '') <> '') then
    lang_code = uxf$enum_code('user_language', lang_code);

  if (trim(coalesce(lang_code, '')) = '') then
    lang_code = uxf$userlang();

  if (list_kind is null) then
    list_kind = 0;

  if (only_is_delphi is null) then
    only_is_delphi = 0;

  if (lang_code = 'ru') then
  begin
    if (list_kind = 0) then
      exit;

    if (list_kind = -1) then
    begin
      is_translated = 0;
      for select trm_id, trm_orig_text, trm_is_delphi, trm_dont_translate,
                 el_id, el_code, trl_translate_text, trl_actual_text, trl_checked, trm_datetime
      from ux$transmain
           join enum_list on (el_kind = 'user_language' and el_id <> 0)
           left join ux$translate on (trm_id = trl_from_trm_id and trl_lang_code = el_code)
      where trm_is_delphi >= :only_is_delphi and
            trl_from_trm_id is null
      order by trm_orig_text, el_id
      into trm_id, trm_orig_text, trm_is_delphi, trm_dont_translate,
           lang_id, trl_lang_code, trl_translate_text, trl_actual_text, trl_checked, trm_datetime
      do
        suspend;
    end else
    if (list_kind = 1) then
    begin
      is_translated = 1;
      for select trm_id, trm_orig_text, trm_is_delphi, trm_dont_translate,
                 el_id, trl_lang_code, trl_translate_text, trl_actual_text, trl_checked, trm_datetime
      from ux$transmain
           join enum_list on (el_kind = 'user_language' and el_id <> 0)
           join ux$translate on (trm_id = trl_from_trm_id and trl_lang_code = el_code)
      where trm_is_delphi >= :only_is_delphi
      order by trm_orig_text, el_id
      into trm_id, trm_orig_text, trm_is_delphi, trm_dont_translate,
           lang_id, trl_lang_code, trl_translate_text, trl_actual_text, trl_checked, trm_datetime
      do
        suspend;
    end

    exit;
  end

  trl_lang_code = lang_code;
  lang_id = uxf$enum_id('user_language', lang_code);
  for select trm_id, trm_orig_text, trm_is_delphi, trm_dont_translate,
             trl_translate_text, trl_actual_text, trl_checked, trm_datetime
  from ux$transmain
       left join ux$translate on (trm_id = trl_from_trm_id and trl_lang_code = :lang_code)
  where trm_is_delphi >= :only_is_delphi
  order by trm_orig_text
  into trm_id, trm_orig_text, trm_is_delphi, trm_dont_translate,
       trl_translate_text, trl_actual_text, trl_checked, trm_datetime
  do
  begin
    is_translated = 0;
    if (coalesce(trl_actual_text, '') <> '' or coalesce(trl_translate_text, '') <> '') then
      is_translated = 1;

    if (coalesce(trl_actual_text, '') = trm_orig_text and coalesce(trl_translate_text, '') = '') then
      is_translated = 0;

    if (list_kind = 1 and is_translated = 0) then
      Continue;

    if (list_kind = -1 and is_translated = 1) then
      Continue;

    suspend;
  end
end
