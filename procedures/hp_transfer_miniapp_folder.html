CREATE PROCEDURE HP_TRANSFER_MINIAPP_FOLDER(
    KF_ID DECIMAL(18,0),
    LO_ID DECIMAL(18,0))
RETURNS (
    RES_TEXT VARCHAR(255),
    KF_NEW_NAME VARCHAR(100))
AS
declare variable reg_id integer;
declare variable kal_from_ka_id decimal(18,0);
declare variable kal_number integer;
declare variable kal_new_no integer;
declare variable kol integer;
declare variable ke_code varchar(100);
declare variable kal_date d_date;
begin
  res_text = '';
  kf_new_name = '';

  if (exists(select *
             from kontract, kontstate
             where ks_from_k_id = k_id and
                   k_from_kf_id = :kf_id and
                   ks_state in (1, 2) and
                   ks_ol_is_delete = 0)) then
  begin
    res_text = 'В кредитной папке имеются договора. Прием-передача возможно через другое окно.';
    suspend;
    exit;
  end

  select kal_from_ka_id, kal_number, kal_new_reg_id, kal_date
  from knaktlist
  where kal_from_kf_id = :kf_id and
        kal_date_end >= '31.12.2999' and
        knaktlist.is_delete = 0
  into kal_from_ka_id, kal_number, reg_id, kal_date;

  if (reg_id is null) then
  begin
    res_text = 'Не определен офис кредитной папки';
    suspend;
    exit;
  end

  kol = 0;
  for select ke_code
      from krexpert
      where ke_from_p_id = :lo_id and
            ke_from_reg_id = :reg_id and
            ke_status = 1 and
            ke_date_end >= '31.12.2999' and
            ke_date <= current_date
      into :ke_code
  do
  begin
    kol = kol + 1;
  end

  if (kol <> 1) then
  begin
    if (kol = 0) then
      res_text = 'У выбранного сотрудника отсутствует код СВЗ в этом офисе';
    else
      res_text = 'У выбранного сотрудника имеется ' || :kol || ' кода СВЗ в этом офисе. Сначала наведите порядок в справочнике СВЗ';
    suspend;
    exit;
  end

  select kal_new_no
  from knaktlist
  where kal_new_officer_code = :ke_code and
        kal_new_reg_id = :reg_id and
        kal_new_kf_officer = :lo_id and
        knaktlist.is_delete = 0
  order by kal_new_officer_code desc, kal_new_no desc
  rows 1
  into :kal_new_no;
  if (kal_new_no is null) then
    kal_new_no = 0;
  kal_new_no = kal_new_no + 1;

  kf_new_name = ke_code || '-' || kal_new_no;

  update knaktlist
  set kal_new_kf_officer = :lo_id,
      kal_new_kf_name = :kf_new_name,
      kal_new_no = :kal_new_no,
      kal_new_officer_code = :ke_code
  where (kal_from_ka_id = :kal_from_ka_id) and
        (kal_number = :kal_number) and
        (kal_from_kf_id = :kf_id);

  execute procedure knaktlist_update(:kf_id, kal_date);
  suspend;
end
