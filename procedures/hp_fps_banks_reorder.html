CREATE PROCEDURE HP_FPS_BANKS_REORDER
AS
declare variable fpb_id d_integer;
declare variable r_no d_integer;
begin
  -- Процедура обновляет позицию банка СБП в соответствии со статистикой.
  -- Берутся банки где более 10 выдач, уходят в топ в соответствии с частотой с которой встречаются
  -- Остальные сортируются по алфавиту
  r_no = 0;
  for with
  tab_top as
  (select fcb_from_fpb_id as fpb_id, count(*) * 1000 as r_order
    from fps_client_banks
    group by 1
    having count(*) >= 10
    order by count(*) desc
  ),
  tab_rest as (
    select fpb_id, row_number() over (order by fpb_name desc) as r_order
    from fps_banks
    where fpb_id not in (select fpb_id from tab_top)
  ),
  tab_all as (
    select fpb_id, r_order
    from tab_top
    union
    select fpb_id, r_order
    from tab_rest
  )
  select fpb_id
  from tab_all
  order by r_order desc
  into fpb_id
  do
  begin
    update fps_banks
    set fpb_order_no = :r_no
    where fpb_id = :fpb_id
      and fpb_order_no <> :r_no;
    r_no = r_no + 1;
  end
end
