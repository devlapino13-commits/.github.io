CREATE PROCEDURE HP_EDOC_BLANK_SET_BLOB(
    EDB_ID D_ID,
    EDB_BLOB D_BLOB_BINARY,
    EBH_HASH D_STR_64,
    EDB_PAGE_COUNT D_SMALLINT = 0)
AS
declare variable edl_status d_smallint;
declare variable edf_print_name d_str_255;
declare variable bls_id d_id;
declare variable edb_need_approve d_smallint;
declare variable edc_msg_kind d_smallint;
begin
  select edb_need_approve, edl_status, edf_print_name, edc_msg_kind
  from edoc_blank, edoc_list, edoc_frx, edoc_cfg
  where edb_id = :edb_id and
        edb_from_edl_id = edl_id and
        edb_from_edf_id = edf_id and
        edl_from_edc_id = edc_id
  into edb_need_approve, edl_status, edf_print_name, edc_msg_kind;

  if (edb_blob is not null) then
  begin
    if (edl_status = 1 and edb_need_approve = 0) then
    begin
      -- Если производится попытка переподготовить бланк документа,
      -- который подписывается через sima.az, ломаемся, это запрещено,
      -- т.к. это уничтожит подписи.
      if (edc_msg_kind = 7) then
        exception e_default 'Re-preparing of signed blanks is not allowed!';

      -- Бланк готов и документ подписан: переносим в blobstore, обновляем статус, hash
      execute procedure hp_insert_into_blobstore ('edoc_' || :edb_id || '.pdf', :edf_print_name, :edb_blob,1000000000000, 4,
              3)
      returning_values :bls_id;

      update edoc_blank
      set edb_from_bls_id = :bls_id,
          edb_blob = null,
          edb_status = 1,
          edb_page_count = :edb_page_count
      where edb_id = :edb_id;

      if (coalesce(ebh_hash, '') <> '') then
      begin
        update or insert into edoc_blank_hash (ebh_from_edb_id, ebh_hash)
        values (:edb_id, :ebh_hash)
        matching (ebh_from_edb_id);
      end
    end
    else
    begin
      -- Бланк готов, документ не подписан/не утвержден: сохраняем блоб, обновляем статус
      update edoc_blank
      set edb_blob = :edb_blob,
          edb_status = 1,
          edb_page_count = :edb_page_count
      where edb_id = :edb_id;
    end
  end
  else
  begin
    -- Бланк не готов: обновляем статус, блоб не трогаем, чтобы не потерять
    update edoc_blank
    set edb_status = -1,
        edb_page_count = -1
    where edb_id = :edb_id;
  end
end
