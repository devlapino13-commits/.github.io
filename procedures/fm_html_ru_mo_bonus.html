CREATE PROCEDURE FM_HTML_RU_MO_BONUS(
    UN_ID DECIMAL(18,0),
    REP_DATE_END DATE)
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable color_black varchar(100);
declare variable color_yellow varchar(100);
declare variable color_green varchar(100);
declare variable color_red varchar(100);
declare variable lo_id decimal(18,0);
declare variable s_text varchar(100);
declare variable res_text_tmp blob sub_type 1 segment size 80;
declare variable line_str varchar(255);
declare variable cm_database_id integer;
declare variable k_id decimal(18,0);
declare variable aarep_bonus_minus decimal(18,4) = 0;
declare variable koef_penalty decimal(18,4) = 0;
declare variable kf_name varchar(100);
declare variable c_name_fio varchar(100);
declare variable cell_no varchar(30);
begin
  res_text = (select res_style
              from fm_html_styles) || '<body>';
  color_black = 'black';
  color_yellow = '#F08205';
  color_green = '#02B316';
  color_red = 'red';

  /*  select cm_database_id from company
  into :cm_database_id;   */
  :cm_database_id = f_get_db_id();

  if (cm_database_id = 81001) then -- для РФ
  begin
    res_text = BLOB_APPEND(res_text, '<h1 class="green_text">Расчет бонуса МО</h1>');

    select un_from_c_id
    from username
    where un_id = :un_id
    into :lo_id;

    if (lo_id is null) then
    begin
      line_str = 'Пользователь не определен';
      line_str = '<font color="' || color_red || '">' || :line_str || '</font>';
      res_text = BLOB_APPEND(res_text, line_str || '<br>');
      suspend;
      exit;
    end
    line_str = 'Дата: <b>' || iif(rep_date_end is not null, right('0' || extract(day from rep_date_end), 2) || '.' || right('0' || extract(month from rep_date_end), 2) || '.' || extract(year from rep_date_end), 'Не указан') || '</b>';
    line_str = '<font color="' || color_black || '">' || :line_str || '</font>';
    res_text = BLOB_APPEND(res_text, line_str || '<br>');

    s_text = '';
    select reg_code || ' ' || reg_name
    from region, hr_pstatus, joblist, jobkind
    where reg_unicode = ps_from_reg_id and
          ps_from_c_id = :lo_id and
          ps_date_begin <= :rep_date_end and
          ps_date_end >= :rep_date_end and
          ps_is_working = 1 and
          ps_status = 1 and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id and
          jk_system_code in ('JobSystem.Region_KM', 'JobSystem.LO_Senior')
    into :s_text;
    if (s_text is null) then
      s_text = '';
    if (s_text = '') then
    begin
      line_str = 'У вас нет действующего приказа МО.';
      line_str = '<font color="' || color_red || '">' || :line_str || '</font>';
      res_text = BLOB_APPEND(res_text, line_str || '<br>');
      suspend;
      exit;
    end

    line_str = 'Офис: <b>' || s_text || '</b>';
    line_str = '<font color="' || color_black || '">' || :line_str || '</font>';
    res_text = BLOB_APPEND(res_text, line_str || '<br>');

    select c_name_fio
    from clients
    where c_id = :lo_id
    into :s_text;

    line_str = s_text;
    line_str = '<font color="' || color_black || '">' || :line_str || '</font>';
    res_text = BLOB_APPEND(res_text, line_str || '<br><br>');

    res_text = BLOB_APPEND(res_text, '<table width="100%">
         <col width="70%">
         <col width="30%">
         <tr>
           <th>Показатели</th>
           <th>Значения</th>
         </tr>');

    if (:rep_date_end < cast('01.08.2016' as date)) then
    begin
      select '<tr><td class="filled" align="left"><b>Бонус за последний день</b></td>' || '<td class="filled" align="right"> ' || iif(aarep_day_bonus_total > 0, '<font color="#02B316">', '<font color="red">') || '<b>' || cast(cast(aarep_day_bonus_total as decimal(15,0)) as varchar(100)) || ' </font>р.</b></td></tr>' || '<tr><td align="left">&nbsp;&nbsp;Доход за погашения</td>
              <td align="right"><b>' || cast(cast(aarep_day_bonus_plus as decimal(15,0)) as varchar(100)) || ' р.</b></td></tr>' || '<tr><td align="left">&nbsp;&nbsp;Штраф за непогашения</td>
              <td align="right"><b>' || cast(cast(aarep_day_bonus_minus as decimal(15,0)) as varchar(100)) || ' р.</b></td></tr>' || '<tr><td align="left">&nbsp;&nbsp;Коэффициент возвратности</td>
              <td align="right"><b>' || cast(cast(aarep_day_returnment_koef * 100.00 as decimal(15,2)) as varchar(100)) || ' %</b></td></tr>' ||

             '<tr><td class="filled" align="left"><b>Бонус с начала месяца</b></td>' || '<td class="filled" align="right"> ' || iif(aarep_bonus_total_fact > 0, '<font color="#02B316">', '<font color="red">') || '<b>' || cast(cast(aarep_bonus_total_fact as decimal(15,0)) as varchar(100)) || ' </font>р.</b></td></tr>' || '<tr><td align="left">&nbsp;&nbsp;Доход за погашения</td>
              <td align="right"><b>' || cast(cast(aarep_bonus_plus as decimal(15,0)) as varchar(100)) || ' р.</b></td></tr>' || '<tr><td align="left">&nbsp;&nbsp;Штраф за непогашения</td>
              <td align="right"><b>' || cast(cast(aarep_bonus_minus as decimal(15,0)) as varchar(100)) || ' р.</b></td></tr>' || '<tr><td align="left">&nbsp;&nbsp;Коэффициент возвратности</td>
              <td align="right"><b>' || cast(cast(aarep_returnment_koef * 100.00 as decimal(15,2)) as varchar(100)) || ' %</b></td></tr>' ||

             '<tr><td align="left">&nbsp;&nbsp;Базовый заработок</td>
              <td align="right"><b>' || cast(cast(aarep_bonus_total as decimal(15,0)) as varchar(100)) || ' р.</b></td></tr>' || '<tr><td align="left">&nbsp;&nbsp;Количество СВЗ для расчета</td>
              <td align="right"><b>' || cast(cast(aarep_lo_count as decimal(15,0)) as varchar(100)) || '</b></td></tr>' || '<tr><td align="left">&nbsp;&nbsp;Количество клиентов этих СВЗ</td>
              <td align="right"><b>' || cast(cast(aarep_client_count as decimal(15,0)) as varchar(100)) || '</b></td></tr>' ||

             iif(aarep_lo_count <> 0, '<tr><td align="left">&nbsp;&nbsp;Количество клиентов на 1 СВЗ</td>
              <td align="right"><b>' || cast(cast(aarep_client_count as float) / cast(aarep_lo_count as float) as decimal(15,1)) || '</b></td></tr>', '') ||

             '<tr><td align="left">&nbsp;&nbsp;Коэф. за клиентов 1 СВЗ</td>
              <td align="right"><b>' || cast(cast(aarep_lo_koef as decimal(15,1)) as varchar(100)) || ' %</b></td></tr>' || '<tr><td class="filled" align="left">&nbsp;&nbsp;<b>При 100% возвратности, Ваш бонус за месяц может составить</b></td>
              <td class="filled" align="right"><b>' || cast(aarep_bonus_ideal / 100.0 * aarep_lo_koef as decimal(15,0)) || ' р.</b></td></tr>'
      from cb_ru_ssvz_201604_one(1000000, :lo_id, :rep_date_end, 1)
      into :res_text_tmp;
    end
    else
    begin
      select '<tr><td class="filled" align="left"><b>Бонус за последний день</b></td>' || '<td class="filled" align="right"> ' || iif(aarep_day_bonus_total > 0, '<font color="#02B316">', '<font color="red">') || '<b>' || cast(cast(aarep_day_bonus_total as decimal(15,0)) as varchar(100)) || ' </font>р.</b></td></tr>' || '<tr><td align="left">&nbsp;&nbsp;Доход за погашения</td>
              <td align="right"><b>' || cast(cast(aarep_day_bonus_plus as decimal(15,0)) as varchar(100)) || ' р.</b></td></tr>' || '<tr><td align="left">&nbsp;&nbsp;Штраф за непогашения</td>
              <td align="right"><b>' || cast(cast(aarep_day_bonus_minus as decimal(15,0)) as varchar(100)) || ' р.</b></td></tr>' || '<tr><td align="left">&nbsp;&nbsp;Коэффициент возвратности</td>
              <td align="right"><b>' || cast(cast(aarep_day_returnment_koef * 100.00 as decimal(15,2)) as varchar(100)) || ' %</b></td></tr>' ||

             '<tr><td class="filled" align="left"><b>Бонус с начала месяца</b></td>' || '<td class="filled" align="right"> ' || iif(aarep_bonus_total_fact > 0, '<font color="#02B316">', '<font color="red">') || '<b>' || cast(cast(aarep_bonus_total_fact as decimal(15,0)) as varchar(100)) || ' </font>р.</b></td></tr>' || '<tr><td align="left">&nbsp;&nbsp;Доход за погашения</td>
              <td align="right"><b>' || cast(cast(aarep_bonus_plus as decimal(15,0)) as varchar(100)) || ' р.</b></td></tr>' || '<tr><td align="left">&nbsp;&nbsp;Штраф за непогашения</td>
              <td align="right"><b>' || cast(cast(aarep_bonus_minus as decimal(15,0)) as varchar(100)) || ' р.</b></td></tr>' || '<tr><td align="left">&nbsp;&nbsp;Коэффициент возвратности</td>
              <td align="right"><b>' || cast(cast(aarep_returnment_koef * 100.00 as decimal(15,2)) as varchar(100)) || ' %</b></td></tr>' ||

             '<tr><td align="left">&nbsp;&nbsp;Базовый заработок</td>
              <td align="right"><b>' || cast(cast(aarep_bonus_total as decimal(15,0)) as varchar(100)) || ' р.</b></td></tr>' || '<tr><td align="left">&nbsp;&nbsp;Количество СВЗ для расчета</td>
              <td align="right"><b>' || cast(cast(aarep_lo_count as decimal(15,0)) as varchar(100)) || '</b></td></tr>' || '<tr><td align="left">&nbsp;&nbsp;Количество клиентов этих СВЗ</td>
              <td align="right"><b>' || cast(cast(aarep_client_count as decimal(15,0)) as varchar(100)) || '</b></td></tr>' ||

             iif(aarep_lo_count <> 0, '<tr><td align="left">&nbsp;&nbsp;Количество клиентов на 1 СВЗ</td>
              <td align="right"><b>' || cast(cast(aarep_client_count as float) / cast(aarep_lo_count as float) as decimal(15,1)) || '</b></td></tr>', '') ||

             '<tr><td align="left">&nbsp;&nbsp;Коэф. за клиентов 1 СВЗ</td>
              <td align="right"><b>' || cast(cast(aarep_lo_koef as decimal(15,1)) as varchar(100)) || ' %</b></td></tr>' || '<tr><td class="filled" align="left">&nbsp;&nbsp;<b>При 100% возвратности, Ваш бонус за месяц может составить</b></td>
              <td class="filled" align="right"><b>' || cast(aarep_bonus_ideal / 100.0 * aarep_lo_koef as decimal(15,0)) || ' р.</b></td></tr>'
      from cb_ru_ssvz_201604_one_2(1000000, :lo_id, :rep_date_end, 1)
      into :res_text_tmp;
    end

    res_text = BLOB_APPEND(res_text, res_text_tmp);
    res_text = BLOB_APPEND(res_text, '</table>');
  end

  if (cm_database_id in (39801, 41701)) then -- для КР
  begin
    res_text = 'Отчет еще не автоматизирован для мобильной программы...';
  end

  if (cm_database_id = 94401) then -- для Аз
  begin
    res_text = 'Отчет еще не автоматизирован для мобильной программы...';
  end
  res_text = BLOB_APPEND(res_text, '</body>');
  suspend;
end
