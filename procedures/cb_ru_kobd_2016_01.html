CREATE PROCEDURE CB_RU_KOBD_2016_01(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    BRANCH_NAME VARCHAR(100),
    C_NAME_FIO VARCHAR(100),
    JL_NAME VARCHAR(100),
    KOL_REPAY INTEGER,
    KOL_DISB INTEGER,
    BONUS INTEGER,
    LO_STAGE_MONTH_FULL DECIMAL(4,2))
AS
declare variable reg_id integer;
declare variable c_id id_type;
declare variable ok_type integer;
declare variable kol_transaction integer;
declare variable un_id id_type;
declare variable jk_system_code varchar(100);
begin
  for select id_list
      from region_list(:rep_region_id, 1, 3, 1)
      into reg_id
  do
  begin
    select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name
    from region r1, region r2
    where r1.reg_unicode = :reg_id and
          r1.parent = r2.reg_unicode
    into reg_name, branch_name;

    for select ps_from_c_id, jl_name, jk_system_code
        from hr_pstatus, joblist, jobkind
        where :rep_date_end between ps_date_begin and ps_date_end and
              ps_is_working = 1 and
              ps_status = 1 and
              ps_from_jl_id = jl_id and
              jl_kind = jk_id and
              jk_system_code = 'JobSystem.Specialist_KOBD' and
              ps_from_reg_id = :reg_id and
              coalesce((select sawt_id
                        from hr_voctravel, hr_orders, sa_worktype
                        where hrv_from_ho_id = ho_id and
                              ho_order_type = sawt_id and
                              hrv_from_c_id = ps_from_c_id and
                              :rep_date_end between hrv_date_begin and hrv_date_end_fact
                        order by hrv_date_begin desc
                        rows 1), 0) not in (16, 17, 18, 8) -- Декретников не включает
        into c_id, jl_name, jk_system_code
    do
    begin
      select lo_stage_month_full
      from hp_get_lo_stage_full(:c_id, :rep_date_end, :jk_system_code)
      into lo_stage_month_full;
--      if (lo_stage_month_full < 0) then lo_stage_month_full = 0;

      un_id = null;
      select un_id
      from username
      where un_from_c_id = :c_id
      into un_id;

      kol_repay = 0;
      kol_disb = 0;
      for select iif(ok_type in (4, 9), 4, 8) as ok_type, count(distinct ol_id)
          from operlist, operkred
          where operlist.is_delete = 0 and
                ol_user_id = :un_id and
                ol_date between :rep_date_begin and :rep_date_end and
                ol_doc_kind in (0, 1) -- кассовый расходный и приходный ордер
                and
                ol_id = ok_from_ol_id and
                ok_os + ok_perc + ok_fine <> 0 and
                ok_type in (4, 8, 9) and
                ((ok_type = 8 and
                ok_os > 0) or
                (ok_type in (4, 9) and
                ok_os + ok_perc + ok_fine < 0))
          group by ok_type
          into ok_type, kol_transaction
      do
      begin
        if (ok_type = 4) then
          kol_repay = kol_transaction;
        if (ok_type = 8) then
          kol_disb = kol_transaction;
      end
      bonus = 0;
      if (lo_stage_month_full > 5.5) then
      begin
        bonus = kol_repay * 1 + kol_disb * 2; -- было  bonus = kol_repay * 2 + kol_disb * 4
      end
      select c_name_fio
      from clients
      where c_id = :c_id
      into c_name_fio;

      suspend;
    end
  end
end
