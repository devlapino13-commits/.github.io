CREATE PROCEDURE HP_OPEREXTRA_GET_EXEXUTION_INFO(
    K_ID D_ID)
RETURNS (
    REP_OPER_NAME D_STR_255,
    REP_OPER_DATE D_STR_CODE,
    REP_AMOUNT D_CURRENCY,
    REP_DETAIL D_STR_LONG,
    REP_UN_FIO D_STR_100)
AS
declare variable RD_ID_CANCEL D_INTEGER;
declare variable RD_ID_RESTORATION D_INTEGER;
begin
  -- RFCKZ-639 отображает историю по контракту в модуле Отмена/Восстановление исполнительной надписи OperExtraCancelAndRestorationForm
  rd_id_cancel = f_get_rd_id('OperExtraKind.LawClientExpenseCancel');
  rd_id_restoration = f_get_rd_id('OperExtraKind.LawClientExpenseRestoration');

  rep_oper_name = 'первичное начисление';
  rep_oper_date = '---';
  rep_amount = 0;
  rep_detail = '---';
  rep_un_fio = '---';

  select f_datetostr(ol.ol_date),
         ox.ox_amount_plus,
         ox.ox_detail,
         un.un_fio
  from operextra ox
  inner join refdata rd on rd.rd_id = ox.ox_from_rd_id
  inner join operlist ol on ol.ol_id = ox.ox_from_ol_id
  inner join username un on un.un_id = ol.ol_user_id
  where ox_object_kind = 9
    and ox_object_id = :k_id
    and ox_ol_is_active = 1
    and rd.rd_code in ('OperExtraKind.ExecutiveInscription')
    and ox_amount_plus > 0
  rows 1
  into rep_oper_date,
       rep_amount,
       rep_detail,
       rep_un_fio;

  suspend;

  rep_oper_name = 'отмена начисления';

  rep_oper_date = '---';
  rep_amount = 0;
  rep_detail = '---';
  rep_un_fio = '---';

  select f_datetostr(ol.ol_date),
         ox.ox_amount_minus,
         ox.ox_detail,
         un.un_fio
  from operextra_execution_note oen
  inner join operextra ox on ox.ox_from_ol_id = oen.oen_from_ol_id_cancel and ox.ox_id = oen.oen_from_ox_id_cancel
  inner join operlist ol on ol.ol_id = oen.oen_from_ol_id_cancel
  inner join username un on un.un_id = ol.ol_user_id
  where oen.oen_object_kind = 9 and
        oen.oen_object_id = :k_id and
        oen.oen_from_rd_id = :rd_id_cancel and
        ox.ox_ol_is_active = 1 and
        ox.ox_amount_minus > 0
  rows 1
  into rep_oper_date,
       rep_amount,
       rep_detail,
       rep_un_fio;

  suspend;

  rep_oper_name = 'восстановление начисления';
  rep_oper_date = '---';
  rep_amount = 0;
  rep_detail = '---';
  rep_un_fio = '---';

  select f_datetostr(ol.ol_date),
         ox.ox_amount_plus,
         ox.ox_detail,
         un.un_fio
  from operextra_execution_note oen
  inner join operextra ox on ox.ox_from_ol_id = oen.oen_from_ol_id_restoration and ox.ox_id = oen.oen_from_ox_id_restoration
  inner join operlist ol on ol.ol_id = oen.oen_from_ol_id_restoration
  inner join username un on un.un_id = ol.ol_user_id
  where oen.oen_object_kind = 9 and
        oen.oen_object_id = :k_id and
        oen.oen_from_rd_id = :rd_id_restoration and
        ox.ox_ol_is_active = 1 and
        ox.ox_amount_plus > 0
  rows 1
  into rep_oper_date,
       rep_amount,
       rep_detail,
       rep_un_fio;

  suspend;

end
