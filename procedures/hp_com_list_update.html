CREATE PROCEDURE HP_COM_LIST_UPDATE(
    CML_ID D_ID,
    CMC_ID D_INTEGER,
    K_ID D_ID,
    ADD_PARAMS D_STR_LONG,
    UN_ID D_ID)
RETURNS (
    RES_CML_ID D_ID,
    RES_TEXT D_STR_255,
    RES_IS_ERROR D_SMALLINT)
AS
declare variable NEW_VALUE D_STR_255;
declare variable OBJ_KIND D_INTEGER;
declare variable PARAM_NAME D_STR_CODE;
declare variable DEFAULT_VALUE D_STR_255;
declare variable DEFAULT_TEXT D_STR_255;
declare variable NEW_STATUS D_INTEGER;
declare variable NEW_STATUS_STR D_STR_CODE;
declare variable CML_DATE D_DATE;
declare variable NEW_DATE D_DATE;
declare variable KF_ID D_ID;
declare variable LO_ID D_ID;
declare variable CMC_CODE D_STR_CODE;
declare variable CMC_COM_KIND D_STR_CODE;
begin
  -- Редактирование параметров реструктуризации из FinStart

  if (coalesce(cml_id, 0) = 0) then
  begin
    if (coalesce(cmc_id, 0) = 0) then
      exception e_default f_format('Параметры не указаны, cml_id %1', coalesce(cmc_id, 'Null'));
  
    if (coalesce(k_id, 0) = 0 and coalesce(cml_id, 0) = 0) then
      exception e_default f_format('Параметры не указаны, k_id %1', coalesce(k_id, 'Null'));
  end

  if (cml_id is null) then
    cml_id = 0;

  if (cmc_id is null) then
    cmc_id = 0;

  if (k_id is null) then
    k_id = 0;

  obj_kind = f_objkind('kontract', 'id');

  res_text = '';
  res_is_error = 0;

  if (cml_id = 0) then
  begin
    select cml_id, res_text, res_is_error
    from hp_com_list_create(:obj_kind, :k_id, :cmc_id, :un_id)
    into :res_cml_id, :res_text, :res_is_error;

    if (res_is_error = 1) then
    begin
      suspend;
      exit;
    end

    select k_from_kf_id
    from kontract
    where k_id = :k_id
    into kf_id;

    select kal_new_kf_officer
    from knaktlist a
    where a.kal_from_kf_id = :kf_id and
          a.kal_date <= current_date and
          a.kal_date_end >= current_date and
          a.is_delete = 0
    into lo_id;

    update com_list
    set cml_init_c_id = :lo_id
    where cml_id = :res_cml_id;

    res_text = '';
  end
  else
  begin
    res_cml_id = cml_id;

    select cml_from_cmc_id, cml_object_id
    from com_list
    where cml_id = :cml_id and
          cml_object_kind = :obj_kind
    into cmc_id, k_id;

    if (row_count = 0) then
      exception e_default f_format('Запись не найдена, cml_id %1', cmc_id);
  end

  select cml_date, cml_last_status, cmc_code, f_enum('com_kind', cmc_com_kind, 'code')
  from com_list, com_cfg
  where cml_id = :res_cml_id
  and cml_from_cmc_id = cmc_id
  into cml_date, new_status, cmc_code, cmc_com_kind;

  new_status_str = f_enum('com_action_status', new_status, 'code');

  if (res_text = '' and new_status_str = 'signing') then
    res_text = 'Документ отправлен на подписание, редактирование запрещается.';

  if (res_text = '' and new_status_str = 'done') then
    res_text = 'Реструктуризация проведена, редактирование запрещается.';

  if (res_text <> '') then
  begin
    res_is_error = 1;
    suspend;
    exit;
  end

  new_date = f_json_str_get_value(add_params, 'com_list_date', current_date);
  if (cml_date <> new_date) then
  begin
    update com_list
    set cml_date = :new_date
    where cml_id = :cml_id;
  end

  for select cmcp_param_name, cmcp_default_value, cmcp_default_text
  from com_cfg_params
  where cmcp_from_cmc_id = :cmc_id
  order by cmcp_order
  into :param_name, default_value, default_text
  do
  begin
    new_value = f_json_str_get_value(add_params, param_name, default_value);
    if (new_value = default_text) then
      new_value = default_value;

    select res_text
    from fm_com_update_params(:res_cml_id, :param_name, :new_value, :un_id)
    into :res_text;

    if (res_text <> '') then
    begin
      res_is_error = 1;
      suspend;
      exit;
    end
  end

  if (cml_id = 0) then
  begin
    select res_text, res_is_error
    from hp_com_action_add(:res_cml_id, null, 'sent', :un_id)
    into :res_text, :res_is_error;

    if (res_is_error = 1) then
    begin
      suspend;
      exit;
    end else
      res_text = '';
  end

  if (cmc_com_kind = 'reschedule') then
  begin
    if (cmc_code in ('kred_kanikul_svo', 'kred_kanikul_standart')) then
    begin
      select res_text from hp_kontvac_tmp(:k_id, :add_params, :un_id)
      into res_text;
      if (res_text <> '') then res_is_error = 1;
    end else
    begin
      select res_is_error, res_text
      from hp_do_reschedule_auto(:k_id, :cml_date, 'auto', 'debug', :un_id)
      into :res_is_error, :res_text;
    end
  end else
  if (cmc_com_kind = 'termination') then
  begin
    select res_is_error, res_text
    from hp_termination_new_graph(:k_id, :add_params)
    into res_is_error, res_text;
  end 

  if (res_is_error = 0 and
    not exists(select * from com_list_resh where cmr_from_cml_id = :res_cml_id)) then
  begin
    -- Сохраняем новый график погашений
    insert into com_list_resh (cmr_from_cml_id, cmr_sh_id, cmr_sh_date, cmr_sh_os, cmr_sh_perc)
    select :res_cml_id, sh_id, sh_date, sh_new_os, sh_new_perc
    from temp_sh
    where sh_new_os <> 0 or sh_new_perc <> 0
    order by sh_date, sh_id;
  end

  if (res_is_error = 0) then
    res_text = 'Параметры успешно сохранены';

  suspend;
end
