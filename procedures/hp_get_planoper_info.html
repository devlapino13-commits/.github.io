CREATE PROCEDURE HP_GET_PLANOPER_INFO(
    REG_ID D_INTEGER,
    C_ID D_ID,
    RD_CODE D_STR_CODE,
    REP_DATE D_DATE,
    PLAN_PERIOD D_INTEGER,
    IS_MGR D_BOOL = 0)
RETURNS (
    PO_AMOUNT D_CURRENCY)
AS
declare variable RD_ID D_INTEGER;
declare variable PM_ID D_INTEGER;
declare variable TMP_AMOUNT D_CURRENCY;
begin
  -- Показывает план по офису или сотруднику

  if (c_id is null) then
    c_id = 0;

  if (is_mgr is null) then
    is_mgr = 0;

  if (plan_period is null) then
    plan_period = extract(year from rep_date) * 100 + extract(month from rep_date);

  po_amount = 0;

  rd_id = f_get_rd_id(rd_code);
  if (c_id = 0) then
  begin
    for select pm_id
        from planmain
        where pm_from_reg_id = :reg_id and
              pm_from_rd_id = :rd_id and
              pm_is_mgr = 0
        into :pm_id
    do
    begin
      tmp_amount = 0;
      select sum(po_calc - po_minus)
      from planoper
      where po_from_pm_id = :pm_id and
            po_period = :plan_period and
            po_date <= :rep_date
      into :tmp_amount;
      if (tmp_amount is null) then
        tmp_amount = 0;

      po_amount = po_amount + tmp_amount;
    end
  end
  else
  begin
    for select pm_id
        from planmain
        where (pm_from_reg_id = iif(:reg_id = 1000000, pm_from_reg_id, :reg_id)) and
              pm_from_c_id = :c_id and
              pm_from_rd_id = :rd_id and
              pm_is_mgr = :is_mgr
        into :pm_id
    do
    begin
      tmp_amount = 0;
      select sum(po_calc - po_minus)
      from planoper
      where po_from_pm_id = :pm_id and
            po_period = :plan_period and
            po_date <= :rep_date
      into :tmp_amount;
      if (tmp_amount is null) then
        tmp_amount = 0;

      po_amount = po_amount + tmp_amount;
    end
  end

--  po_amount = round(po_amount, 0);

  suspend;
end
