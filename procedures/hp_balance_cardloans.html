CREATE PROCEDURE HP_BALANCE_CARDLOANS(
    OBJ_KIND D_STR_20,
    OBJ_ID D_ID,
    DT_OPER D_DATE = current_date)
RETURNS (
    AMOUNT D_CURRENCY)
AS
declare variable k_id d_id;
begin
  amount = 0;
  for
    select
      distinct clo_from_k_id
    from cardlistloan
    where (:obj_kind = 'kontract' and clo_from_k_id = :obj_id)
      or (:obj_kind = 'clients' and clo_from_c_id = :obj_id)
      or (:obj_kind = 'cardlist' and clo_card_id = :obj_id)
    union
    select
      distinct clo_from_k_id
    from cardlistloan
      inner join cardlist
        on cal_card_id = clo_card_id
          and :obj_kind = 'cardacc'
          and cal_from_cca_id = :obj_id
  into
    :k_id
  do
  begin
    amount = amount + coalesce((
      select
        oks_balance
      from oksummary
      where oks_from_k_id = :k_id
        and oks_close_state = 0
        and :dt_oper between oks_date_begin and oks_date_end), 0);
  end
  suspend;
end
