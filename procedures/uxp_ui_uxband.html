CREATE PROCEDURE UXP$UI_UXBAND(
    GRID_UUID D_UUID)
RETURNS (
    BND_GRID_UUID D_UUID,
    BND_ID D_INTEGER,
    BND_PARENT_ID D_INTEGER,
    BND_INDEX D_INTEGER,
    BND_PARENT_INDEX D_INTEGER,
    BND_POSITION D_INTEGER,
    BND_CAPTION D_STR_100,
    BND_HIDDEN D_BOOL,
    BND_HOLD_OWN D_BOOL,
    BND_WIDTH D_INTEGER,
    BND_FIXED_KIND D_SMALLINT,
    BND_CHILDS D_SMALLINT)
AS
begin
  bnd_grid_uuid = grid_uuid;

  for select bnd_id, bnd_parent_id, bnd_index, bnd_childs, bnd_position, bnd_caption, bnd_hidden,
             bnd_hold_own, bnd_width, bnd_fixed_kind
  from ux$band
  where bnd_grid_uuid = :grid_uuid
  order by bnd_index
  into :bnd_id, :bnd_parent_id, :bnd_index, :bnd_childs, :bnd_position, :bnd_caption, :bnd_hidden,
       :bnd_hold_own, :bnd_width, bnd_fixed_kind
  do
  begin
    bnd_parent_index = -1;
    bnd_caption = uxf$lang(bnd_caption);

    if (bnd_parent_id is not null) then
    begin
      select bnd_index
      from ux$band
      where bnd_grid_uuid = :grid_uuid and
            bnd_id = :bnd_parent_id
      into bnd_parent_index;
    end

    if (bnd_hidden = 0 and bnd_childs = 0) then
    begin
      if (not exists(select * from ux$gritem
      where ugi_ugr_uuid = :grid_uuid and
            ugi_bnd_id = :bnd_id)) then
        bnd_hidden = 1;
    end

    suspend;
  end
end
