CREATE PROCEDURE CM_KG_0048_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
DECLARE VARIABLE C_ID ID_TYPE;
DECLARE VARIABLE KC_CIRCLE INTEGER;
DECLARE VARIABLE MA_APP_ID INTEGER;
DECLARE VARIABLE KBM_ID ID_TYPE;
DECLARE VARIABLE KBM_ID_OLD ID_TYPE;
DECLARE VARIABLE MAX_SUM CURR_TYPE;
DECLARE VARIABLE MIN_SUM CURR_TYPE;
DECLARE VARIABLE KF_ID ID_TYPE;
DECLARE VARIABLE REG_ID INTEGER;
DECLARE VARIABLE K_KOLICH_REPAY INTEGER;
DECLARE VARIABLE K_KOLICH_CLIENT_APPROVED INTEGER;
DECLARE VARIABLE K_PERCENT_PER_DAY PERC_TYPE;
DECLARE VARIABLE PR_PRODUCT_KIND INTEGER;
DECLARE VARIABLE K_CIRCLE INTEGER;
DECLARE VARIABLE K_KOMMIS_PERC PERC_TYPE;
DECLARE VARIABLE K_AMOUNT_APPROVED CURR_TYPE;
DECLARE VARIABLE PR_NAME VARCHAR(50);
DECLARE VARIABLE PR_CODE VARCHAR(100);
DECLARE VARIABLE PR_CODE_OLD VARCHAR(100);
DECLARE VARIABLE PERCENT_SNIJEN PERC_TYPE;
DECLARE VARIABLE PERCENT_SNIJEN_PLAN PERC_TYPE;
DECLARE VARIABLE KOL_KIB_CLI INTEGER;
DECLARE VARIABLE KH_CLIENTS_FIO VARCHAR(1024);
DECLARE VARIABLE C_IF VARCHAR(1024);
DECLARE VARIABLE C_IF_OLD VARCHAR(100);
DECLARE VARIABLE NEW_CL INTEGER;
DECLARE VARIABLE CL INTEGER;
DECLARE VARIABLE KOMMIS_SNIJ DECIMAL(15,2);
DECLARE VARIABLE KOMMIS_SNIJ_PLAN DECIMAL(15,2);
begin
 
  FOR select k_disb_date, k_from_kf_id, k_last_reg_id,
             k_kolich_repay, k_kolich_client_approved, k_percent_per_day,
             pr_product_kind, k_circle, k_kommis_perc,
             k_amount_approved, pr_name, pr_code, k_from_ma_app_id
    from kontract, product
    where k_id=:k_id
    and k_from_pr_id = pr_id
    and pr_code containing '.A.'
    and pr_code not containing '.PAR.'
  INTO :KF_DISB_DATE, :KF_ID, :REG_ID,
       :K_KOLICH_REPAY, :K_KOLICH_CLIENT_APPROVED,
       K_PERCENT_PER_DAY, :PR_PRODUCT_KIND, K_CIRCLE,
       K_KOMMIS_PERC, K_AMOUNT_APPROVED, PR_NAME, PR_CODE, MA_APP_ID
  DO
  if (KF_DISB_DATE between '19.09.16' and '31.10.16') then
  BEGIN
    --проверка по клиентам группы
    res_info_text = null;
    for select kncl.kc_from_c_id,
             kncl.kc_amount_approved,
             kncl.kc_circle,
             cl.c_name_fio
    from knclient kncl, clients cl
    where kncl.kc_amount_approved > 0
    and cl.c_id = kncl.kc_from_c_id
    and kncl.kc_from_k_id = :k_id
    into :c_id,
       :kc_amount,
       :kc_circle,
       :res_info_text
    do
    begin
      if (kc_circle = 1) then new_cl = coalesce(new_cl+1, 1);
      else
      begin
        kc_circle = null;
        select count(*)+1 kc_circle
        from knclient, kontstate
        where kc_amount_approved > 0
          and kc_from_c_id = :c_id
          and kc_from_k_id = ks_from_k_id
          and kc_from_k_id <> :k_id
          and ks_ol_is_delete = 0
          and ks_state = 1
        into kc_circle;
        if (kc_circle is null) then new_cl = coalesce(new_cl+1, 1); else cl = coalesce(cl+1,1);
      end
      if(new_cl is null) then
      new_cl = 0;
      if(cl is null) then
      cl = 0;
    
      --история по киб
      kbm_id = null;
      if (kc_circle < 2 or kc_circle is null) then
      begin
        select kbm_id
        from hp_kibminiapp_last(:c_id, :ma_app_id)
        into kbm_id;

        if (kc_amount > 50000 and
                not exists (select *
                            from kibhistory kh
                            where kh_from_kbm_id = :kbm_id
                            and kh_creditstatus not containing('отказано')
                            and kh_creditstatus not containing('рассмотрение заявки')
                            and kh_is_our = 0
                            and kh_is_debtor = 1)) then
        begin
          --res_error_text = 'У клиента нет истории киб. максимальная сумма может быть 50000 c';
          --res_error_is_critical =0;
          --suspend;
        end
        
        if (kbm_id is not null) then
        begin
          for select kh_clients_fio
            from kibhistory kh
            where kh_from_kbm_id = :kbm_id
            and kh_creditstatus not containing('отказано')
            and kh_creditstatus not containing('рассмотрение заявки')
            and kh_is_our = 0
            and kh_is_debtor = 1
            order by kh_id rows 1
            into kh_clients_fio
          do
          begin     
            for select (select txt2||' '||txt1 from hp_fio_clear(c_name_fio)) as c_if
              from clients, knclient
              where kc_amount_approved > 0
              and kc_from_k_id = :k_id
              and kc_from_c_id = c_id
              and kc_from_c_id <> :c_id
            into c_if_old
            do
            begin
              c_if = iif(c_if is null, '', c_if||iif(c_if = '', '', '|'))||c_if_old;
            end
        
            if (kh_clients_fio similar to '%('||c_if||')%') then
              kol_kib_cli = coalesce(kol_kib_cli + 1, 1); else kol_kib_cli = coalesce(kol_kib_cli, 0);
            c_if = '';
          end
        end
      end         
        
      if (kol_kib_cli is null) then kol_kib_cli = 0;
    END
    PERCENT_SNIJEN = ROUND(K_PERCENT_PER_DAY * 365, 1);
    KOMMIS_SNIJ = k_kommis_perc;
    --Первый вариант
    if (PR_CODE containing '.GR.') then
    begin
      if (NEW_CL > CL and KOL_KIB_CLI < 2) then
      begin
        if (PR_CODE containing '.VYGOD.') then KOMMIS_SNIJ_PLAN = 4;
          else KOMMIS_SNIJ_PLAN = 3;
      end
      if (NEW_CL > CL and KOL_KIB_CLI > 1) then
      begin
        if (PR_CODE containing '.VYGOD.') then KOMMIS_SNIJ_PLAN = 2;
        else KOMMIS_SNIJ_PLAN = 1;
      end
      if (NEW_CL <= CL) then
      begin
        if (PR_CODE containing '.VYGOD.') then KOMMIS_SNIJ_PLAN = 2;
        else KOMMIS_SNIJ_PLAN = 1;
      end
    end
    if (PR_CODE containing '.IND.') then
    begin
      if (PR_CODE containing '.EC.') then
      BEGIN
      KOMMIS_SNIJ_PLAN = case
                           when KC_CIRCLE > 1 then 3.5
                             else 4.5
                         end;
      END ELSE
      BEGIN
        KOMMIS_SNIJ_PLAN = KOMMIS_SNIJ;
        res_error_is_critical = 2;
        res_error_text = 'Продукт ИНД-стандарт не участвует в акции!';
        SUSPEND;
      END
    END
--if(k_id = 109010010218100401) then  KOMMIS_SNIJ_PLAN = 1;
     
    if (KOMMIS_SNIJ <> KOMMIS_SNIJ_PLAN) then
    begin
      res_error_text = 'Доп.процент по акции должен быть: '||KOMMIS_SNIJ_PLAN;
      res_error_is_critical =1;
      suspend;
    end
  END
  else
  begin 
    res_error_is_critical = 2;
    res_error_text = 'Выбран не правильный продукт' ;
    SUSPEND;
  end 
END
