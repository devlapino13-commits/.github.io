CREATE PROCEDURE AAREP_FIND_CELLAKT(
    REPORT_REGION_ID INTEGER,
    REPORT_TEXT VARCHAR(100),
    REPORT_IS_SKLAD INTEGER)
RETURNS (
    CAL_FROM_CAM_ID DECIMAL(18,0),
    CAL_ID INTEGER,
    CAL_DATE_BEGIN DATE,
    CAL_DATE_END DATE,
    CAL_FROM_C_ID DECIMAL(18,0),
    CAL_CELL_NO VARCHAR(20),
    CAL_STATE INTEGER,
    CAL_IS_CONTACT INTEGER,
    CAL_IS_CORPORATE INTEGER,
    CAL_HAS_LIMIT INTEGER,
    CAL_FROM_COP_ID INTEGER,
    CAL_PRIOR_C_ID DECIMAL(18,0),
    CAM_DATE DATE,
    CAM_FROM_REG_ID INTEGER,
    CAM_MEMO VARCHAR(255),
    CAM_DATETIME TIMESTAMP,
    CAM_USER_ID DECIMAL(18,0),
    CAM_OFFICE VARCHAR(100),
    CAL_CLIENT_FIO VARCHAR(100),
    CAL_FIO_FROM VARCHAR(100),
    CAM_TYPE_STR VARCHAR(100),
    CAM_USER_NAME VARCHAR(100),
    CLK_LIMIT_KIND VARCHAR(100),
    CAL_JOB_NAME VARCHAR(100),
    CAL_CLIENT_OFFICE_NAME VARCHAR(100))
AS
declare variable cam_type integer;
declare variable st_staff integer;
declare variable st_client integer;
declare variable find_fio varchar(100);
declare variable find_cell varchar(100);
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  if ((report_text <> '') or (report_is_sklad = 1)) then
  begin
    if (:report_text similar to '[+]?[0-9]+') then
      find_cell = report_text;
    else
      find_fio = report_text;

    for select cal.cal_from_cam_id, cal_id, cal_date_begin, cal_date_end, cal_from_c_id, cal_cell_no, cal_state,
               cal_is_contact, cal_is_corporate, cal_has_limit, cal_from_cop_id, cal_prior_c_id, cam_date, cam_type,
               cam_from_reg_id, cam_memo, cam_datetime, cam_user_id, reg_code || ' ' || reg_name, c1.c_name_fio,
               c2.c_name_fio, c3.c_name_fio, clk_name
        from
        (
          select cal.*
          from hp_find_from_namelist(0, 0, :find_fio) nl
            inner join cellaktlist cal on cal.cal_from_c_id = nl.OBJECT_ID

          union all

          select cal.*
          from hp_find_from_namelist(0, 0, :find_fio) nl
            inner join cellaktlist cal on cal.cal_prior_c_id = nl.OBJECT_ID

          union all

          select cal.*
          from cellaktlist cal
          where
            :find_cell is not null
            and cal_cell_no like '%' || :find_cell || '%'
        ) cal inner join cellaktmain cam on cal.cal_from_cam_id = cam.cam_id
          inner join region r on cam.cam_from_reg_id = r.reg_unicode
          inner join region_list(:report_region_id, 1, -1, 1) rl on cam.cam_from_reg_id = rl.id_list
          inner join clients c1 on cal.cal_from_c_id = c1.c_id
          inner join username un on cam.cam_user_id = un.un_id
          inner join clients c3 on un.un_from_c_id = c3.c_id
          inner join cellimitkind clk on cal.cal_has_limit = clk.clk_id
          left join clients c2 on cal.cal_prior_c_id = c2.c_id
        where
          (1 = 1) --cam.cam_type <> -1   kvo времено отключил иначе при передаче со склада сотруднику не видит номера
          and (
            :report_is_sklad = 0
            or (
                  :report_is_sklad = 1
                  and cal_date_end = '31.12.9999'
                  and cal_state = 0
                )
          )
        order by cal_cell_no, cal_date_begin, cal_date_end
        into :cal_from_cam_id, :cal_id, :cal_date_begin, :cal_date_end, :cal_from_c_id, :cal_cell_no, :cal_state,
             :cal_is_contact, :cal_is_corporate, :cal_has_limit, :cal_from_cop_id, :cal_prior_c_id, :cam_date,
             :cam_type, :cam_from_reg_id, :cam_memo, :cam_datetime, :cam_user_id, :cam_office, :cal_client_fio,
             :cal_fio_from, :cam_user_name, :clk_limit_kind
    do
    begin
      cal_job_name = null;
      cal_client_office_name = null;

      select jl.jl_name, reg.reg_name
      from hr_pstatus hps
      inner join joblist jl on jl.jl_id = ps_from_jl_id
      inner join region reg on reg.reg_unicode = hps.ps_from_reg_id
      where current_date between ps_date_begin and ps_date_end and
            ps_is_working = 1 and
            hps.ps_from_c_id = :cal_from_c_id and
            hps.ps_status = 1
      into :cal_job_name, :cal_client_office_name;

      if (cal_job_name is null) then
        cal_job_name = 'Не работает';

      if (cam_type = 0) then
        cam_type_str = 'Прием на склад';
      if (cam_type = 1) then
        cam_type_str = 'Личную симку на контакты';
      if (cam_type = 2) then
        cam_type_str = 'Личную симку на корпоратив';
      if (cam_type = 3) then
        cam_type_str = 'Со склада на пользование';
      if (cam_type = 4) then
        cam_type_str = 'Передача между сотрудниками';
      if (cam_type = 5) then
        cam_type_str = 'Изменение статуса';
      if (cam_type = 6) then
        cam_type_str = 'Аннулирование действ.симки';
      suspend;
    end
  end
end
