CREATE PROCEDURE HP_SMEVPASS_SEND_MSG(
    SPM_KIND D_SMALLINT,
    SBC_ID D_ID,
    UN_ID D_ID,
    MA_ID D_ID = null)
AS
declare variable c_id d_id;
declare variable app_summ d_currency;
declare variable spm_id d_id;
declare variable res_text d_str_255;
begin
  select sbc_from_c_id
  from smevpass_blockclients
  where sbc_id = :sbc_id
  into c_id;

  if (c_id is null) then
    exception e_default 'Not found record for sbc_id = ' || coalesce(sbc_id, '<NULL>');

  if (spm_kind = 1) then
  begin
    select mac_amount
    from miniappclient
    where mac_from_ma_id = :ma_id
      and mac_from_c_id = :c_id
    into app_summ;

    if (coalesce(app_summ, 0) = 0) then
      exception e_default 'Not found app_summ for ma_id = ' || coalesce(ma_id, '<NULL>');
  end
  else if (spm_kind not in (2,3)) then
    exception e_default 'Unknown spm_kind = ' || coalesce(spm_kind, '<NULL>');

  insert into smevpass_msg (spm_from_sbc_id, spm_kind, spm_from_ma_id)
  values (:sbc_id, :spm_kind, :ma_id)
  returning spm_id
  into spm_id;

  select res_text
  from hp_edoc_ins('smevpass.notify', f_objkind('smevpass_msg', 'id'), :spm_id, :un_id)
  into res_text;

  if (coalesce(res_text, '') <> '') then
    exception e_default 'EDoc insert error: ' || res_text;

  if (spm_kind = 2) then
    update smevpass_blockclients
    set sbc_unblock_datetime = current_timestamp,
        sbc_unblock_un_id = :un_id
    where sbc_id = :sbc_id;

end
