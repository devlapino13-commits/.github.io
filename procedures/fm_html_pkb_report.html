CREATE PROCEDURE FM_HTML_PKB_REPORT(
    MA_ID D_ID,
    C_ID D_ID)
RETURNS (
    RES_HTML D_BLOB_TEXT)
AS
declare variable KH_CREATE_DATE D_DATE;
declare variable UN_FIO D_STR_100;
declare variable KF_CLIENT_NAME D_STR_100;
declare variable KF_DATE_BIRTH D_DATE;
declare variable KF_CLIENT_PASS_NO D_STR_100;
declare variable KF_INN D_STR_20;
declare variable KF_AMOUNT decimal(18,2);
declare variable MA_KOLICH_REPAY D_INTEGER;
declare variable BASIC_AMOUNT D_STR_100;
declare variable ACTIVE_CNT D_STR_100;
declare variable MONTHLY_REPAY decimal(18,2);
declare variable ALL_CNT D_STR_100;
declare variable SHOWN_CNT D_INTEGER;
declare variable SCORE_RATE D_STR_100;
declare variable RISK_RATE D_STR_255;
declare variable RISK_RATE_DESCRIPTION D_STR_CODE;
declare variable COMMON_POINTS D_INTEGER;
declare variable IPF_POINTS D_INTEGER;
declare variable KH_NO D_INTEGER;
declare variable KH_COMPANY D_STR_CODE;
declare variable KH_CURRENCY D_STR_10;
declare variable KF_IS_PORUCH_STR D_STR_20;
declare variable KH_IS_ACTIV_TXT D_STR_CODE;
declare variable KH_CREDITTYPE D_STR_CODE;
declare variable KH_OWNAMOUNT D_STR_100;
declare variable KH_UNPAIDAMOUNT D_STR_100;
declare variable KH_CREDITSTATUS D_STR_CODE;
declare variable KH_DATEAWARD D_DATE;
declare variable KH_AMOUNT D_STR_100;
declare variable KH_UNPAIDPERCENAMOUNT D_STR_100;
declare variable KH_REPAYSTATUS D_STR_255;
declare variable KH_DATESTOP D_DATE;
declare variable KH_PERIOD_IN_MONTH D_INTEGER;
declare variable KH_DELIQDAYSCOUNT D_INTEGER;
declare variable KH_CLIENTS_FIO D_STR_LONG;
declare variable KH_PORUCH_FIO D_STR_LONG;
declare variable REGION_ID D_INTEGER;
declare variable COLOR_BLACK varchar(100);
declare variable COLOR_YELLOW varchar(100);
declare variable COLOR_RED varchar(100);
declare variable COLOR_LIGHT_RED varchar(100);
declare variable COLOR_GREEN varchar(100);
declare variable COLOR_LIGHT_GREEN varchar(100);
declare variable BLS_ID D_ID;
declare variable BLS_ID_STOP_CR D_ID;
declare variable IS_GAMBLING D_SMALLINT;
declare variable IS_STOP_CREDIT D_SMALLINT;
declare variable IS_BANKRUPTCY D_SMALLINT;
declare variable IS_TAX_DEBT D_SMALLINT;
declare variable IS_FAKE_BUSINESS D_SMALLINT;
declare variable IS_BAD_SUPPLIER D_SMALLINT;
declare variable IS_BLACK_LIST D_SMALLINT;
declare variable IS_WANTED_PERSON D_SMALLINT;
declare variable IS_WANTED_ALIMENT D_SMALLINT;
declare variable JSON_TEXT D_BLOB_TEXT;
declare variable TYPE_OF_FOUNDING D_STR_CODE;
declare variable TYPE_OF_GUARANTEE D_STR_CODE;
declare variable TYPE_CONTRACT D_STR_255;
declare variable FINANCIAL_INSTITUTION D_STR_255;
declare variable STATUS_TEMP D_STR_10;
declare variable STATUS_STOP_CR D_STR_10;
declare variable PORUCH_NAME D_STR_255;
begin
  select eil.eil_from_bls_id
  from extinfolist eil
  inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
  where eic.eic_code = 'pkb_cib_report'
      and eil.eil_object_kind = f_objkind('miniapp', 'id')
      and eil.eil_from_c_id = :c_id
      and eil.eil_status = 2
  order by eil.eil_status_datetime desc
  rows 1
  into :bls_id;

  if (:bls_id is null) then
    exit;
  color_black = 'black';
  color_yellow = '#f08205';
  color_red = '#ff0000';
  color_light_red = '#ff9999';
  color_green = '#008000';
  color_light_green = '#00ff00';

  res_html = (select res_style
              from fm_html_styles) || '<body>';

  res_html = blob_append(res_html, '<b>Отчет по проверке в БДКИ</b><br><br>');
  ------------------------------------------------------------------------------
  kh_create_date = current_timestamp;
  select cl.c_name_fio, maa.ma_kolich_repay
  from miniapp maa
  inner join username un on un.un_id = maa.ma_from_un_id
  inner join clients cl on cl.c_id = un.un_from_c_id
  where maa.ma_id = :ma_id
    and maa.ma_is_active = 1
    and maa.ma_date_end >= '31.12.2999'
  rows 1
  into :un_fio, ma_kolich_repay;

  select doc_number, principal_amount, number_of_contracts, all_credit_history_count, dob,
        is_gambling, is_bankruptcy, is_tax_debt, is_fake_business, is_bad_supplier,
        is_black_list, is_wanted_person, is_wanted_aliment, c_name_fio
  from hp_parse_pkb_report (:bls_id)
  rows 1
  into :kf_client_pass_no, :basic_amount, :active_cnt, :all_cnt, :kf_date_birth,
       :is_gambling, :is_bankruptcy, :is_tax_debt, :is_fake_business, :is_bad_supplier,
       :is_black_list, :is_wanted_person, :is_wanted_aliment, kf_client_name;
  ----------------------------------RFCKG-4405----------------------------------
  select eil.eil_from_bls_id
  from extinfolist eil
  where eil.eil_from_c_id = :c_id
        and eil.eil_from_eic_id = 4 -- стоп-кредит
        and eil.eil_status = 2
  order by eil.eil_id desc
  rows 1
  into :bls_id_stop_cr;

  if (:bls_id_stop_cr is not null) then
  begin
    select blx_blob
    from hp_get_blobex_blob(:bls_id_stop_cr)
    into json_text;

    status_temp = f_json_str_get_value(:json_text, 'result', '');

    status_stop_cr = (case status_temp
                      when 'true' then 'Да'
                      when 'false' then 'Нет'
                      else ''
                      end);
  end
  else
    status_stop_cr = '';

  select mac.mac_amount
  from miniappclient mac
  where mac.mac_from_ma_id = :ma_id
        and mac.mac_amount > 0
  into :kf_amount;
  common_points = f_point(:c_id, :kh_create_date, 0);

  monthly_repay =  :kf_amount / ma_kolich_repay;

  res_html = blob_append(res_html, '<table width="100%"><col width="40%"><col width="60%">' ||
               f_html_table_row('Дата проверки', :kh_create_date, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Пользователь', :un_fio, '', '', -1, -1, -1, -1, 2, '') || '</table><br>');

  res_html = blob_append(res_html, '<table width="100%"><col width="40%"><col width="60%">' ||
               f_html_table_row('Клиент', :kf_client_name, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Дата рождения', :kf_date_birth, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Документ', :kf_client_pass_no, '', '', -1, -1, -1, -1, 2, '') || '</table><br>');

  res_html = blob_append(res_html, '<b>Предполагаемые суммы по всем активным кредитам<b><br>');
  res_html = blob_append(res_html, '<table width="100%"><col width="40%"><col width="60%">' ||
               f_html_table_row('Сумма в заявке', :kf_amount, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Срок кредита в заявке', :ma_kolich_repay, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Основная сумма', :basic_amount, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Кол-во', :active_cnt, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Ежемесяч. платеж', :monthly_repay, '', '', -1, -1, -1, -1, 2, '') || --***
               f_html_table_row('Всего кредитных историй', :all_cnt, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Отображено', :active_cnt, '', '', -1, -1, -1, -1, 2, '') || '</table><br>');

  res_html = blob_append(res_html, '<b>Внутренний скоринг Компании<b><br>');
  res_html = blob_append(res_html, '<table width="100%"><col width="40%"><col width="60%">' ||
               f_html_table_row('Баллы по классике', :common_points, '', '', -1, -1, -1, -1, 2, '') || '</table><br>');

  res_html = blob_append(res_html, '<b>Информация из дополнительных источников<b><br>');
  res_html = blob_append(res_html, '<table width="100%"><col width="80%"><col width="20%">' ||
               f_html_table_row('Был ли вовлечен в игорный бизнес', iif(:is_gambling = 1, 'Да', 'Нет'), '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Стоп Кредит',:status_stop_cr, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Были применены процедуры банкротство', iif(:is_bankruptcy = 1, 'Да', 'Нет'), '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Есть задолженность по таможенным платежам, налогам и пеням, не погашенную в течение 4 месяцев со дня ее возникновения', iif(:is_tax_debt = 1, 'Да', 'Нет'), '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Налогоплательщик, осуществляющий лжепредпринимательскую деятельность', iif(:is_fake_business = 1, 'Да', 'Нет'), '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Недобросовестный участник государственных закупок', iif(:is_bad_supplier = 1, 'Да', 'Нет'), '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Входит в список о налогоплательщиках - ЮЛ, имеющих налоговую задолженность более 150 МРП и ИП, частных судебных' ||
                                'исполнителях, адвокатах, профессиональных медиаторах, юмеющих налоговую задолженность более 10 МРП, не погашенную' ||
                                'в течении 4 месяцев со дня ее возникновния; агентов, имеющих налоговую задолженность по пенсионным взносам, обязательным' ||
                                'профессиональным пенсионным взносам, не погашенную в срок более чем 6 месяцев', iif(:is_black_list = 1, 'Да', 'Нет'), '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Найден в розыске преступников, должников, без вести пропавшее лиц Комитетом правовой статистике и специальным учетам ГП РК', iif(:is_wanted_person = 1, 'Да', 'Нет'), '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Найден в розыске алиментщиков Комитетом по правовой статистике и специальным учетам ГП РК', iif(:is_wanted_aliment = 1, 'Да', 'Нет'), '', '', -1, -1, -1, -1, 2, '') || '</table><br>');

  ------------------------------------------------------------------------------
  for select date_of_credit_start, date_of_credit_end, total_amount,
             loan_term, ex_overdue_max, outstanding_amount,
             contract_status, payment_status, c_name_fio, type_contract, financial_institution,
             poruch_name
  from hp_parse_pkb_report(:bls_id)
  into :kh_dateaward, kh_datestop, kh_amount,
       :kh_period_in_month, kh_deliqdayscount, kh_unpaidamount,
       kh_creditstatus, kh_repaystatus, kh_clients_fio, type_contract, financial_institution,
       poruch_name
  do
  begin

    res_html = blob_append(res_html, '<b>Кредит № '|| :kh_no || '<b><br>');
    res_html = blob_append(res_html, '<table width="100%"><col width="25%"><col width="25%"><col width="25%"><col width="25%">' ||
               '<tr><th>ФКУ</th><th>Валюта</th><th>Участник</th><th>Статус</th></tr>' ||
               f_html_table_row(iif(:financial_institution is null, '', iif(:financial_institution not containing 'МБулак', 'В другом ФКУ', 'В МБулак')),'KZT',
               iif(poruch_name = '', 'Заемщик', poruch_name), iif(:kh_creditstatus = 1, 'Активный', 'Неактивный'), -1, -1, -1, -1, 4, '') ||
               '<br></table>');


    res_html = blob_append(res_html, '<table width="100%"><col width="40%"><col width="60%">' ||
               f_html_table_row('Тип', :type_contract, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Дата выдачи', f_datetostr(:kh_dateaward), '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Дата окончания', f_datetostr(:kh_datestop), '', '', -1, -1, -1, -1, 2, '') ||

               f_html_table_row('Общая сумма', :kh_amount, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Срок кредита',:kh_period_in_month, '', '', -1, -1, -1, -1, 2, '') ||
            
               f_html_table_row('Остаток ОС', :kh_unpaidamount, '', '', -1, -1, -1, -1, 2, '') ||   -- хардкод
               f_html_table_row('Статус кредита', iif(:kh_creditstatus = 1, 'Активный', 'Неактивный'), '', '', -1, -1, -1, -1, 2, '') ||

               f_html_table_row('Статус погашения', :kh_repaystatus, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Макс. дн. просроч.', :kh_deliqdayscount, '', '', -1, -1, -1, -1, 2, '') ||

               f_html_table_row('Заемщики', :kh_clients_fio, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Поручители', :poruch_name, '', '', -1, -1, -1, -1, 2, '')|| '</table><br>');
  end
  res_html = blob_append(res_html, '</body>');
  suspend;
end
