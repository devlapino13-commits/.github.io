CREATE PROCEDURE MC_DEFINE_LOAN_PROCESS(
    MCL_ID D_ID)
RETURNS (
    DO_PROCESS D_STR_CODE)
AS
declare variable C_ID ID_TYPE;
declare variable IS_ASP_SIGNED D_BOOL;
declare variable IS_NEW_CLIENT D_BOOL;
declare variable KBM_ID D_ID;
begin
  do_process = '';

  select mcl.mcl_from_c_id
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into :c_id;

  is_asp_signed = F_ASP_SIGNED(:c_id, current_date);

  if (is_asp_signed = 0) then -- если у клиента етсь подпсианный договор ЭЦП
    do_process = 'asp_add';
  else
  begin
    select is_new_client
    from hp_is_new_client_for_cred_limit(:c_id)
    into :is_new_client;

    if (is_new_client = 0) then -- если это повторник
      do_process = 'countinue_standard_process';
    else
    begin -- если новый клиент
      select kbm_id
      from mc_online_loan_get_kib_id(:c_id)
      into :kbm_id;

      if (kbm_id is null) then
        do_process = 'limit_check';
      else
        do_process = 'countinue_standard_process';
    end
  end
  suspend;
end
