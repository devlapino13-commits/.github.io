CREATE PROCEDURE HP_AGREEMENT_IU(
    AG_ID_TMP D_INTEGER,
    AG_PARENT_ID D_INTEGER,
    AG_FROM_RD_ID D_INTEGER,
    AG_OBJECT_KIND D_INTEGER,
    AG_OBJECT_ID D_ID,
    AG_FROM_REG_ID D_INTEGER,
    AG_CAPTION D_STR_255,
    AG_AGREEMENT_NO VARCHAR(30),
    AG_AGREEMENT_DATE D_DATE,
    AG_DATE_BEGIN D_DATE,
    AG_DATE_END D_DATE,
    AG_MEMO D_STR_255,
    AG_FROM_UN_ID D_ID,
    AG_DATETIME D_DATETIME,
    AG_STATUS D_INTEGER,
    AG_ELSOM_ACCOUNT D_STR_20,
    AG_ELSOM_BANK_ID D_INTEGER,
    AG_FROM_INIT_REG_ID D_INTEGER,
    AG_ELCARD_ACCOUNT D_STR_20,
    AG_FROM_ACTIV_RD_ID D_INTEGER,
    AG_FROM_AL_ID D_STR_20,
    AG_COMMIS_CARDS_MB D_PERCENT,
    AG_COMMIS_CARDS_OTHER D_PERCENT,
    AG_MBANK_ACCOUNT D_STR_20,
    AG_FROM_PAY_RD_ID D_INTEGER,
    AEA_SIGN D_BOOL,
    AEO_STATUS D_SMALLINT = -5,
    AG_TAX_PERCENT D_PERCENT = null)
RETURNS (
    AG_ID D_INTEGER)
AS
declare variable db_id d_integer;
declare variable old_aeo_status d_smallint;
declare variable commis d_percent;
begin
  db_id = f_get_db_id();
  if (db_id = 41701) then
  begin
    commis = coalesce(ag_commis_cards_other, 0);
    if (ag_from_rd_id = f_get_rd_id('AgreementKinds.IPF_WithCommission')
      and not (commis > 0 and commis < 100))
    then exception e_message 'Некорректное значение "Комиссии ТСП на другие карты".';
  end

  if (:ag_id_tmp is null) then
  begin
    insert into agreements (ag_parent_id, ag_from_rd_id, ag_object_kind, ag_object_id, ag_from_reg_id, ag_caption,
                            ag_agreement_no, ag_agreement_date, ag_date_begin, ag_date_end, ag_memo, ag_from_un_id,
                            ag_datetime, ag_status, ag_elsom_account, ag_elsom_bank_id, ag_from_init_reg_id,
                            ag_elcard_account, ag_from_activ_rd_id, ag_from_al_id, ag_commis_cards_mb,
                            ag_commis_cards_other, ag_mbank_account, ag_from_pay_rd_id, ag_tax_percent)
    values (:ag_parent_id, :ag_from_rd_id, :ag_object_kind, :ag_object_id, :ag_from_reg_id, :ag_caption,
            :ag_agreement_no, :ag_agreement_date, :ag_date_begin, :ag_date_end, :ag_memo, :ag_from_un_id, :ag_datetime,
            :ag_status, :ag_elsom_account, :ag_elsom_bank_id, :ag_from_init_reg_id, :ag_elcard_account,
            :ag_from_activ_rd_id, :ag_from_al_id, :ag_commis_cards_mb, :ag_commis_cards_other, :ag_mbank_account, :ag_from_pay_rd_id, :ag_tax_percent)
    returning ag_id
    into :ag_id;

    insert into agreement_esign_access (aea_from_ag_id, aea_sign, aea_from_un_id)
    values (:ag_id, iif(:aea_sign = 1, true, false), :ag_from_un_id);

    if (:aeo_status <> -1) then
        insert into agreement_eds_original (aeo_from_ag_id, aeo_status, aeo_from_un_id)
        values (:ag_id_tmp, :aeo_status, :ag_from_un_id);

    ag_id_tmp = ag_id;
  end
  else
  begin
    ag_id = ag_id_tmp;

    update agreements
    set ag_parent_id = :ag_parent_id,
        ag_from_rd_id = :ag_from_rd_id,
        ag_object_kind = :ag_object_kind,
        ag_object_id = :ag_object_id,
        ag_from_reg_id = :ag_from_reg_id,
        ag_caption = :ag_caption,
        ag_agreement_no = :ag_agreement_no,
        ag_agreement_date = :ag_agreement_date,
        ag_date_begin = :ag_date_begin,
        ag_date_end = :ag_date_end,
        ag_memo = :ag_memo,
        ag_from_un_id = :ag_from_un_id,
        ag_datetime = :ag_datetime,
        ag_status = :ag_status,
        ag_elsom_account = :ag_elsom_account,
        ag_elsom_bank_id = :ag_elsom_bank_id,
        ag_from_init_reg_id = :ag_from_init_reg_id,
        ag_elcard_account = :ag_elcard_account,
        ag_from_activ_rd_id = :ag_from_activ_rd_id,
        ag_from_al_id = :ag_from_al_id,
        ag_commis_cards_mb = :ag_commis_cards_mb,
        ag_commis_cards_other = :ag_commis_cards_other,
        ag_mbank_account = :ag_mbank_account,
        ag_from_pay_rd_id = :ag_from_pay_rd_id,
        ag_tax_percent = :ag_tax_percent
    where (ag_id = :ag_id_tmp);
  end

  if (not exists(select *
                 from agreement_esign_access aea
                 where aea.aea_from_ag_id = :ag_id_tmp)) then
    insert into agreement_esign_access (aea_from_ag_id, aea_sign, aea_from_un_id)
    values (:ag_id_tmp, iif(:aea_sign = 1, true, false), :ag_from_un_id);
  else
    update agreement_esign_access
    set aea_sign = iif(:aea_sign = 1, true, false),
        aea_from_un_id = :ag_from_un_id
    where (aea_from_ag_id = :ag_id_tmp);

  if (aeo_status <> -5) then
  begin
    select aeo.aeo_status
    from agreement_eds_original aeo
    where aeo.aeo_from_ag_id = :ag_id_tmp
    order by aeo.aeo_datetime desc
    rows 1
    into :old_aeo_status;

    if ((old_aeo_status is null and
        aeo_status = -1) or (:old_aeo_status = :aeo_status)) then
    begin
    end
    else
      insert into agreement_eds_original (aeo_from_ag_id, aeo_status, aeo_from_un_id)
      values (:ag_id_tmp, :aeo_status, :ag_from_un_id);
  end

  ag_id = coalesce(ag_id, 0);
  suspend;
end
