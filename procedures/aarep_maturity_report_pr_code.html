CREATE PROCEDURE AAREP_MATURITY_REPORT_PR_CODE(
    REP_DATE DATE)
RETURNS (
    YEAR_NO INTEGER,
    MONTH_NO INTEGER,
    DIFF_NO INTEGER,
    PLAN_OS DECIMAL(18,4),
    K_BALANCE DECIMAL(18,4),
    PLAN_OS_ST DECIMAL(18,4),
    K_BALANCE_ST DECIMAL(18,4),
    PLAN_OS_EMP DECIMAL(18,4),
    K_BALANCE_EMP DECIMAL(18,4),
    PLAN_OS_IPF DECIMAL(18,4),
    K_BALANCE_IPF DECIMAL(18,4),
    REPAY_NAME VARCHAR(50),
    MONTH_END DATE)
AS
declare variable tmp_balance decimal(18,4);
declare variable tmp_balance_st decimal(18,4);
declare variable tmp_balance_emp decimal(18,4);
declare variable tmp_balance_ipf decimal(18,4);
begin
  for select rep_year, rep_month, rep_line, sum(sh_os), sum(kont_balance),
             sum(iif(pr_code not containing '.IPF.' and pr_product_kind <> 4, sh_os, 0)),
             sum(iif(pr_code not containing '.IPF.' and pr_product_kind <> 4, kont_balance, 0)),
             sum(iif(pr_code not containing '.IPF.' and pr_product_kind = 4, sh_os, 0)),
             sum(iif(pr_code not containing '.IPF.' and pr_product_kind = 4, kont_balance, 0)),
             sum(iif(pr_code containing '.IPF.', sh_os, 0)), sum(iif(pr_code containing '.IPF.', kont_balance, 0))

      from aarep_maturity_rep_tmp(:rep_date)
      group by rep_year, rep_month, rep_line
      order by rep_line
      into :year_no, :month_no, :diff_no, :plan_os, :tmp_balance, :plan_os_st, --st - standart
           :tmp_balance_st, :plan_os_emp, :tmp_balance_emp, :plan_os_ipf, :tmp_balance_ipf

  do
  begin
    if (diff_no = -1) then
    begin
      k_balance = tmp_balance;
      k_balance_st = tmp_balance_st;
      k_balance_emp = tmp_balance_emp;
      k_balance_ipf = tmp_balance_ipf;
    end
    else
    begin
      k_balance = k_balance - plan_os;
      k_balance_st = k_balance_st - plan_os_st;
      k_balance_emp = k_balance_emp - plan_os_emp;
      k_balance_ipf = k_balance_ipf - plan_os_ipf;
    end

    if (diff_no = -1) then
      repay_name = 'Показатели на начало';
    if (diff_no = 0) then
      repay_name = 'Погашение просроченных';
    if (diff_no > 0) then
      repay_name = 'Через ' || diff_no || ' мес.';

    if (month_no < 12) then
      month_end = cast(('01.' ||(month_no + 1) || '.' || year_no) as date) - 1;
    else
      month_end = cast(('31.12.' || year_no) as date);
    suspend;
  end
end
