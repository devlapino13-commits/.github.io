CREATE PROCEDURE AAREP_PKB_PENDING(
    REP_REGION_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    C_NAME_FIO D_STR_255,
    KONTRACT_NO D_STR_100,
    REPORT_DATE D_DATE,
    PENDING_CNT D_INTEGER)
AS
declare variable C_ID D_ID;
declare variable OBJECT_ID D_ID;
declare variable BLS_ID D_ID;
declare variable CREATED D_STR_10;
begin
  for select 
        eil.eil_object_id,
        eil.eil_from_c_id,
        cast(eil.eil_status_datetime as date)
      from extinfolist eil
      inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
      inner join miniapp ma on ma.ma_id = eil.eil_object_id
      where eic.eic_code = 'pkb_cib_report' 
        and eil.eil_status = 2 
        and cast(eil.eil_status_datetime as date) between :rep_date_begin and :rep_date_end
        and (:rep_region_id is null or ma.ma_office_id in (select id_list from region_list(:rep_region_id, 1, 3, 1)))
        
      order by eil.eil_status_datetime desc
  into :object_id, :c_id, :report_date
  do
  begin
    select cl.c_name_fio
    from clients cl
    where cl.c_id = :c_id
    into :c_name_fio;

    kontract_no = null;
    select max(cast(kn.k_kontract_no as varchar(100)))
    from kontract kn
    where kn.k_from_ma_app_id = :object_id
    into :kontract_no;

    pending_cnt = 0;

    select pmi.pmi_bls_id
    from pkb_miniapp_info pmi
    where pmi.pmi_from_ma_id = :object_id
      and pmi.pmi_is_sent = 1
    order by pmi.pmi_last_datetime desc rows 1
    into :bls_id;

    if (bls_id is not null) then
    begin
      select total_created_miniapp
      from hp_parse_pkb_miniapp_info(:bls_id)
      into :created;

      if (created is not null and created <> '') then
        pending_cnt = cast(created as d_integer);
    end
    
    suspend;
  end
end
