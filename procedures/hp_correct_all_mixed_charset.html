CREATE PROCEDURE HP_CORRECT_ALL_MIXED_CHARSET
RETURNS (
    C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(100),
    C_PASS_NO VARCHAR(30),
    RESULT_FIO VARCHAR(100),
    RESULT_PASS_NO VARCHAR(100),
    ERR_FIO INTEGER,
    ERR_PASS_NO INTEGER,
    DB_CHANGED INTEGER,
    CHANGED_KOL INTEGER)
AS
declare variable err_txt varchar(255);
declare variable res_is_changed integer;
declare variable m_id decimal(18,0);
declare variable m_number integer;
begin
  changed_kol = 0;
  for select c_id, c_name_fio, c_pass_no
      from clients
      where c_type = 0
      into :c_id, :c_name_fio, :c_pass_no
--  as cursor cur
  do
  begin
    c_name_fio = trim(coalesce(c_name_fio, ''));
    c_pass_no = trim(coalesce(c_pass_no, ''));
    select err_txt, result, res_is_changed
    from hp_check_mix_charset(:c_name_fio, 1, 2)
    into :err_txt, result_fio, res_is_changed;
    if (err_txt <> '') then
      err_fio = 1;
    else
      err_fio = 0;

    select err_txt, result, res_is_changed
    from hp_check_mix_charset(:c_pass_no, 1, 1)
    into :err_txt, result_pass_no, res_is_changed;
    if (err_txt <> '') then
      err_pass_no = 1;
    else
      err_pass_no = 0;

    if ((c_name_fio <> result_fio) or (c_pass_no <> result_pass_no) or (err_fio = 1) or (err_pass_no = 1)) then
    begin
      db_changed = 0;
      if (((c_name_fio <> result_fio) or (c_pass_no <> result_pass_no)) and
          ((err_fio = 0) and
          (err_pass_no = 0))) then
      begin
        update clients
        set c_name_fio = :result_fio,
            c_pass_no = :result_pass_no
        where c_id = :c_id;
--        where current of cur;
        db_changed = 1;
      end
      changed_kol = changed_kol + db_changed;
      if (db_changed = 0) then
        suspend;
    end
  end

  for select m_id, m_number, m_client_pass_no
      from moratory
      where coalesce(m_client_pass_no, '') <> ''
      into :m_id, :m_number, :c_pass_no
--  as cursor cur
  do
  begin
    c_pass_no = trim(coalesce(c_pass_no, ''));
    select err_txt, result, res_is_changed
    from hp_check_mix_charset(:c_pass_no, 1, 1)
    into :err_txt, result_pass_no, res_is_changed;
    if (err_txt <> '') then
      err_pass_no = 1;
    else
      err_pass_no = 0;

    if ((c_pass_no <> result_pass_no) or (err_pass_no = 1)) then
    begin
      db_changed = 0;
      if ((c_pass_no <> result_pass_no) and
          (err_pass_no = 0)) then
      begin
        update moratory
        set m_client_pass_no = :result_pass_no
        where m_id = :m_id and
              m_number = :m_number;
--        where current of cur;
        db_changed = 1;
      end
      changed_kol = changed_kol + db_changed;
      if (db_changed = 0) then
        suspend;
    end
  end

  suspend;
end
