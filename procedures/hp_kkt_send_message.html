CREATE PROCEDURE HP_KKT_SEND_MESSAGE(
    KKM_ID INTEGER)
AS
declare variable link d_str_255;
declare variable db_is_actual boolean;
declare variable c_id d_id;
declare variable cell_no d_str_code;
declare variable sms_text d_str_255;
declare variable msg_subject d_str_255;
declare variable msg_text d_str_long;
declare variable db_id d_integer;
declare variable err_mail d_str_code;
declare variable err_prefix d_str_10;
declare variable k_id d_id;
declare variable mcl_project d_integer;
declare variable msg_sender_code d_str_code;
begin
  -- Отправка чека через СМС клиенту
  -- Тестовые платежи не отправляем
  -- Ломаться нельзя, т.к. пробитый чек не сохранится
  db_is_actual = exists(select 1 from hp_db_is_actual);
  db_id = f_get_db_id();

  if (db_id = 81001 and db_is_actual and exists(
        select 1 from kkt_fn, kkt_main
        where kkm_id = :kkm_id
          and kkm_from_kkf_id = kkf_id
          and kkf_code in (9999078902012025, 9999078902011387, 9999078902015301)))
  then
    exit;

  if (db_id = 39801 and db_is_actual and exists(
          select 1 from kkt_devices, kkt_fn, kkt_main
          where kkf_from_kkd_id = kkd_id
            and kkm_from_kkf_id = kkf_id
            and kkm_id = :kkm_id
            and kkd_code = 'SWK00033830'))
  then
    exit;

  select kc_from_c_id, kc_from_k_id
  from knclient, kkt_main
  where kkm_id = :kkm_id
    and kc_from_k_id = kkm_from_k_id
    and kc_amount_approved > 0
    and kkm_duplicated_kkm_id is null
    and coalesce(kkm_send_to, '') = ''
  order by kc_status desc, kc_id desc
  rows 1
  into c_id, k_id;

  -- если продан по цессии, то чеки не формируем
  if (exists(select 1 from sud_cession_packet, sud_cession_loans
            where scl_from_scp_id = scp_id
              and scl_from_k_id = :k_id
              and scp_status > 0))
  then
    exit;

  if (c_id is not null) then
  begin
    if (cell_no = '') then
      cell_no = coalesce(f_cont_info(0, c_id, 'cell', 'ok'), '');

    if (cell_no = '') then
      cell_no = coalesce(f_cont_info(0, c_id, 'cell', 'last'), '');

    if (cell_no <> '') then
    begin
      select link from hp_kkt_get_ofd_link(:kkm_id)
      into link;
      sms_text = 'El. chek na %%: ' || link;

      mcl_project = f_get_mcl_project(f_objkind('kontract', 'id'), :k_id);
      msg_sender_code = f_enum_code('sms_sender', iif (:mcl_project = 2, 1, 0));

      f_send_sms(cell_no, sms_text, 'clients', c_id, 'kkt_check', 365, 1000000000000, 0, msg_sender_code);
    end
  end

when any do
  begin
    err_mail = trim(iif(db_id = 81001, 'ru.finstart@molbulak.com', 'finstart@mbfplanet.com'));
    err_prefix = case db_id
                   when 81001 then 'RU: '
                   when 39801 then 'KZ: '
                   when 41701 then 'KG: '
                   when 94401 then 'AZ: '
                   else ''
                 end;

    msg_subject = err_prefix || 'ККТ: Ошибка в hp_kkt_send_message.';
    msg_text  = 'Ошибка в hp_kkt_send_message: ' || 'gdscode: ' || gdscode || ascii_char(10) || ascii_char(13) ||
                'Параметры:' || ascii_char(10) || ascii_char(13) ||
                ' kkm_id = ' || coalesce(kkm_id, '<NULL>');
    in autonomous transaction do
      insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id, msg_command_kind)
      values (0, :err_mail, :msg_subject, :msg_text, 7, 1000000000001, '');
  end
end
