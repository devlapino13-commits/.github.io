CREATE PROCEDURE MC_DOCRECOG_CHECK(
    MCL_ID D_ID,
    INN D_STR_LONG,
    DOC_NO D_STR_LONG)
RETURNS (
    RES_ID D_INTEGER,
    RES_TEXT D_STR_LONG)
AS
declare variable c_id_by_phone d_id;
declare variable c_id_by_inn d_id;
declare variable phone d_str_20;
declare variable phone_by_inn d_str_long;
declare variable is_verified d_integer;
declare variable c_inn d_str_long;
declare variable c_doc_no d_str_long;
begin
  res_id = 0;
  res_text = '';

  select mcl.mcl_login
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into :phone;

  select c.c_id
  from clients c
  where c.c_inn = :inn
  into :c_id_by_inn;

  phone_by_inn = f_cont(:c_id_by_inn, 'cell', 'ok', 'main');

  select res_object_id, res_is_verified
  from hp_contdata_find_object(:phone, 'cell')
  where res_is_verified = 1
  into :c_id_by_phone, :is_verified;

  if (res_id = 0 and
      coalesce(c_id_by_phone, 0) <> 0) then
  begin
    select c.c_inn, c.c_pass_no
    from clients c
    where c.c_id = :c_id_by_phone
    into c_inn, c_doc_no;

    if (res_id = 0 and
        coalesce(c_inn, '') <> '') then
      if (c_inn <> inn) then
      begin
        res_id = 1;
        res_text = 'ИНН текущего документа паспорта отличается от ИНН ранее закрепленного паспорта. Обратитесь в офис компании';
      end

    if (res_id = 0 and
        coalesce(c_doc_no, '') <> '') then
      if (c_doc_no = doc_no) then
      begin
        res_id = 1;
        res_text = 'ID нового паспорта совпадает с ID старого. Скорее всего, этот паспорт уже добавлен.';
      end

    if (coalesce(:phone_by_inn, '') <> '') then
      if (res_id = 0 and
          phone <> phone_by_inn) then
      begin
        res_id = 1;
        res_text = 'Телефонный номер не соответствует ранее верифицированному. Рекомендуется пройти регистрацию с номером ' || phone_by_inn || ' или обратиться в офис компании';
      end
  end

  suspend;
end
