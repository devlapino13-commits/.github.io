CREATE PROCEDURE PRINT_SCHED_CLI_TOTAL(
    KONT_ID DECIMAL(18,0))
RETURNS (
    SH_OS DECIMAL(18,4),
    SH_PERC DECIMAL(18,4),
    SH_KOMMIS DECIMAL(18,4),
    SH_PERC_TAX DECIMAL(18,4),
    SH_OS_PERC DECIMAL(18,4),
    SH_PERC_KOMMIS DECIMAL(18,4),
    SH_NAME_FIO VARCHAR(100),
    SH_KOMIS_TAX DECIMAL(18,4),
    SH_TAX_TOTAL DECIMAL(18,4),
    SH_PERC_WITH_TAX DECIMAL(18,4),
    SH_KOMMIS_WITH_TAX DECIMAL(18,4),
    SH_PERC_KOMIS_WITH_TAX DECIMAL(18,4))
AS
declare variable total_os decimal(18,4); /* SH_NAME_FIO */
declare variable st_staff integer;
declare variable st_client integer;
declare variable cl_count integer;
declare variable total_perc decimal(18,4);
declare variable total_kommis decimal(18,4);
declare variable total_perc_tax decimal(18,4);
declare variable k_amount_approved decimal(18,4);
declare variable k_disb_date date;
declare variable k_nsp_percent decimal(18,7);
declare variable sh_perc_tax_1 decimal(18,4);
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  /* Получаем количество клиентов в контракте, для корректировки процентов на последнем клиенте */

  k_nsp_percent = 0;
  select k_amount_approved, k_disb_date
  from kontract
  where k_id = :kont_id
  into :k_amount_approved, :k_disb_date;

  if (exists(select *
             from nspconfig_data
             where nsd_from_nsm_id = 0 and
                   nsd_date_begin <= :k_disb_date and
                   nsd_date_end >= :k_disb_date)) then
  begin
    select nsd_percent
    from nspconfig_data
    where nsd_from_nsm_id = -1 -- нсп от процента
          and
          nsd_date_begin <= :k_disb_date and
          nsd_date_end >= :k_disb_date
    into :k_nsp_percent;
    if (k_nsp_percent is null) then
      k_nsp_percent = 0;
  end

  select count(*)
  from knclient
  where kc_amount_approved > 0 and
        kc_from_k_id = :kont_id
  into cl_count;
  /*Получаем общие суммы процента, доп. процента и комиссии, для корректировки*/
  select sum(sh_perc), sum(sh_kommis)
  from schedule
  where sh_from_k_id = :kont_id
  into :total_perc, :total_kommis;
  total_perc_tax = round((total_perc + total_kommis) * k_nsp_percent / 100);

  for select c_name_fio, k_amount_approved, kc_amount_approved
      from knclient, clients, kontract
      where kc_from_c_id = c_id and
            kc_from_k_id = :kont_id and
            kc_amount_approved <> 0 and
            k_id = :kont_id
      order by kc_number
      into :sh_name_fio, :total_os, :sh_os
  do
  begin
    total_os = coalesce(total_os, 0);
    sh_os = coalesce(sh_os, 0);
    select sum(sh_perc), sum(sh_kommis)
    from schedule
    where sh_from_k_id = :kont_id
    into :sh_perc, :sh_kommis;
    sh_perc = coalesce(sh_perc, 0);
    sh_kommis = coalesce(sh_kommis, 0);
--sh_perc_tax=coalesce(sh_perc_tax,0);
--sh_perc_tax=0;
--sh_perc_tax=round((sh_kommis+sh_perc_tax)*k_nsp_percent/100);
--sh_perc_tax_1=round((sh_kommis+sh_perc)*k_nsp_percent/100);
    if (cl_count <> 1) then
    begin
      sh_perc = cast(sh_perc * (sh_os / total_os) as decimal(18,2));
      sh_perc_tax = cast((sh_perc * k_nsp_percent / 100) as decimal(18,2));

      sh_kommis = cast(sh_kommis * (sh_os / total_os) as decimal(18,2));
      sh_komis_tax = cast((sh_kommis * k_nsp_percent / 100) as decimal(18,2));

      sh_tax_total = cast(((sh_kommis + sh_perc) * k_nsp_percent / 100) as decimal(18,2));

      sh_perc_with_tax = sh_perc + sh_perc_tax;
      sh_kommis_with_tax = sh_kommis + sh_komis_tax;
      sh_perc_komis_with_tax = sh_perc_with_tax + sh_kommis_with_tax;
--sh_perc_tax=Cast(sh_perc_tax_1*(sh_os/total_os) as decimal(18,2));
      total_perc = total_perc - sh_perc;
      total_kommis = total_kommis - sh_kommis;
      total_perc_tax = total_perc_tax - sh_tax_total;
    end
    else
    /* Последний клиент, корректируем сумму процентов */
    begin
      sh_perc = total_perc;
      sh_kommis = total_kommis;
      sh_perc_tax = total_perc_tax;
    end
    sh_os_perc = coalesce(sh_os, 0) + coalesce(sh_perc, 0);
    sh_perc_kommis = coalesce(sh_perc, 0) + coalesce(sh_kommis, 0);
    suspend;
--cl_count=cl_count-1;
  end
end
