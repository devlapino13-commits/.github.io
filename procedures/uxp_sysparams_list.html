CREATE PROCEDURE UXP$SYSPARAMS_LIST
RETURNS (
    UFP_CODE D_STR_CODE,
    UFP_UUID D_UUID,
    UFP_UFR_UUID D_UUID,
    UFP_UF_UUID D_UUID,
    UF_IS_PARAM D_BOOL,
    UF_PK_FIELD D_SMALLINT,
    UF_POSITION D_INTEGER,
    UF_DOMAIN D_STR_CODE,
    UF_DATA_TYPE D_STR_CODE,
    UF_NOT_NULL D_BOOL,
    UF_DEFAULT D_STR_100,
    UF_NAME D_STR_100,
    UFP_REFRESH D_BOOL,
    UFP_NOT_NULL D_BOOL,
    UFP_EDITOR D_STR_CODE,
    UFP_DEFAULT D_STR_100,
    UFP_LOCKED D_BOOL,
    RES_FIELD_NAME D_STR_100,
    RES_IN_USE D_BOOL,
    RES_EDITOR_INFO D_STR_LONG,
    RES_CAPTION D_STR_100,
    RES_UCM_CODE D_STR_100,
    RES_UCM_IS_OWN D_BOOL,
    RES_BUTTONEDIT_UXA D_STR_100,
    RES_INDEX_TXT D_STR_CODE,
    RES_FIELD_KIND D_STR_10,
    RES_IN_UI D_SMALLINT,
    UEX_EXPRESSION D_STR_4096,
    RES_CATEGORY D_STR_255,
    RES_IS_FK D_BOOL,
    UFP_PUBLIC D_BOOL,
    RES_FK_TABLES D_STR_LONG,
    UF_EXPRESSION D_STR_LONG,
    EDI_SOURCE D_STR_CODE,
    EDI_FIELD_NAME D_STR_CODE,
    EDI_INDEX D_INTEGER,
    EDI_USE_LAST D_BOOL,
    UMA_MASK D_STR_LONG,
    UMA_MESSAGE D_STR_255)
AS
declare variable us_code d_str_code;
declare variable us_uuid d_uuid;
begin
  -- Список параметров для добавления в uxFrItem.

  us_uuid = uxf$sys_id();
  us_code = uxf$code(us_uuid, 'code');

  for select ufp_code, ufp_uuid, ufp_ufr_uuid, ufp_uf_uuid, uf_is_param, uf_pk_field, uf_position, uf_domain,
             uf_data_type, uf_not_null, uf_default, uf_name, ufp_refresh, ufp_not_null, ufp_editor, ufp_default,
             ufp_locked, res_field_name, res_in_use, res_editor_info, res_caption, res_ucm_code, res_ucm_is_own,
             res_buttonedit_uxa, res_index_txt, res_field_kind, res_in_ui, uex_expression, res_category,
             res_is_fk, ufp_public, uma_mask, uma_message
      from uxp$uxfrparam_list(:us_uuid, null)
      into :ufp_code, :ufp_uuid, :ufp_ufr_uuid, :ufp_uf_uuid, :uf_is_param, :uf_pk_field, :uf_position, :uf_domain,
           :uf_data_type, :uf_not_null, :uf_default, :uf_name, :ufp_refresh, :ufp_not_null, :ufp_editor, :ufp_default,
           :ufp_locked, :res_field_name, :res_in_use, :res_editor_info, :res_caption, :res_ucm_code, :res_ucm_is_own,
           :res_buttonedit_uxa, :res_index_txt, :res_field_kind, :res_in_ui, :uex_expression, :res_category,
           :res_is_fk, :ufp_public, :uma_mask, :uma_message
  do
  begin
    res_fk_tables = '';
    select list(uix_fk_source || iif(uix_virtual = 1, ' [V]', ''), ', ')
    from ux$index
    where uix_source = :us_code and
          uix_fk = 1 and
          uix_fields || ';' containing :res_field_name || ';'
    into res_fk_tables;
  
    if (res_fk_tables is null) then
      res_fk_tables = '';

    select exp_expression_all
    from uxp$uxexpression_text(:ufp_uf_uuid)
    into uf_expression;

    edi_source = '';
    edi_field_name = '';
    edi_index = null;
    edi_use_last = 0;

    select edi_source, edi_field_name, edi_index, edi_use_last
    from ux$editor
    where edi_uuid = :ufp_uuid
    into :edi_source, :edi_field_name, :edi_index, edi_use_last;

    if (uf_domain = 'd_date') then
      uf_data_type = 'date';

    suspend;
  end
end
