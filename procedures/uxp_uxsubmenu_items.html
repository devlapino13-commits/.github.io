CREATE PROCEDURE UXP$UXSUBMENU_ITEMS(
    UX_UUID D_UUID,
    UXA_UUID D_UUID,
    HIDE_EMPTY D_BOOL = 0,
    SHOW_ROOT D_BOOL = 0,
    UX_UUID_CODE D_STR_CODE = null,
    UXA_UUID_CODE D_STR_CODE = null)
RETURNS (
    RES_CODE D_STR_20,
    RES_PARENT_CODE D_STR_20,
    RES_CAPTION D_STR_100,
    RES_HINT D_STR_255,
    RES_UA_UUID D_UUID,
    RES_UXA_UUID D_UUID,
    RES_INACTIVE D_BOOL,
    RES_CHECKED D_BOOL,
    RES_IMAGE D_INTEGER,
    RES_DISABLE_ON_EDIT D_BOOL,
    RES_IS_DEFAULT D_BOOL,
    RES_IMG_UUID_CODE D_STR_CODE,
    RES_IS_REPORT D_BOOL,
    RES_IS_COMMAND D_BOOL,
    RES_IS_INSERT D_BOOL,
    RES_UXA_OWNER D_UUID,
    RES_DATA_LINKED D_BOOL,
    RES_SOFT_LINKED D_BOOL,
    RES_HOTKEY D_STR_CODE,
    RES_ADD_DELIMITER D_SMALLINT,
    RES_IS_GROUP D_BLOB_BINARY,
    RES_ALIAS_GROUP D_STR_8100,
    RES_UXA_FCODE D_STR_100,
    RES_LINKED_FRAMES D_STR_LONG,
    RES_TRIGGER_CODE D_STR_CODE,
    RES_IS_BLANK D_BOOL)
AS
declare variable res_order_code d_str_20;
declare variable ux_kind d_str_code;
declare variable uxa_kind d_integer;
declare variable ua_usr_id_field d_str_code;
declare variable item_count d_integer;
declare variable selected_count d_integer;
declare variable ua_trigger_kind d_integer;
declare variable blank_position d_integer;
declare variable uap_locked d_bool;
begin
  if (ux_uuid is null) then
    ux_uuid = uxf$uniq(ux_uuid_code);

  if (uxa_uuid is null) then
    uxa_uuid = uxf$uniq(uxa_uuid_code);

  if (not exists(select * from ux$action
                 where ua_owner = :ux_uuid and
                       (ua_usr_id_field = '' or ua_preblank_kind <> 0))) then
    exit;

  for select 'a_00', 'a_', null, 'Действия', 0, 0, '', 0, '', '', null, null, null, '', 0, 0, '', '', 0, 0
  from rdb$database
  union select 'b_00', 'b_', null, 'Отчеты', 0, 0, '', 0, '', '', null, null, null, '', 0, 0, '', '', 0, 0
  from rdb$database
  union
  select usm_menu_index || '00', usm_menu_index, null, usm_caption, 0, 0, '', 0, '', '', null, null, null, '', 0, 0, '', '', 0, 0
  from ux$submenu
  where usm_ufm_uuid = :ux_uuid
  union
  select ua_menu_index, ua_menu_index, ua_uuid, ua_name, ua_inactive, uxa_kind,
         uu_kind, ua_disable_on_edit, ua_hint, ua_usr_id_field, ua_uxa_uuid, uxa_owner, uuid_to_char(ua_img_uuid),
         ua_hotkey, ua_add_delimiter, ua_is_group, '', ua_uxa_fcode, ua_trigger_kind, iif(ua_preblank_kind <> 0, 1, 0)
  from ux$action, ux$alias, ux$code, ux$uniq
  where ua_owner = :ux_uuid and
        ua_uxa_uuid = uxa_uuid and
        uxa_owner = ux_uuid and
        ux_uuid = uu_id and
        ua_is_group = 0 and
        ua_usr_id_field = '' and
        ua_preblank_kind = 0
  union
  select ua_menu_index, ua_menu_index, ua_uuid, ua_name, ua_inactive, -1,
         -1, ua_disable_on_edit, ua_hint, ua_usr_id_field, ua_uxa_uuid, null, uuid_to_char(ua_img_uuid),
         ua_hotkey, ua_add_delimiter, ua_is_group, ua_alias_group, '', 0, 0
  from ux$action
  where ua_owner = :ux_uuid and
        ua_uxa_uuid is null and
        ua_is_group = 1 and
        ua_usr_id_field = ''
  order by 1
  into :res_order_code, :res_code, res_ua_uuid, res_caption, :res_inactive, uxa_kind,
       ux_kind, res_disable_on_edit, res_hint, ua_usr_id_field, res_uxa_uuid, res_uxa_owner, res_img_uuid_code,
       res_hotkey, res_add_delimiter, res_is_group, res_alias_group, res_uxa_fcode, ua_trigger_kind, res_is_blank
  do
  begin
    res_checked = 1;
    res_parent_code = null;
    res_is_default = 0;
    res_data_linked = 0;
    res_soft_linked = 1;
    res_linked_frames = '';
    res_is_command = 0;
    res_is_insert = 0;
    res_trigger_code = '';

    if (ua_trigger_kind <> 0) then
      res_trigger_code = uxf$enum_code('uxaction_trigger_kind', ua_trigger_kind);

    if (res_alias_group is null) then
      res_alias_group = '';

    if (res_ua_uuid = '') then
      res_ua_uuid = null;

    if (res_ua_uuid is null) then
    begin
      if (show_root = 0 and res_code in ('a_', 'b_')) then
        continue;

      if (not(show_root = 1 and res_code in ('a_', 'b_'))) then
      begin
        if (not exists(select * from ux$action
        where ua_owner = :ux_uuid and
              ua_menu_index starting :res_code and
              ua_menu_index <> :res_code)) then
          Continue;
      end

      if (uxa_uuid is not null) then
      begin
        if (not exists(select * from ux$action, ux$aliasaction
        where ua_owner = :ux_uuid and
              ua_uuid = uaa_ua_uuid and
              uaa_uxa_uuid = :uxa_uuid and
              ua_menu_index starting :res_code and
              ua_menu_index <> :res_code)) then
        begin
          if (hide_empty = 1) then
            continue;
          else
            res_checked = 0;
        end
      end

      select count(*)
      from ux$action
      where ua_owner = :ux_uuid and
            ua_menu_index starting :res_code
      into item_count;

      select count(*)
      from ux$aliasaction, ux$action
      where uaa_uxa_uuid = :uxa_uuid and
            ua_uuid = uaa_ua_uuid and
            ua_menu_index starting :res_code
      into selected_count;

      res_checked = 0;
      if (item_count > 0) then
      begin
        if (item_count = selected_count) then
          res_checked = 1;
        else
        if (selected_count = 0) then
          res_checked = 0;
        else
          res_checked = null;
      end

      if (char_length(res_code) > 4) then
        res_parent_code = left(res_code, char_length(res_code) - 2) || '00';

      res_code = res_order_code;

      res_image = 36; -- папка
    end else
    begin
      if (char_length(res_code) > 4) then
        res_parent_code = left(res_code, char_length(res_code) - 2) || '00';

      if (uxa_uuid is not null) then
      begin
        select uaa_is_default
        from ux$aliasaction
        where uaa_uxa_uuid = :uxa_uuid and
              uaa_ua_uuid = :res_ua_uuid
        into res_is_default;

        if (row_count = 0) then
        begin
          if (hide_empty = 1) then
            continue;
          else
            res_checked = 0;
        end
      end

      ux_kind = uxf$kind(ux_kind, 'code');

      res_image = 19; -- Форма

      res_image = case
        when (ux_kind = 'uxframe' and ua_usr_id_field <> '') then 4 -- поисковик
        when (ux_kind = 'uxframe' and ua_usr_id_field = '') then 17 -- Фрейм
        when (ux_kind = 'uxform' and uxa_kind = 0) then 19 -- Форма
        when (ux_kind = 'uxform' and uxa_kind = 2) then 55 -- Создание записи
        when ux_kind = 'uxcommand' then 25 -- Команда
        when ux_kind = 'uxreport' then 26 -- Команда
      end;

      if (ux_kind = 'uxcommand') then
        res_is_command = 1;

      if (uxa_kind = 2) then
        res_is_insert = 1;

      select list(distinct uuid_to_char(ufp_ufr_uuid)), max(uap_locked)
      from ux$actionparam, ux$frparam, ux$field
      where uap_ua_uuid = :res_ua_uuid and
            uap_owner_ufp_uuid = ufp_uuid and
            uap_is_output = 0 and
            ufp_uf_uuid = uf_uuid and
            uf_is_param = 0
      into res_linked_frames, uap_locked;

      if (uap_locked is null) then
        uap_locked = 0;

      if (coalesce(res_linked_frames, '') <> '') then
      begin
        res_data_linked = 1;

        if (uap_locked = 1) then
          res_soft_linked = 0;
      end
    end

    if (res_is_group = 1) then
      res_image = 139; -- Группа

    if (show_root = 1 and res_parent_code is null) then
      res_parent_code = left(res_code, 2) || '00';

    if (res_code in ('a_00', 'b_00')) then
      res_parent_code = null;

    res_is_report = 0;
    if (res_code starting 'b_') then
      res_is_report = 1;

    suspend;
  end

  if (show_root = 0) then -- Список бланков
  begin
    res_parent_code = null;
    res_is_report = 0;
    res_is_command = 0;
    res_is_insert = 0;
    res_is_group = 0;
    res_alias_group = '';
    res_is_blank = 1;
    blank_position = 0;

    for select ua_menu_index, ua_menu_index, ua_uuid, ua_name, ua_inactive, uxa_kind,
           uu_kind, ua_disable_on_edit, ua_hint, ua_usr_id_field, ua_uxa_uuid, uxa_owner, uuid_to_char(ua_img_uuid),
           ua_hotkey, ua_add_delimiter, ua_uxa_fcode, ua_trigger_kind
    from ux$action, ux$alias, ux$code, ux$uniq
    where ua_owner = :ux_uuid and
          ua_uxa_uuid = uxa_uuid and
          uxa_owner = ux_uuid and
          ux_uuid = uu_id and
          ua_preblank_kind <> 0
    order by 1
    into :res_order_code, :res_code, res_ua_uuid, res_caption, :res_inactive, uxa_kind,
         ux_kind, res_disable_on_edit, res_hint, ua_usr_id_field, res_uxa_uuid, res_uxa_owner, res_img_uuid_code,
         res_hotkey, res_add_delimiter, res_uxa_fcode, ua_trigger_kind
    do
    begin
      res_checked = 1;
      res_is_default = 0;
      res_data_linked = 0;
      res_linked_frames = '';
      res_soft_linked = 1;
      res_is_command = 0;
      res_is_insert = 0;
      res_trigger_code = '';

      blank_position = blank_position + 1;
      res_order_code = 'c_' || lpad(blank_position, 2, '0');
      res_code = res_order_code;

      if (ua_trigger_kind <> 0) then
        res_trigger_code = uxf$enum_code('uxaction_trigger_kind', ua_trigger_kind);
  
      ux_kind = uxf$kind(ux_kind, 'code');

      res_image = 19; -- Форма

      res_image = case
        when (ux_kind = 'uxframe' and ua_usr_id_field <> '') then 4 -- поисковик
        when (ux_kind = 'uxframe' and ua_usr_id_field = '') then 17 -- Фрейм
        when (ux_kind = 'uxform' and uxa_kind = 0) then 19 -- Форма
        when (ux_kind = 'uxform' and uxa_kind = 2) then 55 -- Создание записи
        when ux_kind = 'uxcommand' then 25 -- Команда
        when ux_kind = 'uxreport' then 26 -- Команда
      end;

      if (ux_kind = 'uxcommand') then
        res_is_command = 1;

      if (uxa_kind = 2) then
        res_is_insert = 1;

      select list(distinct uuid_to_char(ufp_ufr_uuid)), max(uap_locked)
      from ux$actionparam, ux$frparam, ux$field
      where uap_ua_uuid = :res_ua_uuid and
            uap_owner_ufp_uuid = ufp_uuid and
            uap_is_output = 0 and
            ufp_uf_uuid = uf_uuid and
            uf_is_param = 0
      into res_linked_frames, uap_locked;

      if (coalesce(res_linked_frames, '') <> '') then
      begin
        res_data_linked = 1;

        if (uap_locked = 1) then
          res_soft_linked = 0;
      end

      if (res_is_group = 1) then
        res_image = 139; -- Группа
  
      suspend;
    end
  end
end
