CREATE PROCEDURE AAREP_REPAYMENTS_DISB
RETURNS (
    REG_NAME D_STR_CODE,
    K_FACT_DISB_DATE D_DATE,
    K_KONTRACT_NO D_STR_20,
    K_AMOUNT_APPROVED D_CURRENCY,
    PRIZNAK D_STR_10)
AS
declare variable K_ID D_ID;
declare variable PAYED_AMMOUNT D_CURRENCY;
declare variable K_FACT_CLOSED_DATE D_DATE;
declare variable REP_DATE D_DATE;
begin
    rep_date = current_date;

    for select r.reg_name,
               ks.ks_date_begin,
               k.k_kontract_no,
               k.k_amount_approved,
               k.k_id,
               k.k_fact_closed_date
    from kontstate ks
    inner join kontract k on k.k_id = ks.ks_from_k_id
    inner join region r on r.reg_unicode = ks.ks_region_id
    where ks.ks_date_begin = :rep_date - 1
    and ks.ks_ol_is_delete = 0
    and ks.ks_state  = 1
    into :reg_name,
         :k_fact_disb_date,
         :k_kontract_no,
         :k_amount_approved,
         :k_id,
         :k_fact_closed_date
    do
    begin
       payed_ammount = null;

       select -sum(ok.ok_os + ok_perc)
       from operkred ok
       where ok.ok_type = 4
       and ok.ok_ol_is_active = 1
       and ok.ok_ol_date = :k_fact_disb_date
       and ok.ok_from_k_id = :k_id
       into :payed_ammount;

       payed_ammount = coalesce(payed_ammount, 0);

       if (payed_ammount <> 0) then
       begin
           priznak = 'ЧДП';

           if (k_fact_closed_date = k_fact_disb_date) then
             priznak = 'ПДП';

           suspend;
       end
    end
end
