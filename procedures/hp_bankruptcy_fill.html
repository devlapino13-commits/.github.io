CREATE PROCEDURE HP_BANKRUPTCY_FILL(
    REP_DATE D_DATE)
AS
declare variable K_ID D_ID;
declare variable C_ID D_ID;
declare variable EIC_ID D_ID;
declare variable IS_BANKRUPTCY D_INTEGER;
begin
  eic_id = (select eic.eic_id from extinfocfg eic where eic.eic_code = 'bankruptcy');

  for select kc.kc_from_k_id,
             kc.kc_from_c_id
      from kontstate ks
      inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
      where ks.ks_ol_is_delete = 0
      and ks.ks_state = 1
      and :rep_date between ks.ks_date_begin and ks.ks_date_end
      and kc.kc_amount_approved > 0
      into k_id,
           c_id
  do
  begin
    select is_bankruptcy
    from get_bankruptcy_status(:c_id)
    into is_bankruptcy;

    if (:is_bankruptcy = 0) then
    begin
        insert into extinfolist (eil_from_c_id, eil_from_eic_id, eil_date, eil_object_kind, eil_object_id, eil_ext_id, eil_is_json, eil_from_bls_id, eil_status, eil_create_datetime, eil_status_datetime, eil_kk_memo_id, eil_kk_un_id, eil_kk_result, eil_kk_datetime, eil_from_eip_id, eil_from_un_id, eil_from_mcl_id)
        values (:c_id, :eic_id, current_date, 9, :k_id, '', 1, null, 0, current_timestamp, current_timestamp, 0, null, 0, null, null, 1000000000000, null);
    end
  end

  post_event 'bankruptcy';
end
