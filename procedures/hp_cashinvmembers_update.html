CREATE PROCEDURE HP_CASHINVMEMBERS_UPDATE(
    CIL_ID D_ID,
    ROLE_KIND D_STR_CODE,
    C_ID D_ID,
    IS_FACT D_BOOL,
    UN_ID D_ID)
AS
begin
  role_kind = f_enum_id('cashmemberrole', role_kind);

  if (role_kind <=1) then
  begin
    update or insert into cashinvmembers (cim_from_cil_id, cim_is_fact, cim_role_kind, cim_from_c_id,
                                          cim_from_un_id)
    values (:cil_id, :is_fact, :role_kind, :c_id, :un_id)
    matching (cim_from_cil_id, cim_is_fact, cim_role_kind);
  end else
  begin
    update or insert into cashinvmembers (cim_from_cil_id, cim_is_fact, cim_role_kind, cim_from_c_id,
                                          cim_from_un_id)
    values (:cil_id, :is_fact, :role_kind, :c_id, :un_id)
    matching (cim_from_cil_id, cim_is_fact, cim_from_c_id);
  end
end
