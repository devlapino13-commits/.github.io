CREATE PROCEDURE HP_AKB_SET_POCKET_ERR(
    POCKET_ID D_INTEGER,
    POCKET_K_ID D_ID,
    SEND_STATUS D_STR_100,
    POCKET_ERR_CODE D_STR_255)
AS
declare variable mp_id d_id;
declare variable k_id d_id;
declare variable kc_num d_integer;
declare variable lp_number d_integer;
declare variable account_no d_str_255;
begin
  if (f_get_db_id() <> 94401) then
    exit;

  if (send_status <> 'err') then
    exit;

  lp_number = 0;
  for select mp.mp_id, mc.mc_from_k_id, kc.kc_number, mc.mc_accountno
      from mkr_poket mp
      inner join mkr_credit mc on mp.mp_id = mc.mc_from_mp_id
      inner join knclient kc on mc.mc_from_k_id = kc.kc_from_k_id
            and kc.kc_amount_approved > 0
      where mp.mp_id = :pocket_id and
            mc.mc_from_k_id = :pocket_k_id
      into :mp_id, :k_id, :kc_num, :account_no
      do
      begin
        lp_number = lp_number + 1;
        begin
            insert into mkr_err_log_credits (lc_id, lc_from_poket_id, lc_from_lp_number, lc_accountno, lc_k_id, lc_knc_number, lc_error_code)
            values (null, :mp_id, :lp_number, :account_no, :k_id, :kc_num, :pocket_err_code);

            update mkr_credit mc
            set mc.mc_is_err = 1
            where mc.mc_from_mp_id = :mp_id and
                  mc.mc_from_k_id = :k_id;
        end
      end
end
