CREATE PROCEDURE HP_SUD_CREATE_EDOC(
    SLD_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable scd_code d_str_code;
  declare variable scd_name d_str_255;
  declare variable claim_date d_date;
  declare variable pr_code d_str_code;
  declare variable smr_id d_id;
  declare variable edl_id d_id;
  declare variable edb_id d_id;
  declare variable smd_proc_type d_smallint;
  declare variable smd_sud_sag_id d_id;
  declare variable smd_respons_sag_id d_id;
begin
  if (sld_id is null) then
    exit;

  res_text = '';

  select pr_code, smd_claim_date, scd_code, scd_name, smr_id, smd_proc_type, smd_sud_sag_id, smd_respons_sag_id
  from sud_loandocs, sud_cfgdockind, sud_maindocs, sud_mainreq, kontract, product
  where sld_id = :sld_id
    and scd_id = sld_doc_kind
    and smd_id = sld_from_smd_id
    and smr_id = smd_id
    and k_id = smr_from_k_id
    and pr_id = k_from_pr_id
  into pr_code, claim_date, scd_code, scd_name, smr_id, smd_proc_type, smd_sud_sag_id, smd_respons_sag_id;

  if (smr_id is null) then
  begin
    res_text = f_format('Не найдена заявка для sld_id = ', sld_id);
    suspend;
    exit;
  end

  if (smd_proc_type is null) then
  begin
    res_text = 'Выберите тип делопроизводства!';
    suspend;
    exit;
  end

  if (smd_sud_sag_id is null) then
  begin
    res_text = 'Суд не выбран!';
    suspend;
    exit;
  end

  if (smd_respons_sag_id is null) then
  begin
    res_text = 'Назначте ответственного!';
    suspend;
    exit;
  end

  if (scd_code = 'court_app') then
    begin
      if (pr_code containing '.zalog' and claim_date is null) then
      begin
        res_text = 'Незаполнена дата отправки претензии';
        suspend;
        exit;
      end

      select edl_id, res_text
      from hp_edoc_ins('sud.app', f_objkind('sud_mainreq', 'id'), :smr_id, :un_id)
      into edl_id, res_text;
  
      if (coalesce(res_text, '') = '') then
      begin
        -- найдем бланк заявления
        edb_id = null;
        select edb_id
        from edoc_blank, edoc_frx
        where edb_from_edl_id = :edl_id
          and edb_from_edf_id = edf_id
          and edf_code = 'sud.app'
        into edb_id;
  
        if (edb_id is null) then
          exception e_default 'Not found edb_id for edl_id = ' || coalesce(edb_id,'<NULL>');
  
        insert into sud_loanbls (slb_from_sld_id, slb_object_kind, slb_object_id, slb_from_un_id)
        values (:sld_id, f_objkind('edoc_blank', 'id'), :edb_id, :un_id);
  
        -- найдем бланк расчета задолженности
        edb_id = null;
        select edb_id
        from edoc_blank, edoc_frx
        where edb_from_edl_id = :edl_id
          and edb_from_edf_id = edf_id
          and edf_code = 'sud.debt_calc'
        into edb_id;
  
        if (edb_id is null) then
          exception e_default 'Not found edb_id for edl_id = ' || coalesce(edb_id,'<NULL>');

        -- найдем расчет среди документов
        select res_sld_id
        from sud_cfgdockind, hp_sud_loandocs_add(:smr_id, null, scd_id, :un_id)
        where scd_code = 'debt_calculation'
        into :sld_id;
  
        insert into sud_loanbls (slb_from_sld_id, slb_object_kind, slb_object_id, slb_from_un_id)
        values (:sld_id, f_objkind('edoc_blank', 'id'), :edb_id, :un_id);
      end
    end
    else
      res_text = f_format('Для документа "%1" не предусмотрена отправка электронного документа.', scd_name);

  suspend;
end
