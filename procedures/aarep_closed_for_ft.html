CREATE PROCEDURE AAREP_CLOSED_FOR_FT(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    "ФИО клиента" D_STR_255,
    "Статус" D_STR_100)
AS
declare variable c_id id_type;
declare variable k_id d_id;
declare variable debt curr_type;
declare variable blob_c_list d_blob_text;
declare variable counter d_integer;
declare variable edl_date_end d_date;
begin
  c_id = null;
  blob_c_list = '';
  counter = 0;

  for select distinct k_id, c_id 
  from
  (
      -- берем закрытые кредиты за период
      select ks.ks_from_k_id as k_id,
             kncl.kc_from_c_id as c_id
      from kontstate ks
      inner join knclient kncl on kncl.kc_from_k_id = ks.ks_from_k_id
      where ks.ks_ol_is_delete = 0 and
            ks.ks_state = 2 and
            ks.ks_date_begin between :rep_date_begin and :rep_date_end and
            kncl.kc_amount_approved > 0
      union all
      -- берем все платежи, списания по внесистемке за период
      select okf.oko_from_k_id, okf.oko_from_c_id
      from operkroff okf
      where okf.oko_ol_is_active = 1 and
            okf.oko_from_k_id = :k_id and
            okf.oko_ol_date between :rep_date_begin and :rep_date_end and
            okf.oko_type in (4, 5) and
            okf.oko_os + okf.oko_sh_os + okf.oko_perc + okf.oko_sh_perc + oko_other + oko_sh_other <> 0
  ) A
  into :k_id, :c_id
  do
  begin
    debt = null;

    --вычисляем остаток, на внестистемке, на конец периода
    select sum(okf.oko_os + okf.oko_sh_os + okf.oko_perc + okf.oko_sh_perc + oko_other + oko_sh_other)
    from operkroff okf
    inner join knclient kncl on kncl.kc_from_k_id = okf.oko_from_k_id
    where okf.oko_ol_is_active = 1 and
          okf.oko_from_k_id = :k_id and
          okf.oko_ol_date <= :rep_date_end and
          kncl.kc_amount_approved > 0 and
          kncl.kc_from_c_id = :c_id
    into :debt;

    if (debt is null or debt <= 0) then --берем всех клиентов у которых нет долга на внесистмке
    begin
      if (debt is null) then
        "Статус" = 'Закрыл займ';

      if (debt <= 0) then
         "Статус" = 'Закрыл долг на внесист';

      if (F_FT_VALID(:c_id, :rep_date_end) = 0) then  -- если нет активного договра АСП (Активным счиатется АСП, который был подписан либо когда у клиента был активный заем либо если после подписания АСП клиент брал заем.
      begin
          if (not exists( -- если нет активного займа
              select ks_from_k_id
              from knclient, kontstate
              where kc_from_c_id = :c_id
              and kc_amount_approved > 0
              and kc_from_k_id = ks_from_k_id
              and ks_ol_is_delete = 0
              and ks_state = 1
              and :rep_date_end between ks_date_begin and  ks_date_end
              union
              select oko_from_k_id   -- и нет займа с долгом на внесистемке
              from operkroff oko, knclient
              where kc_from_c_id = :c_id
              and kc_amount_approved > 0
              and kc_from_k_id = oko_from_k_id
              and oko_ol_is_active = 1
              and oko.oko_ol_date <= :rep_date_end
              group by oko_from_k_id
              having  sum(oko_os + oko_sh_os) > 0 or sum(oko_perc + oko_sh_perc) >0
          )) then
          begin
              blob_c_list = blob_append(blob_c_list, c_id,  '.');

              select cl.c_name_fio
              from clients cl
              where cl.c_id = :c_id
              into "ФИО клиента";

              counter = counter + 1;

              suspend;
          end
     end
    end
  end

  c_id = null;
  "Статус" = null;

  for select eds.eds_from_c_id, edl.edl_date_end
      from edoc_list edl
      inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id
      where edl.edl_status = 1 and
            eds.eds_status = 1 and
            edl.edl_date_end between :rep_date_begin and :rep_date_end and --все клиенты с которыми был приостановлен договор АСП
            edl.edl_from_edc_id = f_edoc_kind('asp.agreement', 'id')
      into :c_id, :edl_date_end
  do
  begin
    if (F_FT_VALID(:c_id, :rep_date_end) = 0) then  -- если нет активного договра АСП (Активным счиатется АСП, который был подписан либо когда у клиента был активный заем либо если после подписания АСП клиент брал заем.
    begin
      if (F_FT_VALID(:c_id, :edl_date_end - 1) = 1) then  -- если за день до  расторжения был активный договр АСП, проверк анужна т.к. может быть ситуация что АСП подписали, но займ клиент не получил, да и АСП был подписан не когда у клиента был активный заем.
      begin
          if (position(c_id in blob_c_list) <= 0) then --если клиента нет в списке закрывших заем за период, проверка нужна чтобы не было дублей в отчете
          begin
           if (not exists( -- если нет активного займа
                  select ks_from_k_id
                  from knclient, kontstate
                  where kc_from_c_id = :c_id
                  and kc_amount_approved > 0
                  and kc_from_k_id = ks_from_k_id
                  and ks_ol_is_delete = 0
                  and ks_state = 1
                  and :rep_date_end between ks_date_begin and  ks_date_end
                  union
                  select oko_from_k_id  -- и нет займа с долгом на внесистемке
                  from operkroff oko, knclient
                  where kc_from_c_id = :c_id
                  and kc_amount_approved > 0
                  and kc_from_k_id = oko_from_k_id
                  and oko_ol_is_active = 1
                  and oko.oko_ol_date <= :rep_date_end
                  group by oko_from_k_id
                  having  sum(oko_os + oko_sh_os) > 0 or sum(oko_perc + oko_sh_perc) >0
            )) then
            begin
                "Статус" = 'Расторгнут договор АСП';
    
                select cl.c_name_fio
                from clients cl
                where cl.c_id = :c_id
                into "ФИО клиента";
    
                counter = counter + 1;
    
                suspend;
            end
          end
       end
    end
  end

  "ФИО клиента" = 'ИТОГО';
  "Статус" = counter;

  suspend;
end
