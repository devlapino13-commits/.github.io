CREATE PROCEDURE HP_GET_HTML_SUSN_STATUS(
    C_ID D_ID,
    MA_ID D_ID,
    AS_TABLE D_SMALLINT)
RETURNS (
    RES_HTML D_STR_8100)
AS
declare variable SUSN_TEXT D_STR_255;
declare variable SUSN_NAMES D_STR_2048;
declare variable SUSN_COLOR D_STR_CODE;
declare variable SUSN_LINK D_STR_LONG;
declare variable RES_STATUS D_SMALLINT;
declare variable IS_SUSN D_SMALLINT;
declare variable TRY_COUNT D_INTEGER;
declare variable MAX_TRY D_INTEGER;
begin
  -- RFCKZ-759, RFCKZ-996
  res_html = '';
  susn_link = '';
  susn_color = 'black';
  max_try = 4;

  select res_status, is_susn, susn_names, try_count, ma_id_out
  from hp_susn_get_status(:c_id, :ma_id)
  into res_status, is_susn, susn_names, try_count, ma_id;

  if (res_status = 0) then
    susn_text = 'Еще не получен';
  else
  if (res_status = 1) then
    susn_text = 'В обработке';
  else
  if (res_status = 2) then
  begin
    susn_text = iif(is_susn = 1, 'Да', 'Нет');
    susn_color = iif(is_susn = 1, 'orange', 'black');
  end
  else
  begin
    susn_color = 'red';

    if (try_count >= max_try) then
      susn_text = 'Сервис СУСН не отвечает';
    else
    begin
      susn_text = 'Ответ от сервиса не получен';
      if (ma_id is not null) then
        susn_link = f_uri_sql('Проверить повторно', 'susn_request', 'miniapp', :ma_id,
                              'none', 'Подтверждаете?', '', 0, 0, '', '', 'dialog', '');
    end
  end

  if (as_table = 1) then
  begin
    if (susn_link <> '') then
    begin
      res_html = '<table width="100%" style="margin: 0; padding 0;">'
                 || '<col width="40%"><col width="40%"><col width="20%">'
                 || f_html_table_row('СУСН',
                      f_format('<span style="color: %1;">%2</span>', susn_color, susn_text),
                      susn_link, -1, -1, -1, -1, -1, 3, '');
    end
    else
    begin
      res_html = '<table width="100%" style="margin: 0; padding 0;">'
                 || '<col width="40%"><col width="60%">'
                 || f_html_table_row('СУСН',
                      f_format('<span style="color: %1;">%2</span>', susn_color, susn_text),
                      '', '', -1, -1, -1, -1, 2, '');
      if (coalesce(susn_names, '') <> '') then
        res_html = res_html || f_html_table_row('Наименование', susn_names,
                                                '', '', -1, -1, -1, -1, 2, '');
    end
    res_html = res_html || '</table>';
  end
  else
  begin
    res_html = f_format('<br>СУСН: <b><span style="color: %1;">%2</span></b>',
                        susn_color, susn_text);
    if (susn_link <> '') then
      res_html = res_html || ' ' || susn_link;
    if (coalesce(susn_names, '') <> '') then
      res_html = res_html || '<br>Наименование: <b>' || susn_names || '</b>';
    res_html = res_html || '<br>';
  end

  suspend;
end
