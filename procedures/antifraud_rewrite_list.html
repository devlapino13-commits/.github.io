CREATE PROCEDURE ANTIFRAUD_REWRITE_LIST
AS
declare variable UID D_STR_CODE;
declare variable INN D_STR_CODE;
declare variable FIRST_NAME D_STR_100;
declare variable SURNAME D_STR_100;
declare variable PATRONIMIC D_STR_100;
declare variable BIRTH_DATE D_DATE;
declare variable LIST_CATEGORY D_STR_CODE;
declare variable PHONE_LIST D_STR_8100;
declare variable FIO D_STR_255;
begin
  delete from finpolis;

  for select uid, inn, first_name, surname, patronimic, birth_date, list_category, phone_list
      from antifraud_tmp
      into :uid, :inn, :first_name, :surname, :patronimic, :birth_date, :list_category, :phone_list
  do
  begin
    fio = trim(coalesce(surname, '') || ' ' || coalesce(first_name, '') || ' ' || coalesce(patronimic, ''));
        
    insert into finpolis (nameu, descript, nd, gr, terr_categ, phones)
    values (:fio, :uid, :inn, :birth_date, :list_category, :phone_list);
  end
end
