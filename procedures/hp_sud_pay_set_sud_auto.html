CREATE PROCEDURE HP_SUD_PAY_SET_SUD_AUTO(
    SLC_ID D_ID,
    W_CODE D_STR_CODE,
    W_NAME D_STR_255,
    W_ADDRESS D_STR_255,
    W_EMAIL D_STR_255,
    W_PHONE D_STR_255,
    W_BANKNAME D_STR_255,
    W_BIK D_STR_10,
    W_INN D_STR_20,
    W_KPP D_STR_10,
    W_ACC D_STR_20,
    W_TREASURE_ACC D_STR_20,
    W_PAYEE_NAME D_STR_255,
    W_KBK D_STR_20,
    W_OKTMO D_STR_20,
    FED_CODE D_STR_CODE,
    FED_NAME D_STR_255,
    FED_ADDRESS D_STR_255,
    FED_EMAIL D_STR_255,
    FED_PHONE D_STR_255,
    FED_BANKNAME D_STR_255,
    FED_BIK D_STR_10,
    FED_INN D_STR_20,
    FED_KPP D_STR_10,
    FED_ACC D_STR_20,
    FED_TREASURE_ACC D_STR_20,
    FED_PAYEE_NAME D_STR_255,
    FED_KBK D_STR_20,
    FED_OKTMO D_STR_20,
    OSP_CODE D_STR_CODE,
    OSP_NAME D_STR_255,
    OSP_ADDRESS D_STR_255,
    OSP_EMAIL D_STR_255,
    OSP_PHONE D_STR_255)
AS
declare variable w_id d_id;
declare variable fed_id d_id;
declare variable osp_id d_id;
begin
  -- процедура записывает результат определения подсудности
  -- мировой суд
  update or insert into sud_agents (sag_kind, sag_code, sag_name, sag_address, sag_email,
                                    sag_phone, sag_bankname, sag_bik, sag_inn, sag_kpp, sag_acc, sag_treasure_acc,
                                    sag_payee_name, sag_kbk, sag_oktmo)
  values (f_enum('sud_agent_kind','world_court', 'id'), :w_code, :w_name, :w_address, :w_email, :w_phone, :w_bankname,
          :w_bik, :w_inn, :w_kpp, :w_acc, :w_treasure_acc, :w_payee_name, :w_kbk, :w_oktmo)
  matching (sag_kind, sag_code)
  returning sag_id
  into w_id;

  -- федеральный суд
  update or insert into sud_agents (sag_kind, sag_code, sag_name, sag_address, sag_email,
                                    sag_phone, sag_bankname, sag_bik, sag_inn, sag_kpp, sag_acc, sag_treasure_acc,
                                    sag_payee_name, sag_kbk, sag_oktmo)
  values (f_enum('sud_agent_kind','federal_court', 'id'), :fed_code, :fed_name, :fed_address, :fed_email, :fed_phone, :fed_bankname,
          :fed_bik, :fed_inn, :fed_kpp, :fed_acc, :fed_treasure_acc, :fed_payee_name, :fed_kbk, :fed_oktmo)
  matching (sag_kind, sag_code)
  returning sag_id
  into fed_id;

  -- ФССП
  update or insert into sud_agents (sag_kind, sag_code, sag_name, sag_address, sag_email, sag_phone)
  values (f_enum('sud_agent_kind','collector', 'id'), :osp_code, :osp_name, :osp_address, :osp_email, :osp_phone)
  matching (sag_kind, sag_code)
  returning sag_id
  into osp_id;

  update sud_loanclient
  set slc_auto_worldcourt = :w_id,
      slc_auto_federalcourt = :fed_id
--      slc_auto_collect = :osp_id
  where slc_id = :slc_id;
end
