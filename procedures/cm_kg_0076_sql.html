CREATE PROCEDURE CM_KG_0076_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable c_id id_type;
declare variable ovs_value str_255;
declare variable pr_code d_str_100;
begin
    for select kncl.kc_from_c_id,
               cl.c_name_fio,
               pr.pr_code
    from knclient kncl
    inner join clients cl on kncl.kc_from_c_id = cl.c_id
    inner join kontract kt on kt.k_id = kncl.kc_from_k_id
    inner join product pr on pr.pr_id = kt.k_from_pr_id
    where kncl.kc_amount_approved >0
    and kncl.kc_from_k_id = :k_id
    into :c_id, :res_info_text, pr_code
    do
    begin
        if (pr_code containing '.auto_kk.' and check_kind = '.send_kk.') then
          exit;
        select ovs_value from  HP_OBJECTPROPS_GET(0, :c_id, -1, 'work_monthly_income', current_date)
        into :ovs_value;
        if (ovs_value is not null) then
        begin
           ovs_value = trim(replace(ovs_value, 'Значение: ', ''));
           if (ovs_value <> '') then
           begin
              if (not (ovs_value similar to '[-]?[[:DIGIT:]]+')) then
              begin
                  res_error_is_critical = 2;
                  res_error_text = 'В Анкете клиента, вопрос: "Общий доход (в месяц)" ошибка в ответе. Укажите сумму в сомах в месяц. Сейчас значение:"'||ovs_value ||'"';
                  suspend;
              end
           end
        end
        ------------
        select ovs_value from  HP_OBJECTPROPS_GET(0, :c_id, -1, 'AnketaQuestions.living_expenses', current_date)
        into :ovs_value;
        if (ovs_value is not null) then
        begin
           ovs_value = trim(replace(ovs_value, 'Значение: ', ''));
           if (ovs_value <> '') then
           begin
              if (not (ovs_value similar to '[-]?[[:DIGIT:]]+')) then
              begin
                  res_error_is_critical = 2;
                  res_error_text = 'В Анкете клиента, вопрос: "Общий расход (в месяц)" ошибка в ответе. Укажите сумму в сомах в месяц. Сейчас значение:'||ovs_value;
                  suspend;
              end
           end
        end
        ----
    end
end
