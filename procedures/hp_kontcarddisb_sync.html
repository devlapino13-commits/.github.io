CREATE PROCEDURE HP_KONTCARDDISB_SYNC(
    K_ID D_ID)
AS
declare variable DB_ID D_INTEGER;
declare variable REG_ID D_INTEGER;
declare variable NEED_ADD D_SMALLINT;
declare variable K_DISB_DATE D_DATE;
declare variable PR_CODE D_STR_100;
begin
  -- Процедура добавляет договор в список выдач на карту
  -- Если условия не готовы, исключает из списка

  db_id = f_get_db_id();

  need_add = 1; -- по умолчанию: добавить в очередь
  select k_last_reg_id, k_disb_date, pr.pr_code
  from kontract
  inner join product pr on k_from_pr_id = pr.pr_id
  where k_id = :k_id and
        k_disb_kind in (5, 6, 7, 8) and
        (:db_id <> 81001 or k_is_online = 1) and -- в РФ только онлайн выдачи попадают в очередь
        k_status = 1
  into reg_id, k_disb_date, pr_code;

  -- если нет утвержденного займа с выдачей на карту, отмечаем на исклюенение из списка
  if (reg_id is null) then
    need_add = 0;

  -- если дата выдачи не сегодня, отмечаем на исклюенение из списка
  if (k_disb_date <> current_date) then
    need_add = 0;

  if (need_add = 1) then
  begin
    -- Если уже выдан, отмечаем на исклюенение из списка
    if (exists(select * from kontstate
    where ks_from_k_id = :k_id and
          ks_state = 1 and
          ks_ol_is_delete = 0)) then need_add = 0;
  end

  if (need_add = 1) then
  begin
    -- Если электронный документ не подписан, отмечаем на исклюенение из списка
    if (f_edoc_status('loan.agreement', k_id, current_date) not in ('inactive', 'signed')) then
      need_add = 0;
  end

  -- RFCRU-3201
  if (need_add = 1 and db_id = 81001 and pr_code containing '.zalog_car.' or pr_code containing '.zalog.') then
  begin
    if (f_edoc_status('loan.agreement', k_id, current_date) <> 'signed' or
        f_edoc_status('loan.collateral', k_id, current_date) <> 'signed') then
            need_add = 0;
  end

  if (need_add = 1) then
  begin
    if (not exists(select * from kontcarddisb where kkd_from_k_id = :k_id)) then
    begin
      insert into kontcarddisb (kkd_from_k_id, kkd_from_reg_id)
      values (:k_id, :reg_id);
    end else
    begin
      update kontcarddisb
      set kkd_from_reg_id = :reg_id
      where kkd_from_k_id = :k_id and
            kkd_from_reg_id <> :reg_id;
    end
  end else
  begin
    delete from kontcarddisb
    where kkd_from_k_id = :k_id;
  end
end
