CREATE PROCEDURE FM_HTML_COM_RESCHEDULE(
    CML_ID D_ID,
    UN_ID D_ID,
    IS_SUBREPORT D_SMALLINT)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable cml_date d_date;
declare variable k_id d_id;
declare variable cml_add_params d_str_8100;
declare variable err_text d_str_255;
declare variable lang d_str_lang;
declare variable scale d_smallint;
declare variable res_is_error smallint;
declare variable tmp_text varchar(32000) character set win1251;
declare variable sh_id d_integer;
declare variable sh_date d_date;
declare variable sh_os d_currency;
declare variable sh_perc d_currency;
declare variable sum_os d_currency;
declare variable sum_perc d_currency;
declare variable cmc_code d_str_code;
begin

  res_is_error = 0;
  scale = 0;
  res_text = '';
  err_text = '';
  lang = f_get_lang_code(un_id, null);

  select cml_date, cml_object_id, cml_add_params, cmc_code
  from com_list, com_cfg
  where cml_id = :cml_id and
        cml_object_kind = f_objkind('kontract', 'id') and
        cml_from_cmc_id = cmc_id
  into :cml_date, :k_id, :cml_add_params, cmc_code;

  if (k_id is null) then
    err_text = f_lang(lang, 'Не найден займ в запросе');

  if (err_text = '' and coalesce(cml_add_params, '') = '') then
    err_text = f_lang(lang, 'Параметры реструктуризации не заполнены');

  if (f_json_str_get_value(cml_add_params, 'date_stop', null) = 'Выберите дату') then
    err_text = 'Вы не выбрали дату, прошу выбрать дату!';

  if (err_text = '') then
  begin
    -- Если нет графика, строим его
    if (not exists(
        select * from com_list_resh
        where cmr_from_cml_id = :cml_id)) then
    begin
      if (cml_date <> current_date) then
      begin
        update com_list
        set cml_date = current_date
        where cml_id = :cml_id;

        select res_text
        from hp_com_memo_add(
          :cml_id,
          f_format(f_lang(:lang, 'Изменена дата запроса: %1 -> %2'), f_datetostr(:cml_date), f_datetostr(current_date)), 0, :un_id)
        into err_text;
        err_text = '';
        cml_date = current_date;
      end

      if (cmc_code in ('kred_kanikul_svo')) then
      begin
        select res_text from hp_kontvac_tmp(:k_id, :cml_add_params,:un_id)
        into err_text;
        if (err_text <> '') then res_is_error = 1;
      end else
      if (cmc_code in ('rastor_dog_sogl_storon', 'rastor_dog_smert_cli', 'rastor_dog_zakon')) then
      begin
        select res_is_error, res_text
        from hp_termination_new_graph(:k_id, :cml_add_params)
        into res_is_error, err_text;
      end else
      begin
        select res_is_error, res_text
        from hp_do_reschedule(:k_id, :cml_date, 'auto', :cml_add_params, :un_id)
        into res_is_error, err_text;
      end

      if (res_is_error = 0) then
      begin
        -- Сохраняем новый график погашений
        insert into com_list_resh (cmr_from_cml_id, cmr_sh_id, cmr_sh_date, cmr_sh_os, cmr_sh_perc)
        select :cml_id, sh_id, sh_date, sh_new_os, sh_new_perc
        from temp_sh
        where sh_new_os <> 0 or sh_new_perc <> 0
        order by sh_date, sh_id;
      end
    end

    -- Новый график погашений
    if (res_is_error = 0) then
    begin
      -- Выясняем требуемую точность округления сумм при форматировании
      if (exists(
          select * from com_list_resh
          where cmr_from_cml_id = :cml_id and
                (round(cmr_sh_os)<>cmr_sh_os or round(cmr_sh_perc) <> cmr_sh_perc)))
      then scale = 2;

      res_text =
        f_format('<h2>%1</h2><table width="100%">'||
        '<col width="10%"><col width="20%"><col width="20%"><col width="20%"><col width="30%">'||
        '<tr><th>№</th><th>%2</th><th>%3</th><th>%%</th><th>%4</th></tr>',
        f_lang(lang, 'Новый график погашений'),
        f_lang(lang, 'Дата'),
        f_lang(lang, 'ОС'),
        f_lang(lang, 'Всего'));

      tmp_text = '';
      sum_os = 0;
      sum_perc = 0;

      for select cmr_sh_id, cmr_sh_date, cmr_sh_os, cmr_sh_perc
      from com_list_resh
      where cmr_from_cml_id = :cml_id
      order by cmr_id
      into sh_id, sh_date, sh_os, sh_perc
      do
      begin
        tmp_text = tmp_text || coalesce(
            '<tr><td align="center">' || sh_id ||
            '</td><td align="center">' || f_datetostr(sh_date) ||
            '</td><td align="right">' || f_numtostr(sh_os, :scale) ||
            '</td><td align="right">' || f_numtostr(sh_perc, :scale) ||
            '</td><td align="right">' || f_numtostr(sh_os + sh_perc, :scale) ||
            '</td></tr>', '');
         sum_os = sum_os + sh_os;
         sum_perc = sum_perc + sh_perc;
      end
  
      res_text = BLOB_APPEND(res_text, tmp_text || coalesce(
        '<tr><td colspan=2>' || f_lang(lang, 'Итого') ||
        '</td><td align="right">' || f_numtostr(sum_os, :scale) ||
        '</td><td align="right">' || f_numtostr(sum_perc, :scale) ||
        '</td><td align="right">' || f_numtostr(sum_os + sum_perc, :scale) ||
        '</td></tr>', '') || '</table>');
    end
    else
      res_text = f_format('<h2>%1: </h2><font color="red">%2</font>', f_lang(lang, 'Ошибки при реструктуризации'), err_text);

    -- Уточняем требуемую точность округления сумм при форматировании
    if (scale = 0 and exists(
        select *
        from aarep_make_zad_list(:k_id, :cml_date, 0, 0)
        where sh_date is not null and
              (zad_os <> 0 or zad_perc <> 0) and
              (round(zad_os) <> zad_os or round(zad_perc) <> zad_perc)))
    then scale = 2;

    -- Текущий график погашений
    res_text = BLOB_APPEND(res_text, f_format('<h2>%1</h2><table width="100%">'||
      '<col width="10%"><col width="20%"><col width="20%"><col width="20%"><col width="30%">'||
      '<tr><th>№</th><th>%2</th><th>%3</th><th>%%</th><th>%4</th></tr>',
      f_lang(lang, 'Текущий график погашений'),
      f_lang(lang, 'Дата'),
      f_lang(lang, 'ОС'),
      f_lang(lang, 'Всего')));

    tmp_text = '';
    sum_os = 0;
    sum_perc = 0;
    for select sh_id, sh_date, zad_os, zad_perc
    from aarep_make_zad_list(:k_id, :cml_date, 0, 0)
    where sh_date is not null and
      (zad_os <> 0 or zad_perc <> 0)
    into sh_id, sh_date, sh_os, sh_perc
    do
    begin
      tmp_text = tmp_text || coalesce(
          '<tr><td align="center">' || sh_id ||
          '</td><td align="center">' || f_datetostr(sh_date) ||
          '</td><td align="right">' || f_numtostr(sh_os, :scale) ||
          '</td><td align="right">' || f_numtostr(sh_perc, :scale) ||
          '</td><td align="right">' || f_numtostr(sh_os + sh_perc, :scale) ||
          '</td></tr>', '');
       sum_os = sum_os + sh_os;
       sum_perc = sum_perc + sh_perc;
    end

    res_text = BLOB_APPEND(res_text, tmp_text || coalesce(
      '<tr><td colspan=2>' || f_lang(lang, 'Итого') ||
      '</td><td align="right">' || f_numtostr(sum_os, :scale) ||
      '</td><td align="right">' || f_numtostr(sum_perc, :scale) ||
      '</td><td align="right">' || f_numtostr(sum_os + sum_perc, :scale) ||
      '</td></tr>', '') || '</table>');

  end
  else -- Выводим ошибки
    res_text = f_format('<h2>%1: </h2><font color="red">%2</font>', f_lang(lang, 'Ошибки'), err_text);

  -- Применяем стили, добавляем <body>
  if (is_subreport = 0) then
    res_text = (select res_style from fm_html_styles) || '<body>' || res_text || '</body>';
  suspend;
end
