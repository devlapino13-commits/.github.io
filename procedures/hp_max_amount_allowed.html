CREATE PROCEDURE HP_MAX_AMOUNT_ALLOWED(
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID,
    L_1 D_CURRENCY)
RETURNS (
    MDS D_CURRENCY,
    L_2 D_CURRENCY,
    ACTIVE_BALANCE D_CURRENCY,
    NO_ZALOG_BALANCE D_CURRENCY)
AS
declare variable pr_code d_str_100;
declare variable c_id d_id;
declare variable ma_id d_id;
declare variable k_disb_date d_date;
declare variable point d_integer;
begin
  if (obj_kind = 9) then
  begin
    select k_disb_date, pr_code, kc_from_c_id, k_from_ma_app_id
    from kontract
    inner join product on pr_id = k_from_pr_id
    inner join knclient on kc_from_k_id = k_id
    where k_id = :obj_id and
          kc_amount_approved > 0
    into k_disb_date, pr_code, c_id, ma_id;
  end

  if (obj_kind = 11) then
  begin
    select ma_date_oper, pr_code, mac_from_c_id, ma_id
    from miniapp
    inner join product on pr_id = ma_from_pr_id
    inner join miniappclient on mac_from_ma_id = ma_id
    where ma_id = :obj_id and
          ma_is_active = 1 and
          ma_date_end = '31.12.9999'
    into k_disb_date, pr_code, c_id, ma_id;
  end

  -- активный баланс клиента
  select sum(oks_balance)
  from knclient, oksummary
  where kc_from_c_id = :c_id and
        kc_amount_approved > 0 and
        kc_from_k_id = oks_from_k_id and
        oks_balance > 0 and
        oks_close_state = 0 and
        :k_disb_date between oks_date_begin and oks_date_end
  into active_balance;
  active_balance = coalesce(active_balance, 0);

  -- баланс клиента без "Залогового" займа
  select sum(oks_balance)
  from knclient, oksummary, kontract, product
  where kc_from_c_id = :c_id and
        kc_amount_approved > 0 and
        kc_from_k_id = oks_from_k_id and
        oks_from_k_id = k_id and
        pr_id = k_from_pr_id and
        pr_code not containing '.zalog.' and
        oks_balance > 0 and
        oks_close_state = 0 and
        :k_disb_date between oks_date_begin and oks_date_end
  into no_zalog_balance;
  no_zalog_balance = coalesce(no_zalog_balance, 0);

  point = f_point_get(11, ma_id, c_id, k_disb_date);

  -- Л2 – лимит, предусмотренный для продукта «Залоговый», согласно баллам клиента по таблице лимитов;
  select amount_max
  from perc_amount_interval_by_point(150, :point, :k_disb_date)
  into l_2;

  mds = 0;
  if (pr_code containing '.zalog.') then
  begin
    -- МДС по продукту «Залоговый» определяется по формуле: Л2 минус баланс всех активных займов клиента  в Компании;
    mds = l_2 - active_balance;
    suspend;
  end
  else
  begin
    -- если баланс клиента без залогового меньше, чем активный баланс, значит, что у клиента есть залоговый
    if (no_zalog_balance < active_balance) then
    begin
      -- МДС, если у клиента есть активный заем «Залоговый»: минимальный из двух показателей: (Л2 минус баланс активного займа «Залоговый») либо (Л1**) - минус баланс всех активных беззалоговых займов;
      mds = minvalue(l_2 - (active_balance - no_zalog_balance), l_1);
      mds = mds - no_zalog_balance;
      suspend;
    end
    else
    begin
      mds = l_1 - active_balance;
      suspend;
    end
  end
end
