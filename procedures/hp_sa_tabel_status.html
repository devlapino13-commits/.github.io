CREATE PROCEDURE HP_SA_TABEL_STATUS
RETURNS (
    CURR_Y INTEGER,
    CURR_M INTEGER,
    MONTH_STATUS INTEGER)
AS
declare variable n integer;
declare variable start_year integer;
declare variable start_month integer;
declare variable dd date;
declare variable finish_year integer;
declare variable finish_month integer;
begin
  select min(ps_date_begin)
  from hr_pstatus
  into :dd;
  if (dd is null) then
    dd = current_date;
  curr_y = extract(year from :dd);
  curr_m = extract(month from :dd);

  select max(ps_date_begin)
  from hr_pstatus
  into :dd;
  if (dd is null) then
    dd = current_date;
  finish_year = extract(year from :dd);
  finish_month = extract(month from :dd);

  while ((curr_y * 12 + curr_m) <= (finish_year * 12 + finish_month)) do
  begin
    select month_status
    from hp_get_sa_tabel_status(:curr_y, :curr_m)
    into :month_status;
    suspend;
    curr_m = curr_m + 1;
    if (curr_m = 13) then
    begin
      curr_m = 1;
      curr_y = curr_y + 1;
    end
  end
end
