CREATE PROCEDURE CARD_OVERPAYMENT_TRANSFER(
    CS_ID D_INTEGER)
AS
declare variable card_id d_integer;
declare variable cca_id d_integer;
declare variable amount d_currency;
declare variable reg_id d_integer;
declare variable un_id d_id;
declare variable k_id d_id;
declare variable k_no d_str_20;
declare variable disb_date d_date;
declare variable c_id d_id;
declare variable c_name d_str_100;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable ol_id d_id;
declare variable phone d_str_20;
declare variable out_balance boolean;
declare variable ol_memo d_str_long;
declare variable ob_detail d_str_long;
begin
  select
    cs_card_id,
    cs_from_cca_id,
    cs_amount,
    cs_from_un_id,
    cs_from_k_id,
    cs_from_c_id,
    c_name_fio,
    k_kontract_no,
    k_disb_date,
    k_last_reg_id
  from cardsync
    inner join kontract
      on k_id = cs_from_k_id
    inner join clients
      on c_id = cs_from_c_id
  where cs_id = :cs_id
    and cs_from_csc_id = f_get_cardsynccfg_id('write-off_on_closing')
    and cs_status = 2
  into
    :card_id,
    :cca_id,
    :amount,
    :un_id,
    :k_id,
    :c_id,
    :c_name,
    :k_no,
    :disb_date,
    :reg_id;

  if (row_count = 1) then
  begin
    acc_dt = (
      select al_id from acc_list
      where al_from_reg_id = :reg_id
        and al_currency = '417'
        and al_account = '20001/1');

    ol_id = (
      select
        ol_id
      from hp_do_repayment (
        :k_id,
        0,              -- c_id
        current_date,   -- rep_date
        :amount,        -- repay_amount
        1,              -- repay_to_os
        1,              -- repay_to_perc
        1,              -- repay_to_fine
        1,              -- repay_to_kommis
        1,              -- need_save
        0,              -- is_full_repay
        0,              -- is_writeoff
        :reg_id,        -- repay_fact_office_id
        3,              -- repay_dt_account_kind
        :acc_dt,        -- repay_dt_account_no
        1,              -- repay_curr_rate
        :un_id,         -- repay_user_id
        :c_id,          -- repay_document_client
        null,           -- repay_doc_client_name_reason
        0,              -- repay_manager_c_id
        0));             -- repay_from_rd_id
    if (ol_id > 0) then
    begin
      out_balance = (exists(select * from operkroff where oko_from_ol_id = :ol_id));
      acc_kt = (
        select al_id from acc_list
        where al_from_reg_id = :reg_id
          and al_currency = '417'
          and al_account = iif(:out_balance, '61099/3', '21199/1101'));
      ol_memo = f_format('Кредитный дог.№%1 от %2 %3 %4',
        :k_no,
        f_datetostr(:disb_date),
        coalesce(f_format('номер ID %1 (сч.%2)', :card_id, :cca_id), ''),
        :c_name);
      ob_detail = 'Обязательство по Возврату остатка с карты МБулак';

      update operlist set
        ol_memo = :ol_memo
      where ol_id = :ol_id;

      update operbook set
        ob_account_kt = :acc_kt,
        ob_detail = :ob_detail
      where ob_from_ol_id = :ol_id;

      if (out_balance) then
      begin
        insert into operbook (
          ob_from_ol_id,
          ob_account_dt,
          ob_account_kt,
          ob_currency,
          ob_curr_rate,
          ob_summa_val,
          ob_summa,
          ob_detail)
        select first 1
          ob_from_ol_id,
          :acc_kt,
          (select
            al_id
           from acc_list
           where al_from_reg_id = ob_ol_reg_id
             and al_currency = ob_currency
             and al_account = '80421/1'),
          ob_currency,
          ob_curr_rate,
          ob_summa_val,
          ob_summa,
          ob_detail
        from operbook
        where ob_from_ol_id = :ol_id;

        insert into operbook (
          ob_from_ol_id,
          ob_account_dt,
          ob_account_kt,
          ob_currency,
          ob_curr_rate,
          ob_summa_val,
          ob_summa,
          ob_detail)
        select first 1
          ob_from_ol_id,
          (select
            al_id
           from acc_list
           where al_from_reg_id = ob_ol_reg_id
             and al_currency = ob_currency
             and al_account = '95002/1'),
          (select
            al_id
           from acc_list
           where al_from_reg_id = ob_ol_reg_id
             and al_currency = ob_currency
             and al_account = '95001/2'),
          ob_currency,
          ob_curr_rate,
          ob_summa_val,
          ob_summa,
          ob_detail
        from operbook
        where ob_from_ol_id = :ol_id;
      end

      phone = coalesce(f_cont_info('clients', :c_id, 'cell_main', 'ok'), '');
      if (phone <> '') then
      execute procedure hp_send_quick_message(
        'sms',
        :phone,
        '',
        f_format('U Vas pereplata po kreditu v summe %1 som. Poluchite dannuyu summu v ofisah MBulak', cast(amount as decimal(18,2))),
        '',
        null,
        1000000000000);
    end
  end
end
