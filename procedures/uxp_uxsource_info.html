CREATE PROCEDURE UXP$UXSOURCE_INFO(
    UX_UUID D_UUID,
    UX_UUID_CODE D_STR_CODE = null)
RETURNS (
    US_UUID D_UUID,
    US_CODE D_STR_CODE,
    US_NAME D_STR_CODE,
    US_LOOKUP_ID D_STR_CODE,
    US_LOOKUP_TEXT D_STR_255,
    US_MEMO D_STR_8100,
    US_IS_TABLE D_BOOL,
    US_KIND D_INTEGER,
    US_IS_DOC D_BOOL,
    US_DIFFERENT D_SMALLINT,
    US_LOOKUP_SQL_TEXT D_STR_LONG,
    US_KEYNAME_SQL_TEXT D_STR_LONG,
    US_SQL_ERROR_TEXT D_STR_255,
    US_MAIN_TABLE D_STR_CODE,
    US_AUTOINCFIELD D_STR_CODE,
    US_SQL D_STR_4096,
    US_TREE_PARENT_FIELD D_STR_CODE,
    US_TREE_POSITION_FIELD D_STR_CODE,
    US_TREE_MOVE_SQL D_STR_8100)
AS
declare variable us_keyfields d_str_code;
begin
  if (coalesce(ux_uuid, '') = '') then
    ux_uuid = uxf$uniq(ux_uuid_code);

  select us_uuid, us_is_table, us_kind, us_is_doc, us_different, us_memo, us_code, us_name,
         us_lookup_id, us_lookup_text, us_main_table, us_autoincfield, us_sql,
         us_tree_parent_field, us_tree_position_field, us_keyfields
  from ux$source
  where us_uuid = :ux_uuid
  into us_uuid, us_is_table, us_kind, us_is_doc, us_different, us_memo, us_code, us_name,
       us_lookup_id, us_lookup_text, us_main_table, us_autoincfield, us_sql,
       us_tree_parent_field, us_tree_position_field, us_keyfields;

  us_lookup_sql_text = '';
  us_keyname_sql_text = '';
  us_sql_error_text = '';

  select lookup_sql, keyname_sql, error_text
  from uxp$uxsource_sql_prepare(:ux_uuid, '')
  into :us_lookup_sql_text, :us_keyname_sql_text, :us_sql_error_text;

  us_tree_move_sql = '';
  select tree_move_sql
  from uxp$tree_move_sql(:us_code, :us_keyfields, :us_tree_parent_field, :us_tree_position_field)
  into :us_tree_move_sql;

  suspend;
end
