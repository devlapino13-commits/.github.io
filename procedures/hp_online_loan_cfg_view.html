CREATE PROCEDURE HP_ONLINE_LOAN_CFG_VIEW(
    IS_ACTUAL_DATA D_SMALLINT = 0)
RETURNS (
    OLC_ID D_INTEGER,
    OLC_DATE_BEGIN D_DATE,
    OLC_DATE_END D_DATE,
    OLC_MIN_POINT D_INTEGER,
    OLC_DESCRIPTION D_STR_LONG,
    OLC_DATETIME D_DATETIME,
    OLC_USER_DATE_END D_DATE,
    UN_NAME D_STR_CODE)
AS
begin
  if (is_actual_data = 1) then
  begin
    for select olc.olc_date_begin, olc.olc_date_end, olc.olc_min_point, olc.olc_id, olc.olc_description
        from online_loan_cfg olc
        where current_date between olc.olc_date_begin and olc.olc_date_end or
              olc.olc_date_begin > current_date
        order by olc.olc_date_begin desc
        into olc_date_begin, olc_date_end, olc_min_point, olc_id, olc_description
     do
       suspend;
  end 
  else
  if (is_actual_data = 0) then
  begin
    for select olc.olc_id, olc.olc_min_point, olc.olc_date_begin, olc.olc_date_end, olc.olc_datetime,
               olc.olc_user_date_end, olc.olc_description, un.un_name
        from online_loan_cfg olc
        inner join username un on un.un_id = olc.olc_from_un_id
        order by olc.olc_id desc
        into olc_id, olc_min_point, olc_date_begin, olc_date_end, olc_datetime,
             olc_user_date_end, olc_description, un_name
    do
      suspend;
  end 
end
