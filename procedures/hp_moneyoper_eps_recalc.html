CREATE PROCEDURE HP_MONEYOPER_EPS_RECALC(
    REP_DATE D_DATE)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable recalc_count d_integer;
declare variable cash_acc_id d_id;
declare variable recalced_date d_integer;
begin
  res_is_error = 0;
  res_text = '';

  recalced_date = 0;

  for select acc_id
  from ref_account, sys_accbalance
  where sab_full_code = 'ПлатСистемы.Погашено' and
        acc_from_sab_id = sab_id
  into cash_acc_id
  do
  begin
    for select kob_id, kob_from_ol_id, kob_ol_date, kob_cashflow_date
    from kn_operbook
    where kob_ol_date = :rep_date and
          kob_ol_is_active = 1 and
          kob_acc_dt + 0 = :cash_acc_id
    as cursor cur
    do
    begin
      if (not exists(select kob_ol_date
      from operperiod, kn_operbook, ref_account, sys_accbalance
      where cur.kob_ol_date between olp_date_begin and olp_date_end and
            kob_ol_date = cur.kob_cashflow_date and
            kob_ol_is_active = 1 and
            olp_from_ol_id = kob_from_ol_id and
            kob_acc_kt + 0 = :cash_acc_id and
            kob_acc_dt = acc_id and
            acc_from_sab_id = sab_id and
            sab_full_code starting 'Деньги.Безнал.Банк.')) then
      begin
        recalced_date = recalced_date + 1;

        update kn_operbook
        set kob_from_rml_id = -2
        where current of cur;
      end
    end
  end

  update kn_operbook
  set kob_from_rml_id = -2
  where kob_ol_date = :rep_date and
        kob_ol_is_active = 1 and
        coalesce(kob_from_rml_id, 0) = 0;

  recalc_count = row_count;

  if (recalc_count + recalced_date = 0) then
    res_text = 'Нет данных для пересчета';
  else
    res_text = 'Пересчитаны коды: ' || recalc_count || ', даты: ' || recalced_date;

  suspend;
end
