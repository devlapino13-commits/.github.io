CREATE PROCEDURE HP_FOR_LATIN_CHECK(
    FIO STR_100)
RETURNS (
    LATIN_QTY INTEGER,
    STATUS INTEGER,
    STATUS_TXT STR_100,
    KIRILICA_QTY INTEGER)
AS
declare variable i integer;
declare variable ch varchar(1);
declare variable has_open_bracket smallint;
declare variable has_close_bracket smallint;
begin
  i = 0;
--min_latin_position = 0;
--max_latin_position = 0;
  latin_qty = 0;
  has_open_bracket = 0;
  has_close_bracket = 0;
  kirilica_qty = 0;
  status = 0;

--FIO = '(akbaralieva) Еркиной (yorqinoi) Муйдиновна (moydinjonovna)';

  while (i <= character_length(fio)) do
  begin
    i = i + 1;
    ch = substring(fio from i for 1);

    if (ch in ('a', 'b', 'c', 'd', 'e', 'f', 'g', 'h', 'i', 'j',
               'k', 'l', 'm', 'n', 'o', 'p', 'q', 'r', 's', 't',
               'u', 'v', 'w', 'x', 'y', 'z')) then
    begin
--if (min_latin_position = 0) then min_latin_position = i;
--max_latin_position = i;
      latin_qty = latin_qty + 1;
    end
    else
    if (ch in ('а', 'б', 'в', 'г', 'д', 'е', 'ё', 'ж', 'з', 'и',
               'й', 'к', 'л', 'м', 'н', 'о', 'п', 'р', 'с', 'т',
               'у', 'ф', 'х', 'ц', 'ч', 'ш', 'щ', 'ь', 'ъ', 'ы',
               'э', 'ю', 'я')) then
    begin
      kirilica_qty = kirilica_qty + 1;
    end

    if (ch = '(') then
      has_open_bracket = 1;
    if (ch = ')') then
      has_close_bracket = 1;
  end

  if (latin_qty > 0) then
  begin
    if ((latin_qty <= 3) and
        (kirilica_qty > 0)) then
    begin
      status = 1;
      status_txt = 'ошибка в буквах (несколько латинских букв)';
    end
    else
    if ((has_open_bracket = 0) and
        (has_close_bracket = 0)) then
    begin
      status = 2;
      status_txt = 'нет скобок';
    end
    else
    if ((has_open_bracket = 0) and
        (has_close_bracket = 1)) then
    begin
      status = 3;
      status_txt = 'нет открывающихся скобок';
    end
    else
    if ((has_open_bracket = 1) and
        (has_close_bracket = 0)) then
    begin
      status = 4;
      status_txt = 'нет закрывающихся скобок';
    end
    else
    if (kirilica_qty > 0) then
    begin
      status = 5;
      status_txt = 'латиница и кирилица';
    end
    else
    if (kirilica_qty = 0) then
    begin
      status = 6;
      status_txt = 'кирилица отсутствует';
    end
    else
    begin
      status = 8;
      status_txt = 'что-то другое';
    end
  end
  if ((kirilica_qty = 0) and
      (latin_qty = 0)) then
  begin
    status = 7;
    status_txt = 'нет ни кирилицы ни латиницы';
  end

  if (status > 0) then
    suspend;
end
