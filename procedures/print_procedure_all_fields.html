CREATE PROCEDURE PRINT_PROCEDURE_ALL_FIELDS(
    PROC_NAME VARCHAR(255))
RETURNS (
    PROC_FIELD VARCHAR(100),
    PROC_DESC BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    PROC_FIELD_TYPE INTEGER,
    PROC_FIELD_ORDER INTEGER,
    PROC_FIELD_NOT_NULL INTEGER,
    PROC_FIELD_SIZE INTEGER,
    PROC_FIELD_TYPE_NAME VARCHAR(100),
    PROC_FIELD_SOURCE VARCHAR(31))
AS
declare variable f_sub_type integer;
declare variable field_type_name varchar(31);
declare variable field_scale integer;
declare variable f_sub_type_name varchar(31);
begin
  proc_name = upper(:proc_name);
  for select trim(rdb$parameter_name), trim(rdb$description), rdb$parameter_type, rdb$parameter_number,
             coalesce(rdb$null_flag, 0), rdb$field_source
      from rdb$procedure_parameters
      where rdb$procedure_name = :proc_name and
            rdb$package_name is null
      order by 3, 4
      into :proc_field, :proc_desc, proc_field_type, proc_field_order, proc_field_not_null, proc_field_source
  do
  begin
    if (trim(coalesce(proc_desc, '')) = '') then
      proc_desc = trim(proc_field);
    proc_field_size = 0;

    select f.rdb$field_length, f.rdb$field_scale, trim(t.rdb$type_name), f.rdb$field_sub_type
    from rdb$fields as f, rdb$types t
    where f.rdb$field_name = :proc_field_source and
          f.rdb$field_type = t.rdb$type and
          t.rdb$field_name = 'RDB$FIELD_TYPE'
    into :proc_field_size, :field_scale, :field_type_name, f_sub_type;

    proc_field_type_name = field_type_name;

    if (proc_field_type_name = 'SHORT') then
      proc_field_type_name = 'SMALLINT';
    if (proc_field_type_name = 'LONG') then
      proc_field_type_name = 'INTEGER';
    if (proc_field_type_name = 'INT64') then
      proc_field_type_name = 'BIGINT';
    if (proc_field_type_name = 'FLOAT') then
      proc_field_type_name = 'FLOAT';
    if (proc_field_type_name = 'DOUBLE') then
      proc_field_type_name = 'DOUBLE PRECISION';

    if (proc_field_type_name = 'INT64') then
    begin
      if (field_scale = -4) then
        proc_field_type_name = 'DECIMAL(18,4)';
      else
      if (field_scale = 0) then
        proc_field_type_name = 'DECIMAL(18,0)';
      else
      if (field_scale = -7) then
        proc_field_type_name = 'DECIMAL(18,7)';
      else
        proc_field_type_name = 'DECIMAL';
    end
    if (proc_field_type_name = 'DATE') then
      proc_field_type_name = 'DATE';
    if (proc_field_type_name = 'TIME') then
      proc_field_type_name = 'TIME';
    if (proc_field_type_name = 'TIMESTAMP') then
      proc_field_type_name = 'DATETIME';
    if (proc_field_type_name = 'VARYING') then
      proc_field_type_name = 'VARCHAR';

    if (field_type_name = 'BLOB') then
    begin
      select trim(t.rdb$type_name)
      from rdb$types t
      where t.rdb$type = :f_sub_type and
            t.rdb$field_name = 'RDB$FIELD_SUB_TYPE'
      into :f_sub_type_name;

      proc_field_type_name = proc_field_type_name || ' ' || f_sub_type_name;
    end

    if (proc_field_type_name <> 'VARCHAR') then
      proc_field_size = 0;

    suspend;
  end
end
