CREATE PROCEDURE FIND_PAYSYS_ITEM(
    PROJECT_ID INTEGER,
    SOURCE_KIND INTEGER,
    SOURCE_ID DECIMAL(18,0),
    CARD_NO VARCHAR(20))
RETURNS (
    PSI_ID DECIMAL(18,0),
    PSI_CODE VARCHAR(20),
    PSI_DATE_BEGIN DATE,
    PSI_DATE_END DATE,
    IS_ERROR INTEGER)
AS
declare variable found_count integer;
begin
  found_count = 0;
  is_error = 0;
  psi_date_begin = null;
  psi_date_end = null;
  psi_code = '';
  if (source_kind = 0) then
  begin
    select psi_id, psi_date_begin, psi_date_end, psi_code
    from paysys_item
    where psi_code = :card_no and
          psi_from_psp_id = :project_id
    order by psi_id desc
    rows 1
    into :psi_id, :psi_date_begin, :psi_date_end, psi_code;

    if (psi_id is not null) then
    begin
      found_count = 0;
      select sum(psl_sign)
      from paysys_list
      where psl_from_psi_id = :psi_id and
            psl_is_delete = 0 and
            psl_object_kind <> 0
      into :found_count;
      if (found_count is null) then
        found_count = 0;
    end
    else
    begin
      psi_id = -1;
      psi_date_begin = null;
      psi_date_end = null;
    end
    if (found_count <> 0) then
      is_error = 1;
  end
  else
  begin
    psi_id = -1;
    for select psi_id, psi_date_begin, psi_date_end, psi_code
        from paysys_item
        where psi_from_psp_id = :project_id and
              psi_code = :card_no
        into :psi_id, :psi_date_begin, :psi_date_end, psi_code
    do
    begin
      select sum(psl_sign)
      from paysys_list
      where psl_object_kind = :source_kind and
            psl_object_id = :source_id and
            psl_from_psi_id = :psi_id and
            psl_is_delete = 0
      group by psl_object_id
      having sum(psl_sign) > 0
      into :found_count;
      if (found_count is null) then
        found_count = 0;
    end
    if (found_count = 0) then
      is_error = 1;
    if (psi_id is null) then
      psi_id = -1;
  end
  suspend;
end
