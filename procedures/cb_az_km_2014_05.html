CREATE PROCEDURE CB_AZ_KM_2014_05(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(60),
    KR_MANAGER_NAME VARCHAR(100),
    JL_NAME VARCHAR(100),
    OKS_BALANCE CURR_TYPE,
    OKS_DELINQ_BALANCE CURR_TYPE,
    PAR7 CURR_TYPE,
    BONUS CURR_TYPE,
    BONUS_UDEJAN CURR_TYPE,
    BONUS_TOTAL CURR_TYPE)
AS
declare variable ps_from_reg_id integer;
declare variable ps_from_c_id id_type;
declare variable bonus_perc curr_type;
begin

  for select ps_from_reg_id, ps_from_c_id, jl_name, iif(ps_from_jl_id = 19, 0.05, 0.08)
      from hr_pstatus, region_list(:rep_region_id, 1, 3, 1), joblist
      where :rep_date_end between ps_date_begin and ps_date_end and
            ps_is_working = 1 and
            ps_status = 1 and
            ps_from_jl_id in (19, 8) and
            ps_from_reg_id = id_list and
            ps_from_jl_id = jl_id
      into ps_from_reg_id, ps_from_c_id, jl_name, bonus_perc
  do
  begin

    reg_name = (select reg_code || ' ' || reg_name
                from region
                where reg_unicode = :ps_from_reg_id);
    kr_manager_name = (select c_name_fio
                       from clients
                       where c_id = :ps_from_c_id);

    oks_balance = null;
    oks_delinq_balance = null;
    select sum(oks_balance), sum(iif(:rep_date_end - oks_delinq_date > 7, oks_balance, 0))
    from oksummary
    where oks_region_id = :ps_from_reg_id and
          oks_close_state = 0 and
          :rep_date_end between oks_date_begin and oks_date_end
    into oks_balance, oks_delinq_balance;
    if (oks_balance is null) then
      oks_balance = 0;
    if (oks_delinq_balance is null) then
      oks_delinq_balance = 0;

    par7 = oks_delinq_balance / oks_balance * 100;
    par7 = round(par7, 2);

    bonus = round(oks_balance * bonus_perc / 100, 0);

    bonus_udejan = case
                     when par7 < 1 then 0
                     when par7 >= 1 and par7 < 2 then bonus * 25 / 100
                     when par7 >= 2 and par7 < 3 then bonus * 50 / 100
                     else bonus
                   end;
    bonus_udejan = round(bonus_udejan, 0);
    bonus_total = bonus - bonus_udejan;

    suspend;
  end
end
