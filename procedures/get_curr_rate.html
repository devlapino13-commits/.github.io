CREATE PROCEDURE GET_CURR_RATE(
    CU_ID VARCHAR(3),
    CU_DATE DATE)
RETURNS (
    CURR_RATE DECIMAL(18,4))
AS
begin
  /* Procedure Text */
  curr_rate = 0;
  select distinct cr_rate
  from opercurr
  where cr_from_cu_id = :cu_id and
        cr_date <= :cu_date and
        cr_date_end >= :cu_date rows 1
  into :curr_rate;
  suspend;
end
