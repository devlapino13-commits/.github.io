CREATE PROCEDURE HP_EDOC_SMS_SIGN(
    CELL_NO D_STR_CODE,
    PIN_CODE D_STR_CODE)
AS
declare variable eds_id d_id;
declare variable eds_status d_integer;
declare variable edm_status d_smallint;
declare variable clt_id d_integer;
declare variable msg_id d_id;
declare variable edl_status d_smallint;
declare variable db_id d_integer;
declare variable need_send_sms d_smallint;
declare variable c_id bigint;
declare variable res_text varchar(255);
declare variable edl_id d_id;
declare variable tmp_text d_str_255;
declare variable res_is_error d_smallint;
declare variable edc_code d_str_code;
declare variable obj_kind d_integer;
declare variable obj_id d_id;
declare variable disb_kind d_integer;
begin
  res_text = '';
  need_send_sms = 0;

  cell_no = replace(cell_no, '+', '');
  db_id = f_get_db_id();
  clt_id = f_cont_id(cell_no, 'null');

  if (clt_id is null) then -- Неизвестный номер телефона
    exit;

  select edm_from_eds_id, edm_from_msg_id, edm_status, eds_from_c_id
  from edoc_msg, edoc_sign
  where edm_from_clt_id = :clt_id and
        edm_hash_pin in (:pin_code, f_md5(:pin_code)) and -- в режиме теста пин-код храним в без шифрования
        edm_from_eds_id = eds_id
  order by edm_status
  rows 1
  into eds_id, msg_id, edm_status, c_id;

  if (eds_id is not null) then
  begin
    if (edm_status = 1) then
      res_text = iif(db_id <> 94401, 'Документы уже подписаны', 'Sənədlər artıq imzalanıb');

    if (res_text = '') then
    begin
      select edl_id, eds_status, iif(coalesce(edl_date_end, current_date) < current_date, -1, edl_status),
        edc_code, edl_object_kind, edl_object_id
      from edoc_sign, edoc_list, edoc_cfg
      where eds_id = :eds_id and
            eds_from_edl_id = edl_id and
            edl_from_edc_id = edc_id
      into edl_id, eds_status, edl_status, edc_code, obj_kind, obj_id;
    
      if (edl_status = -1) then
        res_text = iif(db_id <> 94401, 'Данные устарели', 'Məlumatlar köhnəlib');
  
      if (eds_status = 1) then
        res_text = iif(db_id <> 94401, 'Документы уже подписаны', 'Sənədlər artıq imzalanıb');

      if (edc_code in ('bankcard.moneta', 'bankcard.elcart')) then
        res_text = iif(db_id <> 94401, 'Для закрепления карты перейдите по ссылке в СМС', 'Kartı göstərmək üçün SMS-lə əlaqəni izləyin');

      if (edc_code = 'loan.application' and f_can_disb_on_card(obj_kind, obj_id, db_id) = 0) then
      begin
        -- Если выдача по СБП, можно подписывать только через КМП.
        select ma_disb_kind
        from miniapp
          inner join product on ma_from_pr_id = pr_id
        where ma_id = :obj_id
          and ma_is_active = 1
          and ma_date_end >= '31.12.2999'
        into disb_kind;
        if (disb_kind in (7,8)) then
          res_text = 'Заявление на заем нужно подписать через приложение M Bulak';
      end
    end

    if (res_text = '') then
    begin
      update edoc_msg
      set edm_status = 1
      where edm_from_eds_id = :eds_id and
            edm_from_msg_id = :msg_id;
  
      update edoc_sign
      set eds_enum_edoc_sign_kind = f_enum('edoc_sign_kind', 'sms', 'id')
      where eds_id = :eds_id and
            eds_enum_edoc_sign_kind <> f_enum('edoc_sign_kind', 'sms', 'id');
  
      res_text = iif(db_id <> 94401, 'Документы успешно подписаны', 'Sənədlər uğurla imzalanıb');
      need_send_sms = 1;

      select res_is_error, res_text
      from hp_edoc_com_action(:edl_id, 1000000000000) -- пользователь fsrobot
      into :res_is_error, :tmp_text;

      if (res_is_error = 1) then
        res_text = tmp_text;
    end
  end else
  begin
    -- если это не eDoc, верифицируем контактные данные клиента

    select res_text, c_id, need_send_sms
    from hp_contveri_sign('cell', :cell_no, 'sms', :pin_code)
    into :res_text, :c_id, :need_send_sms;
  end

  if (need_send_sms = 1 and coalesce(res_text, '') <> '') then
  begin
    insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id,
                                msg_command_kind, msg_object_kind, msg_object_id)
    values (3, :cell_no, '', :res_text, 7, 1000000000000, 'quick sms', f_objkind('clients', 'id'), :c_id);
  end
end
