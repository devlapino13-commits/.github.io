CREATE PROCEDURE HP_APPREGISTRTERM_PRINT(
    ITA_ID D_ID)
RETURNS (
    HEAD_REG D_STR_8100,
    DOC_NO D_STR_20,
    DOC_DATE D_DATE,
    TERM_TYPE D_SMALLINT,
    TERM_NAME D_STR_255,
    TERM_NAME_LAT D_STR_255,
    TERM_IPC_ID D_STR_20,
    TERM_SERIAL_NO D_STR_255,
    ADDRESS_STR D_STR_255,
    CONTACT_NAME D_STR_100,
    CONTACT_PHONE D_STR_20,
    LONGITUDE D_GEO,
    LATITUDE D_GEO,
    HOURS D_STR_20,
    WEEKEND_HOURS D_STR_20,
    MCC_CODE D_STR_100,
    SERVICE_STR D_STR_255,
    COMPANY_NAME D_STR_255,
    STATUS D_SMALLINT)
AS
declare variable object_id d_id;
declare variable partner_id d_id;
declare variable trm_id d_integer;
declare variable ito_id d_integer;
declare variable address_prev_str d_str_255;
declare variable nr d_str_lang;
begin
  for
    select
      ito_id,
      trm_id,
      ita_doc_no,
      ita_doc_date,
      ita_type,
      ita_term_name,
      ita_term_name_lat,
      trm_ipc_id,
      trm_serial_no,
      ito_object_id,
      adl_address_str,
      adl_long,
      adl_lat,
      c_name_fio,
      ita_status
    from ipc_terminal_app
      inner join ipc_terminal_ownership
        on ito_trm_id = ita_from_trm_id
          and ita_doc_date between ito_date_begin and ito_date_end
      inner join ipc_terminals
        on trm_id = ito_trm_id
      left join addrlist
        on adl_id = ita_from_adl_id
      inner join clients
        on c_id = ita_contact_person
    where ita_id = :ita_id
  into
    :ito_id,
    :trm_id,
    :doc_no,
    :doc_date,
    :term_type,
    :term_name,
    :term_name_lat,
    :term_ipc_id,
    :term_serial_no,
    :object_id,
    :address_str,
    :longitude,
    :latitude,
    :contact_name,
    :status
  do
  begin
    nr = ascii_char(13) || ascii_char(10);
    if (term_type = 1) then
    begin
      select
        cs_from_c_id,
        cs_longitude,
        cs_latitude,
        cs_hours,
        cs_weekend_hours
      from clientshops
      where cs_id = :object_id
      into
        :partner_id,
        :longitude,
        :latitude,
        :hours,
        :weekend_hours;

      select first 1
        rd_str_id,
        rd_memo
      from agreements
        inner join agreement_feac
          on afe_from_ag_id = ag_id
        inner join refdata
          on rd_id = afe_from_rd_id
      where ag_object_kind = 0
        and ag_object_id = :partner_id
        and ag_from_rd_id = f_get_rd_id('AgreementKinds.tsp')
        and ag_status = 1
        and :doc_date between coalesce(ag_date_begin, '01.01.1900') and coalesce(ag_date_end, '31.12.9999')
      order by ag_id desc, afe_from_rd_id
      into
        :mcc_code,
        :service_str;
    end
    else
    begin
      hours = '08:00-20:00';
      weekend_hours = '08:00-20:00';
    end

    company_name = f_company_name('short', 'rus');
    contact_phone = '08000007777';

    address_prev_str = (
      select first 1
        adl_address_str
      from ipc_terminal_ownership
        inner join clientshops
          on cs_id = ito_object_id
        inner join addrlist
          on adl_id = cs_from_adl_id
      where ito_trm_id = :trm_id
        and ito_id < :ito_id
        and ito_status = -1
      order by ito_id desc);
    head_reg = iif(address_prev_str is null,
      f_format(
        '<b>Заявка на подключение POS-терминала в ТСП в продуктивной среде</b>' || nr ||
        'Настоящим %1 просит подключить POS-терминал, расположенный в торгово-сервисном предприятии:', company_name),
      f_format(
        '<b>Заявка на изменение регистрационных параметров/настроек POS-терминала, установленного в ТСП.</b>' || nr ||
        'Настоящим %1 просит произвести изменение регистрационных параметров/настроек POS-терминала ID %2, ' ||
        'установленного в ТСП, ранее расположенного по адресу: %3', company_name, term_ipc_id, address_prev_str));

    suspend;
  end
end
