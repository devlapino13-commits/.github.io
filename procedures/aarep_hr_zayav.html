CREATE PROCEDURE AAREP_HR_ZAYAV(
    REP_REGION DECIMAL(18,0),
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    HO_ORDER_NO VARCHAR(255),
    HO_DATE DATE,
    C_NAME_FIO VARCHAR(255),
    JL_NAME VARCHAR(255),
    REG_NAME VARCHAR(255),
    PS_BRANCH VARCHAR(255),
    HO_TEXT VARCHAR(255),
    NUMBER INTEGER,
    SUBNUMBER INTEGER,
    HO_ORDER_TYPE VARCHAR(255),
    HRV_DATE_BEGIN DATE,
    HRV_DATE_END DATE,
    UN_FIO VARCHAR(255),
    C_INN D_STR_20,
    VACATION_DAYS D_INTEGER)
AS
declare variable pos0 integer;
declare variable pos2 integer;
declare variable len integer;
declare variable text0 varchar(255);
declare variable st_staff integer;
declare variable st_client integer;
declare variable hrv_from_c_id decimal(18,0);
declare variable db_id integer;
declare variable ho_status_user_id decimal(18,0);
begin
  un_fio = null;
  ho_status_user_id = null;

  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  :db_id = f_get_db_id();

  for select ho_order_no, ho_date, c_name_fio, r1.reg_code || ' ' || r1.reg_name, /*Офис сотрудника*/
             r2.reg_name, ho_text, sawt_name, hrv_date_begin, hrv_date_end_fact, hrv_from_c_id, ho_status_user_id, un_fio, c_inn
      from hr_orders ho, sa_worktype, hr_voctravel, clients, region r1, region r2, region_list(:rep_region, 1, 3, 1), username
      where ho_date between :rep_date_begin and :rep_date_end and
            hrv_from_ho_id = ho_id and
            sawt_id = ho_order_type and
            hrv_from_c_id = c_id and
            ho_from_reg_id = r1.reg_unicode and
            r1.parent = r2.reg_unicode and
            (r1.reg_unicode = id_list) and
            r1.reg_status = 3 and
            ho.ho_status <> -1 and
            un_id = ho.ho_from_un_id
      order by ho_order_no, r2.reg_unicode
      into :ho_order_no, :ho_date, :c_name_fio, :reg_name, /*Офис сотрудника*/
           :ps_branch, :ho_text, :ho_order_type, :hrv_date_begin, :hrv_date_end, :hrv_from_c_id ,ho_status_user_id, un_fio, c_inn
  do
  begin

    :vacation_days = :hrv_date_end - :hrv_date_begin + 1; --подсчет кол-во дней отпуска

    if ((ho_order_no is not null) and
        (ho_order_no <> '')) then
    begin
      jl_name = '';
      select jl_name
      from hr_pstatus, joblist
      where ps_from_c_id = :hrv_from_c_id and
            ps_date_begin <= :rep_date_end and
            ps_date_end >= :rep_date_end and
            ps_is_working = 1 and
            ps_from_jl_id = jl_id
      order by ps_is_working desc
      rows 1
      into :jl_name;

      if (db_id <> 94401) then
      begin
        pos0 = position('-', ho_order_no) - 1;
        --S.K.
        if (pos0 < 0 and db_id in (39801, 41701)) then
          Exception E_Message(
            'Ошибка в номере приказа: ' || ASCII_CHAR(13) || ASCII_CHAR(13) ||
              reg_name || ASCII_CHAR(13) ||
              c_name_fio || ASCII_CHAR(13) ||
              ho_order_type || ASCII_CHAR(13) ||
              '№ ' || ho_order_no || ' от ' || f_datetostr(ho_date));

        text0 = left(ho_order_no, pos0);

        len = char_length(text0);

        pos2 = position('/', text0) - 1;

        if ((pos2 < pos0) and
            (pos2 <> -1)) then
        begin
          number = left(text0, pos2);
        end
        else
        begin
          number = left(text0, pos0);
        end

        if (pos2 <> -1) then
        begin
          pos2 = len - pos2 - 1;
          subnumber = right(text0, pos2);
        end
        else
          subnumber = null;

        if (pos0 = -1) then
          number = null;
        suspend;
      end
      else
        suspend;
    end
    else
    begin
      number = 0;
      subnumber = 0;
      suspend;
    end
  end
end
