CREATE PROCEDURE EPASS_PARSE_RES_JSON(
    ER_ID D_INTEGER)
RETURNS (
    FIO D_STR_100,
    SEX D_SMALLINT,
    INN D_STR_20,
    BIRTH_DATE D_DATE,
    CITIZEN D_INTEGER,
    NATION D_INTEGER,
    PASS_NO D_STR_100,
    PASS_DATE D_DATE,
    PASS_DATE_END D_DATE,
    DOC_TYPE D_INTEGER,
    AUTHORITY D_STR_255,
    UN_ID D_ID,
    SURNAME D_STR_100,
    NAME D_STR_100,
    PATRONYMIC D_STR_100)
AS
declare variable PDF_JSON D_STR_LONG;
declare variable SEX_STR D_STR_10;
declare variable NATION_STR D_STR_20;
begin
  select epr.erp_json, er.er_from_un_id
  from epass_response epr
  inner join epass_req er on epr.erp_from_er_id = er.er_id
  where epr.erp_from_er_id = :er_id
  into :pdf_json, :un_id;

  if (:pdf_json is null) then
  begin
    suspend;
    exit;
  end

  citizen = 8;
  inn = f_json_str_get_value(:pdf_json, 'personal_number', '');
  sex_str = f_json_str_get_value(:pdf_json, 'sex', '');
  surname = f_json_str_get_value(:pdf_json, 'surname_ru', '');
  name =  f_json_str_get_value(:pdf_json, 'name_ru', '');
  patronymic = f_json_str_get_value(:pdf_json, 'patronymic_ru', '');
  fio = surname || ' ' || name || ' ' || patronymic;

  if (sex_str <> '') then
    sex = iif(:sex_str = 'male', 1, 0);
  else
    sex = iif(:sex_str = '2', 1, 0);

  nation_str = f_json_str_get_value(:pdf_json, 'nationality', '');
  nation = (select
                   case :nation_str
                     when 'ОРУС' then f_get_rd_id('Nation.RU')
                     when 'КЫРГЫЗ' then f_get_rd_id('Nation.RU')
                     when 'КАЗАК' then f_get_rd_id('Nation.KZ')
                     else f_get_rd_id('Nation.Other')
                   end
            from rdb$database);
  birth_date = cast(f_json_str_get_value(:pdf_json, 'date_of_birth', '') as date);
  pass_no = f_json_str_get_value(:pdf_json, 'document_no', '');
  doc_type = iif(left(:pass_no, 2) = 'ID', 296, 8);
  pass_date = cast(f_json_str_get_value(:pdf_json, 'date_of_issue', '') as date);
  pass_date_end = cast(f_json_str_get_value(:pdf_json, 'date_of_expiry', '') as date);
  authority = f_json_str_get_value(:pdf_json, 'authority', '');
  suspend;
end
