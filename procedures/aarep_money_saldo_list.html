CREATE PROCEDURE AAREP_MONEY_SALDO_LIST(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    RES_REGION_NAME VARCHAR(100),
    RES_ACCOUNT VARCHAR(20),
    RES_ACCOUNT_NAME VARCHAR(200),
    RES_CURRENCY VARCHAR(3),
    RES_SALDO DECIMAL(18,4),
    RES_SALDO_EKV DECIMAL(18,4),
    RES_SALDO_CASH DECIMAL(18,4),
    RES_SALDO_CASH_VAL DECIMAL(18,4),
    RES_SALDO_BANK DECIMAL(18,4),
    RES_HQ_NAME VARCHAR(100),
    RES_BRANCH_NAME VARCHAR(100))
AS
declare variable res_region_id integer;
declare variable res_hq integer;
begin
  for select sum(res_saldo_cash), sum(res_saldo_bank), sum(res_saldo_cash + res_saldo_bank), res_reg_name, reg_account,
             reg_account_name, res_region_id, res_currency, sum(res_saldo_cash_val),
             sum(res_saldo_bank_val + res_saldo_cash_val)
      from aarep_money_temp(:report_region_id, :report_date_end)
      group by res_reg_name, reg_account, reg_account_name, res_currency, res_region_id
      having(sum(res_saldo_cash + res_saldo_bank) <> 0 or sum(res_saldo_bank_val + res_saldo_cash_val) <> 0)
      into :res_saldo_cash, :res_saldo_bank, :res_saldo_ekv, :res_region_name, :res_account, :res_account_name,
           :res_region_id, :res_currency, :res_saldo_cash_val, :res_saldo
  do
  begin
    select r2.reg_code || ' ' || r2.reg_name, r3.reg_code || ' ' || r3.reg_name, r3.reg_unicode
    from region r1, region r2, region r3
    where r1.reg_unicode = :res_region_id and
          r2.reg_unicode = r1.parent and
          r3.reg_unicode = r2.parent
    into :res_branch_name, :res_hq_name, :res_hq;
    if (res_hq = 1000000) then
    begin
      res_hq_name = res_branch_name;
    end
    suspend;
  end
end
