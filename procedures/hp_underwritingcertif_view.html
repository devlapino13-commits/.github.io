CREATE PROCEDURE HP_UNDERWRITINGCERTIF_VIEW
RETURNS (
    ID D_INTEGER,
    UCF_ID D_INTEGER,
    UCF_FROM_REG_ID D_INTEGER,
    UCF_DATE_BEGIN D_DATE,
    UCF_DATE_END D_DATE,
    UCF_ACTIVE D_BOOL,
    FIO D_STR_100,
    C_ID D_ID,
    UN_ID D_ID)
AS
declare variable reg_status d_smallint;
begin
  id = 0;
  for select iif(ucf_id is null, -1, ucf_id), iif(ucf_from_reg_id is null, reg.reg_unicode, ucf_from_reg_id),
             ucf_date_begin, ucf_date_end, iif(ucf_active is null, 0, ucf_active), ucf_from_un_id
      from region reg
      left join underwritingcertif ucf on ucf.ucf_from_reg_id = reg.reg_unicode
        and ucf.ucf_system = f_enum_id('access_sytem', 'undervriter_certif')
      where reg.reg_status = 3 and
            reg.is_delete = 0 and
            current_date between coalesce(reg.reg_date_open, '01.01.1900') and coalesce(reg.reg_date_close, '31.12.9999')
      into :ucf_id, :ucf_from_reg_id, :ucf_date_begin, :ucf_date_end, :ucf_active, :un_id
  do
  begin
    if (:ucf_from_reg_id in (1060100, 1090100)) then
      continue;
    id = id + 1;
    fio = '';
    c_id = 0;
    suspend;
    if (:ucf_active = 0) then
      continue;

    for select c.c_name_fio, ucfs.ucfs_date_begin, ucfs.ucfs_date_end, coalesce(ucfs.ucfs_active, 0), c.c_id,
               ucfs.ucfs_from_un_id
        from hr_pstatus ps
        inner join joblist jl on ps.ps_from_jl_id = jl.jl_id
        inner join jobkind jk on jl.jl_kind = jk.jk_id and
              jk.jk_system_code in ('JobSystem.LO_Senior', 'JobSystem.LO_Standart')
        inner join clients c on ps.ps_from_c_id = c.c_id
        left join underwritingcertifspf ucfs on c.c_id = ucfs.ucfs_from_c_id
        where ps.ps_is_working = 1 and
              ps.ps_status = 1 and
              ps.ps_from_reg_id = :ucf_from_reg_id and
              current_date between ps.ps_date_begin and coalesce(ps.ps_date_end, '31.12.2999')
        into :fio, :ucf_date_begin, :ucf_date_end, :ucf_active, :c_id, :un_id
    do
    begin
      id = id + 1;
      suspend;
    end
  end
end
