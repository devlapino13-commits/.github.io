CREATE PROCEDURE AAREP_SIGNED_QUEST_R_PERIOD(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    RES_REP_DATE D_DATE,
    RES_REG_NAME D_STR_CODE,
    RES_LOAN_OFFICER D_STR_100,
    RES_APP_DATE D_DATE,
    RES_C_NAME_FIO D_STR_100)
AS
begin
  -- для отчета "Подписанные анкеты-опросники по отказникам за период"
  while (rep_date_begin <= rep_date_end)
  do
  begin
    for select res_rep_date, res_reg_name, res_loan_officer, res_app_date, res_c_name_fio
    from aarep_signed_questionnaires_r(:rep_region_id, :rep_date_begin)
    into :res_rep_date, :res_reg_name, :res_loan_officer, :res_app_date, :res_c_name_fio
    do
      suspend;

    rep_date_begin = rep_date_begin + 1;
  end
end
