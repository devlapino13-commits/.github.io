CREATE PROCEDURE HP_PRE_SUD_COMMUNICATE(
    REP_K_ID ID_TYPE)
RETURNS (
    COM_TYPE STR_255,
    REP_TYPE D_INTEGER,
    COM_DATE D_DATE,
    COM_DATETIME D_DATETIME,
    FIO_CONT STR_100,
    PHONE_NO STR_100,
    UN_NAME STR_100,
    LAST_COM_DATE D_DATE,
    QTY_DELINQ D_INTEGER,
    QTY_LAST_30 D_INTEGER,
    QTY_LAST_7 D_INTEGER,
    QTY_TODAY D_INTEGER,
    RES_FILENAME D_STR_LONG,
    ADD_PARAMS D_STR_LONG)
AS
declare variable C_ID D_ID;
declare variable DELINQ_DATE D_DATE;
declare variable KC_ID D_ID;
declare variable MSG_TYPE integer;
declare variable COM_SUBTYPE varchar(50);
declare variable AUTOINFORMER_FILE D_STR_100;
begin
  rep_type = 0;
  com_subtype = '';

  for select f_cont_code(pnc_dst_phone_id), cast(pnc_start as date), pnc_start, src.c_name_fio,
             dst.c_name_fio || iif(pnc_is_contperson = 1, ' - Доп.номер', ' - Осн. номер'),
             'Телефонные переговоры КЦ'
      from phonecalls
      inner join clients src on src.c_id = pnc_src_c_id
      inner join clients dst on dst.c_id = pnc_dst_c_id
      where pnc_object_kind = 9 and
            pnc_object_id = :rep_k_id and
            pnc_direction = 1 and
            pnc_call_status_id = 1 and
            pnc_is_delinq = 1 and
            pnc_source = 1
      order by pnc_source, pnc_start desc
      into phone_no, com_date, com_datetime, un_name, fio_cont, com_type
  do
  begin
    suspend;
  end

  ------------------------------ Автоинформатор
  com_type = 'Автоинформатор';

  -- старая схема
  for select oic_call_date, oic_call_datetime, ois_object_id, 'Автоинформатор'
      from okt_ivr_calls
      inner join okt_ivr_calls_svod on ois_id = oic_from_ois_id
      inner join pantera_kontracts on pant_id = ois_from_pant_id
      inner join refdata on oic_call_result = rd_id
      where pant_from_k_id = :rep_k_id and
            oic_duration > 0
      order by oic_call_datetime desc
      into com_date, com_datetime, c_id, un_name
  do
  begin
    phone_no = '';

    select c_name_fio, f_cont_info('clients', c_id, 'cell_main', 'ok')
    from clients
    where c_id = :c_id
    into fio_cont, phone_no;

    autoinformer_file = 'auto_1'; -- RFCRU-4792
    res_filename = 'https://[basic_auth_ru_voice]record.molbulak.ru/' || autoinformer_file || '.mp3';
    add_params = autoinformer_file;

    suspend;
  end

  -- новая схема
  com_type = 'Автоинформатор (API)';  -- RFCRU-5197

  for select oac.oas_call_date, oac.oas_call_datetime, oac.oas_object_id, 'Автоинформатор'
      from okt_abonent_calls oac
      inner join pantera_kontracts on pant_id = oac.oas_from_pant_id
      inner join refdata on oac.oas_result = rd_id
      where pant_from_k_id = :rep_k_id and
            oac.oas_duration > 0 and
            oac.oas_is_ivr = 1
      order by oac.oas_call_datetime desc
      into com_date, com_datetime, c_id, un_name
  do
  begin
    phone_no = '';

    select c_name_fio, f_cont_info('clients', c_id, 'cell_main', 'ok')
    from clients
    where c_id = :c_id
    into fio_cont, phone_no;

    autoinformer_file = 'auto_1'; -- RFCRU-4792
    res_filename = 'https://[basic_auth_ru_voice]record.molbulak.ru/' || autoinformer_file || '.mp3';
    add_params = autoinformer_file;

    suspend;
  end
  --------------------------------

  com_type = 'Сообщения';
  add_params = '';
  un_name = 'Robot';
  for select kc_id, kc_from_c_id, c_name_fio
      from knclient
      inner join clients on kc_from_c_id = c_id
      where kc_from_k_id = :rep_k_id and
            kc_amount_approved > 0
      into kc_id, c_id, fio_cont
  do
  begin
    for select cast(msg_datetime as date), msg_datetime, phone_no, msg_body, msg_type
        from (select msg_datetime, msg_mail_to as phone_no, msg_body, msg_type
              from fsrobot_msgbox
              where msg_object_kind = f_objkind('knclient', 'id') and
                    msg_object_id = :kc_id and
                    msg_type = 1 -- СМС-сообщения. (рассылки не используют quick_sms)
              union all
              select msg_datetime, 'КМП' as phone_no,
                     f_json_str_get_value('{' || msg_body || '}', 'notification.body', ''), msg_type
              from fsrobot_msgbox
              where msg_object_kind = f_objkind('knclient', 'id') and
                    msg_object_id = :kc_id and
                    msg_type = 2 and -- PUSH-сообщения
                    msg_lifetime_days > 365 and
                    msg_command_kind <> 'fsrobot_mailing_sms_repay_ok'
              union all
              select msg_datetime, msg_mail_to as phone_no, msg_body, msg_type
              from sud_mainreq
              inner join edoc_list on edl_object_kind = 74 and
                    edl_object_id = smr_id
              inner join edoc_sign on eds_from_edl_id = edl_id
              inner join edoc_msg on edm_from_eds_id = eds_id
              inner join fsrobot_msgbox on msg_id = edm_from_msg_id
              where smr_from_k_id = :rep_k_id and
                    edl_status >= 0)
        order by msg_datetime desc
        into com_date, com_datetime, phone_no, add_params, msg_type
    do
    begin
      com_subtype = case msg_type
                      when 0 then 'Email'
                      when 1 then 'Sms'
                      when 2 then 'Push'
                      when 3 then 'Sms' -- quick_sms
                      when 4 then 'Email' -- quick_email
                      else 'Другое'
                    end;
      com_type = 'Сообщения (' || trim(com_subtype) || ')';
      suspend;
    end
  end
  add_params = '';

  delinq_date = null;

  select oks_delinq_date
  from oksummary
  where current_date between oks_date_begin and oks_date_end and
        oks_close_state = 0 and
        oks_from_k_id = :rep_k_id
  into delinq_date;

  if (delinq_date is null) then
  begin
    select oks_delinq_date
    from oksummary
    where oks_from_k_id = :rep_k_id and
          oks_delinq_date < '31.12.9999'
    order by oks_date_begin desc rows 1
    into delinq_date;
  end

  rep_type = 1;
  com_type = 'Телефонные переговоры КЦ';
  last_com_date = null;
  qty_delinq = 0;
  qty_today = 0;
  qty_last_7 = 0;
  qty_last_30 = 0;

  for select cast(pnc_start as date)
      from (select pnc_start
            from phonecalls
            inner join clients src on src.c_id = pnc_src_c_id
            inner join clients dst on dst.c_id = pnc_dst_c_id
            where pnc_object_kind = 9 and
                  pnc_object_id = :rep_k_id and
                  pnc_direction = 1 and
                  pnc_call_status_id = 1 and
                  pnc_is_delinq = 1 and
                  pnc_source = 1)
      order by pnc_start desc
      into com_date
  do
  begin
    if (com_date >= :delinq_date) then
    begin
      qty_delinq = qty_delinq + 1;
      last_com_date = maxvalue(coalesce(last_com_date, com_date), com_date);
    end
    if (com_date = current_date) then
      qty_today = qty_today + 1;
    if (com_date between current_date - 7 and current_date) then
      qty_last_7 = qty_last_7 + 1;
    if (com_date between current_date - 30 and current_date) then
      qty_last_30 = qty_last_30 + 1;
  end
  suspend;

  -------------------- Автоинформатор
  com_type = 'Автоинформатор';
  last_com_date = null;
  qty_delinq = 0;
  qty_today = 0;
  qty_last_7 = 0;
  qty_last_30 = 0;

  -- старая схема
  for select oic_call_date
      from (select oic_call_date
            from okt_ivr_calls
            inner join okt_ivr_calls_svod on ois_id = oic_from_ois_id
            inner join pantera_kontracts on pant_id = ois_from_pant_id
            inner join refdata on oic_call_result = rd_id
            where pant_from_k_id = :rep_k_id and
                  oic_duration > 0)
      order by oic_call_date desc
      into com_date
  do
  begin
    if (com_date >= :delinq_date) then
    begin
      qty_delinq = qty_delinq + 1;
      last_com_date = maxvalue(coalesce(last_com_date, com_date), com_date);
    end
    if (com_date = current_date) then
      qty_today = qty_today + 1;
    if (com_date between current_date - 7 and current_date) then
      qty_last_7 = qty_last_7 + 1;
    if (com_date between current_date - 30 and current_date) then
      qty_last_30 = qty_last_30 + 1;
  end
  suspend;

  -- новая схема
  com_type = 'Автоинформатор (API)';  -- RFCRU-5197
  last_com_date = null;
  qty_delinq = 0;
  qty_today = 0;
  qty_last_7 = 0;
  qty_last_30 = 0;

  for select oas_call_date
      from (select oac.oas_call_date
            from okt_abonent_calls oac
            inner join pantera_kontracts on pant_id = oac.oas_from_pant_id
            inner join refdata on oac.oas_result = rd_id
            where pant_from_k_id = :rep_k_id and
                  oac.oas_duration > 0 and
                  oac.oas_is_ivr = 1)
      order by oas_call_date desc
      into com_date
  do
  begin
    if (com_date >= :delinq_date) then
    begin
      qty_delinq = qty_delinq + 1;
      last_com_date = maxvalue(coalesce(last_com_date, com_date), com_date);
    end
    if (com_date = current_date) then
      qty_today = qty_today + 1;
    if (com_date between current_date - 7 and current_date) then
      qty_last_7 = qty_last_7 + 1;
    if (com_date between current_date - 30 and current_date) then
      qty_last_30 = qty_last_30 + 1;
  end
  suspend;
  ----------------

  com_type = 'Сообщения';
  for select kc_id, kc_from_c_id, c_name_fio
      from knclient
      inner join clients on kc_from_c_id = c_id
      where kc_from_k_id = :rep_k_id and
            kc_amount_approved > 0
      into kc_id, c_id, fio_cont
  do
  begin
    last_com_date = null;
    qty_delinq = 0;
    qty_today = 0;
    qty_last_7 = 0;
    qty_last_30 = 0;
    for select cast(msg_datetime as date)
        from (select msg_datetime
              from fsrobot_msgbox
              where msg_object_kind = f_objkind('knclient', 'id') and
                    msg_object_id = :kc_id and
                    msg_type = 1 -- СМС-сообщения. (рассылки не используют quick_sms)
              union all
              select msg_datetime
              from fsrobot_msgbox
              where msg_object_kind = f_objkind('knclient', 'id') and
                    msg_object_id = :kc_id and
                    msg_type = 2 and -- PUSH-сообщения
                    msg_lifetime_days > 365 and
                    msg_command_kind <> 'fsrobot_mailing_sms_repay_ok'
              union all
              select msg_datetime
              from sud_mainreq
              inner join edoc_list on edl_object_kind = 74 and
                    edl_object_id = smr_id
              inner join edoc_sign on eds_from_edl_id = edl_id
              inner join edoc_msg on edm_from_eds_id = eds_id
              inner join fsrobot_msgbox on msg_id = edm_from_msg_id
              where smr_from_k_id = :rep_k_id and
                    edl_status >= 0)
        order by msg_datetime desc
        into com_date
    do
    begin
      if (com_date >= :delinq_date) then
      begin
        qty_delinq = qty_delinq + 1;
        last_com_date = maxvalue(coalesce(last_com_date, com_date), com_date);
      end
      if (com_date = current_date) then
        qty_today = qty_today + 1;
      if (com_date between current_date - 7 and current_date) then
        qty_last_7 = qty_last_7 + 1;
      if (com_date between current_date - 30 and current_date) then
        qty_last_30 = qty_last_30 + 1;
    end
    suspend;
  end

end
