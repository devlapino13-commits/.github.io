CREATE PROCEDURE HP_KLADR_GET_LIST_UX
RETURNS (
    PKI_ID D_ID,
    PKI_NAME D_STR_255,
    PKI_TYPE D_STR_CODE,
    PKI_CODE D_STR_20,
    PKI_OKATD D_STR_20,
    PKI_PARENT D_ID,
    STATUS D_STR_20)
AS
declare variable state d_integer;
begin
  for select pki_id, pki_name, pki_type, pki_code, pki_okatd, pki_parent
  from places_kladr_import
  order by pki_code
  into pki_id, pki_name, pki_type, pki_code, pki_okatd, pki_parent
  do
  begin
    state = right(pki_code, 2);
    status = case
               when state = 0 then 'Актуальный'
               when state < 51 then 'Переименован'
               when state = 51 then 'Переподчинен'
               when state < 99 then 'Резерв'
               else 'Не существует'
             end;

    suspend;
  end
end
