CREATE PROCEDURE HP_GET_KONTUR_EXT_INFO(
    C_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    DATE_SUD_RD D_DATE,
    DATE_SUD_RI D_DATE,
    DATE_SUD_RIZ D_DATE,
    DATE_SUD_KPZ D_DATE,
    DATE_SUD_RDZ D_DATE,
    DATE_NSUD_VPVB D_DATE,
    DATE_NSUD_ZPVB D_DATE,
    DATE_NSUD_PPVB D_DATE,
    STATUS_OTHER D_STR_100,
    STATUS VARCHAR(50),
    STATUS_DATE DATE,
    IS_JUDICIAL SMALLINT,
    EIL_ID BIGINT,
    LINK D_STR_255,
    PUB_DATE D_DATE,
    DELO_NO D_STR_255)
AS
begin
  -- Свернутая история по банкротству клиента
  /*
    Документация:
    https://developer.kontur.ru/doc/focus/method?type=get&path=/api3/personBankruptcy
    Тестовое API:
    https://focus-api.kontur.ru/api3/personBankruptcy/userform

    Статус процедуры внесудебного банкротства:
      Возбуждена процедура внесудебного банкротства,
      Заявление о банкротстве возвращено гражданину,
      Прекращена процедура внесудебного банкротства,
      Завершена процедура внесудебного банкротства

    Стадии судебного банкротства:
      Наблюдение,
      Финансовое оздоровление,
      Внешнее управление,
      Конкурсное производство,
      Конкурсное производство завершено,
      Реструктуризация долгов,
      Реструктуризация долгов завершена,
      Реализация имущества,
      Реализация имущества завершена,
      Отказано в признании должника банкротом,
      Производство по делу прекращено,
      Не удалось определить стадию

  */

  -- Даты по статусам, попадающие в отчет
  date_sud_rd = null; -- Реструктуризация долгов
  date_sud_ri = null; -- Реализация имущества
  date_sud_riz = null; -- Реализация имущества завершена
  date_sud_kpz = null; -- Конкурсное производство завершено
  date_sud_rdz = null; -- Реструктуризация долгов завершена
  date_nsud_vpvb = null; -- Возбуждена процедура внесудебного банкротства
  date_nsud_zpvb = null; -- Завершена процедура внесудебного банкротства
  date_nsud_ppvb = null; -- Прекращена процедура внесудебного банкротства
  link = null; -- Ссылка на публикацию в ЕФРСБ

  -- Получаем все результативные запросы в Контур по данному клиенту
  for select eil_id
  from extinfocfg
    left join extinfolist on eil_from_eic_id = eic_id
  where eic_code = 'kontur'
    and eil_from_c_id = :c_id
    and eil_status = 2
    and eil_from_bls_id is not null
    and eil_date between :rep_date_begin and :rep_date_end
  order by 1
  into eil_id do
  begin
    select status, status_date, is_judicial, pub_date, delo_no, 'https://fedresurs.ru/bankruptmessages/' || message_id
    from hp_get_kontur_info(:eil_id)
    into status, status_date, is_judicial, pub_date, delo_no, link;

    if (status_date  not between rep_date_begin and rep_date_end) then
      continue; -- Более ранние статусы не интересуют

    -- Судебные
    if (status = 'Реструктуризация долгов' and date_sud_rd is null) then
      date_sud_rd = status_date;
    if (status = 'Реализация имущества' and date_sud_ri is null) then
      date_sud_ri = status_date;
    if (status = 'Реализация имущества завершена' and date_sud_riz is null) then
      date_sud_riz = status_date;
    if (status = 'Конкурсное производство завершено' and date_sud_kpz is null) then
      date_sud_kpz = status_date;
    if (status = 'Реструктуризация долгов завершена' and date_sud_rdz is null) then
      date_sud_rdz = status_date;

    -- Внесудебные
    if (status = 'Возбуждена процедура внесудебного банкротства' and date_nsud_vpvb is null) then
      date_nsud_vpvb = status_date;
    if (status = 'Завершена процедура внесудебного банкротства' and date_nsud_zpvb is null) then
      date_nsud_zpvb = status_date;
    if (status = 'Прекращена процедура внесудебного банкротства' and date_nsud_ppvb is null) then
      date_nsud_ppvb = status_date;

    if (is_judicial = 1 and status not in (
        'Реструктуризация долгов',
        'Реализация имущества',
        'Реализация имущества завершена',
        'Конкурсное производство завершено',
        'Реструктуризация долгов завершена')) then
      status_other = status;
  end

  suspend;
end
