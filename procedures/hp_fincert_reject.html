CREATE PROCEDURE HP_FINCERT_REJECT(
    MA_ID D_ID,
    K_ID D_ID)
AS
begin
  --RFCRU-3955, 4015 автоматический отказ при совпадении со списками финЦЕРТ - номер паспорта, ИНН, снилс
  if (:k_id is not null) then
  begin
    update kontract
    set k_status = -1,
      k_user_id = 1000000000000
    where k_id = :k_id;

    insert into kontreslist (krl_from_kr_id, krl_from_un_id, krl_from_k_id, krl_ma_reg_id, krl_ma_id,
                            krl_datetime, krl_date_old, krl_date_new, krl_from_fs)
    values (9, 1000000000000, :k_id, 0, null, current_timestamp, current_date, current_date, 0);

    select k.k_from_ma_app_id
    from kontract k
    where k.k_id = :k_id
    into ma_id;
  end
  else
  begin
    update miniapp
    set ma_is_active = 0
    where ma_id = :ma_id;

    insert into kontreslist (krl_from_kr_id, krl_from_un_id, krl_from_k_id, krl_ma_reg_id, krl_ma_id,
                            krl_datetime, krl_date_old, krl_date_new, krl_from_fs)
    values (9, 1000000000000, null, 0, :ma_id, current_timestamp, current_date, current_date, 0);
  end
end
