CREATE PROCEDURE HP_FIO_CLEAR(
    FIO_TXT VARCHAR(100))
RETURNS (
    FULL_TEXT VARCHAR(100),
    TXT1 VARCHAR(100),
    TXT2 VARCHAR(100),
    TXT3 VARCHAR(100),
    SHORT_FIO VARCHAR(100),
    NAME_SURNAME VARCHAR(100))
AS
declare variable kk integer;
begin
  txt1 = '';
  txt2 = '';
  txt3 = '';

  fio_txt = (select res_text
             from hp_txt_clear(:fio_txt));
  full_text = fio_txt;
  if (fio_txt = '') then
  begin
    suspend;
    exit;
  end

  kk = position(' ', :fio_txt);
  if (kk > 0) then
  begin
-- ФИО делим на два или три куска
    txt1 = substring(:fio_txt from 1 for kk - 1);
    fio_txt = substring(:fio_txt from kk + 1 for 100);
    fio_txt = trim(:fio_txt);
    kk = position(' ', fio_txt);
    if (kk > 0) then
    begin
      txt2 = substring(:fio_txt from 1 for kk - 1);
      fio_txt = substring(:fio_txt from kk + 1 for 100);
      fio_txt = trim(:fio_txt);
      txt3 = :fio_txt;
    end
    else
      txt2 = :fio_txt;
  end
  else
  begin
    txt1 = :fio_txt;
  end
  short_fio = left(txt1, 1) || lower(substring(txt1 from 2 for 100));
  if (txt2 <> '') then
    short_fio = short_fio || ' ' || left(txt2, 1) || '.';
  if (txt3 <> '') then
    short_fio = short_fio || ' ' || left(txt3, 1) || '.';

  name_surname = left(txt2, 1) || lower(substring(txt2 from 2 for 100)) || ' ' || left(txt1, 1) || lower(substring(txt1 from 2 for 100));
  suspend;
end
