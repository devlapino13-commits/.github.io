CREATE PROCEDURE HP_DOCPROP_DPK_LIST(
    PK_ID D_INTEGER,
    ONLY_LINKED D_BOOL,
    UN_ID D_ID = null)
RETURNS (
    DP_REC_ID D_INTEGER,
    DP_ID D_INTEGER,
    DP_CODE D_STR_CODE,
    DP_NAME D_STR_255,
    DP_MEMO D_STR_255,
    DP_POSITION D_INTEGER,
    ORIG_GROUP_NO D_INTEGER,
    DPK_GROUP_NO D_INTEGER,
    DPK_CHECKED_0 D_INTEGER,
    DPK_CHECKED_1 D_INTEGER,
    DPK_CHECKED_2 D_INTEGER)
AS
declare variable citizen_kind d_integer;
declare variable valid_kind d_integer;
begin
  if (pk_id is null) then
    pk_id = 0;

  for select dp_id, dp_code, dp_name, dp_memo, dp_position
  from docprop
  order by dp_position, dp_id
  into :dp_id, :dp_code, :dp_name, :dp_memo, :dp_position
  do
  begin
    for select 0
    from rdb$database
    union
    select distinct dpk_group_no
    from docproppk
    where dpk_from_dp_id = :dp_id and
          dpk_from_pk_id = :pk_id and
          dpk_group_no <> 0 and
          dpk_is_delete = 0
    into dpk_group_no
    do
    begin
      dpk_checked_0 = -1;
      dpk_checked_1 = -1;
      dpk_checked_2 = -1;

      dp_rec_id = dp_id * 100 + dpk_group_no;
  
      for select dpk_citizen_kind, dpk_valid_kind
      from docproppk
      where dpk_from_dp_id = :dp_id and
            dpk_from_pk_id = :pk_id and
            dpk_group_no = :dpk_group_no and
            dpk_is_delete = 0
      into citizen_kind, valid_kind
      do
      begin
        if (citizen_kind = 0) then
          dpk_checked_0 = valid_kind;
        if (citizen_kind = 1) then
          dpk_checked_1 = valid_kind;
        if (citizen_kind = 2) then
          dpk_checked_2 = valid_kind;
      end

      orig_group_no = dpk_group_no;

      if (dpk_group_no = 0) then
        dpk_group_no = null;

      if (only_linked = 0 or
          dpk_checked_0 <> -1 or
          dpk_checked_1 <> -1 or
          dpk_checked_2 <> -1) then
        suspend;
    end
  end
end
