CREATE PROCEDURE AAREP_KO_NBKI_GUARANTEE(
    JSA D_INTEGER)
RETURNS (
    UID D_STR_255,
    GUARANTEE_SUM D_CURRENCY,
    GUARANTEE_DATE D_DATE,
    END_DATE D_DATE,
    FACT_END_DATE D_DATE,
    END_REASON_STR D_STR_255)
AS
declare variable jso d_integer;

declare function getcatval(kind d_str_code, code d_str_code)
  returns d_str_255
  as
  begin
    return coalesce((select bct_name from bki_catalog where bct_kind = :kind and bct_code = :code), 'Неопределен');
  end

begin
  if (coalesce(jsa, 0) = 0) then
    exit;

  for select jso
  from hp_bki_json_get_list(:jsa, 1)
  into jso
  do
  begin
    uid = json.obj_get_val(jso, 'guarantee_uuid');
    guarantee_sum = f_bki_json_get_num(jso, 'guarantee_volume');
    guarantee_date = f_bki_json_get_date(jso, 'guarantee_agreement_dt');
    end_date = f_bki_json_get_date(jso, 'guarantee_expiration_date');
    fact_end_date = f_bki_json_get_date(jso, 'guarantee_fact_expiration_date');
    end_reason_str = getcatval('guarantee_endReason', json.obj_get_val(jso, 'guarantee_end_reason'));

    if (uid is not null or guarantee_sum is not null or guarantee_date is not null) then
      suspend;
  end
end
