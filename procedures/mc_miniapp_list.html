CREATE PROCEDURE MC_MINIAPP_LIST(
    MCL_ID D_ID,
    C_ID D_ID)
RETURNS (
    MA_ID D_INTEGER,
    OWN_DISB_AMOUNT D_CURRENCY,
    DISB_DATE D_STR_10,
    DISB_AMOUNT D_CURRENCY,
    MA_STATUS D_SMALLINT,
    MA_STATUS_STR D_STR_100,
    KF_NAME D_STR_100,
    DATE_APP D_STR_10,
    SVZ_FIO D_STR_100,
    OFFICE_DATA D_STR_LONG,
    CELL_NO D_STR_100,
    SHOW_REJECTION_BUTTON D_BOOL)
AS
declare variable KF_ID D_ID;
declare variable DISB_OL D_ID;
declare variable CLOSE_OL D_ID;
declare variable MCL_PROJECT D_STR_20;
begin
  /*
      МП для клиентов.
      ХП предоставляет список заявок.
      Заявка отображается если находится в статусе: Отказ, На рассмотрении, Утвержден
    
      DISB_DATE - дата выдачи
      DISB_AMOUNT - общая сумма займа (всей группы)
      OWN_DISB_AMOUNT - собственная сумма займа
      MA_STATUS - статус, в котором находится заявка в числовом выражении
      MA_STATUS_STR - статус, в котором находится заявка в текстовом представлении
      KF_NAME - номер группы
      DATE_APP - дата первичной регистрации заявки
      SVZ_FIO - ФИО СВЗ за которым закреплена группа
      OFFICE_DATA - данные по офису где числится заявка в формате: 0205, Б.Ордынка, адрес: Москва, ул. Б.Ордынка 5
      CELL_NO - телефонный номер
      SHOW_REJECTION_BUTTON - показывать/не показывать" кнопку "Отказаться"
  */

  select f_enum('mcl_project', mcl.mcl_project, 'code', 0)
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into :mcl_project;

  -- RFCRU-3766
  for select maa.ma_id, macl.mac_amount, maa.ma_from_kf_id, f_datetostr(maa.ma_date_oper), -- планируемая дата выдачи
             maa.ma_summa_minus,
-- планиуремая сумма выдачи
             (
             case maa.ma_is_active
               when -1 then -1 --добавил на основании задачи RFCRU-1589
               when 0 then -1
               when 1 then 0
             end)
      from miniappclient macl
      inner join miniapp maa on maa.ma_id = macl.mac_from_ma_id
      inner join product pr on maa.ma_from_pr_id = pr.pr_id
        and iif(:mcl_project = 'oivo', pr.pr_code containing '.oivo_pack.', pr.pr_code not containing '.oivo_pack.')
      where macl.mac_from_c_id = :c_id and
            macl.mac_amount > 0 and
            maa.ma_date_end = '31.12.9999'
      order by ma_id desc
      into :ma_id, :own_disb_amount, :kf_id, :disb_date, :disb_amount, :ma_status
  do
  begin
-- получаю дату первичной регистрации заявки
    select f_datetostr(maa.ma_date_app)
    from miniapp maa
    where maa.ma_id = :ma_id
    order by maa.ma_id
    rows 1
    into :date_app;

    select kal.kal_new_kf_name, cl.c_name_fio,
           reg.reg_code || ', ' || reg.reg_name || coalesce(', адрес: ' || adl.adl_address_str, ''),
           f_cont_info('clients', c_id, 'cell_main', 'ok')
    from knaktlist kal
    inner join clients cl on cl.c_id = kal.kal_new_kf_officer
    inner join region reg on reg.reg_unicode = kal.kal_new_reg_id
    left join addrlist adl on adl.adl_id = reg.reg_address_fact
    where kal.kal_from_kf_id = :kf_id and
          kal.is_delete = 0 and
          current_date between kal.kal_date and kal.kal_date_end
    into :kf_name, :svz_fio, :office_data, :cell_no;

--if (ma_status = 0) then  -- если заявка не аннулирована, тогда прохожусь по кредиту
--begin
    disb_ol = null;
    close_ol = null;

    select iif(coalesce(kont.k_amount_approved, 0) > 0, kont.k_amount_approved, kont.k_amount_applicate),
           iif(coalesce(kncl.kc_amount_approved, 0) > 0, kncl.kc_amount_approved, kncl.kc_amount_applicate),
           f_datetostr(kont.k_disb_date), kont.k_status, ol_disb.ol_id, ol_close.ol_id
    from kontract kont
    inner join knclient kncl on kncl.kc_from_k_id = kont.k_id
    left join operlist ol_disb on ol_disb.ol_id = kont.k_fact_disb_ol_id
    left join operlist ol_close on ol_close.ol_id = kont.k_fact_closed_ol_id
    where kont.k_from_ma_app_id = :ma_id and
          kncl.kc_amount_approved > 0 and
          kncl.kc_from_c_id = :c_id
    into :disb_amount, :own_disb_amount, :disb_date, :ma_status, :disb_ol, :close_ol;
--end

    if ((disb_ol is null) and
        (:close_ol is null)) then
    begin
      ma_status_str = (case ma_status
                         when -1 then 'Отказ'
                         when 0 then 'На рассмотрении'
                         when 1 then 'Утержден'
                       end);

      show_rejection_button = f_freeze_get_reject_status(ma_id);

      suspend;
    end
  end
end
