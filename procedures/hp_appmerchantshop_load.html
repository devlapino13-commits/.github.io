CREATE PROCEDURE HP_APPMERCHANTSHOP_LOAD(
    IQA_ID D_ID NOT NULL)
RETURNS (
    NUM_ADD D_INTEGER,
    NOSIGN_LST D_STR_8100,
    MSG_TEXT D_STR_255)
AS
declare variable dt_app d_date;
declare variable c_id d_id;
declare variable c_name_fio d_str_100;
declare variable cs_id d_id;
declare variable cs_name d_str_255;
declare variable adl_id d_id;
declare variable adl_str d_str_255;
declare variable ag_id d_integer;
declare variable dogovor_tsp d_integer;
declare variable rd_id_tsp d_integer;
begin
  dt_app = (select iqa_date from ipc_qr_app where iqa_id = :iqa_id);
  num_add = 0;
  nosign_lst = '';
  msg_text = '';
  rd_id_tsp = f_get_rd_id('AgreementKinds.tsp');

  for
    select
      c_id,
      c_name_fio,
      cs_id,
      cs_name,
      cs_from_adl_id
    from clientshops
      inner join clients
        on c_id = cs_from_c_id
    where cs_qr_data is null and
        cs_from_rd_id = :rd_id_tsp
  into
    :c_id,
    :c_name_fio,
    :cs_id,
    :cs_name,
    :adl_id
  do
  begin
    ag_id = (
      select first 1 ag_id
      from agreements
      where ag_object_kind = 0
        and ag_object_id = :c_id
        and ag_from_rd_id = :rd_id_tsp
        and :dt_app between coalesce(ag_date_begin, '01.01.1900') and coalesce(ag_date_end, '31.12.9999')
        and ag_status = 1
        order by ag_id desc);

    if (ag_id > 0) then
    begin
      if (not exists(
        select *
        from ipc_qr_app_shops
          inner join ipc_qr_app
            on iqa_id = iqs_from_iqa_id
              and iqa_status >= 0
        where iqs_from_cs_id = :cs_id)) then
      begin
        if (f_edoc_status('tsp.agreement', :ag_id, :dt_app) <> 'signed') then
       begin
          dogovor_tsp = 0;
          select cdc.cds_id
          from clientdocs cdc
          inner join passkind pask on pask.pk_id = cdc.cds_from_pk_id
          inner join passdoctype pdt on pdt.pdt_id = pask.pk_from_pdt_id
          where cdc.cds_from_c_id = :c_id and
                cdc.cds_is_delete = 0 and
                pdt.pdt_code = 'tsp_agreement' and
                :dt_app between coalesce(cdc.cds_doc_date_begin, '01.01.1900') and coalesce(cdc.cds_doc_date_end, '31.12.9999')
          order by cdc.cds_datetime desc
          rows 1
          into dogovor_tsp;
          if (coalesce(dogovor_tsp, 0) = 0) then
        begin
          nosign_lst = nosign_lst || iif(nosign_lst = '', '', ';') || cs_id;
          continue;
        end
       end

        if (adl_id > 0) then
          adl_str = (select adl_address_str from addrlist where adl_id = :adl_id);
        else
          adl_str = '';
        cs_name = cs_name || '||' || cs_id;
        insert into ipc_qr_app_shops (
          iqs_from_iqa_id,
          iqs_from_cs_id,
          iqs_tcp_data,
          iqs_shop_name
        )
        values(
          :iqa_id,
          :cs_id,
          :c_name_fio || iif(:adl_str = '', '', ', ' || :adl_str),
          :cs_name
        );
        num_add = num_add + 1;
      end
    end
  end
  msg_text = iif(num_add = 0, 'ТСП для загрузки не определены', f_format('Загружено ТСП без QR-кода: %1', num_add));
  suspend;
end
