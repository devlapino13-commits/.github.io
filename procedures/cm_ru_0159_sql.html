CREATE PROCEDURE CM_RU_0159_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable MA_EXPERT_ID ID_TYPE;
declare variable MA_OFFICE_ID integer;
declare variable PR_CODE varchar(100);
declare variable C_ID D_ID;
begin
  for select pr_code, ma.ma_expert_id
      from miniapp ma, product pr
      where pr.pr_id = ma.ma_from_pr_id and
            ma_id = :ma_id and
            ma.ma_date_end = '31.12.9999' and
            pr_code containing '.zalog_car.'
      into pr_code, ma_expert_id
  do
  begin
    if (not exists(select * from car_disb_svz_list cd
      where cd.c_id = :ma_expert_id and
      current_date between cd.c_date_start and cd.c_date_end)) then
    begin
      res_error_text = 'Выдача по продуку "Автозайм" запрещен (Приказ № 41-03-20)';
      res_error_is_critical = 1;
      suspend;
    end 

    for
    select mac_from_c_id from miniappclient
    where mac_from_ma_id = :ma_id
    and mac_amount > 0
    into c_id
    do
    begin
      if (exists(
      select * from knclient, kontstate
      where kc_from_c_id = :c_id
      and kc_amount_approved > 0
      and kc_from_k_id = ks_from_k_id
      and ks_ol_is_delete = 0
      and ks_state = 1
      and current_date between ks_date_begin and ks_date_end)) then
      begin
        res_error_text = 'У клиента есть активный заем в Компании - выдача Автозайма запрещена!';
        res_error_is_critical = 2;
        suspend;
      end
    end
  end
end
