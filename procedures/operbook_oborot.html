CREATE PROCEDURE OPERBOOK_OBOROT(
    DATE_BEGIN DATE,
    DATE_END DATE,
    REGION_ID INTEGER,
    IS_NOMINAL SMALLINT,
    IS_OBOROT SMALLINT)
RETURNS (
    OBR_ID VARCHAR(20),
    OBR_CURRENCY VARCHAR(3),
    OBR_BEGIN_DT DECIMAL(18,4),
    OBR_BEGIN_KT DECIMAL(18,4),
    OBR_OBOROT_DT DECIMAL(18,4),
    OBR_OBOROT_KT DECIMAL(18,4),
    OBR_LAST_DATE DATE)
AS
declare variable res_saldo decimal(18,4);
begin
  if (is_nominal = 0) then
  begin
    obr_begin_dt = 0;
    obr_begin_kt = 0;
    obr_oborot_dt = 0;
    obr_oborot_kt = 0;

    for select al_id, al_currency
        from acc_list, region_list(:region_id, 1, -1, 1)
        where al_from_reg_id = id_list
        into :obr_id, :obr_currency
    do
    begin
      res_saldo = f_al_bal(obr_id, date_begin - 1);
      obr_begin_dt = 0;
      obr_begin_kt = 0;

      select obs_date
      from obsummary
      where obs_from_al_id = :obr_id and
            obs_date < :date_begin
      order by obs_from_al_id desc, obs_date desc
      rows 1
      into :obr_last_date;

      if (res_saldo is null) then
        res_saldo = 0;
      if (res_saldo >= 0) then
        obr_begin_dt = res_saldo;
      else
        obr_begin_kt = -res_saldo;

      if (res_saldo <> 0) then
        suspend;
    end

    if (is_oborot = 1) then
    begin
      obr_begin_dt = 0;
      obr_begin_kt = 0;
      obr_oborot_dt = 0;
      obr_oborot_kt = 0;

      for select distinct obs_from_al_id, al_currency
      from obsummary, acc_list, region_list(:region_id, 1, -1, 1)
      where obs_from_al_id = al_id and
            al_from_reg_id = id_list and
            obs_date between :date_begin and :date_end
      into :obr_id, :obr_currency
      do
      begin
        select dt_ekv, kt_ekv, max_date
        from hp_al_obor(:obr_id, :date_begin, :date_end)
        into :obr_oborot_dt, :obr_oborot_kt, :obr_last_date;

        if (obr_oborot_dt = 0 and obr_oborot_kt = 0) then
          Continue;

        suspend;
      end
    end
  end
  else
  begin
    obr_begin_dt = 0;
    obr_begin_kt = 0;
    obr_oborot_dt = 0;
    obr_oborot_kt = 0;

    for select al_id, al_currency
        from acc_list, region_list(:region_id, 1, -1, 1)
        where al_from_reg_id = id_list
        into :obr_id, :obr_currency
    do
    begin
      res_saldo = f_al_bal(obr_id, date_begin - 1);

      obr_begin_dt = 0;
      obr_begin_kt = 0;
      select obs_date
      from obsummary
      where obs_from_al_id = :obr_id and
            obs_date < :date_begin
      order by obs_from_al_id desc, obs_date desc
      rows 1
      into :obr_last_date;

      if (res_saldo >= 0) then
        obr_begin_dt = res_saldo;
      else
        obr_begin_kt = -res_saldo;
      if (res_saldo <> 0) then
        suspend;
    end

    if (is_oborot = 1) then
    begin
      obr_begin_dt = 0;
      obr_begin_kt = 0;
      obr_oborot_dt = 0;
      obr_oborot_kt = 0;

      for select distinct obs_from_al_id, al_currency
      from obsummary, acc_list, region_list(:region_id, 1, -1, 1)
      where obs_from_al_id = al_id and
            al_from_reg_id = id_list and
            obs_date between :date_begin and :date_end
      into :obr_id, :obr_currency
      do
      begin
        select dt_ekv, kt_ekv, max_date
        from hp_al_obor(:obr_id, :date_begin, :date_end)
        into :obr_oborot_dt, :obr_oborot_kt, :obr_last_date;

        if (obr_oborot_dt = 0 and obr_oborot_kt = 0) then
          Continue;

        suspend;
      end
    end
  end
end
