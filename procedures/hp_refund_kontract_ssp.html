CREATE PROCEDURE HP_REFUND_KONTRACT_SSP(
    REP_K_ID ID_TYPE,
    REP_DATE D_DATE)
RETURNS (
    CALC_MONTH_REPAY_FINAL D_CURRENCY)
AS
declare variable kc_id id_type;
declare variable credenddate d_date;
declare variable calc_period_month d_integer;
declare variable debtcurr_opsum d_currency;
declare variable debtcurr_percentsum d_currency;
declare variable future_percent_sum d_currency;
declare variable debtoverdue_opsum d_currency;
declare variable debtoverdue_percentsum d_currency;
begin
  -- возвращает ССП (среднемесячная сумма платежа) потенциального для рефинансирования внутреннего займа на дату
  -- используется в МП в окне "Займы для рефинансирования"

  select kc_id, k_kont_last_date
  from kontract, knclient
  where kc_from_k_id = k_id
    and k_id = :rep_k_id
  into kc_id, credenddate;

  calc_period_month = 0;
  if (rep_date <= credenddate) then
    calc_period_month = (extract(year from credenddate) - extract(year from rep_date)) * 12
                       + extract(month from credenddate) - extract(month from rep_date)
                       + iif(extract(day from credenddate) > extract(day from rep_date), 1, 0);

  select res_delinq_os, res_delinq_perc, res_future_os, res_future_calc_perc, res_future_perc_without_calc
  from hp_pdn_kontract_info(:kc_id, :rep_date)
  into debtOverdue_opSum, debtOverdue_percentSum, debtcurr_opSum, debtcurr_percentSum, future_percent_sum;

  calc_month_repay_final = (debtcurr_opSum + debtcurr_percentsum + future_percent_sum)/iif(calc_period_month > 0, calc_period_month, 1);
  calc_month_repay_final = calc_month_repay_final + debtoverdue_opsum + debtoverdue_percentsum;
  calc_month_repay_final = round(calc_month_repay_final , 2);

  suspend;
end
