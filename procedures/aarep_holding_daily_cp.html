CREATE PROCEDURE AAREP_HOLDING_DAILY_CP(
    REP_REGION_ID ID_TYPE,
    REP_DATE_END D_DATE)
RETURNS (
    OFFICE_NAME D_STR_100,
    PERIOD_DATE D_DATE,
    QUANTITY_WORKERS D_INTEGER,
    TOTAL_CNT D_INTEGER,
    TOTAL_SUM D_INTEGER,
    BALANCE_OFFICE D_CURRENCY,
    PAR1 D_CURRENCY,
    PAR30 D_CURRENCY,
    PAR_PERCENT1 D_CURRENCY,
    PAR_PERCENT30 D_CURRENCY)
AS
declare variable office_id id_type;
declare variable month_begin d_date;
declare variable d d_date;
declare variable period_begin d_date;
declare variable period_end d_date;
declare variable period_last_day d_date;
declare variable cur_period d_date;
declare variable sum_total_cnt d_integer;
declare variable sum_total_sum d_integer;
declare variable total_cnt_all_day d_integer;
declare variable total_sum_all_day d_integer;
declare variable total_works_all_day d_integer;
declare variable balance_office_all_day d_currency;
declare variable par1_all_day d_currency;
declare variable par30_all_day d_currency;
declare variable base_begin d_date;
declare variable base_month d_date;
declare variable rep_month d_date;
declare variable months_diff d_integer;
declare variable shift_blocks d_integer;
declare variable shift_months d_integer;
begin
    -- первый день текущего месяца
    month_begin = dateadd(1 - extract(day from :rep_date_end) day to :rep_date_end);

    -- прошлые месяцы
    base_begin = cast('01.09.2025' as date);

    period_end = :month_begin - 1;

    -- Приводим обе даты к 1 числу месяца
    base_month = dateadd(1 - extract(day from base_begin) day to base_begin);
    rep_month  = dateadd(1 - extract(day from :rep_date_end) day to :rep_date_end);

    -- Сколько ПОЛНЫХ месяцев прошло
    months_diff = datediff(month from :base_month to :rep_month);

    if (months_diff < 9) then
    begin
        period_begin = base_month;
    end
    else
    begin
        -- Сколько полных 9-месячных блоков прошло
        shift_blocks = months_diff / 9;
    
        -- Каждый блок = +3 месяца
        shift_months = shift_blocks * 3;
    
        period_begin = dateadd(shift_months month to base_month);
    end

    -- ======== Отчет по каждому офису =========
    for select reg.reg_unicode, reg.reg_name
        from region reg
        where reg.reg_unicode in (1020100, 1030100, 1040200)
        into :office_id, :office_name
    do
    begin
        sum_total_cnt = 0;
        sum_total_sum = 0;

        -- ==== Прошлые месяцы для офиса ====
        cur_period = :period_begin;
        while (cur_period <= period_end) do
        begin
            period_last_day = dateadd(1 month to :cur_period) - 1;

            -- количество сотрудников
            select count(distinct hrp.ps_from_c_id)
            from hr_pstatus hrp
            where hrp.ps_from_reg_id = :office_id and
                 :period_last_day between hrp.ps_date_begin and hrp.ps_date_end and
                 hrp.ps_from_jl_id = 7 and
                 hrp.ps_status = 1 and
                 hrp.ps_is_working = 1
            into :quantity_workers;

            select count(ks.ks_from_k_id),
                   coalesce(sum(kc.kc_amount_approved),0)
            from kontstate ks
            inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
               and kc.kc_amount_approved > 0
            where ks.ks_region_id = :office_id
              and ks.ks_state = 1
              and ks.ks_ol_is_delete = 0
              and ks.ks_date_begin >= :cur_period
              and ks.ks_date_begin < dateadd(1 month to :cur_period)
            into :total_cnt, :total_sum;

            select sum(oks_balance),
                   sum(iif(:period_last_day - oks_delinq_date > 0, oks_balance, 0)),
                   sum(iif(:period_last_day - oks_delinq_date > 30, oks_balance, 0))
            from oksummary
            where oks_region_id = :office_id
              and :period_last_day between oks_date_begin and oks_date_end
              and oks_close_state = 0
            into :balance_office,
                 :par1,
                 :par30;

            balance_office = round(coalesce(balance_office,0)/1000000,2);
            par1 = round(coalesce(par1,0)/1000000,2);
            par30 = round(coalesce(par30,0)/1000000,2);
            par_percent1  = iif(balance_office > 0, (par1  / balance_office) * 100, 0);
            par_percent30 = iif(balance_office > 0, (par30 / balance_office) * 100, 0);

            sum_total_cnt = sum_total_cnt + coalesce(total_cnt,0);
            sum_total_sum = sum_total_sum + coalesce(total_sum,0);

            period_date = :cur_period;
            suspend;

            cur_period = dateadd(1 month to :cur_period);
        end

        -- ==== Текущий месяц по дням для офиса ====
        d = :month_begin;
        while (d <= :rep_date_end) do
        begin
            -- количество сотрудников
            select count(distinct hrp.ps_from_c_id)
            from hr_pstatus hrp
            where hrp.ps_from_reg_id = :office_id and
                 :d between hrp.ps_date_begin and hrp.ps_date_end and
                 hrp.ps_from_jl_id = 7 and
                 hrp.ps_status = 1 and
                 hrp.ps_is_working = 1
            into :quantity_workers;

            select count(ks.ks_from_k_id),
                   coalesce(sum(kc.kc_amount_approved),0)
            from kontstate ks
            inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
               and kc.kc_amount_approved > 0
            where ks.ks_region_id = :office_id
              and ks.ks_state = 1
              and ks.ks_ol_is_delete = 0
              and ks.ks_date_begin = :d
            into :total_cnt, :total_sum;

            select sum(oks_balance),
                   sum(iif(:d - oks_delinq_date > 0, oks_balance, 0)),
                   sum(iif(:d - oks_delinq_date > 30, oks_balance, 0))
            from oksummary
            where oks_region_id = :office_id
              and :d between oks_date_begin and oks_date_end
              and oks_close_state = 0
            into :balance_office,
                 :par1,
                 :par30;

            balance_office = round(coalesce(balance_office,0)/1000000,2);
            par1 = round(coalesce(par1,0)/1000000,2);
            par30 = round(coalesce(par30,0)/1000000,2);
            par_percent1  = iif(balance_office > 0, (par1  / balance_office) * 100, 0);
            par_percent30 = iif(balance_office > 0, (par30 / balance_office) * 100, 0);

            sum_total_cnt = sum_total_cnt + coalesce(total_cnt,0);
            sum_total_sum = sum_total_sum + coalesce(total_sum,0);

            period_date = :d;
            suspend;

            d = d + 1;
        end

        -- ==== Итог по офису ====
        office_name = 'Итого по офису';
        quantity_workers = null;
        total_cnt = sum_total_cnt;
        total_sum = sum_total_sum;
        balance_office = null;
        par1 = null;
        par30 = null;
        par_percent1 = null;
        par_percent30 = null;
        period_date = null;
        suspend;
    end

    -- ======== Итоги по всем офисам по каждой дате =========
    -- Прошлые месяцы
    cur_period = :period_begin;
    while (cur_period <= period_end) do
    begin
        period_last_day = dateadd(1 month to :cur_period) - 1;

        total_cnt_all_day = 0;
        total_sum_all_day = 0;
        total_works_all_day = 0;
        balance_office_all_day = 0;
        par1_all_day = 0;
        par30_all_day = 0;

        for  select reg.reg_unicode
            from region reg
            where reg.reg_unicode in (1020100, 1030100, 1040200)
            into :office_id
        do
        begin
            -- количество сотрудников
            select count(distinct hrp.ps_from_c_id)
            from hr_pstatus hrp
            where hrp.ps_from_reg_id = :office_id and
                 :period_last_day between hrp.ps_date_begin and hrp.ps_date_end and
                 hrp.ps_from_jl_id = 7 and
                 hrp.ps_status = 1 and
                 hrp.ps_is_working = 1
            into :quantity_workers;

            -- суммы за месяц офиса
            select count(ks.ks_from_k_id),
                   coalesce(sum(kc.kc_amount_approved),0)
            from kontstate ks
            inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
               and kc.kc_amount_approved > 0
            where ks.ks_region_id = :office_id
              and ks.ks_state = 1
              and ks.ks_ol_is_delete = 0
              and ks.ks_date_begin >= :cur_period
              and ks.ks_date_begin < dateadd(1 month to :cur_period)
            into :total_cnt, :total_sum;

            select sum(oks_balance),
                   sum(iif(:period_last_day - oks_delinq_date > 0, oks_balance, 0)),
                   sum(iif(:period_last_day - oks_delinq_date > 30, oks_balance, 0))
            from oksummary
            where oks_region_id = :office_id
              and :period_last_day between oks_date_begin and oks_date_end
              and oks_close_state = 0
            into :balance_office,
                 :par1, 
                 :par30;

            total_cnt_all_day = total_cnt_all_day + coalesce(total_cnt, 0);
            total_sum_all_day = total_sum_all_day + coalesce(total_sum, 0);
            total_works_all_day = total_works_all_day + coalesce(quantity_workers, 0);

            balance_office_all_day = balance_office_all_day + coalesce(balance_office,0);
            par1_all_day = par1_all_day + coalesce(par1,0);
            par30_all_day = par30_all_day + coalesce(par30,0);
        end

        office_name = 'Итого по всем офисам';
        period_date = :period_last_day;
        quantity_workers = total_works_all_day;
        total_cnt = total_cnt_all_day;
        total_sum = total_sum_all_day;
        balance_office = round(balance_office_all_day / 1000000, 2);
        par1 = round(par1_all_day / 1000000, 2);
        par30 = round(par30_all_day / 1000000, 2);
        par_percent1 = iif(balance_office_all_day > 0,
                            round((par1_all_day / balance_office_all_day) * 100,4),
                            0);
        par_percent30 = iif(balance_office_all_day > 0,
                            round((par30_all_day / balance_office_all_day) * 100,4),
                            0);
        suspend;

        cur_period = dateadd(1 month to :cur_period);
    end

    -- Дни текущего месяца
    d = :month_begin;
    while (d <= :rep_date_end) do
    begin
        total_cnt_all_day = 0;
        total_sum_all_day = 0;
        total_works_all_day = 0;
        balance_office_all_day = 0;
        par1_all_day = 0;
        par30_all_day = 0;

        for select reg.reg_unicode
            from region reg
            where reg.reg_unicode in (1020100, 1030100, 1040200)
            into :office_id
        do
        begin
            -- количество сотрудников
            select count(distinct hrp.ps_from_c_id)
            from hr_pstatus hrp
            where hrp.ps_from_reg_id = :office_id and
                 :d between hrp.ps_date_begin and hrp.ps_date_end and
                 hrp.ps_from_jl_id = 7 and
                 hrp.ps_status = 1 and
                 hrp.ps_is_working = 1
            into :quantity_workers;

            select count(ks.ks_from_k_id),
                   coalesce(sum(kc.kc_amount_approved),0)
            from kontstate ks
            inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
               and kc.kc_amount_approved > 0
            where ks.ks_region_id = :office_id
              and ks.ks_state = 1
              and ks.ks_ol_is_delete = 0
              and ks.ks_date_begin = :d
            into :total_cnt, :total_sum;

            select sum(oks_balance),
                   sum(iif(:d - oks_delinq_date > 0, oks_balance, 0)), 
                   sum(iif(:d - oks_delinq_date > 30, oks_balance, 0))
            from oksummary
            where oks_region_id = :office_id
              and :d between oks_date_begin and oks_date_end
              and oks_close_state = 0
            into :balance_office,
                 :par1,
                 :par30;

            total_cnt_all_day = total_cnt_all_day + coalesce(total_cnt,0);
            total_sum_all_day = total_sum_all_day + coalesce(total_sum,0);
            total_works_all_day = total_works_all_day + coalesce(quantity_workers, 0);

            balance_office_all_day = balance_office_all_day + coalesce(balance_office,0);
            par1_all_day = par1_all_day + coalesce(par1,0);
            par30_all_day = par30_all_day + coalesce(par30,0);
        end

        office_name = 'Итого по всем офисам';
        period_date = :d;
        quantity_workers = total_works_all_day;
        total_cnt = total_cnt_all_day;
        total_sum = total_sum_all_day;
        balance_office = round(balance_office_all_day / 1000000, 2);
        par1 = round(par1_all_day / 1000000, 2);
        par30 = round(par30_all_day / 1000000, 2);
        par_percent1 = iif(balance_office_all_day > 0,
                            round((par1_all_day / balance_office_all_day) * 100,4),
                            0);
        par_percent30 = iif(balance_office_all_day > 0,
                            round((par30_all_day / balance_office_all_day) * 100,4),
                            0);
        suspend;

        d = d + 1;
    end
end
