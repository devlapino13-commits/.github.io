CREATE PROCEDURE HP_CLIENTSHOPS_SAVE(
    CS_ID D_ID,
    CS_FROM_C_ID D_ID,
    CS_NAME D_STR_255,
    CS_NAME_LAT D_STR_255,
    CS_FROM_ADL_ID D_ID,
    CS_CONTACT_FIO D_STR_100,
    CS_PHONE D_STR_20,
    CS_HOURS D_STR_20,
    CS_WEEKEND_HOURS D_STR_20,
    CS_FROM_UN_ID D_ID,
    CS_LATITUDE D_GEO,
    CS_LONGITUDE D_GEO,
    CS_FROM_RD_ID D_INTEGER,
    CS_FROM_AL_ID D_STR_20,
    CS_FROM_REG_ID D_INTEGER,
    CS_DISCOUNT D_PERCENT)
RETURNS (
    RES_CS_ID D_ID)
AS
begin
  if (:cs_discount is not null and :cs_discount not between 0 and 7) then
    exception e_message 'Размер скидки должен быть в пределах 0 и 7.';

  update or insert into clientshops(
    cs_id,
    cs_from_c_id,
    cs_name,
    cs_name_lat,
    cs_from_adl_id,
    cs_latitude,
    cs_longitude,
    cs_contact_fio,
    cs_phone,
    cs_hours,
    cs_weekend_hours,
    cs_from_un_id,
    cs_from_rd_id,
    cs_from_al_id,
    cs_from_reg_id,
    cs_discount
    )
  values(
    iif(:cs_id >= 0, :cs_id, null),
    :cs_from_c_id,
    :cs_name,
    :cs_name_lat,
    :cs_from_adl_id,
    :cs_latitude,
    :cs_longitude,
    :cs_contact_fio,
    :cs_phone,
    :cs_hours,
    :cs_weekend_hours,
    :cs_from_un_id,
    :cs_from_rd_id,
    :cs_from_al_id,
    :cs_from_reg_id,
    :cs_discount
    )
  matching (cs_id)
  returning cs_id
  into :res_cs_id;

  suspend;
end
