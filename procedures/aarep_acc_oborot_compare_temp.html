CREATE PROCEDURE AAREP_ACC_OBOROT_COMPARE_TEMP(
    REPORT_REGION_ID INTEGER,
    REPORT_YEAR INTEGER,
    REPORT_MONTH INTEGER,
    REPORT_PERIOD_KIND INTEGER,
    REPORT_PERIOD_KOLICH INTEGER)
RETURNS (
    REP_ACCOUNT VARCHAR(20),
    REP_SUMMA DECIMAL(18,4),
    REP_PERIOD_ID INTEGER)
AS
declare variable ACC_DT varchar(20);
declare variable ACC_KT varchar(20);
declare variable OB_SUMMA decimal(18,4);
declare variable PER_KOLICH integer;
declare variable NN integer;
declare variable DD1 date;
declare variable DD2 date;
declare variable MM integer;
declare variable YY integer;
declare variable OB_DATE date;
declare variable AL_ID varchar(20);
declare variable REG_ID integer;
declare variable BG_KIND integer;
begin
  rep_period_id = 0;
  if (report_period_kolich > 12) then
    report_period_kolich = 12;
  if (report_period_kind = 0) then
    per_kolich = :report_period_kolich;
  if (report_period_kind = 1) then
    per_kolich = :report_period_kolich * 3;
  if (report_period_kind = 2) then
    per_kolich = :report_period_kolich * 6;
  if (report_period_kind = 3) then
    per_kolich = :report_period_kolich * 12;

  yy = :report_year;
  if (report_month < 12) then
    mm = :report_month + 1;
  else
  begin
    mm = 1;
    yy = :report_year + 1;
  end
  dd2 = cast(('01.' || :mm || '.' || :yy) as date) - 1;

  nn = :report_year * 12 + :report_month - per_kolich + 1;
  yy = cast(:nn / 12 as integer);
  if (12 * :yy >= :nn) then
    yy = :yy - 1;
  mm = :nn - :yy * 12;
  dd1 = cast(('01.' || :mm || '.' || :yy) as date);

  for select obs_date, obs_from_al_id, bg_kind, sum(obs_dt_ekv - obs_kt_ekv)
      from obsummary, region_list(:report_region_id, 1, -1, 1), acc_list, balgroup
      where obs_date between :dd1 and :dd2 and
            obs_from_al_id = al_id and
            al_from_reg_id = id_list and
            al_from_bg_id = bg_id
      group by obs_date, obs_from_al_id, bg_kind
      having sum(obs_dt_ekv - obs_kt_ekv) <> 0
      into :ob_date, :rep_account, :bg_kind, :rep_summa
  do
  begin
    if (bg_kind = 4) then
      rep_summa = :rep_summa;

    yy = extract(year from ob_date);
    mm = extract(month from ob_date);

    if (report_period_kind = 0) then
    begin
      rep_period_id = (:report_year * 12 + :report_month) - (:yy * 12 + :mm);
    end
    if (report_period_kind = 1) then
    begin
      rep_period_id = (((:report_year * 12 + :report_month) - (:yy * 12 + :mm)) / 3);
    end
    if (report_period_kind = 2) then
    begin
      rep_period_id = (((:report_year * 12 + :report_month) - (:yy * 12 + :mm)) / 6);
    end
    if (report_period_kind = 3) then
    begin
      rep_period_id = (((:report_year * 12 + :report_month) - (:yy * 12 + :mm)) / 12);
    end
    rep_period_id = 12 - :rep_period_id;
    suspend;
  end

  for select al_id, al_from_reg_id
      from balgroup, acc_list, region_list(:report_region_id, 1, -1, 1)
      where bg_kind = 3 and
            al_from_bg_id = bg_id and
            al_from_reg_id = id_list
      into :al_id, :reg_id
  do
  begin
    for select ob_ol_date, ob_account_dt, sum(ob_summa)
        from operbook
        where ob_ol_is_active = 1 and
              ob_ol_date between :dd1 and :dd2 and
              ob_ol_reg_id = :reg_id and
              ob_account_kt = :al_id
        group by ob_ol_date, ob_account_dt
        into :ob_date, :rep_account, :ob_summa
    do
    begin
      yy = extract(year from ob_date);
      mm = extract(month from ob_date);

      if (report_period_kind = 0) then
      begin
        rep_period_id = (:report_year * 12 + :report_month) - (:yy * 12 + :mm);
      end
      if (report_period_kind = 1) then
      begin
        rep_period_id = (((:report_year * 12 + :report_month) - (:yy * 12 + :mm)) / 3);
      end
      if (report_period_kind = 2) then
      begin
        rep_period_id = (((:report_year * 12 + :report_month) - (:yy * 12 + :mm)) / 6);
      end
      if (report_period_kind = 3) then
      begin
        rep_period_id = (((:report_year * 12 + :report_month) - (:yy * 12 + :mm)) / 12);
      end

      rep_period_id = 12 - :rep_period_id;

      rep_summa = -:ob_summa;
      suspend;
    end
    for select ob_ol_date, ob_account_kt, sum(ob_summa)
        from operbook
        where ob_ol_is_active = 1 and
              ob_ol_date between :dd1 and :dd2 and
              ob_ol_reg_id = :reg_id and
              ob_account_dt = :al_id
        group by ob_ol_date, ob_account_kt
        into :ob_date, :rep_account, :ob_summa
    do
    begin
      yy = extract(year from ob_date);
      mm = extract(month from ob_date);

      if (report_period_kind = 0) then
      begin
        rep_period_id = (:report_year * 12 + :report_month) - (:yy * 12 + :mm);
      end
      if (report_period_kind = 1) then
      begin
        rep_period_id = (((:report_year * 12 + :report_month) - (:yy * 12 + :mm)) / 3);
      end
      if (report_period_kind = 2) then
      begin
        rep_period_id = (((:report_year * 12 + :report_month) - (:yy * 12 + :mm)) / 6);
      end
      if (report_period_kind = 3) then
      begin
        rep_period_id = (((:report_year * 12 + :report_month) - (:yy * 12 + :mm)) / 12);
      end

      rep_period_id = 12 - :rep_period_id;

      rep_summa = :ob_summa;
      suspend;
    end

  end
end
