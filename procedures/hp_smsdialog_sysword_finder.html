CREATE PROCEDURE HP_SMSDIALOG_SYSWORD_FINDER(
    SDR_SMS_TEXT VARCHAR(255))
RETURNS (
    SDSW_ID INTEGER)
AS
declare variable sdsw_done integer;
begin
  sdsw_done = 0;
  sdsw_id = 0;

  if (sdr_sms_text similar to '[[:DIGIT:]]{4}') then -- если входящий СМС состоит из 4 цифр
  begin
    sdsw_id = -1;
    suspend;
    exit;
  end

  sdr_sms_text = sdr_sms_text || ' ';
-- ищет командные слова из текста СМС на соответствие системных комманд
-- командные слова в SMSDIALOGSYSWORDS должны быть заполнены через квадратные скобки
  for select sdsw_id
      from smsdialogsyswords
      where sdsw_is_active = 1 and
            sdsw_id > 0 and
            :sdr_sms_text starting sdsw_command_names
      order by sdsw_command_names desc
      rows 1
      into :sdsw_id
  do
  begin
    sdsw_done = 1;
    suspend;
  end
  if (sdsw_done = 0) then
  begin
    sdsw_id = 0;
    suspend;
  end
end
