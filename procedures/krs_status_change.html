CREATE PROCEDURE KRS_STATUS_CHANGE(
    KE_KEXPERT_ID DECIMAL(18,0),
    KE_REGION_ID INTEGER)
AS
declare variable k decimal(18,0);
declare variable d date;
declare variable chet integer;
declare variable d_prior date;
begin
  /* Procedure Text */
  chet = 0;
  for select ke_id, ke_date
      from krexpert
      where ke_from_p_id = :ke_kexpert_id and
            ke_from_reg_id = :ke_region_id
      order by ke_date desc
      into :k, :d
  do
  begin
    chet = :chet + 1;
    if (chet > 1) then
    begin
      update krexpert
      set ke_date_end = :d_prior - 1
      where ke_id = :k and
            ke_date_end <> :d_prior - 1;
    end
    else
    begin
      update krexpert
      set ke_date_end = '31.12.9999'
      where ke_id = :k and
            ke_date_end <> '31.12.9999';
    end
    d_prior = d;
  end
end
