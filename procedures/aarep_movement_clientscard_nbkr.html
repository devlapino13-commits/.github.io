CREATE PROCEDURE AAREP_MOVEMENT_CLIENTSCARD_NBKR(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REG_ID D_INTEGER,
    REG_NAME D_STR_CODE,
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    CARD_ID D_INTEGER,
    CCA_ID D_INTEGER,
    DATE_ISSUE D_DATE,
    K_ID D_ID,
    K_KONTRACT_NO D_STR_20,
    K_DISB_DATE D_DATE,
    PR_NAME D_STR_100,
    CREDIT_LIMIT D_CURRENCY,
    BALANCE_BEG D_CURRENCY,
    SUMMA_DT D_CURRENCY,
    SUMMA_KT D_CURRENCY,
    BALANCE_END D_CURRENCY)
AS
declare variable reg_lst d_str_8100;
declare variable acc d_str_10;
declare variable dt_buff d_date;
declare variable delta d_currency;
declare variable clo_id d_integer;
declare variable k_from_pr_id d_id;
declare variable dt_temp d_date;
begin
  select
    coalesce(list(id_list, ';'), '')
  from region_list(:rep_region_id, 1, 3, 1)
  into
    :reg_lst;

  for
    select
      clo_id,
      reg_unicode,
      reg_name,
      c_id,
      c_name_fio,
      cal_card_id,
      cal_from_cca_id,
      cal_date_issue,
      k_id,
      k_kontract_no,
      k_disb_date,
      k_from_pr_id,
      k_amount_approved,
      pr_name
    from cardlistloan
      inner join cardlist
        on cal_card_id = clo_card_id
      inner join clients
        on c_id = clo_from_c_id
      inner join cardplace
        on cpl_id = cal_last_cpl_id
      inner join region
        on reg_unicode = cpl_from_reg_id
      inner join kontract
        on k_id = clo_from_k_id
          and k_disb_date between :rep_date_begin and :rep_date_end
      inner join product 
        on pr_id = k_from_pr_id
  into
    :clo_id,
    :reg_id,
    :reg_name,
    :c_id,
    :c_name_fio,
    :card_id,
    :cca_id,
    :date_issue,
    :k_id,
    :k_kontract_no,
    :k_disb_date,
    :k_from_pr_id,
    :credit_limit,
    :pr_name
  do
  begin
    if (reg_lst not containing reg_id)
      then continue;

    acc = '20001/1';
    with cards as
    (
      select distinct clo2.clo_card_id as card_id
      from cardlistloan as clo1
        inner join cardlistloan as clo2
          on clo2.clo_from_k_id = clo1.clo_from_k_id
      where clo1.clo_card_id = :card_id
    )
    select
      sum(iif(ol_date < :rep_date_begin, iif(ob_short_acc_kt = :acc, ob_summa, 0) - iif(ob_short_acc_dt = :acc, ob_summa, 0), 0)),
      sum(iif(ol_date between :rep_date_begin and :rep_date_end and ob_short_acc_dt = :acc, ob_summa, 0)),
      sum(iif(ol_date between :rep_date_begin and :rep_date_end and ob_short_acc_kt = :acc, ob_summa, 0)),
      sum(iif(ol_date <= :rep_date_end, iif(ob_short_acc_kt = :acc, ob_summa, 0) - iif(ob_short_acc_dt = :acc, ob_summa, 0), 0))
    from operbook
      inner join (
        select distinct ol_id as olo_id
          from operkred
            inner join operlist
              on ol_id = (select coalesce(ol_parent_id, ol_id) from operlist where ol_id = ok_from_ol_id)
          where ok_from_k_id = :k_id
        union
        select ol_id as olo_id
          from pvnopers
              inner join cards
                on card_id = po_card_id
              inner join operlist
                on ol_parent_id = (select coalesce(ol_parent_id, ol_id) from operlist where ol_id = po_from_ol_id )
                  or ol_id = (select coalesce(ol_parent_id, ol_id) from operlist where ol_id = po_from_ol_id )
            where po_card_k_id = :k_id
              and po_is_mb_card = 1
              and po_from_ol_id > 0
              and po_is_delete = 0
        union
        select ol_id as olo_id
          from ipc_exd_detail
            inner join cards
              on card_id = ied_from_card_id
            inner join operlist
              on ied_from_ol_id in (ol_id, ol_parent_id)
          where ied_from_k_id = :k_id
            and ied_from_ol_id > 0
        ) as a
      on a.olo_id = ob_from_ol_id
      inner join operlist
        on ol_id = ob_from_ol_id
          and is_delete = 0
    where ob_short_acc_dt = :acc
      or ob_short_acc_kt = :acc
    into
      :balance_beg,
      :summa_dt,
      :summa_kt,
      :balance_end;

    if (exists(
      select *
      from cardlistloan
      where clo_from_k_id = :k_id
        and clo_id < :clo_id))
    then credit_limit = null;

    -- заплатки
    if (card_id in (2028, 2030, 2043, 2015)) then
    begin
      dt_buff = iif(card_id in (2028, 2030), '23.04.2021', '28.04.2021');
      balance_beg = 0;
      summa_kt = iif(dt_buff between rep_date_begin and rep_date_end, credit_limit, 0);
      summa_dt = summa_kt;
      balance_end = balance_beg + summa_kt - summa_dt;
    end
    if (rep_date_end >= '30.12.2021') then
    begin
      delta =
        case card_id
          when 22263 then 4400
          when 16184 then 50
        else 0 end;
      if (delta > 0) then
      begin
        if (balance_end > 0) then
        begin
          balance_end = balance_end - delta;
          summa_dt = summa_dt + delta;
        end else begin
          balance_end = balance_end + delta;
          summa_kt = summa_kt + delta;
        end
      end
    end

    dt_temp = :k_disb_date - 1;
    if (not exists(
      select *
      from kontstate
        inner join cardlistloan
          on clo_card_id = :card_id
            and clo_from_k_id = ks_from_k_id
      where ks_from_k_id <> :k_id
        and :dt_temp between ks_date_begin and ks_date_end
        and ks_ol_is_delete = 0
        and ks_state = 1))
    then balance_end = 0;

    if (credit_limit is not null) then
      suspend;
  end
end
