CREATE PROCEDURE CM_AZ_0034_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable KAL_NEW_KF_OFFICER ID_TYPE;
declare variable UN_FROM_C_ID ID_TYPE;
declare variable KC_FROM_C_ID ID_TYPE;
declare variable UN_NAME varchar(30);
declare variable PR_CODE varchar(100);
declare variable JK_SYSTEM_CODE varchar(100);
declare variable JK_NAME varchar(100);
begin
  if (K_ID = 0) then
    exit;
  for
  select KAL_NEW_KF_OFFICER, KC.KC_AMOUNT_APPROVED, KC.KC_FROM_C_ID, K_DISB_DATE
  from KONTRACT K
  inner join KNAKTLIST KAL on (K_FROM_KF_ID = KAL_FROM_KF_ID and
        KAL.IS_DELETE = 0 and
        K.K_DISB_DATE between KAL_DATE and KAL_DATE_END)
  inner join KNCLIENT KC on (K_ID = KC_FROM_K_ID and
        KC_AMOUNT_APPROVED > 0)
  where K_ID = :K_ID
  into KAL_NEW_KF_OFFICER, KC_AMOUNT, KC_FROM_C_ID, KF_DISB_DATE
  do
  begin
      if (coalesce(KAL_NEW_KF_OFFICER, 0) <> 0) then
      begin
        select JK_SYSTEM_CODE, JK.JK_NAME
        from HR_PSTATUS, HR_ORDERS, JOBLIST JL, JOBKIND JK
        where PS_STATUS = 1 and
              PS_FROM_HO_ID = HO_ID and
              PS_FROM_JL_ID = JL_ID and
              PS_FROM_C_ID = :KAL_NEW_KF_OFFICER and
              :KF_DISB_DATE between PS_DATE_BEGIN and PS_DATE_END and
              JK.JK_ID = JL.JL_KIND
        order by PS_DATE_BEGIN desc rows 1
        into JK_SYSTEM_CODE, JK_NAME;
    
        if (JK_SYSTEM_CODE in ('JobSystem.LO_Standart') and
            JK_NAME = 'Kiçik kredit işçisi') then
        begin
          /*
            SYSDBA
            |Для КС со стажем 90 дней и менее нужно поставить блок на сумму чтобы не смогли принять Заявку на сумму более 2001 манат на 1 клиента.
        */
          if (KC_AMOUNT > 2000) then
          begin
            res_info_text = (select C_NAME_FIO
                             from CLIENTS
                             where C_ID = :KC_FROM_C_ID);
            res_error_text = 'Kiçik kredit işçisi üçün veriləcək kreditin məbləği 2000 manatdan yuxarı ola bilməz';
            res_error_is_critical = 1;
            suspend;
          end
    
        end
      end
  end
end
