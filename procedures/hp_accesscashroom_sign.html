CREATE PROCEDURE HP_ACCESSCASHROOM_SIGN(
    ACR_ID D_INTEGER,
    C_ID D_ID = null)
RETURNS (
    ERR_TEXT D_STR_255)
AS
declare variable dt_curr d_datetime;
declare variable status d_smallint;
declare variable edl_id d_id;
declare variable issuer_id d_id;
declare variable cashier_id d_id;
declare variable person_id d_id;
declare variable director_id d_id;
declare variable issuer_datesign d_datetime;
declare variable cashier_datesign d_datetime;
declare variable person_datesign d_datetime;
declare variable director_datesign d_datetime;
declare variable num_sign d_smallint;
begin
  err_text = '';
  dt_curr = current_timestamp;

  for
    select
      acr_status,
      acr_edl_id,
      acr_issuer_id,
      acr_cashier_id,
      acr_person_id,
      acr_director_id,
      acr_issuer_datesign,
      acr_cashier_datesign,
      acr_person_datesign,
      acr_director_datesign
    from accesscashroom
    where acr_id = :acr_id
  into
    :status,
    :edl_id,
    :issuer_id,
    :cashier_id,
    :person_id,
    :director_id,
    :issuer_datesign,
    :cashier_datesign,
    :person_datesign,
    :director_datesign
  do
  begin
    if (c_id is null) then
    begin
      if (cashier_datesign is null) then
        cashier_datesign = (
          select eds_status_datetime
          from edoc_sign
          where eds_from_edl_id = :edl_id
            and eds_from_c_id = :cashier_id
            and eds_status = 1);
      if (person_datesign is null) then
        person_datesign = (
          select eds_status_datetime
          from edoc_sign
          where eds_from_edl_id = :edl_id
            and eds_from_c_id = :person_id
            and eds_status = 1);
      if (director_datesign is null) then
        director_datesign = (
          select eds_status_datetime
          from edoc_sign
          where eds_from_edl_id = :edl_id
            and eds_from_c_id = :director_id
            and eds_status = 1);
    end else
    begin
      if (issuer_id = c_id and issuer_datesign is null and status = 0)
        then issuer_datesign = dt_curr;
      else
      if (cashier_id = c_id and cashier_datesign is null and status in (1, 2)) then
      begin
        cashier_datesign = dt_curr;
        if (edl_id > 0) then
          update edoc_sign set
            eds_status = 1,
            eds_enum_edoc_sign_kind = 5,
            eds_status_datetime = :cashier_datesign
          where eds_from_edl_id = :edl_id
            and eds_from_c_id = :cashier_id
            and eds_status = 0;
      end else
      if (person_id = c_id and person_datesign is null and status in (1, 2)) then
      begin
        person_datesign = dt_curr;
        if (edl_id > 0) then
          update edoc_sign set
            eds_status = 1,
            eds_enum_edoc_sign_kind = 5,
            eds_status_datetime = :person_datesign
          where eds_from_edl_id = :edl_id
            and eds_from_c_id = :person_id
            and eds_status = 0;
      end else
      if (director_id = c_id and director_datesign is null and status in (1, 2)) then
      begin
        director_datesign = dt_curr;
        if (edl_id > 0) then
          update edoc_sign set
            eds_status = 1,
            eds_enum_edoc_sign_kind = 5,
            eds_status_datetime = :director_datesign
          where eds_from_edl_id = :edl_id
            and eds_from_c_id = :director_id
            and eds_status = 0;
      end else
        err_text = f_format('Ошибка подписания "распоряжения о допуске" №%1', :acr_id);
    end

    if (err_text = '') then
    begin
      num_sign =
        iif(issuer_datesign is null, 0, 1) +
        iif(cashier_datesign is null, 0, 1) +
        iif(person_datesign is null, 0, 1) +
        iif(director_datesign is null, 0, 1);
      update accesscashroom set
        acr_issuer_datesign = :issuer_datesign,
        acr_cashier_datesign = :cashier_datesign,
        acr_person_datesign = :person_datesign,
        acr_director_datesign = :director_datesign,
        acr_status =
          case :num_sign
            when 1 then 1
            when 4 then 3
          else 2 end
      where acr_id = :acr_id;
    end
    suspend;
  end
end
