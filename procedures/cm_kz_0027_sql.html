CREATE PROCEDURE CM_KZ_0027_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable C_ID D_ID;
declare variable POINT D_INTEGER;
declare variable KC_AMOUNT_APPROVED D_CURRENCY;
declare variable pr_id d_integer;
begin
    for select kc.kc_from_c_id, kc.kc_amount_approved, k.k_from_pr_id
    from knclient kc
    inner join kontract k on k.k_id = kc.kc_from_k_id
    where kc.kc_from_k_id = :k_id
        and kc.kc_amount_approved > 0
    into :c_id , :kc_amount_approved, :pr_id
    do
    begin
        :point = f_point(:c_id, current_date, 0);
        if ((coalesce(:point, 0) between 0 and 6) and :kc_amount_approved > 650000) then
        begin
             if (:point>= 1 and
                 :pr_id in (524,525,526,527) and
                 exists(select * from knclient kc
                        inner join kontstate ks on ks.ks_from_k_id = kc.kc_from_k_id and
                                                   ks.ks_ol_is_delete = 0 and
                                                   ks.ks_state in (1,2) and
                                                   current_date between ks.ks_date_begin and ks.ks_date_end
                        where kc.kc_from_c_id = :c_id and
                              kc.kc_amount_approved > 0)) then
                              exit;
            res_error_is_critical = 2;
            res_error_text = 'невозможно выдать сумму свыше 650 000 тенге — при текущем количестве баллов клиенту доступно только до 650 000 тенге.';
            suspend;
            exit;
        end
    end
end
