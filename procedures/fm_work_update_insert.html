CREATE PROCEDURE FM_WORK_UPDATE_INSERT(
    KIW_ID D_ID,
    KIW_FROM_K_ID D_ID,
    KIW_FROM_C_ID D_ID,
    KIW_MEMO D_STR_255,
    KIW_DAYS D_INTEGER,
    KIW_PRICE D_CURRENCY,
    UN_ID D_ID)
RETURNS (
    ERR_TEXT D_STR_255)
AS
begin
  err_text = '';
  if (coalesce(kiw_from_c_id, 0) = 0) then
    err_text = 'Не указан подрядчик';
  else
  if (coalesce(kiw_memo, '') = '') then
    err_text = 'Не указано наименование работ';
  else
  if (coalesce(kiw_days, '') = '') then
    err_text = 'Не указан срок выполнения';
  else
  if (coalesce(kiw_price, 0) = 0) then
    err_text = 'Не указана стоимость работ';

  -- Вставка/обновление работ контракта (kiw_istisnaa_work)
  if (err_text = '') then
  begin
    if (kiw_id is null) then -- Вставка
    begin
      insert into kontract_istisnaa_work (kiw_from_k_id, kiw_from_c_id, kiw_memo, kiw_days, kiw_price, kiw_is_delete)
      values (:kiw_from_k_id, :kiw_from_c_id, :kiw_memo, :kiw_days, :kiw_price, 0)
      returning kiw_id
      into kiw_id;
    end
    else -- Обновление
    begin
      /*
      if (exists (select *
                  from kontract_istinaa_prepay kip
                  inner join actionlist acl on acl.acl_id = kip.kip_prepay_acl_id
                                                )) then
        err_text = 'Операция запрещена. Сначала удалите бухгалтерскую проводку.';
      */
      update kontract_istisnaa_work
      set kiw_from_c_id = :kiw_from_c_id,
          kiw_memo = :kiw_memo,
          kiw_days = :kiw_days,
          kiw_price = :kiw_price
      where kiw_id = :kiw_id;
    end
  end
  suspend;
end
