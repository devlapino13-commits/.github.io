CREATE PROCEDURE HP_FIND_ERROR_VERIF(
    C_ID DECIMAL(18,0))
RETURNS (
    REZ_TEXT D_STR_100)
AS
declare variable acl_status d_smallint;
begin
  rez_text = '';

  if (exists(select *
             from clients cl
             where cl.c_id = :c_id and
                   cl.c_type <> 0)) then
  begin
    rez_text = 'Не физ. лицо';
  end
  if (exists(select *
             from grs_clients gc
             where gc.gc_from_c_id = :c_id and
                   gc.gc_status = -4 and
                   gc.gc_is_last = 1)) then
  begin
    rez_text = 'Паспорт недействительный (ответ ГРС), отправка на ручную верификацию запрещена';
  end

  if (rez_text = '') then
  begin
    select coalesce(acl_status, -9)
    from actionlist
    where acl_object_kind = 0 and
          acl_object_id = :c_id and
          acl_from_rd_id in (f_get_rd_id('ActionGroup.Passport_Verify'), f_get_rd_id('ActionGroup.Passport_Verify_OiVO'))
    order by acl_datetime desc
    rows 1
    into acl_status;

    if (acl_status in (0, 1)) then
      rez_text = 'Запрос принят в обработку, подождите.';
  end
  suspend;
end
