CREATE PROCEDURE PRINT_NEW_SCHEDULE_CLI(
    PRINT_K_ID DECIMAL(18,0))
RETURNS (
    C_NAME_FIO VARCHAR(255),
    SH_FROM_K_ID DECIMAL(18,0),
    SH_NUMBER INTEGER,
    SH_DATE DATE,
    SHCL_OS DECIMAL(18,4),
    SHCL_PERC DECIMAL(18,4),
    SHCL_KOMMIS DECIMAL(18,4),
    SHCL_PERC_TAX DECIMAL(18,4),
    SHCL_TOTAL DECIMAL(18,4),
    SHCL_BALANCE_BEFORE DECIMAL(18,4),
    SHCL_BALANCE_AFTER DECIMAL(18,4),
    SH_DAYS INTEGER,
    IS_HEADER SMALLINT)
AS
declare variable sh_dd date;
declare variable k_amount_approved decimal(18,4);
declare variable kc_amount_approved decimal(18,4);
declare variable prop_perc decimal(18,7);
declare variable sh_balance_before_main decimal(18,4);
declare variable sh_balance_after_main decimal(18,4);
declare variable c_name_fio_temp varchar(255);
declare variable cm_is_client_mode integer;
declare variable c_id id_type;
begin
-- operkred_fix
  select cm_is_client_mode
  from company
  into cm_is_client_mode;

  if (cm_is_client_mode = 0) then
  begin
    shcl_kommis = 0;
    shcl_perc_tax = 0;
    sh_from_k_id = print_k_id;
    sh_days = null;
    select sum(ok_os), max(ok_ol_date)
    from operkred
    where ok_ol_is_active = 1 and
          ok_from_c_id = 0 and
          ok_from_k_id = :print_k_id

    into :sh_balance_before_main, :sh_dd;
    sh_balance_after_main = :sh_balance_before_main;

    k_amount_approved = 0;
    select kont.k_amount_approved
    from kontract kont
    where kont.k_id = :print_k_id
    into :k_amount_approved;

    c_name_fio_temp = '';
    for select cl.c_name_fio, kncl.kc_amount_approved
        from knclient kncl
        inner join clients cl on cl.c_id = kncl.kc_from_c_id
        where kncl.kc_from_k_id = :print_k_id and
              kncl.kc_amount_approved > 0
        into :c_name_fio, :kc_amount_approved
    do
    begin
--IS_HEADER = IS_HEADER + 1;
      shcl_balance_after = sh_balance_after_main;
      shcl_balance_before = sh_balance_before_main;
      sh_days = null;

      for select ok_sh_id, ok_sh_date, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
          from operkred
          where ok_from_k_id = :print_k_id and
                ok_ol_is_active = 1 and
                ok_type < 7 and
                ok_from_c_id = 0
          group by ok_sh_id, ok_sh_date
          having sum(ok_os + ok_sh_os) <> 0 or sum(ok_perc + ok_sh_perc) <> 0
          order by ok_sh_date
          into :sh_number, :sh_date, :shcl_os, :shcl_perc
      do
      begin
        prop_perc = kc_amount_approved * 100 / k_amount_approved;
        shcl_os = round(shcl_os * prop_perc / 100, 2);
        shcl_perc = round(shcl_perc * prop_perc / 100, 2);
        shcl_perc_tax = round(shcl_perc_tax * prop_perc / 100, 2);

        shcl_balance_after = :shcl_balance_after - :shcl_os;
        if (sh_days is not null) then
          sh_days = cast((:sh_date - :sh_dd) as integer);
        else
          sh_days = null;
        shcl_total = shcl_os + shcl_perc + shcl_perc_tax;

        if (c_name_fio_temp <> c_name_fio) then
        begin
          is_header = 1;
          c_name_fio_temp = c_name_fio;
        end
        else
          is_header = 0;

        if (shcl_total <> 0) then
          suspend;
        sh_days = 0;
        sh_dd = sh_date;
        shcl_balance_before = :shcl_balance_before - :shcl_os;
      end
    end
    exit;
  end

  shcl_kommis = 0;
  shcl_perc_tax = 0;
  sh_from_k_id = print_k_id;
  sh_days = null;
  select sum(ok_os), max(ok_ol_date)
  from operkred
  where ok_ol_is_active = 1 and
        ok_from_c_id = 0 and
        ok_from_k_id = :print_k_id

  into :sh_balance_before_main, :sh_dd;
  sh_balance_after_main = :sh_balance_before_main;

  k_amount_approved = 0;
  select kont.k_amount_approved
  from kontract kont
  where kont.k_id = :print_k_id
  into :k_amount_approved;

  c_name_fio_temp = '';
  for select cl.c_name_fio, kncl.kc_amount_approved, kc_from_c_id
      from knclient kncl
      inner join clients cl on cl.c_id = kncl.kc_from_c_id
      where kncl.kc_from_k_id = :print_k_id and
            kncl.kc_amount_approved > 0
      into :c_name_fio, :kc_amount_approved, c_id
  do
  begin
--IS_HEADER = IS_HEADER + 1;
    shcl_balance_after = sh_balance_after_main;
    shcl_balance_before = sh_balance_before_main;
    sh_days = null;

    for select ok_sh_id, ok_sh_date, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
        from operkred
        where ok_from_k_id = :print_k_id and
              ok_ol_is_active = 1 and
              ok_type < 7 and
              ok_from_c_id = :c_id
        group by ok_sh_id, ok_sh_date
        having sum(ok_os + ok_sh_os) <> 0 or sum(ok_perc + ok_sh_perc) <> 0
        order by ok_sh_date
        into :sh_number, :sh_date, :shcl_os, :shcl_perc
    do
    begin
      /*
        PROP_PERC = kc_amount_approved * 100/k_amount_approved;
        SHCL_OS = round(SHCL_OS * PROP_PERC/100,2);
        SHCL_PERC = round(SHCL_PERC * PROP_PERC/100,2);
        SHCL_PERC_TAX = round(SHCL_PERC_TAX * PROP_PERC/100,2);
        */
      shcl_balance_after = :shcl_balance_after - :shcl_os;

      if (sh_days is not null) then
        sh_days = cast((:sh_date - :sh_dd) as integer);
      else
        sh_days = null;
      shcl_total = shcl_os + shcl_perc + shcl_perc_tax;

      if (c_name_fio_temp <> c_name_fio) then
      begin
        is_header = 1;
        c_name_fio_temp = c_name_fio;
      end
      else
        is_header = 0;

      if (shcl_total <> 0) then
        suspend;
      sh_days = 0;
      sh_dd = sh_date;
      shcl_balance_before = :shcl_balance_before - :shcl_os;

    end
  end

end
