CREATE PROCEDURE AAREP_CALC_AMORT_KOMMIS(
    REPORT_REGION_ID DECIMAL(18,0),
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_NEED_PERC INTEGER)
RETURNS (
    RES_AMOUNT_BEGIN DECIMAL(18,4),
    RES_AMOUNT_END DECIMAL(18,4),
    RES_AMOUNT_PERIOD DECIMAL(18,4),
    RES_KOMMIS_AMOUNT DECIMAL(18,4),
    AAREP_CURRENCY VARCHAR(3),
    AAREP_REGION_NAME VARCHAR(100),
    AAREP_PR_ID INTEGER,
    AAREP_REGION_ID INTEGER,
    AAREP_K_FOND VARCHAR(20),
    RES_KOMMIS_PAYED DECIMAL(18,4),
    AAREP_ACCOUNT_PROFIT VARCHAR(20),
    AAREP_ACCOUNT_PAYED VARCHAR(20),
    AAREP_PRODUCT_NAME VARCHAR(50),
    RES_KRED_AMOUNT DECIMAL(18,4),
    RES_NSP_PERIOD DECIMAL(18,2),
    K_KONTRACT_NO D_STR_100,
    K_DISB_DATE DATE)
AS
declare variable curr_days1 decimal(18,4);
declare variable curr_days2 decimal(18,4);
declare variable max_days decimal(18,4);
declare variable kont_state integer;
declare variable aarep_loan_t integer;
declare variable st_staff integer;
declare variable st_client integer;
declare variable cash_id decimal(18,0);
declare variable rep_period integer;
declare variable rep_period_prior integer;
declare variable k_kredline_end date;
declare variable res_disb_date date;
declare variable res_last_date date;
declare variable aarep_k_id decimal(18,0);
declare variable res_perc_daily decimal(18,9);
declare variable aarep_kontract_no varchar(20);
declare variable aarep_client_name varchar(100);
declare variable aarep_loan_type varchar(50);
declare variable last_office_id integer;
declare variable last_office_name varchar(100);
declare variable kcp_kommis_transfer decimal(18,4) = 0;
declare variable kcp_kommis_transfer_payed decimal(18,4) = 0;
declare variable res_amount_begin1 decimal(18,4) = 0;
declare variable kcp_kommis_transfer_payed1 decimal(18,4) = 0;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  rep_period = extract(year from :report_date_end) * 100 + extract(month from :report_date_end);
  rep_period_prior = extract(year from dateadd(month, -1, :report_date_end)) * 100 + extract(month from dateadd(month, -1, :report_date_end));

  res_amount_begin = 0;
  res_amount_end = 0;
  res_amount_period = 0;
  res_nsp_period = 0;
  res_kred_amount = 0;
  res_kommis_amount = 0;
  res_kommis_payed = 0;
  aarep_client_name = '';

  for select distinct reg_code || ' ' || reg_name, pr_id, reg_unicode, k_from_f_id, pr_name, k_currency, k_disb_date,
                      k_kontract_no
      from kontcalcperc, product, region, region_list(:report_region_id, 1, -1, 1), kontract
      where k_id = kcp_from_k_id and
            kcp_period = :rep_period and
            k_from_pr_id = pr_id and
            reg_unicode = id_list and
            kcp_from_reg_id = reg_unicode and
            k_disb_kind <> 6
      union
      select reg_code || ' ' || reg_name, pr_id, reg_unicode, k_from_f_id, pr_name, k_currency, k_disb_date,
             k_kontract_no
      from kontcalcperc, product, region, region_list(:report_region_id, 1, -1, 1), kontract
      where k_id = kcp_from_k_id and
            kcp_period = :rep_period and
            k_from_pr_id = pr_id and
            kcp_from_reg_id = id_list and
            kcp_transfer_reg_id = reg_unicode and
            kcp_from_reg_id <> kcp_transfer_reg_id and
            k_disb_kind <> 6
      order by 3, 2, 4
      into :aarep_region_name, :aarep_pr_id, :aarep_region_id, :aarep_k_fond, :aarep_product_name, :aarep_currency,
           :k_disb_date, :k_kontract_no
  do
  begin
    select res_account
    from hp_make_kredit_acclist(:aarep_pr_id, :aarep_region_id, 6) -- Начисленный доп.процент
    into :aarep_account_profit;

    select res_account
    from hp_make_kredit_acclist(:aarep_pr_id, :aarep_region_id, 7) -- Выплаченный доп.процент
    into :aarep_account_payed;

    suspend;
  end
  aarep_account_payed = '';
  aarep_account_profit = '';

  for select k_currency, kcp_from_k_id, pr_name, cur.reg_code || ' ' || cur.reg_name, pr_id, cur.reg_unicode,
             k_from_f_id, reg_last.reg_unicode, reg_last.reg_code || ' ' || reg_last.reg_name, k_disb_date,
             k_kontract_no
      from kontcalcperc, product, region cur, region_list(:report_region_id, 1, -1, 1), kontract, region reg_last
      where k_id = kcp_from_k_id and
            pr_id = k_from_pr_id and
            kcp_period = :rep_period and
            kcp_from_reg_id = id_list and
            id_list = cur.reg_unicode and
            kcp_transfer_reg_id = reg_last.reg_unicode and
            k_disb_kind <> 6
      into :aarep_currency, :aarep_k_id, :aarep_product_name, :aarep_region_name, :aarep_pr_id, :aarep_region_id,
           :aarep_k_fond, last_office_id, last_office_name, :k_disb_date, k_kontract_no
  do
  begin
    res_amount_begin = 0;
    res_amount_end = 0;
    res_amount_period = 0;
    res_nsp_period = 0;
    res_kred_amount = 0;
    res_kommis_amount = 0;
    res_kommis_payed = 0;
    if (report_need_perc <> 0) then
    begin
      select kcp_kommis_saldo
      from kontcalcperc
      where kcp_period = :rep_period_prior and
            kcp_from_k_id = :aarep_k_id
      into :res_amount_begin;

      select kcp_kommis_calced + iif(kcp_from_reg_id = kcp_transfer_reg_id, 0, kcp_kommis_begin_saldo),
             kcp_kommis_payed, kcp_kommis_saldo, kcp_kommis_transfer_payed
      from kontcalcperc
      where kcp_period = :rep_period and
            kcp_from_k_id = :aarep_k_id
      into :res_amount_period, :res_kommis_payed, :res_amount_end, :kcp_kommis_transfer_payed;

      if (aarep_region_id = last_office_id) then
      begin
        res_nsp_period = 0;
        if (k_disb_date >= '26.10.2013') then
          res_nsp_period = res_amount_period * 2 / 100;
        suspend;
      end
      else
      begin
        res_amount_begin1 = res_amount_begin;
        res_amount_begin = 0;
        res_kommis_payed = res_kommis_payed - kcp_kommis_transfer_payed;
        suspend;
        aarep_region_id = last_office_id;
        aarep_region_name = last_office_name;
        res_amount_begin = 0;
        res_amount_end = 0;
        res_amount_period = 0;
        res_kred_amount = 0;
        res_kommis_amount = 0;
        res_kommis_payed = 0;
        res_amount_begin = res_amount_begin1;
        res_kommis_payed = kcp_kommis_transfer_payed;
        res_amount_period = -res_amount_begin1;

        res_nsp_period = 0;
        if (k_disb_date >= '26.10.2013') then
          res_nsp_period = res_amount_period * 2 / 100;

        suspend;
      end
    end
  end
end
