CREATE PROCEDURE KONTR_ACC_LIST(
    K_ID DECIMAL(18,0),
    K_DATE DATE)
RETURNS (
    AC_TYPE INTEGER,
    AC_NO VARCHAR(20),
    AC_IS_LOCAL SMALLINT,
    AC_DELINQ VARCHAR(20),
    AC_OUTBALANCE VARCHAR(20),
    AC_WRITEOFF VARCHAR(20))
AS
declare variable reg_id integer;
declare variable pr_id integer;
begin
  /* Procedure Text */
  select kal_new_reg_id, k_from_pr_id
  from kontract, knaktlist
  where k_id = :k_id and
        k_from_kf_id = kal_from_kf_id and
        kal_date <= :k_date and
        kal_date_end >= :k_date and
        knaktlist.is_delete = 0
  into :reg_id, :pr_id;

  ac_is_local = 0;

  for select at_id
      from acctype
      where at_kind = 1 -- Кредитные счета
      into :ac_type
  do
  begin
    select res_account, res_acc_delinq, res_acc_outbalance, res_acc_writeoff
    from hp_make_kredit_acclist(:pr_id, :reg_id, :ac_type)
    into :ac_no, :ac_delinq, :ac_outbalance, :ac_writeoff;

    suspend;
  end
end
